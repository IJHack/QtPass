73 const QStringList args = app.arguments();
74 bool consumeNextArg = false;
75 for (int i = 1; i < args.count(); ++i) {
76 const auto &arg = args.at(i);
77
78 if (arg == "--") {
79 consumeNextArg = false;
80 appendWithSpaceIfSuffixNotEmpty(text, joinRemainingArgs(args, i + 1));
81 break;
82 }
83
84 if (consumeNextArg) {
85 consumeNextArg = false;
86 continue;
87 }

89 if (arg.startsWith('-')) {
90 // We only collect positional arguments into `text`.
91 // For options in the form `--option value`, skip the separate value
92 // token. Options in the form `--option=value` are fully contained in
93 // `arg`.
94 const bool optionTakesSeparateValue =
95 !arg.contains('=') && i + 1 < args.count() && arg.startsWith("--") &&
96 !arg.startsWith("---") && !args[i + 1].startsWith('-');
97 if (optionTakesSeparateValue)
98 consumeNextArg = true;
99 continue;
100 }
101
102 if (!text.isEmpty())
103 text += " ";
104 text += arg;
105 }
