333auto ConfigDialog::isFingerprintList(const QString &setting) -> bool {
334 // 40 hex characters for OpenPGP v4 keys, 64 for v5/v6 (gpg 2.5+).
335 static const QRegularExpression fingerprint(
336 QStringLiteral("^(?:[0-9A-Fa-f]{40}|[0-9A-Fa-f]{64})$"));
337 const QStringList keys = setting.split(QLatin1Char(' '), Qt::SkipEmptyParts);
338 return std::all_of(keys.cbegin(), keys.cend(), [](const QString &key) {
339 return fingerprint.match(key).hasMatch();
340 });
341}

343// Profile names must be unique: the settings key profiles by name, so a
344// duplicate would silently overwrite the other.
345auto ConfigDialog::problemsOf(const ProfileEntry &entry,
346 const QHash<QString, int> &names) const
347 -> ProfileProblems {
348 ProfileProblems problems;
349 if (entry.name.isEmpty()) {
350 problems.name = tr("This field is required");
351 } else if (names.value(entry.name) > 1) {
352 problems.name = tr("Another profile already has this name");
353 }
354 if (entry.profile.path.isEmpty()) {
355 problems.path = tr("This field is required");
356 }
357 // pass compares PASSWORD_STORE_SIGNING_KEY against the fingerprints in
358 // gpg's VALIDSIG line, so anything shorter than a fingerprint would sign
359 // and then never verify. Refuse it here instead of failing at first use.
360 if (!isFingerprintList(entry.profile.signingKey)) {
361 problems.key = tr("Full key fingerprints only (40 or 64 hexadecimal "
362 "characters), separated by spaces");
363 }
364 return problems;
365}

385auto ConfigDialog::signingStores() const -> SigningStores {
386 // One folder spelled two ways (through a link, another case on a
387 // case-insensitive file system) is one store: compared as the file system
388 // resolves it where it exists. Resolved once per validate(), not once per
389 // pair of profiles.
390 SigningStores resolved;
391 // Keys are looked up in the keyring that signs: the configured GPG home,
392 // as profiles sign with it, not whatever GNUPGHOME QtPass inherited. The
393 // dialog does not edit the home, so the settings are read once, not on
394 // every keystroke; the environment is built each time, as the home may
395 // only appear meanwhile (a key generated from this dialog).
396 if (!m_gpgSettings) {
397 m_gpgSettings = QtPassSettings::load();
398 }
399 const QProcessEnvironment env = ProfileInit::gpgEnvironment(*m_gpgSettings);
400 for (const ProfileEntry &entry : std::as_const(m_entries)) {
401 const QString real = QFileInfo(entry.profile.path).canonicalFilePath();
402 resolved.stores << (real.isEmpty() ? QDir::cleanPath(entry.profile.path)
403 : real);
404 resolved.keys << signerKeys(entry.profile.signingKey, env);
405 }
406 return resolved;
407}

409auto ConfigDialog::signerKeys(const QString &signingKey,
410 const QProcessEnvironment &env) const
411 -> QSet<QString> {
412 // Verification takes the primary or the signing subkey's fingerprint
413 // from VALIDSIG, and gpg signs for a primary key with its subkey: a key
414 // written as the subkey in one profile and as the primary in another is
415 // one signer (#2087). Only complete IDs are asked about, so typing a key
416 // does not start gpg for every prefix.
417 static const QRegularExpression complete(QStringLiteral(
418 "^(?:0X)?(?:[0-9A-F]{8}|[0-9A-F]{16}|[0-9A-F]{40}|[0-9A-F]{64})$"));
419 const QString gpg = ui->gpgPath->text();
420 QSet<QString> keys;
421 for (const QString &key :
422 signingKey.toUpper().split(QLatin1Char(' '), Qt::SkipEmptyParts)) {
423 keys.insert(key);
424 if (gpg.isEmpty() || !complete.match(key).hasMatch()) {
425 continue;
426 }
427 const QString cacheKey = gpg + QLatin1Char('\n') +
428 env.value(QStringLiteral("GNUPGHOME")) +
429 QLatin1Char('\n') + key;
430 auto cached = m_primaryOf.constFind(cacheKey);
431 if (cached == m_primaryOf.cend()) {
432 QString out;
433 // Blocking, once per key: no trustdb check, which a large keyring
434 // would make slow.
435 const int rc = Executor::executeBlocking(
436 env, gpg,
437 {QStringLiteral("--no-tty"), QStringLiteral("--batch"),
438 QStringLiteral("--no-auto-check-trustdb"),
439 QStringLiteral("--with-colons"),
440 QStringLiteral("--with-fingerprint"),
441 QStringLiteral("--with-subkey-fingerprint"),
442 QStringLiteral("--list-keys"), QStringLiteral("--"), key},
443 &out);
444 cached = m_primaryOf.insert(
445 cacheKey, rc == 0 ? primaryFingerprintOf(out, key) : QString());
446 }
447 if (!cached->isEmpty()) {
448 keys.insert(*cached);
449 }
450 }
451 return keys;
452}

454void ConfigDialog::changeEvent(QEvent *event) {
455 QDialog::changeEvent(event);
456 if (event->type() != QEvent::ActivationChange) {
457 return;
458 }
459 // A key is imported somewhere else: a terminal, a key manager, or the
460 // key generator this dialog opens. Whatever keyring gpg uses, the dialog
461 // is in the background meanwhile, so a key gpg did not know is asked
462 // about again on return.
463 if (!isActiveWindow()) {
464 const auto forgotten = m_primaryOf.removeIf(

466 return lookup.second.isEmpty();
467 });
468 m_recheckSigners = m_recheckSigners || forgotten > 0;
469 } else if (m_recheckSigners) {
470 m_recheckSigners = false;
471 validate();
472 }

475auto ConfigDialog::sharingTheKey(int row, const SigningStores &resolved) const
476 -> QStringList {
477 // A list signed for one store verifies in the other when the stores share
478 // a signing key; the folder line names a folder, not a store (#2050).
479 QStringList others;
480 const QSet<QString> &mine = resolved.keys.at(row);
481 if (mine.isEmpty()) {
482 return others;
483 }
484 for (int other = 0; other < m_entries.size(); ++other) {
485 if (other == row || resolved.stores.at(other) == resolved.stores.at(row)) {
486 continue;
487 }
488 if (resolved.keys.at(other).intersects(mine)) {
489 others << m_entries.at(other).name;
490 }
491 }
492 others.sort();
493 return others;
494}

496void ConfigDialog::validate() {
497 QHash<QString, int> names;
498 for (const ProfileEntry &entry : std::as_const(m_entries)) {
499 names[entry.name]++;
500 }
501 bool status = true;
502 const SigningStores resolved = signingStores();
503 for (int row = 0; row < m_entries.size(); ++row) {
504 ProfileProblems problems = problemsOf(m_entries.at(row), names);
505 if (const QStringList others = sharingTheKey(row, resolved);
506 !others.isEmpty()) {
507 problems.sharedKey =
508 tr("%1 signs with the same key for another store. A recipient "
509 "list signed for one of these stores is accepted in the other: "
510 "use a separate signing key for each store whose recipients "
511 "differ.")
512 .arg(others.join(QStringLiteral(", ")));
513 }
514 showProblems(row, problems);
515 status = status && problems.none();
516 }
517 ui->buttonBox->button(QDialogButtonBox::Ok)->setEnabled(status);
518 updatePassGitWarning();
519}

575void ConfigDialog::followActiveProfile(const Profiles &profiles) {
576 // As MainWindow::on_profileBox_currentTextChanged does when switching.
577 AppSettings s = QtPassSettings::load();
578 const QString wasActive = s.activeProfile;
579 QString nowActive;
580 for (const ProfileEntry &entry : std::as_const(m_entries)) {
581 if (!wasActive.isEmpty() && entry.originalName == wasActive) {
582 nowActive = entry.name;
583 }
584 }
585 s.activeProfile = nowActive;
586 const auto active = profiles.constFind(nowActive);
587 if (active != profiles.constEnd()) {
588 s.useGit = active->useGit.value_or(s.useGit);
589 s.autoPush = active->autoPush.value_or(s.autoPush);
590 s.autoPull = active->autoPull.value_or(s.autoPull);
591 }
592 QtPassSettings::save(s);
593}

841void ConfigDialog::initialiseProfileStore(const QString &name,
842 const Profile &profile,
843 const QString &path) {
844 // Only the recipients come through the active backend. passStore is the
845 // profile, or getGpgIdPath() would pre-tick the active store's recipients;
846 // signing keys are per profile, exactly as the profile switch applies them
847 // (MainWindow::on_profileBox_currentTextChanged).
848 const AppSettings settings = QtPassSettings::load();
849 AppSettings profileSettings = settings;
850 profileSettings.passStore = Util::normalizeFolderPath(path);
851 profileSettings.passSigningKey = profile.signingKey;
852 // needsInit() found no .gpg-id: nothing to preselect, and none that
853 // appears meanwhile is taken (it would be read under the active
854 // profile's signing policy).
855 UsersDialog usersDialog(QtPassSettings::getPass(), profileSettings,
856 profileSettings.passStore,
857 UsersDialog::Preselect::Nothing, this);
858 usersDialog.setInitOnAccept(false);
859 usersDialog.setWindowTitle(tr("Select recipients for %1").arg(name));
860 if (usersDialog.exec() != QDialog::Accepted) {
861 return;
862 }
863 QString note;
864 const bool ok = ProfileInit::initialise(
865 path, usersDialog.selectedUsers(), profileSettings,
866 profile.useGit.value_or(settings.useGit), &note);
867 if (!ok) {
868 QMessageBox::warning(this, tr("Could not initialise profile %1").arg(name),
869 note);
870 } else if (!note.isEmpty()) {
871 QMessageBox::information(this, tr("Profile %1").arg(name), note);
872 }
873}

875void ConfigDialog::initializeNewProfiles(const Profiles &existingProfiles) {
876 const Profiles newProfiles = getProfiles();
877 // QMap iterates in name order.
878 for (auto it = newProfiles.cbegin(); it != newProfiles.cend(); ++it) {
879 const QString path = QDir::cleanPath(it.value().path);
880 // A store that was already a profile when the dialog opened, under this
881 // or another name: a rename is not a new store.
882 const bool known = std::any_of(
883 existingProfiles.cbegin(), existingProfiles.cend(),
884 [&it](const Profile &old) { return old.path == it.value().path; });
885 // needsInit is false for a missing folder, so it is created first.
886 if (!known && ensureProfileFolder(path) && ProfileInit::needsInit(path)) {
887 initialiseProfileStore(it.key(), it.value(), path);
888 }
889 }
