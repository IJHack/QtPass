34auto classifyRecord(const QString &record_type) -> GpgRecordType {
35 if (record_type == "pub") {
36 return GpgRecordType::Pub;
37 }
38 if (record_type == "sec") {
39 return GpgRecordType::Sec;
40 }
41 if (record_type == "uid") {
42 return GpgRecordType::Uid;
43 }
44 if (record_type == "fpr") {
45 return GpgRecordType::Fpr;
46 }
47 if (record_type == "sub") {
48 return GpgRecordType::Sub;
49 }
50 if (record_type == "ssb") {
51 return GpgRecordType::Ssb;
52 }
53 if (record_type == "grp") {
54 return GpgRecordType::Grp;
55 }
56 return GpgRecordType::Unknown;

59void handlePubSecRecord(const QStringList &props, bool secret,
60 UserInfo &current_user) {
61 if (props.size() < GPG_MIN_FIELDS) {
62 return;
63 }
64 current_user.key_id = props[GPG_FIELD_KEY_ID];
65 current_user.name = props[GPG_FIELD_USERID];
66 if (!props[GPG_FIELD_VALIDITY].isEmpty()) {
67 current_user.validity = props[GPG_FIELD_VALIDITY][0].toLatin1();
68 }
69
70 // An empty field (GnuPG's "no expiry") leaves the QDateTime null, so
71 // callers distinguish "unset" from "set" with isValid() alone.
72 bool okCreated = false;
73 const qint64 createdSecs = props[GPG_FIELD_CREATED].toLongLong(&okCreated);
74 if (okCreated) {
75 current_user.created = QDateTime::fromSecsSinceEpoch(createdSecs);
76 }
77
78 bool okExpiry = false;
79 const qint64 expirySecs = props[GPG_FIELD_EXPIRY].toLongLong(&okExpiry);
80 if (okExpiry) {
81 current_user.expiry = QDateTime::fromSecsSinceEpoch(expirySecs);
82 }
83
84 current_user.have_secret = secret;
85}

96void handleFprRecord(const QStringList &props, UserInfo &current_user) {
97 if (props.size() < GPG_MIN_FIELDS) {
98 return;
99 }
100 // The key ID from the pub/sec record is the fingerprint's tail; take the
101 // whole fingerprint, but only something shaped like one (40 hex for v4,
102 // 64 for v5/v6): it becomes a recipient and a signing identity later.
103 static const QRegularExpression shape(
104 QStringLiteral("^(?:[0-9A-Fa-f]{40}|[0-9A-Fa-f]{64})$"));
105 const QString &fingerprint = props[GPG_FIELD_FINGERPRINT];
106 if (!current_user.key_id.isEmpty() && shape.match(fingerprint).hasMatch() &&
107 fingerprint.endsWith(current_user.key_id, Qt::CaseInsensitive)) {
108 current_user.key_id = fingerprint;
109 }
110}

112auto parseGpgColonOutput(const QString &output, bool secret)
113 -> QList<UserInfo> {
114 QList<UserInfo> users;
115
116 const QStringList lines =
117 output.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
118
119 UserInfo current_user;
120
121 for (const QString &key : lines) {
122 QStringList props = key.split(':');
123 if (props.size() < GPG_MIN_FIELDS) {
124 continue;
125 }
126
127 const QString &record_type = props[0];
128 const GpgRecordType type = classifyRecord(record_type);
129
130 switch (type) {
131 case GpgRecordType::Pub:
132 case GpgRecordType::Sec:
133 if (!current_user.key_id.isEmpty()) {
134 users.append(current_user);
135 }
136 current_user = UserInfo();
137 handlePubSecRecord(props, secret && (type == GpgRecordType::Sec),
138 current_user);
139 break;
140 case GpgRecordType::Uid:
141 if (current_user.name.isEmpty()) {
142 handleUidRecord(props, current_user);
143 }
144 break;
145 case GpgRecordType::Fpr:
146 handleFprRecord(props, current_user);
147 break;
148 default:
149 break;
150 }
151 }
152
153 if (!current_user.key_id.isEmpty()) {
154 users.append(current_user);
155 }
156
157 return users;
158}

160auto primaryFingerprintOf(const QString &output, const QString &key)
161 -> QString {
162 QString wanted = key.trimmed().toUpper();
163 if (wanted.startsWith(QLatin1String("0X"))) {
164 wanted.remove(0, 2);
165 }
166 if (wanted.isEmpty()) {
167 return {};
168 }
169 const bool isId = wanted.size() == 8 || wanted.size() == 16;
170 QStringList holders;
171 QString primary;
172 bool expectPrimary = false;
173 bool holds = false;
174 // A pub or sub record carries the key's own ID: the fingerprint's tail
175 // for v4 keys, its head for v5/v6. A short ID is the long one's tail.
176 const auto namesKeyId = [&](const QStringList &props) {
177 return isId && props[GPG_FIELD_KEY_ID].toUpper().endsWith(wanted);
178 };
179 const auto endBlock = [&] {
180 if (holds && !primary.isEmpty() && !holders.contains(primary)) {
181 holders << primary;
182 }
183 primary.clear();
184 holds = false;
185 };
186 const QStringList lines =
187 output.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
188 for (const QString &line : lines) {
189 const QStringList props = line.split(QLatin1Char(':'));
190 if (props.size() < GPG_MIN_FIELDS) {
191 continue;
192 }
193 switch (classifyRecord(props[0])) {
194 case GpgRecordType::Pub:
195 endBlock();
196 expectPrimary = true;
197 holds = holds || namesKeyId(props);
198 break;
199 case GpgRecordType::Sub:
200 expectPrimary = false;
201 holds = holds || namesKeyId(props);
202 break;
203 case GpgRecordType::Fpr:
204 if (expectPrimary) {
205 primary = props[GPG_FIELD_FINGERPRINT].toUpper();
206 expectPrimary = false;
207 }
208 if (!isId && props[GPG_FIELD_FINGERPRINT].toUpper() == wanted) {
209 holds = true;
210 }
211 break;
212 default:
213 break;
214 }
215 }
216 endBlock();
217 return holders.size() == 1 ? holders.first() : QString();
218}
