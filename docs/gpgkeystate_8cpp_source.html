36auto classifyRecord(const QString &record_type) -> GpgRecordType {
37 if (record_type == "pub") {
38 return GpgRecordType::Pub;
39 }
40 if (record_type == "sec") {
41 return GpgRecordType::Sec;
42 }
43 if (record_type == "uid") {
44 return GpgRecordType::Uid;
45 }
46 if (record_type == "fpr") {
47 return GpgRecordType::Fpr;
48 }
49 if (record_type == "sub") {
50 return GpgRecordType::Sub;
51 }
52 if (record_type == "ssb") {
53 return GpgRecordType::Ssb;
54 }
55 if (record_type == "grp") {
56 return GpgRecordType::Grp;
57 }
58 return GpgRecordType::Unknown;
59}

61void handlePubSecRecord(const QStringList &props, bool secret,
62 UserInfo &current_user) {
63 if (props.size() < GPG_MIN_FIELDS) {
64 return;
65 }
66 current_user.key_id = props[GPG_FIELD_KEY_ID];
67 current_user.name = props[GPG_FIELD_USERID];
68 if (!props[GPG_FIELD_VALIDITY].isEmpty()) {
69 current_user.validity = props[GPG_FIELD_VALIDITY][0].toLatin1();
70 }
71
72 // An empty field (GnuPG's "no expiry") leaves the QDateTime null, so
73 // callers distinguish "unset" from "set" with isValid() alone.
74 bool okCreated = false;
75 const qint64 createdSecs = props[GPG_FIELD_CREATED].toLongLong(&okCreated);
76 if (okCreated) {
77 current_user.created = QDateTime::fromSecsSinceEpoch(createdSecs);
78 }
79
80 bool okExpiry = false;
81 const qint64 expirySecs = props[GPG_FIELD_EXPIRY].toLongLong(&okExpiry);
82 if (okExpiry) {
83 current_user.expiry = QDateTime::fromSecsSinceEpoch(expirySecs);
84 }
85
86 current_user.have_secret = secret;
87}

100auto uidNameRank(const QString &validity) -> int {
101 switch (validity.isEmpty() ? '\0' : validity.at(0).toLatin1()) {
102 case 'u':
103 case 'f':
104 return 3;
105 case 'm':
106 return 2;
107 case '-':
108 case 'q':
109 return 1;
110 default:
111 return 0;
112 }
113}

121auto decodeColonField(const QString &field) -> QString {
122 if (!field.contains(QLatin1String("\\x"))) {
123 return field;
124 }
125 const QByteArray in = field.toUtf8();
126 const auto isHex = [](char c) {
127 return std::isxdigit(static_cast<unsigned char>(c)) != 0;
128 };
129 QByteArray out;
130 out.reserve(in.size());
131 for (int i = 0; i < in.size(); ++i) {
132 if (in.at(i) == '\\' && i + 3 < in.size() && in.at(i + 1) == 'x' &&
133 isHex(in.at(i + 2)) && isHex(in.at(i + 3))) {
134 out.append(static_cast<char>(in.mid(i + 2, 2).toInt(nullptr, 16)));
135 i += 3;
136 } else {
137 out.append(in.at(i));
138 }
139 }
140 return QString::fromUtf8(out);
141}

150void handleUidRecord(const QStringList &props, UserInfo &current_user,
151 int &nameRank) {
152 current_user.uids.append(decodeColonField(props[GPG_FIELD_USERID]));
153 const int rank = uidNameRank(props[GPG_FIELD_VALIDITY]);
154 if (current_user.name.isEmpty() || rank > nameRank) {
155 current_user.name = props[GPG_FIELD_USERID];
156 nameRank = rank;
157 }
158}

160void handleFprRecord(const QStringList &props, UserInfo &current_user) {
161 if (props.size() < GPG_MIN_FIELDS) {
162 return;
163 }
164 // GnuPG reports the v4 key ID from the fingerprint's tail, and the v5/v6
165 // key ID from its head. Promote only an exact 16-hex key ID with the
166 // version-specific fingerprint shape and location; malformed or mismatched
167 // input stays a key ID rather than becoming a false full identity.
168 static const QRegularExpression keyIdShape(
169 QStringLiteral("^[0-9A-Fa-f]{16}$"));
170 static const QRegularExpression v4FingerprintShape(
171 QStringLiteral("^[0-9A-Fa-f]{40}$"));
172 static const QRegularExpression v5v6FingerprintShape(
173 QStringLiteral("^[0-9A-Fa-f]{64}$"));
174 const QString &keyId = current_user.key_id;
175 if (!keyIdShape.match(keyId).hasMatch()) {
176 return;
177 }
178
179 const QString &fingerprint = props[GPG_FIELD_FINGERPRINT];
180 const bool isV4 = v4FingerprintShape.match(fingerprint).hasMatch() &&
181 fingerprint.endsWith(keyId, Qt::CaseInsensitive);
182 const bool isV5V6 = v5v6FingerprintShape.match(fingerprint).hasMatch() &&
183 fingerprint.startsWith(keyId, Qt::CaseInsensitive);
184 if (isV4 || isV5V6) {
185 current_user.key_id = fingerprint;
186 }
187}

189auto parseGpgColonOutput(const QString &output, bool secret)
190 -> QList<UserInfo> {
191 QList<UserInfo> users;
192
193 const QStringList lines =
194 output.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
195
196 UserInfo current_user;
197 bool expectPrimaryFingerprint = false;
198 // A subkey's fpr record follows its sub or ssb record.
199 bool expectSubkeyFingerprint = false;
200 // uidNameRank() of current_user.name; a name GnuPG 1 put on the pub
201 // record itself counts as unknown.
202 int nameRank = 0;
203
204 for (const QString &key : lines) {
205 QStringList props = key.split(':');
206 if (props.size() < GPG_MIN_FIELDS) {
207 expectPrimaryFingerprint = false;
208 expectSubkeyFingerprint = false;
209 continue;
210 }
211
212 const QString &record_type = props[0];
213 const GpgRecordType type = classifyRecord(record_type);
214 // Short records, uids and subkey records end the primary key's head. A
215 // designated revoker's rvk record sits between pub and its fpr; a
216 // keygrip's grp record follows the fpr.
217 if (type == GpgRecordType::Uid || type == GpgRecordType::Sub ||
218 type == GpgRecordType::Ssb) {
219 expectPrimaryFingerprint = false;
220 }
221
222 switch (type) {
223 case GpgRecordType::Pub:
224 case GpgRecordType::Sec:
225 if (!current_user.key_id.isEmpty()) {
226 users.append(current_user);
227 }
228 current_user = UserInfo();
229 handlePubSecRecord(props, secret && (type == GpgRecordType::Sec),
230 current_user);
231 nameRank = current_user.name.isEmpty() ? 0 : 1;
232 expectPrimaryFingerprint = true;
233 expectSubkeyFingerprint = false;
234 break;
235 case GpgRecordType::Uid:
236 handleUidRecord(props, current_user, nameRank);
237 expectSubkeyFingerprint = false;
238 break;
239 case GpgRecordType::Sub:
240 case GpgRecordType::Ssb:
241 if (!props[GPG_FIELD_KEY_ID].isEmpty()) {
242 current_user.subkeys.append(props[GPG_FIELD_KEY_ID]);
243 }
244 expectSubkeyFingerprint = true;
245 break;
246 case GpgRecordType::Fpr:
247 if (expectPrimaryFingerprint) {
248 handleFprRecord(props, current_user);
249 expectPrimaryFingerprint = false;
250 } else if (expectSubkeyFingerprint &&
251 !props[GPG_FIELD_FINGERPRINT].isEmpty()) {
252 current_user.subkeys.append(props[GPG_FIELD_FINGERPRINT]);
253 }
254 expectSubkeyFingerprint = false;
255 break;
256 default:
257 break;
258 }
259 }
260
261 if (!current_user.key_id.isEmpty()) {
262 users.append(current_user);
263 }
264
265 return users;
266}

268auto primaryFingerprintOf(const QString &output, const QString &key)
269 -> QString {
270 QString wanted = key.trimmed().toUpper();
271 if (wanted.startsWith(QLatin1String("0X"))) {
272 wanted.remove(0, 2);
273 }
274 if (wanted.isEmpty()) {
275 return {};
276 }
277 const bool isId = wanted.size() == 8 || wanted.size() == 16;
278 QStringList holders;
279 QString primary;
280 bool expectPrimary = false;
281 bool holds = false;
282 // A pub or sub record carries the key's own ID: the fingerprint's tail
283 // for v4 keys, its head for v5/v6. A short ID is the long one's tail.
284 const auto namesKeyId = [&](const QStringList &props) {
285 return isId && props[GPG_FIELD_KEY_ID].toUpper().endsWith(wanted);
286 };
287 const auto endBlock = [&] {
288 if (holds && !primary.isEmpty() && !holders.contains(primary)) {
289 holders << primary;
290 }
291 primary.clear();
292 holds = false;
293 };
294 const QStringList lines =
295 output.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
296 for (const QString &line : lines) {
297 const QStringList props = line.split(QLatin1Char(':'));
298 if (props.size() < GPG_MIN_FIELDS) {
299 continue;
300 }
301 switch (classifyRecord(props[0])) {
302 case GpgRecordType::Pub:
303 endBlock();
304 expectPrimary = true;
305 holds = holds || namesKeyId(props);
306 break;
307 case GpgRecordType::Sub:
308 expectPrimary = false;
309 holds = holds || namesKeyId(props);
310 break;
311 case GpgRecordType::Fpr:
312 if (expectPrimary) {
313 primary = props[GPG_FIELD_FINGERPRINT].toUpper();
314 expectPrimary = false;
315 }
316 if (!isId && props[GPG_FIELD_FINGERPRINT].toUpper() == wanted) {
317 holds = true;
318 }
319 break;
320 default:
321 break;
322 }
323 }
324 endBlock();
325 return holders.size() == 1 ? holders.first() : QString();
326}
