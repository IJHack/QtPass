31void QtPass::init() {
32 QString passStore = QtPassSettings::getPassStore(Util::findPasswordStore());
33 QtPassSettings::setPassStore(passStore);
34
35 QtPassSettings::initExecutables();
36
37 QString version = QtPassSettings::getVersion();
38
39 // Config updates
40 if (version.isEmpty()) {
41 qCDebug(lcQtPass) << "assuming fresh install";
42 AppSettings s = QtPassSettings::load();
43 if (s.autoclearSeconds < 5)
44 s.autoclearSeconds = 10;
45 if (s.autoclearPanelSeconds < 5)
46 s.autoclearPanelSeconds = 10;
47 s.usePwgen = !s.pwgenExecutable.isEmpty();
48 s.passTemplate = QStringLiteral("login\nurl\nOTP");
49 QtPassSettings::save(s);
50 // A fresh profile already has the new useOtp default, so record the
51 // migration as done. Without this, turning OTP off during the first session
52 // would be undone by the migration branch on the next launch.
53 QtPassSettings::getInstance()->setValue(
54 SettingsConstants::otpMigratedToNative, true);
55 } else {
56 AppSettings s = QtPassSettings::load();
57 bool changed = false;
58 if (s.passTemplate.isEmpty()) {
59 s.passTemplate = QStringLiteral("login\nurl\nOTP");
60 changed = true;
61 }
62 // `useOtp` used to gate the Unix-only pass-otp extension and now gates
63 // built-in TOTP, so an existing profile's stored false is stale rather than
64 // a preference. The new default of true never reaches these users: the
65 // fresh-install branch above already persisted the key, and the serializer
66 // writes it unconditionally. Enable it once, remembered by its own key so a
67 // deliberate opt-out is not overridden on the next launch.
68 if (!QtPassSettings::getInstance()
69 ->value(SettingsConstants::otpMigratedToNative, false)
70 .toBool()) {
71 s.useOtp = true;
72 QtPassSettings::getInstance()->setValue(
73 SettingsConstants::otpMigratedToNative, true);
74 changed = true;
75 }
76 if (changed) {
77 QtPassSettings::save(s);
78 }
79 }
80
81 QtPassSettings::setVersion(VERSION);
82}

114void QtPass::processFinished(const QString &output, const QString &errout) {
115 // A silent command (git push with nothing to push, say) has nothing to
116 // show; re-setting the browser's HTML for it would only reset its scroll.
117 if (!output.isEmpty()) {
118 emit outputReady(formatOutput(output));
119 }
120 reportError(0, errout);
121 emit operationFinished();
122}

142auto QtPass::formatOutput(QString output, const QString &prefix,
143 const QString &postfix) -> QString {
144 // Escapes and links only launchable http(s) URLs; anything else stays
145 // plain text (the text browser opens external links on click).
146 output = Util::linkifyUrls(output);
147 output.replace(QStringLiteral("\n"), "<br />");

157void QtPass::showTextAsQRCode(const QString &text) {
158 const AppSettings s = QtPassSettings::load();
159 const QString qrExe = s.qrencodeExecutable.isEmpty()
160 ? QStringLiteral("/usr/bin/qrencode")
161 : s.qrencodeExecutable;
162 QProcess qrencode;
163 Executor::detachFromTerminal(qrencode);
164 qrencode.start(qrExe, QStringList() << "-o-"
165 << "-tPNG");
166 // A missing or non-executable binary never "finishes": exitStatus() stays
167 // NormalExit and exitCode() 0, which used to fall through to an empty popup.
168 if (!qrencode.waitForStarted()) {
169 emit statusMessage(tr("Could not start qrencode: %1").arg(qrExe), 2000);
170 return;
171 }
172 qrencode.write(text.toUtf8());
173 qrencode.closeWriteChannel();
174 // A hung qrencode also leaves exitStatus()/exitCode() at their defaults.
175 if (!qrencode.waitForFinished()) {
176 qrencode.kill();
177 emit statusMessage(tr("qrencode did not finish in time"), 2000);
178 return;
179 }
180 QByteArray output(qrencode.readAllStandardOutput());
181
182 const bool crashed = qrencode.exitStatus() != QProcess::NormalExit;
183 // exitCode() is meaningless after a crash, so only consult it otherwise.
184 const bool failed = !crashed && qrencode.exitCode() != 0;
185 if (crashed || failed) {
186 QString error(qrencode.readAllStandardError());
187 if (error.trimmed().isEmpty()) {
188 error = crashed
189 ? tr("qrencode crashed")
190 : tr("qrencode exited with code %1").arg(qrencode.exitCode());
191 }
192 emit statusMessage(error, 2000);
193 } else {
194 QPixmap image;
195 image.loadFromData(output, "PNG");
196 QDialog *popup = createQRCodePopup(image);
197 popup->exec();
198 }
199}
