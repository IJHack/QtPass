61Pass::Pass() : env(QProcessEnvironment::systemEnvironment()) {
62 connect(&exec, &Executor::finished, this, &Pass::finished);
63 connect(&exec, &Executor::error, this, &Pass::finished);
64
65 connect(&exec, &Executor::starting, this, &Pass::startingExecuteWrapper);
66 connect(&exec, &Executor::drained, this, [this] {
67 // Idle means the queue is empty and nothing runs, so no insert is in
68 // flight: force the pending count to zero. A report that never arrived
69 // (the "No such transaction" path in ImitatePass::finished(), or any
70 // later change that drops an insert without reporting) would otherwise
71 // leave the count one too high for the rest of the session, and every
72 // later dialog would take its own report for an earlier one and wait for
73 // ever. Idle cannot fall between a dialog's insertsPending() read and its
74 // own report, since that insert keeps the executor busy.
75 m_insertsPending = 0;
76 emit idle();
77 });
78 // Each queued process starts in the environment as the store is then:
79 // the Git pin is judged at that moment (dequeuing() comes just before the
80 // start, on this thread).
81 connect(&exec, &Executor::dequeuing, this, [this](int, bool completes) {
82 if (completes) {
83 exec.setEnvironment(withCurrentGitPin(env));
84 }
85 });
86 // Merge our vars into WSLENV rather than blindly appending a duplicate
87 // entry. WSLENV flags: /p translates the path, /u forwards from Windows
88 // into WSL only; /w would be the other direction, from WSL out, and leaves
89 // pass inside WSL without the value. Everything updateEnv() sets for pass
90 // has to cross, the signing key above all: without it pass does not check
91 // the .gpg-id signature.
92 const QStringList wslenvVars = {
93 QStringLiteral("PASSWORD_STORE_DIR/p"),
94 QStringLiteral("PASSWORD_STORE_SIGNING_KEY/u"),
95 QStringLiteral("PASSWORD_STORE_GENERATED_LENGTH/u"),
96 QStringLiteral("PASSWORD_STORE_CHARACTER_SET/u"),
97 QStringLiteral("PASSWORD_STORE_GPG_OPTS/u"),
98 QStringLiteral("GIT_CEILING_DIRECTORIES/p"),
99 QStringLiteral("GIT_LITERAL_PATHSPECS/u")};
100 const QString existing = env.value(QStringLiteral("WSLENV"));
101 if (existing.isEmpty()) {
102 env.insert(QStringLiteral("WSLENV"), wslenvVars.join(':'));
103 } else {
104 QStringList parts = existing.split(':', Qt::SkipEmptyParts);
105 for (const QString &v : wslenvVars) {
106 if (!parts.contains(v))
107 parts.append(v);
108 }
109 env.insert(QStringLiteral("WSLENV"), parts.join(':'));
110 }

128auto Pass::loggableArgs(const QStringList &args) -> QStringList {
129 // Secrets travel on stdin (Insert's content, KeygenDialog's batch file),
130 // never in argv; this keeps the log honest should that ever change.
131 static const QStringList secretOptions{
132 QStringLiteral("--passphrase"), QStringLiteral("--override-session-key")};
133 QStringList out;
134 bool hide = false;
135 for (const QString &arg : args) {
136 if (hide) {
137 out << QStringLiteral("<redacted>");
138 hide = false;
139 continue;
140 }
141 if (arg.startsWith(QStringLiteral("otpauth://"))) {
142 out << QStringLiteral("<redacted>");
143 continue;
144 }
145 if (arg.startsWith(QStringLiteral("--passphrase=")) ||
146 arg.startsWith(QStringLiteral("--override-session-key="))) {
147 out << arg.left(arg.indexOf(u'=') + 1) + QStringLiteral("<redacted>");
148 continue;
149 }
150 hide = secretOptions.contains(arg);
151 out << arg;
152 }
153 return out;
154}

167 const QString currentPath = env.value(QStringLiteral("PATH"));

178 // one (#1711) fails every gpg call with "No secret key". Clearing the
179 // setting restores the inherited value rather than keeping the old path.
180 const QString inheritedHome = QProcessEnvironment::systemEnvironment().value(
181 QStringLiteral("GNUPGHOME"));
182 const auto useInheritedHome = [this, &inheritedHome]() {
183 if (inheritedHome.isEmpty()) {
184 env.remove(QStringLiteral("GNUPGHOME"));
185 } else {
186 env.insert(QStringLiteral("GNUPGHOME"), inheritedHome);
187 }
188 };
189 if (m_settings.gpgHome.isEmpty()) {
190 useInheritedHome();
191 } else {
192 QDir absHome(m_settings.gpgHome);
193 absHome.makeAbsolute();
194 if (absHome.exists()) {
195 env.insert(QStringLiteral("GNUPGHOME"), absHome.path());
196 } else {
197 if (inheritedHome.isEmpty()) {
198 qCWarning(lcQtPass) << "gpgHome" << absHome.path()
199 << "does not exist; using the default GnuPG home";
200 emit statusMsg(tr("Configured GPG home %1 does not exist, using the "
201 "default keyring")
202 .arg(absHome.path()),
203 5000);
204 } else {
205 qCWarning(lcQtPass)
206 << "gpgHome" << absHome.path() << "does not exist; using GNUPGHOME"
207 << inheritedHome << "from the environment";
208 emit statusMsg(tr("Configured GPG home %1 does not exist, using "
209 "GNUPGHOME %2 from the environment")
210 .arg(absHome.path(), inheritedHome),
211 5000);
212 }
213 useInheritedHome();
214 }

218auto Pass::generatePassword(unsigned int length, const QString &charset)
219 -> QString {
220 if (length == 0) {
221 emit critical(tr("Invalid password length"),
222 tr("Can't generate password with zero length."));
223 return {};
224 }
225 QString passwd;
226 if (m_settings.usePwgen) {
227 // --secure goes first as it overrides --no-* otherwise
228 QStringList args;
229 args.append("-1");
230 if (!m_settings.lessRandom) {
231 args.append("--secure");
232 }
233 args.append(m_settings.avoidCapitals ? "--no-capitalize" : "--capitalize");
234 args.append(m_settings.avoidNumbers ? "--no-numerals" : "--numerals");
235 if (m_settings.useSymbols) {
236 args.append("--symbols");
237 }
238 args.append(QString::number(length));
239 if (Executor::executeBlocking(m_settings.pwgenExecutable, args, &passwd) ==
240 0) {
241 static const QRegularExpression literalNewLines{"[\\n\\r]"};
242 passwd.remove(literalNewLines);
243 } else {
244 passwd.clear();
245 qCDebug(lcQtPass) << "pwgen fail";
246 // Report it, as the other generation failures do: otherwise Generate
247 // does nothing silently and OK then saves an empty (or, on Edit, the
248 // old) password.
249 emit critical(
250 tr("Password generation failed"),
251 tr("Could not run pwgen (\"%1\"). Check the pwgen program in the "
252 "settings, or turn \"Use pwgen\" off.")
253 .arg(m_settings.pwgenExecutable));
254 }
255 } else {
256 // An empty CUSTOM charset falls back to ALLCHARS, not a weak password
257 // (#780).
258 const QString cs = fallbackCharset(
259 charset, m_settings.passwordConfiguration
260 .Characters[PasswordConfiguration::ALLCHARS]);
261 if (cs.length() > 0) {
262 passwd = generateRandomPassword(cs, length);
263 } else {
264 emit critical(
265 tr("No characters chosen"),
266 tr("Can't generate password, there are no characters to choose from "
267 "set in the configuration!"));
268 }
269 }
270 return passwd;
271}

273bool Pass::gpgSupportsEd25519(const QString &gpgExecutable) {
274 const QString exe =
275 gpgExecutable.isEmpty() ? QStringLiteral("gpg") : gpgExecutable;
276 QString out, err;
277 if (Executor::executeBlocking(exe, {"--version"}, &out, &err) != 0) {
278 return false;
279 }
280 QRegularExpression versionRegex(R"(gpg \‍(GnuPG\) (\d+)\.(\d+))");
281 QRegularExpressionMatch match = versionRegex.match(out);
282 if (!match.hasMatch()) {
283 return false;
284 }
285 int major = match.captured(1).toInt();
286 int minor = match.captured(2).toInt();
287 return major > 2 || (major == 2 && minor >= 1);
288}

354auto Pass::resolveGpgconfCommand(const QString &gpgPath)
355 -> ResolvedGpgconfCommand {
356 if (gpgPath.trimmed().isEmpty()) {
357 return {"gpgconf", {}};
358 }
359
360 // WSL: gpgconf next to the configured gpg, same distribution, via --exec
361 // not the shell. An unparsed form (`wsl sh -c ...`, bare `wsl`) falls back
362 // to gpgconf on the Windows PATH.
363 if (const auto wsl = Executor::parseWslCommand(gpgPath)) {
364 if (!QFileInfo(wsl->command).fileName().startsWith("gpg")) {
365 return {"gpgconf", {}};
366 }
367 const auto gpgconf = wsl->with(resolveWslGpgconfPath(wsl->command));
368 return {gpgconf.launcher, gpgconf.argv({})};
369 }
370
371 const QStringList parts = QProcess::splitCommand(gpgPath);
372 if (parts.isEmpty()) {
373 return {"gpgconf", {}};
374 }
375 const QString first = parts.first();
376 if (QFileInfo(first).fileName().compare("wsl", Qt::CaseInsensitive) == 0 ||
377 QFileInfo(first).fileName().compare("wsl.exe", Qt::CaseInsensitive) ==
378 0) {
379 return {"gpgconf", {}};
380 }
381
382 if (!first.contains('/') && !first.contains('\\')) {
383 return {"gpgconf", {}};
384 }
385
386 QString gpgconfPath = findGpgconfInGpgDir(first);
387 if (!gpgconfPath.isEmpty()) {
388 return {gpgconfPath, {}};
389 }
390
391 return {"gpgconf", {}};
392}

394void Pass::GenerateGPGKeys(QString batch) {
395 const QString gpgPath = m_settings.gpgExecutable;
396 if (gpgPath.isEmpty()) {
397 // The executor would fail an empty executable with a generic process
398 // error; the keygen dialog waits for generateGPGKeysFailed and a reason
399 // it can show. Queued so we do not re-enter KeygenDialog::done(), which
400 // drives key generation synchronously.
401 QMetaObject::invokeMethod(
402 this,
403 [this]() {
404 emit generateGPGKeysFailed(tr("No GPG executable configured"));
405 emit processErrorExit(1, tr("No GPG executable configured"));
406 },
407 Qt::QueuedConnection);
408 return;
409 }
410
411 // A stale gpg-agent holding the key database lock causes "database locked"
412 // timeouts during key generation.
413 ResolvedGpgconfCommand resolvedGpgconf = resolveGpgconfCommand(gpgPath);
414 QStringList killArgs = resolvedGpgconf.arguments;
415 killArgs << "--kill";
416 killArgs << "gpg-agent";
417 // Use same environment as key generation to target correct gpg-agent
418 if (Executor::executeBlocking(env, resolvedGpgconf.program, killArgs) != 0) {
419 qCWarning(lcQtPass) << "Failed to kill gpg-agent";
420 }
421
422 executeWrapper(GPG_GENKEYS, gpgPath, {"--gen-key", "--no-tty", "--batch"},
423 std::move(batch), true, true);
424}

471auto gpgErrorMessage(const QString &err) -> QString {
472 // Machine-readable status tokens added by --status-fd 2
473 if (containsAny(err, {QStringLiteral("[GNUPG:] KEYEXPIRED"),
474 QStringLiteral("[GNUPG:] INV_RECP 5 ")}))
475 return QCoreApplication::translate(
476 "Pass", "Encryption failed: GPG key has expired. Please renew or "
477 "replace it.");
478 if (containsAny(err, {QStringLiteral("[GNUPG:] KEYREVOKED"),
479 QStringLiteral("[GNUPG:] INV_RECP 4 ")}))
480 return QCoreApplication::translate(
481 "Pass", "Encryption failed: GPG key has been revoked.");
482 if (containsAny(err, {QStringLiteral("[GNUPG:] NO_PUBKEY"),
483 QStringLiteral("[GNUPG:] INV_RECP")}))
484 return QCoreApplication::translate(
485 "Pass", "Encryption failed: recipient GPG key not found or invalid. "
486 "Check that the key ID in .gpg-id is correct and imported.");
487 if (err.contains(QStringLiteral("[GNUPG:] FAILURE")))
488 return QCoreApplication::translate(
489 "Pass", "Encryption failed. Check that your GPG key is valid.");
490
491 // Locale-dependent fallbacks
492 if (containsAnyCaseInsensitive(err, {QLatin1String("key has expired"),
493 QLatin1String("key expired")}))
494 return QCoreApplication::translate(
495 "Pass", "Encryption failed: GPG key has expired. Please renew or "
496 "replace it.");
497 if (containsAnyCaseInsensitive(err, {QLatin1String("key has been revoked"),
498 QLatin1String("revoked")}))
499 return QCoreApplication::translate(
500 "Pass", "Encryption failed: GPG key has been revoked.");
501 // pass passes no --status-fd, so this is all gpg says about a key that is
502 // present but cannot encrypt: expired, revoked, without an encryption
503 // subkey, or not trusted ("There is no assurance this key belongs to the
504 // named user"). "Not found ... imported" would send the user the wrong
505 // way; gpg's own lines, shown below this, say which it is.
506 if (containsAnyCaseInsensitive(err, {QLatin1String("unusable public key")}))
507 return QCoreApplication::translate(
508 "Pass", "Encryption failed: a recipient's GPG key cannot be used. It "
509 "may have expired or been revoked, have no encryption subkey, "
510 "or not be trusted; gpg's message below says which.");
511 if (containsAnyCaseInsensitive(err, {QLatin1String("no public key"),
512 QLatin1String("no secret key")}))
513 return QCoreApplication::translate(
514 "Pass", "Encryption failed: recipient GPG key not found or invalid. "
515 "Check that the key ID in .gpg-id is correct and imported.");
516 if (containsAnyCaseInsensitive(err, {QLatin1String("encryption failed")}))
517 return QCoreApplication::translate(
518 "Pass", "Encryption failed. Check that your GPG key is valid.");
519
520 return {};
521}

523void Pass::finished(int id, int exitCode, const QString &out,
524 const QString &err) {
525 auto pid = static_cast<PROCESS>(id);
526
527 // Pop the Show() this completion belongs to whether it worked or not, or
528 // the next result would be attributed to the wrong file.
529 QString shownFile;
530 if (pid == PASS_SHOW && !m_pendingShows.isEmpty()) {
531 shownFile = m_pendingShows.dequeue();
532 }
533 // Count this insert off, success or failure, before its result goes out, so
534 // insertsPending() tracks the inserts still awaiting a report. Each insert
535 // reaches here once as PASS_INSERT.
536 if (pid == PASS_INSERT && m_insertsPending > 0) {
537 --m_insertsPending;
538 }
539
540 if (exitCode != 0) {
541 handleProcessError(pid, exitCode, out, err);
542 return;
543 }
544
545 if (pid == PASS_SHOW) {
546 emit finishedShow(out, shownFile);
547 return;
548 }
549 emitProcessFinishedSignal(pid, out, err);
550}

552void Pass::handleProcessError(PROCESS pid, int exitCode, const QString &out,
553 const QString &err) {
554 Q_UNUSED(out);
555
556 if (pid == PASS_INSERT) {
557 const QString friendly = gpgErrorMessage(err);
558 const QString why =
559 friendly.isEmpty() ? err : formatInsertError(friendly, err);
560 const bool written = insertWritten(err);
561 emit insertFailed(why, written);
562 if (written) {
563 // The entry is in the store; a git step after it failed (a commit
564 // that could not be signed, with pass.signcommits, say). The dialog
565 // closes on this, so say it here rather than only in the output
566 // panel.
567 emit critical(tr("Saved, not committed"),
568 tr("The entry is saved in the store, but recording it "
569 "in git failed. What was reported:\n\n%1")
570 .arg(err.trimmed()));
571 }
572 emit processErrorExit(exitCode, why);
573 return;
574 }
575
576 if (pid == GPG_GENKEYS) {
577 emit generateGPGKeysFailed(err);
578 }
579 emit processErrorExit(exitCode, err);
580}

582auto Pass::formatInsertError(const QString &friendly, const QString &err)
583 -> QString {
584 QStringList humanLines;
585 for (const QString &line : err.split('\n')) {
586 QString cleanedLine = line;
587 cleanedLine.remove('\r');
588 if (!cleanedLine.startsWith(QLatin1String("[GNUPG:]")))
589 humanLines.append(cleanedLine);
590 }
591 const QString humanErr = humanLines.join('\n').trimmed();
592 return humanErr.isEmpty() ? friendly : friendly + "\n\n" + humanErr;
593}

595void Pass::emitProcessFinishedSignal(PROCESS pid, const QString &out,
596 const QString &err) {
597 // Only output that cannot contain a secret reaches the generic listeners
598 // (the output panel among them); Enums::processInfo is the one table.
599 if (pid >= 0 && pid < Enums::PROCESS_COUNT &&
600 !Enums::processInfo(pid).secret) {
601 emit finishedAnyWithPid(out, err, pid);
602 }
603
604 // The kind-specific signal. PASS_SHOW is emitted by finished(), which
605 // knows the file. Content search runs in-process (NativeGrep), not
606 // through `pass grep`, and emits finishedGrep itself.
607 using Signal = void (Pass::*)(const QString &, const QString &);
608 static const QHash<int, Signal> kSignals{
609 {GIT_INIT, &Pass::finishedGitInit},
610 {GIT_PULL, &Pass::finishedGitPull},
611 {GIT_PUSH, &Pass::finishedGitPush},
612 {PASS_INSERT, &Pass::finishedInsert},
613 {PASS_REMOVE, &Pass::finishedRemove},
614 {PASS_INIT, &Pass::finishedInit},
615 {PASS_MOVE, &Pass::finishedMove},
616 {PASS_COPY, &Pass::finishedCopy},
617 {GPG_GENKEYS, &Pass::finishedGenerateGPGKeys},
618 };
619 if (const Signal signal = kSignals.value(pid, nullptr)) {
620 (this->*signal)(out, err);
621 }
622}

631void Pass::updateEnv() {
632 setEnvVar(QStringLiteral("PASSWORD_STORE_SIGNING_KEY"),
633 m_settings.passSigningKey);
634 setEnvVar(QStringLiteral("PASSWORD_STORE_DIR"), m_settings.passStore);
635 // Every path QtPass hands git names one file or folder: an entry called
636 // "a*.gpg" must not make `git rm -f -- a*.gpg` remove every entry that
637 // matches. Also reaches the git that pass runs, which passes names the
638 // same way.
639 setEnvVar(QStringLiteral("GIT_LITERAL_PATHSPECS"), QStringLiteral("1"));
640
641 const PasswordConfiguration &passConfig = m_settings.passwordConfiguration;
642 setEnvVar(QStringLiteral("PASSWORD_STORE_GENERATED_LENGTH"),
643 QString::number(passConfig.length));
644
645 setEnvVar(QStringLiteral("PASSWORD_STORE_CHARACTER_SET"),
646 effectiveCharset(passConfig));
647
648 // pass hands PASSWORD_STORE_GPG_OPTS to every gpg it runs: the user's own
649 // options (split as pass's shell splits them: spaces, tabs, newlines),
650 // then the output limit ImitatePass passes directly.
651 QStringList gpgOpts = QProcessEnvironment::systemEnvironment()
652 .value(QStringLiteral("PASSWORD_STORE_GPG_OPTS"))
653 .split(QRegularExpression(QStringLiteral("\\s+")),
654 Qt::SkipEmptyParts);
655 gpgOpts << maxOutputArgs();
656 setEnvVar(QStringLiteral("PASSWORD_STORE_GPG_OPTS"),
657 gpgOpts.join(QLatin1Char(' ')));
658
659 // As pass does: git never looks for a repository above the store's own
660 // parent. A store that is not a repository itself but lies inside one (a
661 // dotfiles repository in the home folder) would otherwise have its entries
662 // committed and pushed there by native git mode. Only in the environment

730auto gpgIdLookup(const QString &for_file, const QString &passStore)
731 -> GpgIdLookup {
732 const QString normalizedFile = QDir::fromNativeSeparators(for_file);
733 // Inside the store means at a path boundary: "/store-other/x" is not
734 // under "/store", however the prefix compares.
735 const QString storeDir =
736 QDir::cleanPath(QDir::fromNativeSeparators(passStore));
737 const QString storePrefix = storeDir.endsWith(QLatin1Char('/'))
738 ? storeDir
739 : storeDir + QLatin1Char('/');
740 const QString cleanFile = QDir::cleanPath(normalizedFile);
741 const bool insideStore = isAtOrUnder(cleanFile, storeDir, storePrefix);
742 // A relative name is one inside the store; an absolute path outside it
743 // stays what it is and the walk stops at once.
744 const QString fullPath = insideStore || QDir::isAbsolutePath(cleanFile)
745 ? cleanFile
746 : storePrefix + cleanFile;
747 // A folder (trailing separator, which cleanPath() dropped, or an existing
748 // directory) wants its own list; an entry wants its folder's.
749 const bool isFolder =
750 normalizedFile.endsWith(QLatin1Char('/')) || QFileInfo(fullPath).isDir();
751 // A folder that does not exist yet (an entry saved into a new folder, a
752 // move to one) starts the walk at its nearest existing ancestor in the
753 // store, as pass's set_gpg_recipients does: otherwise the walk ended at
754 // once and the store root's list was taken over the one in between.
755 QString start = isFolder ? fullPath : QFileInfo(fullPath).absolutePath();
756 while (!QFileInfo(start).exists() &&
757 isAtOrUnder(QDir::cleanPath(start), storeDir, storePrefix) &&
758 QDir::cleanPath(start) != storeDir) {
759 start = QFileInfo(start).absolutePath();
760 }
761 return {QDir(start), storeDir, storePrefix};
762}

775 QDir dir = lookup.start;

835auto Pass::unfinishedGitOperationReason() -> QString {
836 if (!m_settings.useGit) {
837 return {};
838 }
839 // One git call names the path of each marker file or directory, whether or
840 // not it exists; the backend runs it in the store (ImitatePass with
841 // `-C <store>`, RealPass through `pass git`). The markers, paired with the
842 // operation each one means, in the order git prints them back.
843 // The names are Git's own commands, which the user then runs with --abort or
844 // --continue; the translator comments say so.
845 const std::array<std::pair<QLatin1String, QString>, 6> markers{{
846 //: The Git command `git rebase`, inserted into "in the middle of a %1".
847 //: Best kept as Git's command name.
848 {QLatin1String("rebase-merge"), tr("rebase")},
849 //: The Git command `git rebase`, inserted into "in the middle of a %1".
850 //: Best kept as Git's command name.
851 {QLatin1String("rebase-apply"), tr("rebase")},
852 //: The Git command `git merge`, inserted into "in the middle of a %1".
853 //: Best kept as Git's command name.
854 {QLatin1String("MERGE_HEAD"), tr("merge")},
855 //: The Git command `git cherry-pick`, inserted into "in the middle of a
856 //: %1". Best kept as Git's command name.
857 {QLatin1String("CHERRY_PICK_HEAD"), tr("cherry-pick")},
858 //: The Git command `git revert`, inserted into "in the middle of a %1".
859 //: Best kept as Git's command name.
860 {QLatin1String("REVERT_HEAD"), tr("revert")},
861 // A multi-commit cherry-pick or revert keeps its sequencer state after a
862 // conflict resolution is committed, when CHERRY_PICK_HEAD/REVERT_HEAD is
863 // already gone but `--continue` has more commits to apply. Checked last,
864 // so the specific head above names the operation when it is still there.
865 //: The Git commands `git cherry-pick` and `git revert`, inserted into
866 //: "in the middle of a %1". Best kept as Git's command names.
867 {QLatin1String("sequencer"), tr("cherry-pick or revert")},
868 }};
869 QStringList args{QStringLiteral("rev-parse")};
870 for (const auto &marker : markers) {
871 args << QStringLiteral("--git-path") << marker.first;
872 }
873 QString out;
874 if (!storeGit(args, &out)) {
875 // Git is off, the store is not a repository, or git could not be asked:
876 // nothing is refused, as the operation would run anyway.
877 return {};
878 }
879 const QStringList paths = out.split(QLatin1Char('\n'), Qt::SkipEmptyParts);
880 for (int i = 0; i < paths.size() && i < static_cast<int>(markers.size());
881 ++i) {
882 const QString path = paths.at(i).trimmed();
883 const QString full = QDir::isAbsolutePath(path)
884 ? path
885 : QDir(m_settings.passStore).filePath(path);
886 if (!QFileInfo::exists(full)) {
887 continue;
888 }
889 //: %1 is a Git command name: rebase, merge, cherry-pick, revert, or
890 //: "cherry-pick or revert".
891 return tr("The password store's Git repository is in the middle of a %1. "
892 "Finish or abort it with Git before changing the password store, "
893 "or what you change may be lost.")
894 .arg(markers.at(static_cast<size_t>(i)).second);
895 }
896 return {};
897}

998auto Pass::skippedGpgId(const QString &forFile, const QString &passStore,
999 QString *recordError) -> std::optional<QString> {
1000 const GpgIdLookup lookup = gpgIdLookup(forFile, passStore);
1001 QDir dir = lookup.start;
1002 // The same walk as nearestGpgId(), stopping at the list it finds, and
1003 // short of the store root: the root's own list has no parent to fall
1004 // back to, and the signature check refuses it when it is missing.
1005 while (dir.exists() && QDir::cleanPath(dir.absolutePath())
1006 .startsWith(lookup.storePrefix, kPathCase)) {
1007 const QFileInfo candidate(dir.absoluteFilePath(".gpg-id"));
1008 if (isOwnGpgId(candidate)) {
1009 return std::nullopt;
1010 }
1011 // A link under the name (dangling or not), or nothing under it where
1012 // this device accepted a list before: both are the folder's list gone,
1013 // not a folder that never had one.
1014 if (candidate.isSymLink() || candidate.isJunction()) {
1015 return candidate.absoluteFilePath();
1016 }
1017 // An unreadable record is not "never seen": whether a list was accepted
1018 // here cannot be told, so this folder is refused too (fail closed), with
1019 // the record's own reason for the caller to show.
1020 QString why;
1021 const std::optional<qint64> seen =
1022 GpgIdGeneration::remembered(candidate.absoluteFilePath(), &why);
1023 if (!seen) {
1024 if (recordError != nullptr) {
1025 *recordError =
1026 why.isEmpty()
1027 ? tr("The generation record of the recipient lists cannot "
1028 "be read.")
1029 : why;
1030 }
1031 return candidate.absoluteFilePath();
1032 }
1033 if (*seen > 0) {
1034 return candidate.absoluteFilePath();
1035 }
1036 if (!dir.cdUp()) {
1037 break;
1038 }
1039 }
1040 return std::nullopt;
1041}

1047auto judgedRecipients(GpgIdGeneration::Verdict verdict,
1048 const QByteArray &contents, const QString &gpgIdPath,
1049 const QString &why) -> Pass::RecipientsForEditing {
1050 using RecipientsForEditing = Pass::RecipientsForEditing;
1051 RecipientsForEditing result;
1052 switch (verdict) {
1053 case GpgIdGeneration::Verdict::Accepted:
1054 result.state = RecipientsForEditing::State::Verified;
1055 result.recipients = Pass::parseRecipients(contents, gpgIdPath);
1056 break;
1057 case GpgIdGeneration::Verdict::Rollback:
1058 // Authentic and this folder's, only older: the recovery goes through
1059 // this dialog, so it is preselected, with the reason in view.
1060 result.state = RecipientsForEditing::State::VerifiedRollback;
1061 result.recipients = Pass::parseRecipients(contents, gpgIdPath);
1062 result.warning = why;
1063 break;
1064 case GpgIdGeneration::Verdict::Unbound:
1065 case GpgIdGeneration::Verdict::Conflict:
1066 case GpgIdGeneration::Verdict::WrongFolder:
1067 case GpgIdGeneration::Verdict::WrongStore:
1068 case GpgIdGeneration::Verdict::Malformed:
1069 case GpgIdGeneration::Verdict::RecordUnavailable:
1070 // Signed, but not shown to be this folder's list, not a list to trust,
1071 // or nothing to judge it by: preselecting it would sign it in.
1072 result.state = RecipientsForEditing::State::Rejected;
1073 result.warning =
1074 Pass::tr("%1 Nothing is preselected: saving would sign whatever is "
1075 "in it. Select the recipients yourself.")
1076 .arg(why);
1077 break;
1078 }
1079 return result;
1080}

1084auto Pass::recipientsForEditing(const QString &dir, const QString &passStore)
1085 -> RecipientsForEditing {
1086 RecipientsForEditing result;
1087 const QString folder = dir.isEmpty() ? QString() : dir;
1088 const QString gpgIdPath = getGpgIdPath(folder, passStore);
1089 const GpgIdSigner signer(
1090 m_settings.gpgExecutable,
1091 GpgIdSigner::keysFromSetting(m_settings.passSigningKey));
1092 if (!signer.enabled()) {
1093 // A list that is there but cannot be read as one (too large, not a
1094 // regular file, unreadable) is not an empty list: preselecting nothing
1095 // without a word would let OK write a new list over it, and later
1096 // re-encryptions drop whoever it named.
1097 // Read once and parsed from those bytes: a second read could find
1098 // something else under the name. A link counts as there, dangling too.
1099 const QFileInfo info(gpgIdPath);
1100 if (!info.exists() && !info.isSymLink()) {
1101 result.state = RecipientsForEditing::State::Unsigned;
1102 return result;
1103 }
1104 const std::optional<QByteArray> bytes =
1105 Util::readRegularFile(gpgIdPath, Util::kMaxGpgIdBytes);
1106 if (!bytes) {
1107 result.state = RecipientsForEditing::State::Rejected;
1108 result.warning =
1109 tr("%1 cannot be read as a recipient list (a regular file of at "
1110 "most %2), so nothing is preselected. Check it before choosing "
1111 "recipients here: OK writes a new list over it.")
1112 .arg(gpgIdPath,
1113 QLocale().formattedDataSize(Util::kMaxGpgIdBytes));
1114 return result;
1115 }
1116 result.state = RecipientsForEditing::State::Unsigned;
1117 result.recipients = parseRecipients(*bytes, gpgIdPath);
1118 return result;
1119 }
1120 QString recordError;
1121 if (const std::optional<QString> skipped =
1122 skippedGpgId(folder, passStore, &recordError)) {
1123 // The folder's own list was removed or replaced by a link (or the record
1124 // that would tell cannot be read): the list the walk found further up
1125 // was not signed for this folder, so preselecting it would sign it in
1126 // here with one click.
1127 result.state = RecipientsForEditing::State::Rejected;
1128 result.warning =
1129 recordError.isEmpty()
1130 ? tr("The recipient list %1 is missing or is not a regular file, "
1131 "so nothing is preselected: the list of a parent folder was "
1132 "not signed for this one. Select this folder's recipients "
1133 "yourself; OK writes and signs a fresh list.")
1134 .arg(*skipped)
1135 : tr("%1 Nothing is preselected: whether %2 was removed cannot "
1136 "be told.")
1137 .arg(recordError, *skipped);
1138 return result;
1139 }
1140 if (!QFileInfo::exists(gpgIdPath)) {
1141 // A folder without a list yet (new store, new profile): nothing to
1142 // verify and nothing to preselect.
1143 result.state = RecipientsForEditing::State::Verified;
1144 return result;
1145 }
1146 QByteArray contents;
1147 if (!signer.verifyFile(gpgIdPath, &contents)) {
1148 result.state = RecipientsForEditing::State::Rejected;
1149 result.warning =
1150 tr("The recipient list %1 does not verify against the signing key, "
1151 "so nothing is preselected: saving would sign whatever is in it. "
1152 "Select the recipients yourself; OK writes and signs a fresh list.")
1153 .arg(gpgIdPath);
1154 return result;
1155 }
1156 QString why;
1157 const GpgIdGeneration::Verdict verdict =
1158 GpgIdGeneration::accept(gpgIdPath, contents, passStore, &why);
1159 return judgedRecipients(verdict, contents, gpgIdPath, why);
1160}

1222auto Pass::boundedRandom(quint32 bound) -> quint32 {
1223 if (bound < 2) {
1224 return 0;
1225 }
1226
1227 quint32 randval;
1228 // arc4random_uniform-style rejection against modulo bias: drop values
1229 // below 2^32 % bound, which (1 + ~bound) % bound computes in quint32.
1230 const quint32 rejectionThreshold = (1 + ~bound) % bound;
1231
1232 do {
1233 randval = QRandomGenerator::system()->generate();
1234 } while (randval < rejectionThreshold);
1235
1236 return randval % bound;
1237}
