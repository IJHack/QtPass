30class SingleLineValidator : public QValidator {

41 auto validate(QString &input, int &pos) const -> State override {
42 if (std::none_of(input.cbegin(), input.cend(),
43 &SecretLineEdit::isLineBreak)) {
44 return Acceptable;
45 }
46 const QStringView head = QStringView(input).left(pos);
47 pos -= static_cast<int>(std::count_if(head.cbegin(), head.cend(),
48 &SecretLineEdit::isLineBreak));
49 input = SecretLineEdit::withoutLineBreaks(input);
50 return Acceptable;
51 }

55auto SecretLineEdit::isLineBreak(QChar ch) -> bool {
56 const char16_t c = ch.unicode();
57 return c == u'\n' || c == u'\r' || c == u'\u0085' || c == u'\u2028' ||
58 c == u'\u2029';
59}

84auto SecretLineEdit::createSecretContextMenu() -> QMenu * {
85 QMenu *menu = createStandardContextMenu();
86 for (QAction *action : menu->actions()) {
87 // The standard actions call QLineEdit::copy() and cut(), which are not
88 // virtual.
89 if (action->objectName() == QLatin1String("edit-copy")) {
90 disconnect(action, &QAction::triggered, this, nullptr);
91 connect(action, &QAction::triggered, this,
92 [this] { copySelection(QClipboard::Clipboard); });
93 } else if (action->objectName() == QLatin1String("edit-cut")) {
94 disconnect(action, &QAction::triggered, this, nullptr);
95 connect(action, &QAction::triggered, this, [this] { cutSelection(); });
96 } else if (action->objectName() == QLatin1String("edit-paste")) {
97 disconnect(action, &QAction::triggered, this, nullptr);
98 connect(action, &QAction::triggered, this,
99 [this] { pasteOneLine(QClipboard::Clipboard); });
100 }
101 }
102 return menu;
103}

105void SecretLineEdit::keyPressEvent(QKeyEvent *event) {
106 if (event->matches(QKeySequence::Copy)) {
107 copySelection(QClipboard::Clipboard);
108 event->accept();
109 return;
110 }
111 if (event->matches(QKeySequence::Cut)) {
112 cutSelection();
113 event->accept();
114 return;
115 }
116 if (event->matches(QKeySequence::Paste)) {
117 // As QWidgetLineControl: Ctrl+Shift+Insert is a Paste key only with the
118 // X11 key bindings, and there it pastes the primary selection.
119 const bool selection =
120 event->key() == Qt::Key_Insert &&
121 event->modifiers() == (Qt::ControlModifier | Qt::ShiftModifier) &&
122 QApplication::clipboard()->supportsSelection();
123 pasteOneLine(selection ? QClipboard::Selection : QClipboard::Clipboard);
124 event->accept();
125 return;
126 }
127 // QLineEdit copies what these delete (Ctrl+K and Ctrl+U with the X11 key
128 // bindings), as a cut.
129 const bool toEnd = event->matches(QKeySequence::DeleteEndOfLine);
130 if (toEnd || event->matches(QKeySequence::DeleteCompleteLine)) {
131 if (!isReadOnly()) {
132 const int from = toEnd ? cursorPosition() : 0;
133 setSelection(from, static_cast<int>(text().size()) - from);
134 cutSelection();
135 }
136 event->accept();
137 return;
138 }
139 withoutPlainSelectionCopy([this, event] { QLineEdit::keyPressEvent(event); });
140 // Where QLineEdit would have copied: after a key it handled, not after a
141 // lone modifier or another key it ignores.
142 if (event->isAccepted()) {
143 publishSelection();
144 }
145}

148 // QLineEdit pastes the primary selection here on a middle click.

152 pasteOneLine(QClipboard::Selection);

154 return;
155 }

158 if (event->button() == Qt::LeftButton) {

169void SecretLineEdit::copySelection(QClipboard::Mode mode) {
170 // As QLineEdit itself: nothing leaves a field that shows dots.
171 if (echoMode() != QLineEdit::Normal || !hasSelectedText()) {
172 return;
173 }
174 const QString text = selectedText();
175 QApplication::clipboard()->setMimeData(buildClipboardMimeData(text), mode);
176 emit copied(text);
177}

188void SecretLineEdit::withoutPlainSelectionCopy(
189 const std::function<void()> &handler) {
190 // QLineEdit fills the primary selection with plain text itself, at the end
191 // of every key it handles and on a left-button release, but only in Normal
192 // echo mode (QWidgetLineControl::copy()). Replacing that afterwards still
193 // leaves a moment in which a clipboard manager could take the unmarked
194 // text, so the base handler runs in Password mode instead: same length,
195 // so cursor and selection stay where they are, and no copy.
196 // Editable, the field keeps Normal mode (see the class description) and
197 // publishSelection() replaces the plain copy.
198 if (echoMode() != QLineEdit::Normal || !isReadOnly()) {
199 handler();
200 return;
201 }
202 setEchoMode(QLineEdit::Password);
203 handler();
204 setEchoMode(QLineEdit::Normal);
205}

213void SecretLineEdit::pasteOneLine(QClipboard::Mode mode) {
214 if (isReadOnly()) {
215 return;
216 }
217 const QString text = withoutLineBreaks(QApplication::clipboard()->text(mode));
218 if (text.isEmpty() && !hasSelectedText()) {
219 return;
220 }
221 // insert() replaces the selection and records the edit for Undo; with no
222 // line break left the validator changes nothing, so the history stays.
223 // QLineEdit::paste() also makes the paste an undo step of its own, apart
224 // from the typing around it; the boundaries do that here.
225 markUndoBoundary();
226 insert(text);
227 markUndoBoundary();
228}

230void SecretLineEdit::markUndoBoundary() {
231 // QWidgetLineControl::separate(), which QLineEdit::paste() calls, is
232 // private; a cursor move calls it too, and the boundary lands before the
233 // next edit. A selection needs none: removing it sets its own.
234 if (hasSelectedText() || text().isEmpty()) {
235 return;
236 }
237 const int pos = cursorPosition();
238 setCursorPosition(pos > 0 ? pos - 1 : pos + 1);
239 setCursorPosition(pos);
240}
