19auto SecretLineEdit::createSecretContextMenu() -> QMenu * {
20 QMenu *menu = createStandardContextMenu();
21 for (QAction *action : menu->actions()) {
22 // The standard actions call QLineEdit::copy() and cut(), which are not
23 // virtual.
24 if (action->objectName() == QLatin1String("edit-copy")) {
25 disconnect(action, &QAction::triggered, this, nullptr);
26 connect(action, &QAction::triggered, this,
27 [this] { copySelection(QClipboard::Clipboard); });
28 } else if (action->objectName() == QLatin1String("edit-cut")) {
29 disconnect(action, &QAction::triggered, this, nullptr);
30 connect(action, &QAction::triggered, this, [this] { cutSelection(); });
31 }
32 }
33 return menu;
34}

36void SecretLineEdit::keyPressEvent(QKeyEvent *event) {
37 if (event->matches(QKeySequence::Copy)) {
38 copySelection(QClipboard::Clipboard);
39 event->accept();
40 return;
41 }
42 if (event->matches(QKeySequence::Cut)) {
43 cutSelection();
44 event->accept();
45 return;
46 }
47 // QLineEdit copies what these delete (Ctrl+K and Ctrl+U with the X11 key
48 // bindings), as a cut.
49 const bool toEnd = event->matches(QKeySequence::DeleteEndOfLine);
50 if (toEnd || event->matches(QKeySequence::DeleteCompleteLine)) {
51 if (!isReadOnly()) {
52 const int from = toEnd ? cursorPosition() : 0;
53 setSelection(from, static_cast<int>(text().size()) - from);
54 cutSelection();
55 }
56 event->accept();
57 return;
58 }
59 withoutPlainSelectionCopy([this, event] { QLineEdit::keyPressEvent(event); });
60 // Where QLineEdit would have copied: after a key it handled, not after a
61 // lone modifier or another key it ignores.
62 if (event->isAccepted()) {
63 publishSelection();
64 }
65}

81void SecretLineEdit::copySelection(QClipboard::Mode mode) {
82 // As QLineEdit itself: nothing leaves a field that shows dots.
83 if (echoMode() != QLineEdit::Normal || !hasSelectedText()) {
84 return;
85 }
86 const QString text = selectedText();

100void SecretLineEdit::withoutPlainSelectionCopy(
101 const std::function<void()> &handler) {
102 // QLineEdit fills the primary selection with plain text itself, at the end
103 // of every key it handles and on a left-button release, but only in Normal
104 // echo mode (QWidgetLineControl::copy()). Replacing that afterwards still
105 // leaves a moment in which a clipboard manager could take the unmarked
106 // text, so the base handler runs in Password mode instead: same length,
107 // so cursor and selection stay where they are, and no copy.
108 // Editable, the field keeps Normal mode (see the class description) and
109 // publishSelection() replaces the plain copy.
110 if (echoMode() != QLineEdit::Normal || !isReadOnly()) {
111 handler();
112 return;
113 }
114 setEchoMode(QLineEdit::Password);
115 handler();
116 setEchoMode(QLineEdit::Normal);
117}
