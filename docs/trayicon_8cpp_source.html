40void TrayIcon::createActions() {

60void TrayIcon::createTrayIcon() {
61 // QMenu needs a widget parent; the main window owns the menu, this object
62 // owns the tray icon.
63 trayIconMenu = new QMenu(parentwin);
64 trayIconMenu->addAction(showAction);
65 trayIconMenu->addAction(hideAction);
66 trayIconMenu->addAction(minimizeAction);
67 trayIconMenu->addAction(maximizeAction);
68 trayIconMenu->addAction(restoreAction);
69 quitSeparator = trayIconMenu->addSeparator();
70 trayIconMenu->addAction(quitAction);
71
72 sysTrayIcon = new QSystemTrayIcon(this);
73 sysTrayIcon->setContextMenu(trayIconMenu);
74}
