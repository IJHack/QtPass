294void MainWindow::focusInput() {
295 // findChild() rather than ui->lineEdit: on a fresh-config first launch
296 // config() calls setWindowFlags(), which rebuilds the native window, and
297 // under Qt 6.11 the QLineEdit was destroyed while ui->lineEdit kept its old
298 // address (SIGSEGV in QWidget::testAttribute). findChild() returns null.
299 if (!isVisible()) {
300 return;
301 }
302 auto *lineEdit = findChild<QLineEdit *>(QStringLiteral("lineEdit"));
303 if (lineEdit == nullptr || !lineEdit->isVisible()) {
304 return;
305 }
306 lineEdit->selectAll();
307 lineEdit->setFocus();
308 // Only once focus has landed: set in showEvent(), an early return above
309 // would consume the one-shot and never retry.
310 m_firstShowCompleted = true;
311}

323void MainWindow::changeEvent(QEvent *event) {
324 QWidget::changeEvent(event);
325 if (event->type() == QEvent::PaletteChange && m_displayPanel != nullptr) {
326 // Desktop switched light/dark. Top-level widgets get PaletteChange
327 // (ApplicationPaletteChange goes to QApplication); stylesheets with
328 // baked-in palette colours must be re-derived by hand.
329 m_displayPanel->refreshPalette();
330 }
331 if (event->type() == QEvent::ActivationChange && isActiveWindow() &&
332 isVisible()) {
333 // Defer a tick so the activation dispatch chain unwinds first: calling
334 // focusInput() inline has segfaulted on half-rebuilt ui state.
335 QMetaObject::invokeMethod(this, &MainWindow::focusInput,
336 Qt::QueuedConnection);
337 }

425void MainWindow::flashText(const QString &text, const bool isError,
426 const bool isHtml) {
427 if (isError) {
428 ui->textBrowser->setTextColor(Qt::red);
429 } else {
430 // setText() re-applies the current char format, red after an error, to
431 // the whole document. Clear the foreground rather than pin a palette
432 // colour, so text follows runtime light/dark switches (#946).
433 QTextCharFormat format = ui->textBrowser->currentCharFormat();
434 format.clearForeground();
435 ui->textBrowser->setCurrentCharFormat(format);
436 }
437
438 if (isHtml) {
439 QString _text = text;
440 if (!ui->textBrowser->toPlainText().isEmpty()) {
441 _text = ui->textBrowser->toHtml() + _text;
442 }
443 ui->textBrowser->setHtml(_text);
444 } else {
445 ui->textBrowser->setText(text);
446 }

466void MainWindow::applyWindowFlagsSettings() {
467 const bool wantOnTop = QtPassSettings::isAlwaysOnTop();
468 const Qt::WindowFlags flags = windowFlags();
469 if (flags.testFlag(Qt::WindowStaysOnTopHint) == wantOnTop) {
470 // setWindowFlags() rebuilds the native window, the teardown that left

605void MainWindow::passShowHandler(const QString &p_output, const QString &file) {
606 // Only the entry we most recently asked to show may repaint the panel; a
607 // slower decrypt of an entry the user has since left is dropped, and so is
608 // one that arrives after the selection was cleared (deselect, a profile
609 // switch): nothing is asked to be shown then.
610 if (!file.isEmpty() && file != m_shownFile) {
611 // A newer show is pending and releases the interface when it lands;
612 // with none (the selection was cleared) nothing else would until the
613 // watchdog. setUiElementsEnabled() keeps a re-encryption's lock.
614 if (m_shownFile.isEmpty()) {
615 setUiElementsEnabled(true);
616 }
617 return;
618 }
619 const AppSettings s = QtPassSettings::load();
620 // What a template marks secret, in the settings or the store's
621 // .templates; read per entry shown, so an edit there counts at once.
622 if (s.hidePassword) {
623 m_displayPanel->setMarkedSecrets(
624 TemplateIO::secretFieldsIn(s.passTemplate) +
625 TemplateIO::readSecretFields(s.passStore));
626 }
627 QStringList templ =
628 s.useTemplate ? TemplateIO::fieldsIn(s.passTemplate) : QStringList();
629 bool allFields = s.useTemplate && s.templateAllFields;
630 // A secret the panel hides is a field even when the template doesn't list
631 // it: left in the text below the fields, it would show in plain view.
632 FileContent fileContent = FileContent::parse(
633 p_output, templ, allFields, [this, &s](const QString &name) {
634 return m_displayPanel->isSecret(name, s);
635 });
636 QString output = p_output;
637 // Display variant: empty when the password line is itself an otpauth URI, so
638 // the shared secret is neither rendered nor copied to the clipboard.
639 QString password = fileContent.getPasswordForDisplay();
640
641 // Skipped for an OTP request: the user asked for a code, and two

672 // is ours, everything else belongs to a tree click or a Ctrl+C.

923void MainWindow::onGrepFinished(
924 const QList<QPair<QString, QStringList>> &results) {
925 const GrepSearchController::FinishOutcome outcome = m_grep.finishSearch();
926 if (outcome.restoreCursor) {
927 QApplication::restoreOverrideCursor();
928 }
929 // Re-enable the UI before the discard check so a cancelled search can never
930 // leave controls disabled.
931 setUiElementsEnabled(true);
932 if (outcome.discard) {
933 return;
934 }
935 if (!m_grep.inGrepMode())
936 return;
937 ui->grepResultsList->clear();
938 if (results.isEmpty()) {
939 ui->statusBar->showMessage(tr("No matches found."), 3000);
940 ui->grepResultsList->setVisible(false);
941 ui->treeView->setVisible(true);
942 return;
943 }
944 const AppSettings s = QtPassSettings::load();
945 const bool hideContent = s.hideContent;
946 int totalLines = 0;
947 int entries = 0;
948 bool stopped = false;
949 for (const auto &pair : results) {
950 if (NativeGrep::isStoppedMarker(pair)) {
951 stopped = true;
952 continue;
953 }
954 ++entries;
955 auto *entryItem = new QTreeWidgetItem(ui->grepResultsList);
956 entryItem->setText(0, pair.first);
957 entryItem->setData(0, Qt::UserRole, pair.first);
958 for (const QString &line : pair.second) {
959 auto *lineItem = new QTreeWidgetItem(entryItem);
960 // The marker is no content: shown as it is with hidden content too,
961 // and not counted as a match.
962 if (NativeGrep::isMoreMarker(line)) {
963 lineItem->setText(0, tr("… more matches in this entry"));
964 } else {
965 lineItem->setText(0, hideContent ? "***" + tr("Content hidden") + "***"
966 : line);
967 ++totalLines;
968 }
969 lineItem->setData(0, Qt::UserRole, pair.first);
970 }
971 }
972 ui->grepResultsList->expandAll();
973 ui->treeView->setVisible(false);
974 ui->grepResultsList->setVisible(true);
975 if (stopped) {
976 // The search stopped there; more entries may match.
977 ui->statusBar->showMessage(
978 tr("Showing the first %n match(es); search for something narrower "
979 "to see the rest.",
980 nullptr, totalLines),
981 10000);
982 } else {
983 ui->statusBar->showMessage(tr("Found %n match(es)", nullptr, totalLines) +
984 " " +
985 tr("in %n entr(ies).", nullptr, entries),
986 3000);
987 }
988 if (s.useAutoclearPanel)
989 clearPanelTimer.start();
990}

1089auto MainWindow::confirmDeletion(const QString &file, bool isDir) -> bool {
1090 const QString path = QDir::separator() + file;
1091 if (!isDir) {
1092 return PlainMessageBox::askYes(
1093 this, tr("Delete password?"),
1094 tr("Are you sure you want to delete %1?").arg(path));
1095 }
1096 // Whole sentences, not a question stitched from pieces: a verb-final
1097 // language cannot put the rest of the question after the path.
1098 // Plain text, as every box quoting the store: the path is a name other
1099 // writers control, and as rich text "a b" would also show as "a b".
1100 QString question =
1101 tr("Are you sure you want to delete %1 and the whole content?").arg(path);
1102 // A link, junction or special file inside is as unexpected as a stray
1103 // plain file: the walker leaves them out of content and reports them.
1104 QStringList skipped;
1105 const QStringList content = Util::regularFilesUnder(
1106 m_tree->fileSystem().rootPath() + path, {QStringLiteral("*")}, &skipped);
1107 const bool unexpected =
1108 !skipped.isEmpty() ||
1109 std::any_of(content.cbegin(), content.cend(), [](const QString &entry) {
1110 return QFileInfo(entry).suffix() != QLatin1String("gpg");
1111 });
1112 if (unexpected) {
1113 question += QStringLiteral("\n\n") +
1114 tr("Attention: there are unexpected files in the given "
1115 "folder, check them before continuing.");
1116 }
1117 return PlainMessageBox::askYes(this, tr("Delete folder?"), question);
1118}
