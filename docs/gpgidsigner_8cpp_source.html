47auto GpgIdSigner::haveSecretKey() const -> bool {
48 if (!enabled()) {
49 return false;
50 }
51 // Only the first key signs (see sign()), so only its secret key matters:
52 // a usable second key would let Init start and fail at the signature.
53 const QString &key = m_keys.first();
54 QString out;
55 const int rc = run({QStringLiteral("--status-fd=1"),
56 QStringLiteral("--list-secret-keys"), key},
57 &out);
58 if (rc != 0) {
59 qCDebug(lcQtPass) << "GPG list-secret-keys failed with code:" << rc;
60 return false;
61 }
62 return out.contains(QStringLiteral("[GNUPG:] KEY_CONSIDERED ") + key);
63}

65auto GpgIdSigner::sign(const QString &gpgIdFile, const QByteArray &contents,
66 QString *error) const -> bool {
67 if (!enabled()) {
68 return true;
69 }
70 if (m_keys.size() > 1) {
71 qCDebug(lcQtPass)
72 << "Multiple signing keys configured; using only the first key:"
73 << m_keys.first();
74 }
75 // The executor feeds stdin as UTF-8 text (see verify()).
76 const QString text = QString::fromUtf8(contents);
77 if (text.toUtf8() != contents) {
78 if (error)
79 *error = QCoreApplication::translate(
80 "GpgIdSigner", "the recipient list is not valid UTF-8");
81 return false;
82 }
83 QTemporaryDir scratch;
84 if (!scratch.isValid()) {
85 if (error)
86 *error = scratch.errorString();
87 return false;
88 }
89 const QString output = scratch.filePath(QStringLiteral("sig"));
90 // --local-user, not --default-key: without that secret key gpg would
91 // print "all values passed to '--default-key' ignored", sign with
92 // whatever secret key it has and exit 0.
93 const int rc = run(
94 {QStringLiteral("--local-user"), m_keys.first(), QStringLiteral("--yes"),
95 QStringLiteral("--detach-sign"), QStringLiteral("--output"),
96 Executor::translatePathForWsl(output, m_gpg), QStringLiteral("-")},
97 nullptr, error, text);
98 if (rc != 0) {
99 qCDebug(lcQtPass) << "GPG signing failed with code:" << rc;
100 return false;
101 }
102 QFile signature(output);
103 if (!signature.open(QIODevice::ReadOnly) || signature.size() <= 0) {
104 if (error)
105 *error =
106 QCoreApplication::translate("GpgIdSigner", "gpg wrote no signature");
107 return false;
108 }
109 const QByteArray bytes = signature.readAll();
110 signature.close();
111 // Checked before it is placed: a signature by any key but the configured
112 // ones (an older gpg, a wrapper, a smartcard picking its own) is not
113 // written, and the one there, if any, stays.
114 if (!verify(contents, output)) {
115 if (error)
116 *error = QCoreApplication::translate(
117 "GpgIdSigner",
118 "the new signature does not verify as one of the signing keys");
119 return false;
120 }
121 return Util::writeFileReplacing(gpgIdFile + QStringLiteral(".sig"), bytes,
122 true, error);
123}

125auto GpgIdSigner::validSigFingerprints(const QString &statusOutput)
126 -> QStringList {
127 static const QRegularExpression re(
128 // 40 hex for v4 keys, 64 for v5/v6 (gpg 2.5+), either position.
129 R"(^\‍[GNUPG:\‍] VALIDSIG ([A-F0-9]{40}(?:[A-F0-9]{24})?) .* )"
130 R"(([A-F0-9]{40}(?:[A-F0-9]{24})?)\r?$)",
131 QRegularExpression::MultilineOption);
132 // gpg reports a signature by a revoked key as REVKEYSIG plus KEYREVOKED,
133 // and still prints VALIDSIG and exits 0. A signing key revoked because it
134 // was compromised must not vouch for a recipient list, so a signature
135 // whose block (from one NEWSIG to the next) says so does not count. One
136 // by an expired key does, as gpg calls it a good signature: expiry is not
137 // compromise, and pass accepts it too.
138 static const QRegularExpression revoked(
139 QStringLiteral(R"(^\‍[GNUPG:\‍] (?:REVKEYSIG|KEYREVOKED)\b)"),
140 QRegularExpression::MultilineOption);
141 static const QRegularExpression newSig(
142 QStringLiteral(R"(^\‍[GNUPG:\‍] NEWSIG\b)"),
143 QRegularExpression::MultilineOption);
144 const QRegularExpressionMatch m = re.match(statusOutput);
145 if (!m.hasMatch()) {
146 return {};
147 }
148 const qsizetype from =
149 statusOutput.lastIndexOf(newSig, m.capturedStart()) < 0
150 ? 0
151 : statusOutput.lastIndexOf(newSig, m.capturedStart());
152 const qsizetype to = statusOutput.indexOf(newSig, m.capturedEnd());
153 const QString block = statusOutput.mid(from, to < 0 ? -1 : to - from);
154 if (revoked.match(block).hasMatch()) {
155 return {};
156 }
157 return {m.captured(1), m.captured(2)};
158}

160auto GpgIdSigner::verify(const QByteArray &contents,
161 const QString &signatureFile) const -> bool {
162 if (!enabled()) {
163 return true;
164 }
165 // The executor feeds stdin as UTF-8 text. A .gpg-id is one: key IDs,
166 // fingerprints, e-mail addresses, comments. Bytes that would not survive
167 // the round trip cannot be verified faithfully, so they are not verified
168 // at all rather than as something else.
169 const QString text = QString::fromUtf8(contents);
170 if (text.toUtf8() != contents) {
171 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
172 << ": the signed file is not valid UTF-8";
173 return false;
174 }
175 // Same for the signature: a valid pair replayed from elsewhere through two
176 // links is not this folder's list.
177 if (Util::isLinkedFolder(signatureFile)) {
178 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
179 << ": the signature file is a link";
180 return false;
181 }
182 // A regular file of bounded size, as the list itself is read: a FIFO
183 // planted under the name would block gpg, and the GUI thread waiting for
184 // it, and a sparse file of gigabytes would be read whole.
185 {
186 QFile probe;
187 if (!Util::openRegularFile(signatureFile, probe) ||
188 probe.size() > kMaxSignatureBytes) {
189 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
190 << ": not a regular file of at most"
191 << kMaxSignatureBytes << "bytes";
192 return false;
193 }
194 }
195 QString out;
196 const QString sig = Executor::translatePathForWsl(signatureFile, m_gpg);
197 // "-" makes gpg read the signed data from stdin: the bytes we hold.
198 const int rc =
199 run({QStringLiteral("--verify"), QStringLiteral("--status-fd=1"),
200 QStringLiteral("--"), sig, QStringLiteral("-")},
201 &out, nullptr, text);
202 if (rc != 0) {
203 qCDebug(lcQtPass) << "GPG verify failed with code:" << rc;
204 return false;
205 }
206 const QStringList fingerprints = validSigFingerprints(out);
207 for (const QString &key : m_keys) {
208 if (fingerprints.contains(key)) {
209 return true;
210 }
211 }
212 return false;
213}

215auto GpgIdSigner::verifyFile(const QString &gpgIdFile,
216 QByteArray *contents) const -> bool {
217 // A link under the name is not the store's .gpg-id, whatever it points
218 // to and however well that is signed.
219 if (Util::isLinkedFolder(gpgIdFile)) {
220 contents->clear();
221 return false;
222 }
223 // Read as the regular file it is and bounded: a FIFO would block, a
224 // sparse file of gigabytes be read whole, before any signature is checked.
225 std::optional<QByteArray> bytes =
226 Util::readRegularFile(gpgIdFile, Util::kMaxGpgIdBytes);
227 if (!bytes) {
228 contents->clear();
229 return false;
230 }
231 *contents = *std::move(bytes);
232 return verify(*contents, gpgIdFile + QStringLiteral(".sig"));
233}
