48auto GpgIdSigner::haveSecretKey() const -> bool {
49 if (!enabled()) {
50 return false;
51 }
52 // Only the first key signs (see sign()), so only its secret key matters:
53 // a usable second key would let Init start and fail at the signature.
54 const QString &key = m_keys.first();
55 QString out;
56 const int rc =
57 run({QStringLiteral("--with-colons"), QStringLiteral("--fixed-list-mode"),
58 QStringLiteral("--list-secret-keys"), QStringLiteral("--"), key},
59 &out);
60 if (rc != 0) {
61 qCDebug(lcQtPass) << "GPG list-secret-keys failed with code:" << rc;
62 return false;
63 }
64 // Records, one per line: a key (sec, or ssb for a subkey) and later its
65 // own fpr (an rvk may come between). A user ID is a record of its own,
66 // with its colons and line breaks escaped, so nothing a key's owner
67 // wrote can pass for either; a search of the whole output for the
68 // fingerprint could be fooled by one. A key signs when it can (s in its
69 // own capabilities), is not revoked, expired or invalid, and its secret
70 // part is here (field 15 is # for a stub).
71 // Configured as a subkey, that subkey must sign. Configured as a primary
72 // key, gpg's --local-user picks a signing key from its whole block, so
73 // the primary or any of its subkeys will do.
74 struct Key {
75 bool signs = false;
76 bool valid = false;
77 bool primary = false;
78 QString fpr;
79 };
80 QList<QList<Key>> blocks;
81 bool awaitingFpr = false;
82 const QStringList lines = out.split(QLatin1Char('\n'));
83 for (const QString &line : lines) {
84 const QStringList fields = line.trimmed().split(QLatin1Char(':'));
85 const QString &type = fields.first();
86 const bool primary = type == QLatin1String("sec");
87 if (primary || type == QLatin1String("ssb")) {
88 if (primary) {
89 blocks.append(QList<Key>());
90 }
91 if (blocks.isEmpty()) {
92 awaitingFpr = false;
93 continue;
94 }
95 const QString validity = fields.value(1);
96 Key k;
97 k.primary = primary;
98 k.valid = validity != QLatin1String("r") &&
99 validity != QLatin1String("e") &&
100 validity != QLatin1String("i");
101 k.signs = k.valid && fields.value(11).contains(QLatin1Char('s')) &&
102 fields.value(14) != QLatin1String("#");
103 blocks.last().append(k);
104 awaitingFpr = true;
105 } else if (type == QLatin1String("fpr") && awaitingFpr) {
106 blocks.last().last().fpr = fields.value(9);
107 awaitingFpr = false;
108 }
109 }
110 for (const QList<Key> &block : std::as_const(blocks)) {
111 // A revoked or expired primary key takes its subkeys with it.
112 if (block.isEmpty() || !block.first().valid) {
113 continue;
114 }
115 for (const Key &k : block) {
116 if (k.fpr.compare(key, Qt::CaseInsensitive) != 0) {
117 continue;
118 }
119 if (k.signs) {
120 return true;
121 }
122 if (k.primary && std::any_of(block.cbegin(), block.cend(),
123 [](const Key &sub) { return sub.signs; })) {
124 return true;
125 }
126 }
127 }
128 return false;
129}

131auto GpgIdSigner::sign(const QString &gpgIdFile, const QByteArray &contents,
132 QString *error) const -> bool {
133 if (!enabled()) {
134 return true;
135 }
136 if (m_keys.size() > 1) {
137 qCDebug(lcQtPass)
138 << "Multiple signing keys configured; using only the first key:"
139 << m_keys.first();
140 }
141 // The executor feeds stdin as UTF-8 text (see verify()).
142 const QString text = QString::fromUtf8(contents);
143 if (text.toUtf8() != contents) {
144 if (error)
145 *error = QCoreApplication::translate(
146 "GpgIdSigner", "the recipient list is not valid UTF-8");
147 return false;
148 }
149 QTemporaryDir scratch;
150 if (!scratch.isValid()) {
151 if (error)
152 *error = scratch.errorString();
153 return false;
154 }
155 const QString output = scratch.filePath(QStringLiteral("sig"));
156 // --local-user, not --default-key: without that secret key gpg would
157 // print "all values passed to '--default-key' ignored", sign with
158 // whatever secret key it has and exit 0.
159 const int rc = run(
160 {QStringLiteral("--local-user"), m_keys.first(), QStringLiteral("--yes"),
161 QStringLiteral("--detach-sign"), QStringLiteral("--output"),
162 Executor::translatePathForWsl(output, m_gpg), QStringLiteral("-")},
163 nullptr, error, text);
164 if (rc != 0) {
165 qCDebug(lcQtPass) << "GPG signing failed with code:" << rc;
166 return false;
167 }
168 QFile signature(output);
169 if (!signature.open(QIODevice::ReadOnly) || signature.size() <= 0) {
170 if (error)
171 *error =
172 QCoreApplication::translate("GpgIdSigner", "gpg wrote no signature");
173 return false;
174 }
175 const QByteArray bytes = signature.readAll();
176 signature.close();
177 // Checked before it is placed: a signature by any key but the configured
178 // ones (an older gpg, a wrapper, a smartcard picking its own) is not
179 // written, and the one there, if any, stays.
180 // The signature is in this call's own scratch folder, outside the store:
181 // checked by a copy of this signer without the store root.
182 GpgIdSigner scratchSigner = *this;
183 scratchSigner.m_storeRoot.clear();
184 if (!scratchSigner.verify(contents, output)) {
185 if (error)
186 *error = QCoreApplication::translate(
187 "GpgIdSigner",
188 "the new signature does not verify as one of the signing keys");
189 return false;
190 }
191 return Util::writeFileReplacing(gpgIdFile + QStringLiteral(".sig"), bytes,
192 true, error);
193}

195auto GpgIdSigner::validSigFingerprints(const QString &statusOutput)
196 -> QStringList {
197 static const QRegularExpression re(
198 // 40 hex for v4 keys, 64 for v5/v6 (gpg 2.5+), either position.
199 R"(^\‍[GNUPG:\‍] VALIDSIG ([A-F0-9]{40}(?:[A-F0-9]{24})?) .* )"
200 R"(([A-F0-9]{40}(?:[A-F0-9]{24})?)\r?$)",
201 QRegularExpression::MultilineOption);
202 // gpg reports a signature by a revoked key as REVKEYSIG plus KEYREVOKED,
203 // and still prints VALIDSIG and exits 0. A signing key revoked because it
204 // was compromised must not vouch for a recipient list, so a signature
205 // whose block (from one NEWSIG to the next) says so does not count. One
206 // by an expired key does, as gpg calls it a good signature: expiry is not
207 // compromise, and pass accepts it too.
208 static const QRegularExpression revoked(
209 QStringLiteral(R"(^\‍[GNUPG:\‍] (?:REVKEYSIG|KEYREVOKED)\b)"),
210 QRegularExpression::MultilineOption);
211 static const QRegularExpression newSig(
212 QStringLiteral(R"(^\‍[GNUPG:\‍] NEWSIG\b)"),
213 QRegularExpression::MultilineOption);
214 const QRegularExpressionMatch m = re.match(statusOutput);
215 if (!m.hasMatch()) {
216 return {};
217 }
218 const qsizetype from =
219 statusOutput.lastIndexOf(newSig, m.capturedStart()) < 0
220 ? 0
221 : statusOutput.lastIndexOf(newSig, m.capturedStart());
222 const qsizetype to = statusOutput.indexOf(newSig, m.capturedEnd());
223 const QString block = statusOutput.mid(from, to < 0 ? -1 : to - from);
224 if (revoked.match(block).hasMatch()) {
225 return {};
226 }
227 return {m.captured(1), m.captured(2)};
228}

237auto GpgIdSigner::verify(const QByteArray &contents,
238 const QString &signatureFile) const -> bool {
239 if (!enabled()) {
240 return true;
241 }
242 // The executor feeds stdin as UTF-8 text. A .gpg-id is one: key IDs,
243 // fingerprints, e-mail addresses, comments. Bytes that would not survive
244 // the round trip cannot be verified faithfully, so they are not verified
245 // at all rather than as something else.
246 const QString text = QString::fromUtf8(contents);
247 if (text.toUtf8() != contents) {
248 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
249 << ": the signed file is not valid UTF-8";
250 return false;
251 }
252 // git signs a commit or tag the same way a .gpg-id is signed: a detached
253 // signature over the object without its own signature header. With
254 // pass.signcommits (or commit.gpgsign) the signing key signs QtPass's
255 // commits, so such a commit and its signature could be pushed as .gpg-id and
256 // .gpg-id.sig and would verify here as a signed recipient list - whose lines
257 // (the commit's headers and message, part of which can be a pushed entry
258 // name) then become recipients. A commit begins "tree <hash>" and a tag
259 // "object <hash>"; refuse either before gpg is asked.
260 static const QRegularExpression commitOrTag(
261 QStringLiteral(R"(\A(?:tree|object) [0-9a-f]{40}(?:[0-9a-f]{24})?\n)"));
262 if (commitOrTag.match(text).hasMatch()) {
263 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
264 << ": the signed data is a git commit or tag object";
265 return false;
266 }
267 // Same for the signature: a valid pair replayed from elsewhere through two
268 // links is not this folder's list.
269 if (Util::isLinkedFolder(signatureFile)) {
270 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
271 << ": the signature file is a link";
272 return false;
273 }
274 // Read as a regular file of bounded size, as the list itself is, and
275 // give gpg a copy of those bytes: a FIFO planted under the name would
276 // block gpg, and the GUI thread waiting for it, a sparse file of
277 // gigabytes would be read whole, and either could be swapped in between
278 // a check of the name and gpg opening it.
279 const std::optional<QByteArray> signature =
280 readOwnFile(signatureFile, kMaxSignatureBytes);
281 if (!signature) {
282 qCDebug(lcQtPass) << "Refusing to verify" << signatureFile
283 << ": not a regular file of at most" << kMaxSignatureBytes
284 << "bytes";
285 return false;
286 }
287 QTemporaryDir scratch;
288 if (!scratch.isValid()) {
289 return false;
290 }
291 const QString copy = scratch.filePath(QStringLiteral("sig"));
292 {
293 QFile file(copy);
294 if (!file.open(QIODevice::WriteOnly) ||
295 file.write(*signature) != signature->size()) {
296 return false;
297 }
298 }
299 QString out;
300 const QString sig = Executor::translatePathForWsl(copy, m_gpg);
301 // "-" makes gpg read the signed data from stdin: the bytes we hold.
302 // --batch stops at the first bad signature; without it gpg checked every
303 // packet in the file, each against the key it names, however slow.
304 // --trust-model pgp, whatever gpg.conf says: under `trust-model always`
305 // gpg skips the trust check that reports KEYREVOKED, so a revoked key that
306 // has also expired came back as merely expired (EXPKEYSIG) and its
307 // signature counted. The model changes no verdict on the signature
308 // itself, only which notes gpg adds to it. --no-auto-check-trustdb: a
309 // keyring kept under another model would otherwise have its trustdb
310 // rebuilt on the first such call; the revoked flag comes from the key
311 // block, not from that computation.
312 const int rc =
313 run({QStringLiteral("--batch"), QStringLiteral("--trust-model=pgp"),
314 QStringLiteral("--no-auto-check-trustdb"),
315 QStringLiteral("--verify"), QStringLiteral("--status-fd=1"),
316 QStringLiteral("--"), sig, QStringLiteral("-")},
317 &out, nullptr, text);
318 if (rc != 0) {
319 qCDebug(lcQtPass) << "GPG verify failed with code:" << rc;
320 return false;
321 }
322 const QStringList fingerprints = validSigFingerprints(out);
323 for (const QString &key : m_keys) {
324 if (fingerprints.contains(key)) {
325 return true;
326 }
327 }
328 return false;
329}

331auto GpgIdSigner::verifyFile(const QString &gpgIdFile,
332 QByteArray *contents) const -> bool {
333 // A link under the name is not the store's .gpg-id, whatever it points
334 // to and however well that is signed.
335 if (Util::isLinkedFolder(gpgIdFile)) {
336 contents->clear();
337 return false;
338 }
339 // Read as the regular file it is and bounded: a FIFO would block, a
340 // sparse file of gigabytes be read whole, before any signature is checked.
341 std::optional<QByteArray> bytes =
342 readOwnFile(gpgIdFile, Util::kMaxGpgIdBytes);
343 if (!bytes) {
344 contents->clear();
345 return false;
346 }
347 *contents = *std::move(bytes);
348 return verify(*contents, gpgIdFile + QStringLiteral(".sig"));
349}
