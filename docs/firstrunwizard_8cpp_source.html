100auto FirstRunWizard::secretKeys(const QString &gpgExecutable)
101 -> QList<UserInfo> {
102 if (!runnable(gpgExecutable)) {
103 return {};
104 }
105 QString out;
106 const int rc = Executor::executeBlocking(
107 gpgExecutable,
108 {QStringLiteral("--no-tty"), QStringLiteral("--with-colons"),
109 QStringLiteral("--with-fingerprint"),
110 QStringLiteral("--list-secret-keys")},
111 &out);
112 if (rc != 0) {
113 return {};
114 }
115 return parseGpgColonOutput(out, true);

123auto FirstRunWizard::initialiseStore(const QString &store) -> bool {
124 QList<UserInfo> recipients;
125 for (const UserInfo &key : std::as_const(m_keys)) {
126 if (key.enabled) {
127 recipients.append(key);
128 }
129 }
130 const bool created = !QDir(store).exists();
131 if (created && !Util::mkpathForStore(store)) {
132 QMessageBox::warning(
133 this, tr("Error"),
134 tr("Failed to create password-store at: %1").arg(store));
135 return false;
136 }
137#ifdef Q_OS_WIN
138 // Only a folder made here is hidden, like ~/.password-store on Unix; a
139 // folder the user picked keeps its attributes.
140 if (created) {
141 SetFileAttributes(store.toStdWString().c_str(), FILE_ATTRIBUTE_HIDDEN);
142 }
143#endif
144 QString note;
145 if (!ProfileInit::initialise(store, recipients, m_settings, m_settings.useGit,
146 &note)) {
147 // ProfileInit writes .gpg-id before it signs and commits; take it back so
148 // a second Finish runs the whole initialisation again instead of finding
149 // a store that looks finished.
150 QFile::remove(QDir(store).filePath(QStringLiteral(".gpg-id")));
151 QFile::remove(QDir(store).filePath(QStringLiteral(".gpg-id.sig")));
152 QMessageBox::warning(this, tr("Password store not initialised"), note);
153 return false;
154 }
155 if (!note.isEmpty()) {
156 // ProfileInit's own note talks of switching profiles; on the first run
157 // the folder simply becomes the store.
158 QMessageBox::information(
159 this, tr("Password store"),
160 tr("%1 already contains encrypted files; they were not "
161 "re-encrypted to the ticked keys. Open Users after the start "
162 "to do that.")
163 .arg(QDir::toNativeSeparators(store)));
164 }
165 return true;
166}

168auto FirstRunWizard::initialiseGit(const QString &store) -> bool {
169 QString note;
170 if (ProfileInit::initGit(store, m_settings, &note)) {
171 return true;
172 }
173 // Take the repository back too, so a second Finish retries the whole thing
174 // instead of skipping a .git that never got its first commit.
175 QDir(QDir(store).filePath(QStringLiteral(".git"))).removeRecursively();
176 QMessageBox::warning(this, tr("Password store not initialised"), note);
177 return false;
178}

180void FirstRunWizard::accept() {
181 // The Finish button validates the page before calling this; a programmatic
182 // accept() gets the same treatment so the last page's choices are in.
183 if (currentPage() != nullptr && !currentPage()->validatePage()) {
184 return;
185 }
186 const QString store = QDir::cleanPath(m_settings.passStore);
187 const bool needsGit =
188 m_settings.useGit && !QDir(store).exists(QStringLiteral(".git"));
189 const bool ready = !isStore(store) ? initialiseStore(store)
190 : needsGit ? initialiseGit(store)
191 : true;
192 if (!ready) {
193 return;
194 }
195 m_settings.passStore = Util::normalizeFolderPath(store);
196 QtPassSettings::save(m_settings);
197 // The backend cached the old paths at construction.
198 PassBackendFactory::invalidate();
199 QWizard::accept();
200}

231void ProgramsWizardPage::initializePage() {
232 AppSettings &s = m_wizard->m_settings;
233 // QtPassSettings::initExecutables() fills empty paths from PATH only; a
234 // stored path whose binary has moved would be shown as an error here, so
235 // look it up again in that case.
236 auto detect = [](QString &stored, const QStringList &names) {
237 if (runnable(stored)) {
238 return;
239 }
240 for (const QString &name : names) {
241 const QString found = Util::findBinaryInPath(name);
242 if (!found.isEmpty()) {
243 stored = found;
244 return;
245 }
246 }
247 };
248 detect(s.gpgExecutable, {QStringLiteral("gpg2"), QStringLiteral("gpg")});
249 detect(s.gitExecutable, {QStringLiteral("git")});
250 detect(s.passExecutable, {QStringLiteral("pass")});
251 detect(s.pwgenExecutable, {QStringLiteral("pwgen")});
252 m_gpg->setText(s.gpgExecutable);
253 m_git->setText(s.gitExecutable);
254 m_pass->setText(s.passExecutable);
255 m_usePass->setChecked(s.usePass || runnable(s.passExecutable));
256 updateStatus();
257}

314void KeyWizardPage::reload() {
315 m_wizard->m_keys =
316 FirstRunWizard::secretKeys(m_wizard->m_settings.gpgExecutable);
317 m_list->clear();
318 const QDateTime now = QDateTime::currentDateTime();
319 for (UserInfo &key : m_wizard->m_keys) {
320 // An expired or otherwise unusable key would make every encryption to
321 // the new store fail; list it, but do not tick it.
322 QString why;
323 if (key.expiry.isValid() && key.expiry < now) {
324 why = tr("expired");
325 } else if (!key.isValid()) {
326 why = tr("not usable");
327 }
328 key.enabled = why.isEmpty();
329 QString label = QStringLiteral("%1\n%2").arg(key.name, key.key_id);
330 if (!why.isEmpty()) {
331 label += QStringLiteral(" (%1)").arg(why);
332 }
333 auto *item = new QListWidgetItem(label, m_list);
334 item->setFlags(key.enabled ? (item->flags() | Qt::ItemIsUserCheckable)
335 : (item->flags() & ~Qt::ItemIsEnabled));
336 item->setCheckState(key.enabled ? Qt::Checked : Qt::Unchecked);
337 }
338 m_status->setText(m_wizard->m_keys.isEmpty()
339 ? tr("GnuPG has no secret key yet. Generate one here, "
340 "or import your existing key with gpg first.")
341 : QString());
342 emit completeChanged();
343}

345void KeyWizardPage::generate() {
346 // A backend of the wizard's own, set up from its copy of the settings, so
347 // the chosen gpg is used and nothing is written before Finish. Each Pass
348 // owns its Executor; the shared backend is not involved.
349 ImitatePass pass;
350 pass.init(m_wizard->m_settings);
351 KeygenDialog d(m_wizard->m_settings.gpgExecutable, &pass, this);
352 d.exec();
353 reload();
354}

387void StoreWizardPage::initializePage() {
388 QString path = m_wizard->m_settings.passStore;
389 if (path.isEmpty()) {
390 path = Util::findPasswordStore();
391 }
392 m_path->setText(QDir::toNativeSeparators(QDir::cleanPath(path)));
393 const QString clean = QDir::cleanPath(path);
394 const bool repository = QDir(clean).exists(QStringLiteral(".git"));
395 bool haveGit = runnable(m_wizard->m_settings.gitExecutable);
396 // A first commit needs a name and an e-mail; without them Finish would
397 // fail at the very end. Ask git now and say what to do instead.
398 const bool identity =
399 haveGit && (repository || ProfileInit::gitIdentityConfigured(
400 QDir::homePath(), m_wizard->m_settings));
401 m_useGit->setToolTip(
402 haveGit && !identity
403 ? tr("Git has no name and e-mail to commit with yet. Run\n"
404 "git config --global user.name \"Your Name\"\n"
405 "git config --global user.email you@example.org\n"
406 "and turn Git on in Settings afterwards.")
407 : tr("Every change becomes a commit; a folder that is no "
408 "repository yet gets one"));
409 haveGit = haveGit && identity;
410 m_useGit->setEnabled(haveGit);
411 // On for a repository or a store still to be made, off for an existing
412 // store that is no repository: turning Git on there would make every
413 // remove fail in a folder git knows nothing about.
414 m_useGit->setChecked(haveGit && (m_wizard->m_settings.useGit || repository ||
415 !FirstRunWizard::isStore(clean)));
416 updateStatus();
417}

427void StoreWizardPage::updateStatus() {
428 const QString path = QDir::cleanPath(m_path->text().trimmed());
429 QString text;
430 if (path.isEmpty()) {
431 text = tr("Enter a folder.");
432 } else if (FirstRunWizard::isStore(path)) {
433 const int entries = static_cast<int>(
434 Util::regularFilesUnder(path, {QStringLiteral("*.gpg")}).size());
435 text = tr("An existing password store with %n entries.", nullptr, entries);
436 } else if (QDir(path).exists()) {
437 text = QDir(path).isEmpty()
438 ? tr("An empty folder; it will be set up as a password store.")
439 : tr("This folder is not a password store yet; a .gpg-id for "
440 "the ticked keys will be written into it.");
441 } else {
442 text = tr("The folder does not exist yet; it will be created.");
443 }
444 if (!path.isEmpty() && !FirstRunWizard::isStore(path) && recipients() == 0) {
445 text += QLatin1Char(' ') +
446 tr("Go back and tick at least one key to encrypt it to.");
447 }
448 m_status->setText(text);
449 emit completeChanged();
450}
