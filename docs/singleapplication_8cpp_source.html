16SingleApplication::SingleApplication(
17 int &argc, char *argv[], // NOLINT(modernize-avoid-c-arrays)
18 QString uniqueKey)
19 : QApplication(argc, argv), _isRunning(false),
20 _uniqueKey(std::move(uniqueKey)) {
21 sharedMemory.setKey(_uniqueKey);
22 if (sharedMemory.attach()) {
23 if (peerIsListening()) {
24 _isRunning = true;
25 return;
26 }
27 // The segment outlived the instance that created it (a crash leaves
28 // System V shared memory behind on Unix). Nobody answers on the socket,
29 // so take over instead of forwarding into the void forever.
30 qCDebug(lcQtPass) << "Stale single instance segment, taking over.";
31 sharedMemory.detach();
32 }
33 becomePrimary();
34}

38void SingleApplication::receiveMessage() {
39 while (QLocalSocket *localSocket = localServer->nextPendingConnection()) {
40 auto buffer = QSharedPointer<QByteArray>::create();
41 connect(localSocket, &QLocalSocket::readyRead, this,
42 [localSocket, buffer]() {
43 buffer->append(localSocket->readAll());
44 if (buffer->size() > maxMessageBytes) {
45 // Not a command line any more; drop the peer and its data.
46 buffer->clear();
47 localSocket->abort();
48 }
49 });
50 connect(localSocket, &QLocalSocket::disconnected, this,
51 [this, localSocket, buffer]() {
52 buffer->append(localSocket->readAll());
53 if (!buffer->isEmpty()) {
54 emit messageAvailable(QString::fromUtf8(buffer->constData()));
55 }
56 localSocket->deleteLater();
57 });
58 }
59}

72 // its segment and take its place, so the window main() opens next is the

74 // later launch fail create() and open yet another window without IPC.
75 _isRunning = false;

82void SingleApplication::becomePrimary() {
83 // create shared memory.
84 if (!sharedMemory.create(1)) {
85 // Another launch claimed the segment first (two launchers recovering
86 // from the same crash). Give it a moment to start listening and become
87 // its secondary, so main() forwards instead of opening a second window.
88 if (sharedMemory.attach()) {
89 for (int attempt = 0; attempt < takeoverProbes; ++attempt) {
90 if (peerIsListening()) {
91 _isRunning = true;
92 return;
93 }
94 QThread::msleep(takeoverProbeIntervalMs);
95 }
96 sharedMemory.detach();
97 }
98 qCDebug(lcQtPass) << "Unable to create single instance.";
99 return;
100 }
101 // create local server and listen to incoming messages from other
102 // instances. A socket file left behind by a crashed instance would make
103 // listen() fail (address in use) and silently disable IPC for good, so
104 // clear it first: no live peer answered on it, or create() would have
105 // failed.
106 QLocalServer::removeServer(_uniqueKey);
107 localServer.reset(new QLocalServer(this));
108 localServer->setSocketOptions(QLocalServer::UserAccessOption);
109 connect(localServer.data(), &QLocalServer::newConnection, this,
110 &SingleApplication::receiveMessage);
111 if (!localServer->listen(_uniqueKey)) {
112 qCWarning(lcQtPass) << "SingleApplication: cannot listen on" << _uniqueKey
113 << ":" << localServer->errorString();
114 // Holding the segment without a server would make every later launch
115 // attach, fail the probe and then fail create(): no instance could ever
116 // take over while this one lives. Release it and run without IPC.
117 localServer.reset();
118 sharedMemory.detach();
119 }
120}

122auto SingleApplication::forwardMessage(const QString &message) -> bool {
123 QLocalSocket localSocket(this);
124 localSocket.connectToServer(_uniqueKey, QIODevice::WriteOnly);
125 if (!localSocket.waitForConnected(timeout)) {
126 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
127 return false;
128 }
129 QByteArray payload = message.toUtf8();
130 if (payload.isEmpty()) {
131 payload = QByteArray(1, '\0');
132 }
133 localSocket.write(payload);
134 if (!localSocket.waitForBytesWritten(timeout)) {
135 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
136 return false;
137 }
138 localSocket.disconnectFromServer();
139 return true;
140}
