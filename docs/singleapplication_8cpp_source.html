45// Named after a hash of the base, so any login name makes a valid file name
46// and the path stays well under the 104 bytes macOS allows a socket path.
47auto SingleApplication::privateKeyIn(const QString &base, const QString &dir)
48 -> QString {
49 if (dir.isEmpty() || !QFileInfo(dir).isDir()) {
50 qCWarning(lcQtPass) << "SingleApplication: no private folder for the"
51 " socket; running without single instance.";
52 return base +
53 QStringLiteral("%1").arg(QRandomGenerator::system()->generate64(),
54 16, 16, QLatin1Char('0'));
55 }
56 const QByteArray hash =
57 QCryptographicHash::hash(base.toUtf8(), QCryptographicHash::Sha256)
58 .toHex()
59 .left(16);
60 return QDir(dir).filePath(QStringLiteral("qtpass-") +
61 QString::fromLatin1(hash));
62}

64auto SingleApplication::lockPathFor(const QString &key) -> QString {
65 if (QDir::isAbsolutePath(key)) {

76void SingleApplication::receiveMessage() {
77 while (QLocalSocket *localSocket = localServer->nextPendingConnection()) {
78 auto buffer = QSharedPointer<QByteArray>::create();
79 connect(localSocket, &QLocalSocket::readyRead, this,
80 [localSocket, buffer]() {
81 buffer->append(localSocket->readAll());
82 if (buffer->size() > maxMessageBytes) {
83 // Not a command line any more; drop the peer and its data.
84 buffer->clear();
85 localSocket->abort();
86 }
87 });
88 connect(localSocket, &QLocalSocket::disconnected, this,
89 [this, localSocket, buffer]() {
90 buffer->append(localSocket->readAll());
91 if (!buffer->isEmpty()) {
92 emit messageAvailable(QString::fromUtf8(buffer->constData()));
93 }
94 localSocket->deleteLater();
95 });
96 }
97}

102auto SingleApplication::sendMessage(const QString &message) -> bool {
103 if (!_isRunning) {
104 return false;
105 }
106 if (forwardMessage(message)) {
107 return true;
108 }
109 // The peer vanished between the probe in the constructor and now. Take its
110 // lock and its place, so the window main() opens next is the reachable
111 // instance; if it still holds the lock, it may be restarting its server.
112 _isRunning = false;
113 claimOrFollow();
114 return false;
115}

118void SingleApplication::claimOrFollow() {
119 const bool leftOver = QFileInfo::exists(_lock.fileName());
120 if (_lock.tryLock(0)) {
121 if (leftOver) {
122 // A crash leaves the lock file behind; its process is gone.
123 qCDebug(lcQtPass) << "Stale single instance lock, taking over.";
124 }
125 becomePrimary();
126 return;
127 }
128 if (_lock.error() != QLockFile::LockFailedError) {
129 qCWarning(lcQtPass) << "SingleApplication: cannot take" << _lock.fileName()
130 << "; running without IPC.";
131 return;
132 }
133 qCDebug(lcQtPass) << "Single instance lock held, probing its server.";
134 waitForPeer();
135}

137void SingleApplication::becomePrimary() {
138 // A socket file left behind by a crashed instance would make listen() fail
139 // (address in use) and silently disable IPC for good, so clear it first:
140 // this instance holds the lock, so no live instance listens on it.
141 QLocalServer::removeServer(_uniqueKey);
142 localServer.reset(new QLocalServer(this));
143 localServer->setSocketOptions(QLocalServer::UserAccessOption);
144 connect(localServer.data(), &QLocalServer::newConnection, this,
145 &SingleApplication::receiveMessage);
146 if (!localServer->listen(_uniqueKey)) {
147 qCWarning(lcQtPass) << "SingleApplication: cannot listen on" << _uniqueKey
148 << ":" << localServer->errorString();
149 // Holding the lock without a server would make every later launch wait
150 // for a server that never comes: no instance could take over while this
151 // one lives. Release it and run without IPC.
152 localServer.reset();
153 _lock.unlock();
154 }
155}

157void SingleApplication::waitForPeer() {
158 for (int attempt = 0; attempt < takeoverProbes; ++attempt) {
159 if (peerIsListening()) {
160 _isRunning = true;
161 return;
162 }
163 QThread::msleep(takeoverProbeIntervalMs);
164 }
165 qCDebug(lcQtPass) << "Unable to create single instance.";
166}

168auto SingleApplication::forwardMessage(const QString &message) -> bool {
169 QLocalSocket localSocket(this);
170 localSocket.connectToServer(_uniqueKey, QIODevice::WriteOnly);
171 if (!localSocket.waitForConnected(timeout)) {
172 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
173 return false;
174 }
175 QByteArray payload = message.toUtf8();
176 if (payload.isEmpty()) {
177 payload = QByteArray(1, '\0');
178 }
179 localSocket.write(payload);
180 if (!localSocket.waitForBytesWritten(timeout)) {
181 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
182 return false;
183 }
184 localSocket.disconnectFromServer();
185 return true;
186}

188auto SingleApplication::peerIsListening() -> bool {
189 QLocalSocket probe;
190 probe.connectToServer(_uniqueKey, QIODevice::WriteOnly);
191 if (!probe.waitForConnected(timeout)) {
192 return false;
193 }
194 probe.disconnectFromServer();
195 return true;
