26SingleApplication::SingleApplication(
27 int &argc, char *argv[], // NOLINT(modernize-avoid-c-arrays)
28 QString uniqueKey)
29 : QApplication(argc, argv), _isRunning(false),
30 _uniqueKey(std::move(uniqueKey)), _lock(lockPathFor(_uniqueKey)) {

50// Named after a hash of the base, so any login name makes a valid file name
51// and the path stays well under the 104 bytes macOS allows a socket path.
52auto SingleApplication::privateKeyIn(const QString &base, const QString &dir)
53 -> QString {
54 if (dir.isEmpty() || !QFileInfo(dir).isDir()) {
55 qCWarning(lcQtPass) << "SingleApplication: no private folder for the"
56 " socket; running without single instance.";
57 return base +
58 QStringLiteral("%1").arg(QRandomGenerator::system()->generate64(),
59 16, 16, QLatin1Char('0'));
60 }
61 const QByteArray hash =
62 QCryptographicHash::hash(base.toUtf8(), QCryptographicHash::Sha256)
63 .toHex()
64 .left(16);
65 return QDir(dir).filePath(QStringLiteral("qtpass-") +
66 QString::fromLatin1(hash));
67}

77auto tokenUser(HANDLE token) -> std::vector<BYTE> {
78 DWORD size = 0;
79 GetTokenInformation(token, TokenUser, nullptr, 0, &size);
80 if (size == 0) {
81 return {};
82 }
83 std::vector<BYTE> record(size);
84 if (!GetTokenInformation(token, TokenUser, record.data(), size, &size)) {
85 return {};
86 }
87 return record;
88}

95auto processUser(HANDLE process) -> std::vector<BYTE> {
96 HANDLE token = nullptr;
97 if (!OpenProcessToken(process, TOKEN_QUERY, &token)) {
98 return {};
99 }
100 std::vector<BYTE> record = tokenUser(token);
101 CloseHandle(token);
102 return record;

108// Windows only: see the header. A server whose process or token this user
109// may not open is not this user's.
110auto SingleApplication::serverRunsAsThisUser(const QLocalSocket &socket)
111 -> bool {
112#ifdef Q_OS_WIN
113 // NOLINTNEXTLINE(performance-no-int-to-ptr): QLocalSocket gives the HANDLE
114 const auto pipe = reinterpret_cast<HANDLE>(socket.socketDescriptor());
115 ULONG pid = 0;
116 if (pipe == nullptr || pipe == INVALID_HANDLE_VALUE ||
117 !GetNamedPipeServerProcessId(pipe, &pid)) {
118 return false;
119 }
120 HANDLE server = OpenProcess(PROCESS_QUERY_LIMITED_INFORMATION, FALSE, pid);
121 if (server == nullptr) {
122 return false;
123 }
124 const std::vector<BYTE> theirs = processUser(server);
125 CloseHandle(server);
126 const std::vector<BYTE> ours = processUser(GetCurrentProcess());
127 if (theirs.empty() || ours.empty()) {
128 return false;
129 }
130 return EqualSid(
131 reinterpret_cast<const TOKEN_USER *>(theirs.data())->User.Sid,
132 reinterpret_cast<const TOKEN_USER *>(ours.data())->User.Sid) !=
133 FALSE;
134#else
135 Q_UNUSED(socket);
136 return true;
137#endif
138}

152void SingleApplication::receiveMessage() {
153 while (QLocalSocket *localSocket = localServer->nextPendingConnection()) {
154 auto buffer = QSharedPointer<QByteArray>::create();
155 connect(localSocket, &QLocalSocket::readyRead, this,
156 [localSocket, buffer]() {
157 buffer->append(localSocket->readAll());
158 if (buffer->size() > maxMessageBytes) {
159 // Not a command line any more; drop the peer and its data.
160 buffer->clear();
161 localSocket->abort();
162 }
163 });
164 connect(localSocket, &QLocalSocket::disconnected, this,
165 [this, localSocket, buffer]() {
166 buffer->append(localSocket->readAll());
167 if (!buffer->isEmpty()) {
168 emit messageAvailable(QString::fromUtf8(buffer->constData()));
169 }
170 localSocket->deleteLater();
171 });
172 }
173}

178auto SingleApplication::sendMessage(const QString &message) -> bool {
179 if (!_isRunning) {
180 return false;
181 }
182 if (forwardMessage(message)) {
183 return true;
184 }
185 // The peer vanished between the probe in the constructor and now. Take its
186 // lock and its place, so the window main() opens next is the reachable
187 // instance; if it still holds the lock, it may be restarting its server.
188 _isRunning = false;
189 claimOrFollow();
190 return false;
191}

193// private functions.
194void SingleApplication::claimOrFollow() {
195 const bool leftOver = QFileInfo::exists(_lock.fileName());
196 if (_lock.tryLock(0)) {
197 if (leftOver) {
198 // A crash leaves the lock file behind; its process is gone.
199 qCDebug(lcQtPass) << "Stale single instance lock, taking over.";
200 }
201 becomePrimary();
202 return;
203 }
204 if (_lock.error() != QLockFile::LockFailedError) {
205 qCWarning(lcQtPass) << "SingleApplication: cannot take" << _lock.fileName()
206 << "; running without IPC.";
207 return;
208 }
209 qCDebug(lcQtPass) << "Single instance lock held, probing its server.";
210 waitForPeer();
211}

213void SingleApplication::becomePrimary() {
214 // A socket file left behind by a crashed instance would make listen() fail
215 // (address in use) and silently disable IPC for good, so clear it first:
216 // this instance holds the lock, so no live instance listens on it.
217 QLocalServer::removeServer(_uniqueKey);
218 localServer.reset(new QLocalServer(this));
219 localServer->setSocketOptions(QLocalServer::UserAccessOption);
220 connect(localServer.data(), &QLocalServer::newConnection, this,
221 &SingleApplication::receiveMessage);
222 if (!localServer->listen(_uniqueKey)) {
223 qCWarning(lcQtPass) << "SingleApplication: cannot listen on" << _uniqueKey
224 << ":" << localServer->errorString();
225 // Holding the lock without a server would make every later launch wait
226 // for a server that never comes: no instance could take over while this
227 // one lives. Release it and run without IPC.
228 localServer.reset();
229 _lock.unlock();
230 }
231}

233void SingleApplication::waitForPeer() {
234 for (int attempt = 0; attempt < takeoverProbes; ++attempt) {
235 if (peerIsListening()) {
236 _isRunning = true;
237 return;
238 }
239 QThread::msleep(takeoverProbeIntervalMs);
240 }
241 qCDebug(lcQtPass) << "Unable to create single instance.";
242}

244auto SingleApplication::forwardMessage(const QString &message) -> bool {
245 QLocalSocket localSocket(this);
246 localSocket.connectToServer(_uniqueKey, QIODevice::WriteOnly);
247 if (!localSocket.waitForConnected(timeout)) {
248 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
249 return false;
250 }
251 if (!serverRunsAsThisUser(localSocket)) {
252 qCWarning(lcQtPass) << "SingleApplication: the server on" << _uniqueKey
253 << "is not this user's; not forwarding.";
254 localSocket.abort();
255 return false;
256 }
257 QByteArray payload = message.toUtf8();
258 if (payload.isEmpty()) {
259 payload = QByteArray(1, '\0');
260 }
261 localSocket.write(payload);
262 if (!localSocket.waitForBytesWritten(timeout)) {
263 qCDebug(lcQtPass) << localSocket.errorString().toLatin1();
264 return false;
265 }
266 localSocket.disconnectFromServer();
267 return true;
268}

270auto SingleApplication::peerIsListening() -> bool {
271 QLocalSocket probe;
272 probe.connectToServer(_uniqueKey, QIODevice::WriteOnly);
273 if (!probe.waitForConnected(timeout)) {
274 return false;
275 }
276 if (!serverRunsAsThisUser(probe)) {
277 qCWarning(lcQtPass) << "SingleApplication: the server on" << _uniqueKey
278 << "is not this user's.";
279 probe.abort();
280 return false;
281 }
282 probe.disconnectFromServer();
283 return true;
284}
