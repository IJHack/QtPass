91void KeygenDialog::replace(const QString &key, const QString &value) {
92 // A pasted line break would put batch directives of its own into the
93 // template, and correcting the field later would leave them there:
94 // nothing such goes in. done() refuses the field itself.
95 if (!isBatchValue(value)) {
96 return;
97 }
98 QStringList clear;
99 QString expert = ui->plainTextEdit->toPlainText();
100 const QStringList lines =
101 expert.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
102 for (QString line : lines) {
103 line.replace(QRegularExpression(key + ":.*"), key + ": " + value);
104 clear.append(line);
105 }
106 ui->plainTextEdit->setPlainText(clear.join("\n"));
107}

119static bool isControlStatement(const QString &line, const QString &control) {
120 const QString trimmed = line.trimmed();
121 if (!trimmed.startsWith(control, Qt::CaseInsensitive))
122 return false;
123 return trimmed.size() == control.size() ||
124 trimmed.at(control.size()).isSpace();
125}

127// Keywords are matched as gpg matches them: an expert's "%No-Protection" left
128// next to the Passphrase: line makes gpg generate an unprotected key anyway.
129// The value is spliced in as a plain string, never through a regex.
130QString KeygenDialog::applyPassphrase(const QString &batch,
131 const QString &passphrase) {
132 static const QString noProtection = QStringLiteral("%no-protection");
133 static const QString commit = QStringLiteral("%commit");
134 static const QString passphraseKey = QStringLiteral("Passphrase:");
135 const QString protection =
136 passphrase.isEmpty() ? noProtection : passphraseKey + " " + passphrase;
137
138 QStringList clear;
139 bool spliced = false;
140 const QStringList lines =
141 batch.split(Util::newLinesRegex(), Qt::SkipEmptyParts);
142 for (QString line : lines) {
143 // gpg reads each line as a C string, so it ends at a NUL: what follows
144 // one would hide "%no-protection" from the check below and not from
145 // gpg. Cut it here as gpg does.
146 if (const qsizetype nul = line.indexOf(QChar::Null); nul >= 0) {
147 line.truncate(nul);
148 }
149 if (isControlStatement(line, noProtection) ||
150 line.trimmed().startsWith(passphraseKey, Qt::CaseInsensitive)) {
151 if (!spliced) {
152 clear.append(protection);
153 spliced = true;
154 }
155 continue;
156 }
157 clear.append(line);
158 }
159
160 if (!spliced && !passphrase.isEmpty()) {
161 // Parameters must precede %commit; append when the template has none.
162 const auto commitLine =
163 std::find_if(clear.cbegin(), clear.cend(), [](const QString &line) {
164 return isControlStatement(line, commit);
165 });
166 clear.insert(commitLine - clear.cbegin(), protection);
167 }
168 return clear.join("\n");
169}

176void KeygenDialog::done(int r) {
177 if (QDialog::Accepted == r) {
178 // The button is disabled without a passphrase choice, but done() can be
179 // reached otherwise (a default button, a script); the rule holds here.
180 // No dialog: the disabled OK already says what is missing.
181 if (!passphraseChosen()) {
182 updateOkState();
183 return;
184 }
185 // Pasted text with a line break would become batch directives of its
186 // own (replace() kept it out of the template); the email is matched
187 // whole by its pattern below.
188 if (!isBatchValue(ui->name->text()) ||
189 !isBatchValue(ui->passphrase1->text())) {
190 QMessageBox::critical(this, tr("Invalid input"),
191 tr("The name and the passphrase cannot contain "
192 "line breaks."));
193 return;
194 }
195 if (ui->name->text().length() < 5) {
196 QMessageBox::critical(this, tr("Invalid name"),
197 tr("Name must be at least 5 characters long."));
198 return;
199 }
200
201 static const QRegularExpression mailre(
202 QRegularExpression::anchoredPattern(
203 R"(\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b)"),
204 QRegularExpression::CaseInsensitiveOption);
205 if (!mailre.match(ui->email->text()).hasMatch()) {
206 QMessageBox::critical(
207 this, tr("Invalid email"),
208 tr("The email address you typed is not a valid email address."));
209 return;
210 }
211
212 ui->widget->setEnabled(false);
213 ui->buttonBox->setEnabled(false);
214 ui->checkBox->setEnabled(false);
215 ui->plainTextEdit->setEnabled(false);
216
217 if (!m_progressIndicator) {
218 m_progressIndicator = std::make_unique<QProgressIndicator>();
219 m_progressIndicator->setParent(this);
220 m_progressIndicator->setSizePolicy(QSizePolicy::Expanding,
221 QSizePolicy::Expanding);
222 this->layout()->addWidget(m_progressIndicator.get());
223 }
224 // A retry after generationFailed() reuses the indicator it stopped and hid.
225 m_progressIndicator->show();
226 m_progressIndicator->startAnimation();
227
228 ui->frame->hide();
229 ui->label->setText(
230 tr("This operation can take some minutes.<br />"
231 "We need to generate a lot of random bytes. It is a good idea to "
232 "perform some other action (type on the keyboard, move the mouse, "
233 "utilize the disks) during the prime generation; this gives the "
234 "random number generator a better chance to gain enough entropy."));
235
236 this->show();
237 startGeneration(applyPassphrase(ui->plainTextEdit->toPlainText(),
238 ui->passphrase1->text()));
239 } else {
240 QObject::disconnect(m_finished);
241 QObject::disconnect(m_failed);
242 QDialog::done(r);
243 return;
244 }
245}

247// Success accepts the dialog, so an exec() caller (the first-run wizard's
248// checkSecretKeys()) sees Accepted; failure gives the form back.
249void KeygenDialog::startGeneration(const QString &batch) {
250 if (m_pass == nullptr) {
251 generationFailed(tr("No password store backend available"));
252 return;
253 }
254 m_finished = connect(m_pass, &Pass::finishedGenerateGPGKeys, this,
255 [this](const QString &, const QString &) {
256 QObject::disconnect(m_finished);
257 QObject::disconnect(m_failed);
258 QDialog::done(QDialog::Accepted);
259 });
260 // Not processErrorExit: that fires for any command that fails while the
261 // key is being generated (a queued git push, say) and would take the
262 // dialog down while gpg is still working.
263 m_failed = connect(m_pass, &Pass::generateGPGKeysFailed, this,
264 [this](const QString &error) {
265 QObject::disconnect(m_finished);
266 QObject::disconnect(m_failed);
267 generationFailed(error);
268 });
269 m_pass->GenerateGPGKeys(batch);
270}

272void KeygenDialog::generationFailed(const QString &error) {
273 if (m_progressIndicator) {
274 m_progressIndicator->stopAnimation();
275 m_progressIndicator->hide();
276 }
277 ui->frame->show();
278 ui->widget->setEnabled(true);
279 ui->checkBox->setEnabled(true);
280 ui->plainTextEdit->setEnabled(true);
281 // OK comes back under the same rule as before the attempt, not
282 // unconditionally: the passphrase choice still has to stand.
283 updateOkState();
284 ui->label->setText(
285 tr("Key generation failed: %1").arg(error.toHtmlEscaped()));
286}
