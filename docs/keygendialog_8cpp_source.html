99static QStringList batchLines(QString batch) {
100 batch.replace(QLatin1Char('\r'), QLatin1Char('\n'));
101 return batch.split(QLatin1Char('\n'), Qt::SkipEmptyParts);

104void KeygenDialog::replace(const QString &key, const QString &value) {
105 // A pasted line break would put batch directives of its own into the
106 // template, and correcting the field later would leave them there:
107 // nothing such goes in. done() refuses the field itself.
108 if (!isBatchValue(value)) {
109 return;
110 }
111 QStringList clear;
112 QString expert = ui->plainTextEdit->toPlainText();
113 const QStringList lines = batchLines(expert);
114 for (QString line : lines) {
115 line.replace(QRegularExpression(key + ":.*"), key + ": " + value);
116 clear.append(line);
117 }
118 ui->plainTextEdit->setPlainText(clear.join("\n"));
119}

131static bool isControlStatement(const QString &line, const QString &control) {
132 const QString trimmed = line.trimmed();
133 if (!trimmed.startsWith(control, Qt::CaseInsensitive))
134 return false;
135 return trimmed.size() == control.size() ||
136 trimmed.at(control.size()).isSpace();
137}

139// Keywords are matched as gpg matches them: an expert's "%No-Protection" left
140// next to the Passphrase: line makes gpg generate an unprotected key anyway.
141// The value is spliced in as a plain string, never through a regex.
142QString KeygenDialog::applyPassphrase(const QString &batch,
143 const QString &passphrase) {
144 static const QString noProtection = QStringLiteral("%no-protection");
145 static const QString commit = QStringLiteral("%commit");
146 static const QString passphraseKey = QStringLiteral("Passphrase:");
147 const QString protection =
148 passphrase.isEmpty() ? noProtection : passphraseKey + " " + passphrase;
149
150 QStringList clear;
151 bool spliced = false;
152 const QStringList lines = batchLines(batch);
153 for (QString line : lines) {
154 // gpg reads each line as a C string, so it ends at a NUL: what follows
155 // one would hide "%no-protection" from the check below and not from
156 // gpg. Cut it here as gpg does.
157 if (const qsizetype nul = line.indexOf(QChar::Null); nul >= 0) {
158 line.truncate(nul);
159 }
160 if (isControlStatement(line, noProtection) ||
161 line.trimmed().startsWith(passphraseKey, Qt::CaseInsensitive)) {
162 if (!spliced) {
163 clear.append(protection);
164 spliced = true;
165 }
166 continue;
167 }
168 clear.append(line);
169 }
170
171 if (!spliced && !passphrase.isEmpty()) {
172 // Parameters must precede %commit; append when the template has none.
173 const auto commitLine =
174 std::find_if(clear.cbegin(), clear.cend(), [](const QString &line) {
175 return isControlStatement(line, commit);
176 });
177 clear.insert(commitLine - clear.cbegin(), protection);
178 }
179 return clear.join("\n");
180}

182auto KeygenDialog::isBatchValue(const QString &value) -> bool {
183 return !hasBatchBreak(value) && QStringView(value).isValidUtf16();
184}
185
186// The value goes into the template box and comes back through
187// toPlainText(), which hands the Unicode line and paragraph separators and
188// Qt's frame markers back as LF. A line edit shows the separators as a
189// space, so a pasted name looked like one line and still added its own
190// batch directives (%pubring <file> has gpg overwrite that file).
191auto KeygenDialog::hasBatchBreak(const QString &value) -> bool {
192 return std::any_of(value.cbegin(), value.cend(), [](QChar c) {
193 switch (c.unicode()) {
194 case u'\n':
195 case u'\r':
196 case u'\0':
197 case QChar::LineSeparator:
198 case QChar::ParagraphSeparator:
199 case 0xFDD0: // QTextBeginningOfFrame
200 case 0xFDD1: // QTextEndOfFrame
201 return true;
202 default:
203 return false;
204 }
205 });
206}
207
208// What gpg's batch reader takes off a value: C isspace() after the colon,
209// spaces and tabs at the end. A trailing \v or \f is refused too, though gpg
210// keeps it; nobody types one.
211auto KeygenDialog::isBatchPassphrase(const QString &passphrase) -> bool {
212 const auto blank = [](QChar c) {
213 return c == QLatin1Char(' ') ||
214 (c >= QLatin1Char('\t') && c <= QLatin1Char('\r'));
215 };
216 return passphrase.isEmpty() ||
217 (!blank(passphrase.front()) && !blank(passphrase.back()));
218}

220void KeygenDialog::done(int r) {
221 if (QDialog::Accepted == r) {
222 // The button is disabled without a passphrase choice, but done() can be
223 // reached otherwise (a default button, a script); the rule holds here.
224 // No dialog: the disabled OK already says what is missing.
225 if (!passphraseChosen()) {
226 updateOkState();
227 return;
228 }
229 // Pasted text with a line break would become batch directives of its
230 // own (replace() kept it out of the template); the email is matched
231 // whole by its pattern below.
232 if (hasBatchBreak(ui->name->text()) ||
233 hasBatchBreak(ui->passphrase1->text())) {
234 QMessageBox::critical(this, tr("Invalid input"),
235 tr("The name and the passphrase cannot contain "
236 "line breaks."));
237 return;
238 }
239 // Half of a character (one surrogate of an emoji, pasted in-process or
240 // from a UTF-16 clipboard) is no text gpg can take: the name would not
241 // be the one typed, and the passphrase would protect the key with
242 // another one.
243 if (!isBatchValue(ui->name->text()) ||
244 !isBatchValue(ui->passphrase1->text())) {
245 QMessageBox::critical(this, tr("Invalid input"),
246 tr("The name or the passphrase holds half of a "
247 "character, such as a broken emoji. Type it "
248 "again."));
249 return;
250 }
251 // gpg would protect the key without the blanks, and then refuse the
252 // passphrase as typed.
253 if (!isBatchPassphrase(ui->passphrase1->text())) {
254 QMessageBox::critical(
255 this, tr("Invalid input"),
256 tr("The passphrase cannot start or end with a space or a tab."));
257 return;
258 }
259 if (ui->name->text().length() < 5) {
260 QMessageBox::critical(this, tr("Invalid name"),
261 tr("Name must be at least 5 characters long."));
262 return;
263 }

265 static const QRegularExpression mailre(
266 QRegularExpression::anchoredPattern(
267 R"(\b[A-Z0-9._%+-]+@[A-Z0-9.-]+\.[A-Z]{2,}\b)"),
268 QRegularExpression::CaseInsensitiveOption);
269 if (!mailre.match(ui->email->text()).hasMatch()) {
270 QMessageBox::critical(
271 this, tr("Invalid email"),
272 tr("The email address you typed is not a valid email address."));
273 return;
274 }
275
276 ui->widget->setEnabled(false);
277 ui->buttonBox->setEnabled(false);
278 ui->checkBox->setEnabled(false);
279 ui->plainTextEdit->setEnabled(false);
280
281 if (!m_progressIndicator) {
282 m_progressIndicator = std::make_unique<QProgressIndicator>();
283 m_progressIndicator->setParent(this);
284 m_progressIndicator->setSizePolicy(QSizePolicy::Expanding,
285 QSizePolicy::Expanding);
286 this->layout()->addWidget(m_progressIndicator.get());
287 }
288 // A retry after generationFailed() reuses the indicator it stopped and hid.
289 m_progressIndicator->show();
290 m_progressIndicator->startAnimation();
291
292 ui->frame->hide();
293 ui->label->setText(
294 tr("This operation can take some minutes.<br />"
295 "We need to generate a lot of random bytes. It is a good idea to "
296 "perform some other action (type on the keyboard, move the mouse, "
297 "utilize the disks) during the prime generation; this gives the "
298 "random number generator a better chance to gain enough entropy."));
299
300 this->show();
301 startGeneration(applyPassphrase(ui->plainTextEdit->toPlainText(),
302 ui->passphrase1->text()));
303 } else {
304 QObject::disconnect(m_finished);
305 QObject::disconnect(m_failed);
306 QDialog::done(r);
307 return;
308 }
309}

311// Success accepts the dialog, so an exec() caller (the first-run wizard's
312// checkSecretKeys()) sees Accepted; failure gives the form back.
313void KeygenDialog::startGeneration(const QString &batch) {
314 if (m_pass == nullptr) {
315 generationFailed(tr("No password store backend available"));
316 return;
317 }
318 m_finished = connect(m_pass, &Pass::finishedGenerateGPGKeys, this,
319 [this](const QString &, const QString &) {
320 QObject::disconnect(m_finished);
321 QObject::disconnect(m_failed);
322 QDialog::done(QDialog::Accepted);
323 });
324 // Not processErrorExit: that fires for any command that fails while the
325 // key is being generated (a queued git push, say) and would take the
326 // dialog down while gpg is still working.
327 m_failed = connect(m_pass, &Pass::generateGPGKeysFailed, this,
328 [this](const QString &error) {
329 QObject::disconnect(m_finished);
330 QObject::disconnect(m_failed);
331 generationFailed(error);
332 });
333 m_pass->GenerateGPGKeys(batch);
334}

336void KeygenDialog::generationFailed(const QString &error) {
337 if (m_progressIndicator) {
338 m_progressIndicator->stopAnimation();
339 m_progressIndicator->hide();
340 }
341 ui->frame->show();
342 ui->widget->setEnabled(true);
343 ui->checkBox->setEnabled(true);
344 ui->plainTextEdit->setEnabled(true);
345 // OK comes back under the same rule as before the attempt, not
346 // unconditionally: the passphrase choice still has to stand.
347 updateOkState();
348 ui->label->setText(
349 tr("Key generation failed: %1").arg(error.toHtmlEscaped()));
350}
