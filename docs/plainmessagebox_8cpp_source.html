28auto slice(const QString &text, qsizetype chars, bool fromEnd) -> QString {
29 if (chars >= text.size()) {
30 return text;
31 }
32 if (fromEnd) {
33 const qsizetype from = text.size() - chars;
34 return text.right(text.at(from).isLowSurrogate() ? chars - 1 : chars);
35 }
36 return text.left(text.at(chars - 1).isHighSurrogate() ? chars - 1 : chars);
37}

48auto part(const QString &text, qsizetype lines, qsizetype chars, bool fromEnd)
49 -> QString {
50 QString piece = slice(text, chars, fromEnd);
51 qsizetype seen = 0;
52 if (fromEnd) {
53 for (qsizetype i = piece.size() - 1; i >= 0; --i) {
54 if (piece.at(i) == QLatin1Char('\n') && ++seen == lines) {
55 return piece.mid(i + 1);
56 }
57 }
58 } else {
59 for (qsizetype i = 0; i < piece.size(); ++i) {
60 if (piece.at(i) == QLatin1Char('\n') && ++seen == lines) {
61 return piece.left(i);
62 }
63 }
64 }
65 return piece;
66}

81auto bounded(const QString &text, bool *cut) -> QString {
82 *cut = text.size() > kMaxChars || text.count(QLatin1Char('\n')) >= kMaxLines;
83 if (!*cut) {
84 return text;
85 }
86 return part(text, kMaxLines / 2, kMaxChars / 2, false) +
87 QStringLiteral("\n\u2026\n") +
88 part(text, kMaxLines / 2, kMaxChars / 2, true);
89}

92auto PlainMessageBox::show(QWidget *parent, QMessageBox::Icon icon,
93 const QString &title, const QString &text,
94 QMessageBox::StandardButtons buttons)
95 -> QMessageBox::StandardButton {
96 QMessageBox box(parent);
97 box.setIcon(icon);
98 // The text quotes names from the store: a bidi override in a folder name
99 // mirrored the rest of a warning, a C1 control cut it short.
100 box.setWindowTitle(Util::displayable(title, false));
101 // Before the text: set afterwards, the text would already have been laid
102 // out as rich text once.
103 box.setTextFormat(Qt::PlainText);
104 const QString shown = Util::displayable(text);
105 bool cut = false;
106 box.setText(bounded(shown, &cut));
107 if (cut) {
108 // All of it under "Show Details...": plain text in a scrolling field,
109 // which does not lay it out at once as the label did. Not all tool
110 // output reaches the Process Output panel (the set-up runs git while
111 // the window waits), so this is where the cause can be read. Past
112 // 256 KB its start and end, for the same reason as the box.
113 box.setDetailedText(shown.size() <= kMaxDetailChars
114 ? shown
115 : slice(shown, kMaxDetailChars / 2, false) +
116 QStringLiteral("\n\u2026\n") +
117 slice(shown, kMaxDetailChars / 2, true));
118 }
119 box.setStandardButtons(buttons);
120 return static_cast<QMessageBox::StandardButton>(box.exec());
121}
