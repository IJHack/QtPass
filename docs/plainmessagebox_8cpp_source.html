10auto PlainMessageBox::show(QWidget *parent, QMessageBox::Icon icon,
11 const QString &title, const QString &text,
12 QMessageBox::StandardButtons buttons)
13 -> QMessageBox::StandardButton {
14 QMessageBox box(parent);
15 box.setIcon(icon);
16 box.setWindowTitle(title);
17 // Before the text: set afterwards, the text would already have been laid
18 // out as rich text once.
19 box.setTextFormat(Qt::PlainText);
20 box.setText(text);
21 box.setStandardButtons(buttons);
22 return static_cast<QMessageBox::StandardButton>(box.exec());
23}
