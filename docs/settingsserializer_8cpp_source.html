15auto SettingsSerializer::loadPasswordConfiguration(QSettings &qs)
16 -> PasswordConfiguration {
17 PasswordConfiguration config;
18
19 int length = qs.value(SettingsConstants::passwordLength, 16).toInt();
20 if (length <= 0) {
21 length = 16;
22 }
23 config.length = length;
24
25 // The ini value is untrusted: anything outside the enum would be used as an
26 // index into PasswordConfiguration::Characters, so clamp it here, once, and
27 // every consumer sees a valid selection.
28 const int selected =
29 qs.value(SettingsConstants::passwordCharsSelection, 0).toInt();
30 if (selected < 0 || selected >= PasswordConfiguration::CHARSETS_COUNT) {
31 qWarning() << "Ignoring out-of-range passwordCharsSelection" << selected;
32 config.selected = PasswordConfiguration::ALLCHARS;
33 } else {
34 config.selected =
35 static_cast<PasswordConfiguration::CharacterSet>(selected);
36 }
37
38 config.Characters[PasswordConfiguration::CUSTOM] =
39 qs.value(SettingsConstants::passwordChars, QString()).toString();
40
41 return config;
42}

44auto SettingsSerializer::load(QSettings &qs) -> AppSettings {
45 AppSettings s;
46
47 // Window / session
48 s.version = qs.value(SettingsConstants::version).toString();
49 s.geometry = qs.value(SettingsConstants::geometry).toByteArray();
50 s.savestate = qs.value(SettingsConstants::savestate).toByteArray();
51 s.activeProfile = qs.value(SettingsConstants::profile).toString();
52
53 // Backend selection / store
54 s.usePass = qs.value(SettingsConstants::usePass, false).toBool();
55 s.passStore = qs.value(SettingsConstants::passStore).toString();
56 s.passSigningKey = qs.value(SettingsConstants::passSigningKey).toString();
57
58 // Executables
59 s.passExecutable = qs.value(SettingsConstants::passExecutable).toString();
60 s.gitExecutable = qs.value(SettingsConstants::gitExecutable).toString();
61 s.gpgExecutable = qs.value(SettingsConstants::gpgExecutable).toString();
62 s.pwgenExecutable = qs.value(SettingsConstants::pwgenExecutable).toString();
63 s.qrencodeExecutable =
64 qs.value(SettingsConstants::qrencodeExecutable).toString();
65 s.gpgHome = qs.value(SettingsConstants::gpgHome).toString();
66 s.sshAuthSockOverride =
67 qs.value(SettingsConstants::sshAuthSockOverride).toString();
68
69 // Clipboard / autoclear
70 s.clipBoardType = static_cast<Enums::ClipBoardType>(
71 qs.value(SettingsConstants::clipBoardType,
72 static_cast<int>(Enums::CLIPBOARD_NEVER))
73 .toInt());
74 s.useSelection = qs.value(SettingsConstants::useSelection, false).toBool();
75 s.useAutoclear = qs.value(SettingsConstants::useAutoclear, false).toBool();
76 s.restoreClipboard =
77 qs.value(SettingsConstants::restoreClipboard, true).toBool();
78 s.autoclearSeconds = qs.value(SettingsConstants::autoclearSeconds, 0).toInt();
79 s.useAutoclearPanel =
80 qs.value(SettingsConstants::useAutoclearPanel, false).toBool();
81 s.autoclearPanelSeconds =
82 qs.value(SettingsConstants::autoclearPanelSeconds, 0).toInt();
83
84 // Content display
85 s.hidePassword = qs.value(SettingsConstants::hidePassword, false).toBool();
86 s.hideSecretFields =
87 qs.value(SettingsConstants::hideSecretFields, true).toBool();
88 s.hideAllFields = qs.value(SettingsConstants::hideAllFields, false).toBool();
89 // Never saved (a new settings file, or one from 1.8): the default set.
90 // Saved, it is what the user left, an empty list included.
91 if (qs.contains(SettingsConstants::secretFieldNames)) {
92 s.secretFieldNames.clear();
93 for (const QString &name :
94 qs.value(SettingsConstants::secretFieldNames).toStringList()) {
95 if (!name.trimmed().isEmpty()) {
96 s.secretFieldNames << name.trimmed();
97 }
98 }
99 } else {
100 s.secretFieldNames = AppSettings::defaultSecretFieldNames();
101 }
102 s.hideContent = qs.value(SettingsConstants::hideContent, false).toBool();
103 s.useMonospace = qs.value(SettingsConstants::useMonospace, false).toBool();
104 s.displayAsIs = qs.value(SettingsConstants::displayAsIs, false).toBool();
105 s.noLineWrapping =
106 qs.value(SettingsConstants::noLineWrapping, false).toBool();
107
108 // Features
109 // Default true: AppSettings::addGPGId defaults to true.
110 s.addGPGId = qs.value(SettingsConstants::addGPGId, true).toBool();
111 s.useGit = qs.value(SettingsConstants::useGit, false).toBool();
112 s.useGrepSearch = qs.value(SettingsConstants::useGrepSearch, false).toBool();
113 // Defaults to true: OTP no longer depends on an external extension, and a
114 // stored value from an older profile still wins over this default.
115 s.useOtp = qs.value(SettingsConstants::useOtp, true).toBool();
116 s.useQrencode = qs.value(SettingsConstants::useQrencode, false).toBool();
117 s.usePwgen = qs.value(SettingsConstants::usePwgen, false).toBool();
118 s.autoPull = qs.value(SettingsConstants::autoPull, false).toBool();
119 s.autoPush = qs.value(SettingsConstants::autoPush, false).toBool();
120 s.gitUseParentRepository =
121 qs.value(SettingsConstants::gitUseParentRepository, false).toBool();
122 s.showProcessOutput =
123 qs.value(SettingsConstants::showProcessOutput, false).toBool();
124 s.showMenuBar = qs.value(SettingsConstants::showMenuBar, false).toBool();
125
126 // Templates
127 s.passTemplate = qs.value(SettingsConstants::passTemplate).toString();
128 s.useTemplate = qs.value(SettingsConstants::useTemplate, false).toBool();
129 s.templateAllFields =
130 qs.value(SettingsConstants::templateAllFields, false).toBool();
131
132 // Password generation
133 s.passwordConfiguration = loadPasswordConfiguration(qs);
134 s.avoidCapitals = qs.value(SettingsConstants::avoidCapitals, false).toBool();
135 s.avoidNumbers = qs.value(SettingsConstants::avoidNumbers, false).toBool();
136 s.lessRandom = qs.value(SettingsConstants::lessRandom, false).toBool();
137 s.useSymbols = qs.value(SettingsConstants::useSymbols, false).toBool();
138
139 // System / tray
140 s.useTrayIcon = qs.value(SettingsConstants::useTrayIcon, false).toBool();
141 s.hideOnClose = qs.value(SettingsConstants::hideOnClose, false).toBool();
142 s.startMinimized =
143 qs.value(SettingsConstants::startMinimized, false).toBool();
144 s.alwaysOnTop = qs.value(SettingsConstants::alwaysOnTop, false).toBool();
145
146 return s;
147}
