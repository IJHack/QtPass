22auto PassBackendFactory::getPass() -> Pass * {
23 if (!pass) {
24 AppSettings s = QtPassSettings::load();
25 // Ensure store directory exists (first-run side effect — load() normalises
26 // the path but does not create the directory).
27 if (!QDir(s.passStore).exists())
28 Util::mkpathForStore(s.passStore);
29 if (s.usePass) {
30 pass = getRealPass();
31 } else {
32 pass = getImitatePass();
33 }
34 if (pass) {
35 pass->init(s);
36 }
37 }
38 return pass;
39}

41auto PassBackendFactory::getRealPass() -> RealPass * {
42 if (realPass.isNull()) {
43 realPass.reset(new RealPass());
44 }
45 return realPass.data();

48auto PassBackendFactory::getImitatePass() -> ImitatePass * {
49 if (imitatePass.isNull()) {
50 imitatePass.reset(new ImitatePass());
51 AppSettings s = QtPassSettings::load();
52 imitatePass->init(s);
53 }
54 return imitatePass.data();
55}

57auto PassBackendFactory::isIdle() -> bool {
58 return (realPass.isNull() || realPass->isIdle()) &&
59 (imitatePass.isNull() || imitatePass->isIdle());
60}
