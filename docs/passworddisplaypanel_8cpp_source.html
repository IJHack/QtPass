43void PasswordDisplayPanel::clear() {
44 while (m_grid->count() > 0) {
45 QLayoutItem *item = m_grid->takeAt(0);
46 if (QWidget *widget = item->widget()) {
47 // A drag out of a field value runs QDrag::exec() inside the value
48 // widget's own mouse handler, and clear() can be reached from that
49 // nested event loop (a drop on the search box, the panel autoclear
50 // timer, a second "qtpass NAME"). Deleting the row then - even with
51 // deleteLater(), which the drag's loop would deliver - frees it under
52 // its own handler and crashes. A row that has a QDrag among its children
53 // is hidden now and deleted once that QDrag is gone, which Qt does after
54 // the drag ends; rows without a drag are deleted at once as before.
55 const QList<QDrag *> drags = widget->findChildren<QDrag *>();
56 if (drags.isEmpty()) {
57 delete widget;
58 } else {
59 widget->hide();
60 connect(drags.last(), &QObject::destroyed, widget,
61 [widget]() { widget->deleteLater(); });
62 }
63 }
64 delete item;
65 }
66 m_container->setSpacing(0);
67}

83void PasswordDisplayPanel::displayFields(const QString &password,
84 const NamedValues &namedValues,
85 const AppSettings &s,
86 const QString &otpConfig) {
87 // Rows are numbered as they are added rather than assuming the password
88 // occupies row 0: an entry written by `pass otp insert` has no password row,
89 // and starting at 1 regardless left an empty grid row above the OTP row.
90 int position = 0;
91 // Defence in depth: getPasswordForDisplay() already blanks a password line
92 // that is an otpauth URI (`pass otp insert`); refuse one here too.
93 if (!password.isEmpty() && !FileContent::isOtpUriValue(password)) {
94 // The password is hidden in addField when needed.
95 addField(position, QObject::tr("Password"), password, s);
96 ++position;
97 }
98 bool otpRendered = false;
99 for (const NamedValue &nv : namedValues) {
100 // Keyed on the value too: a field called anything whose value is an otpauth
101 // URI is still a shared secret, and isLineHidden() hides it from the text
102 // browser, so suppressing only OTP/TOTP names leaked `2fa:` and friends.
103 if (FileContent::isOtpFieldName(nv.name) ||
104 FileContent::isOtpUriValue(nv.value)) {
105 // Never render an OTP field verbatim: addField() would display the
106 // shared secret and hand it to a copy button. Unconditional, so it stays
107 // hidden when OTP support is off and otpConfig is empty.
108 if (!otpRendered && !otpConfig.isEmpty()) {
109 addOtpField(position, otpConfig, s);
110 ++position;
111 otpRendered = true;
112 }
113 continue;
114 }
115 addField(position, nv.name, nv.value, s);
116 ++position;
117 }
118 // The configuration came from the entry body rather than from a field.
119 if (!otpConfig.isEmpty() && !otpRendered) {
120 addOtpField(position, otpConfig, s);
121 }
122 m_container->setSpacing(m_grid->count() == 0 ? 0 : 6);
123}

141auto valueBrowser(const QString &field, const QString &value,
142 const QString &lineStyle) -> SecretTextBrowser * {
143 auto *browser = new SecretTextBrowser();
144 browser->setOpenExternalLinks(true);
145 browser->setOpenLinks(true);
146 browser->setMaximumHeight(kFieldHeight);
147 browser->setMinimumHeight(kFieldHeight);
148 browser->setSizePolicy(
149 QSizePolicy(QSizePolicy::Expanding, QSizePolicy::Minimum));
150 browser->setObjectName(field);
151 bool linked = false;
152 const QString linkedText = Util::linkifyUrls(value, &linked);
153 if (linked) {
154 // Always HTML here: the text is escaped and carries anchor tags, so do
155 // not leave the interpretation to setText()'s auto-detection.
156 browser->setHtml(linkedText);
157 } else {
158 // Plain text: escaping plus setText() showed `a&b` as `a&amp;b` unless
159 // the value happened to contain a '<'.
160 browser->setPlainText(value);
161 }
162 browser->setReadOnly(true);
163 browser->setStyleSheet(lineStyle);
164 browser->setContentsMargins(0, 0, 0, 0);
165 return browser;
166}

169auto urlButton(const QString &url, QWidget *parent) -> QPushButton * {
170 auto *button = new QPushButton(parent);
171 button->setIcon(QIcon::fromTheme(QStringLiteral("applications-internet"),
172 QIcon(":/icons/open-url.svg")));
173 // Escape for the tooltip only; the launched URL stays the validated value.
174 // <qt> forces rich text: auto-detection would show `&amp;` literally, and
175 // plain text misreads a URL containing `&lt;`. Rich text wraps, breaking
176 // the URL at `/` and `?`; white-space:nowrap keeps it on one line (<nobr>
177 // is not enough: Qt only makes its spaces non-breaking).
178 button->setToolTip(
179 QStringLiteral("<qt style=\"white-space:nowrap\">%1</qt>")
180 .arg(QObject::tr("Open %1 in browser").arg(url.toHtmlEscaped())));
181 button->setStyleSheet(QString::fromLatin1(kButtonStyle));
182 button->setCursor(Qt::PointingHandCursor);
183 QObject::connect(button, &QPushButton::clicked, button, [url]() {
184 // Re-validate: never hand an unvalidated string to the OS URL handler.
185 if (Util::isLaunchableWebUrl(url)) {
186 QDesktopServices::openUrl(QUrl(url));
187 }
188 });
189 return button;
190}

202 isSecret(field, s));

205auto PasswordDisplayPanel::isSecret(const QString &field,
206 const AppSettings &s) const -> bool {
207 if (!s.hidePassword) {
208 return false;
209 }
210 const bool marked =
211 std::any_of(m_markedSecrets.cbegin(), m_markedSecrets.cend(),
212 [&field](const QString &name) {
213 return field.compare(name, Qt::CaseInsensitive) == 0;
214 });
215 return marked || (s.hideSecretFields &&
216 FileContent::isSecretFieldName(field, s.secretFieldNames));
217}

221 const QString &value,

224 if (s.clipBoardType != Enums::CLIPBOARD_NEVER) {

239 // must not reach a tooltip or the browser. A field shown in plain view

242 Util::isLaunchableWebUrl(value)) {
243 layout->addWidget(urlButton(value, m_widgetParent));

270void PasswordDisplayPanel::addField(int position, const QString &field,
271 const QString &value,
272 const AppSettings &s) {
273 const QString trimmedField = field.trimmed();
274 // The password row keeps its value exactly: a leading or trailing space is
275 // part of the password, so the copy button, QR code and revealed text must
276 // return what is stored, as Ctrl+C on the tree and the edit dialog do. Only
277 // a trailing CR from a CRLF-stored entry is dropped — it is a line ending,
278 // not part of the secret. Every other row is `name: value`, where the
279 // whitespace around the value is layout and is trimmed.
280 QString displayValue;
281 if (trimmedField == QObject::tr("Password")) {
282 displayValue = value;
283 if (displayValue.endsWith(QLatin1Char('\r'))) {
284 displayValue.chop(1);
285 }
286 } else {
287 displayValue = value.trimmed();
288 }
289 QFrame *frame = createFieldFrame();
290 addActionButtons(frame->layout(), trimmedField, displayValue, s);
291 const QString lineStyle =
292 s.useMonospace
293 ? "QLineEdit, QTextBrowser { border-style: none; background: "
294 "transparent; font-family: monospace; }"
295 : "QLineEdit, QTextBrowser { border-style: none; background: "
296 "transparent; }";
297 // The password, and with it the other secrets an entry holds (a PIN, a
298 // CVV, an API key, what a template marks): behind the eye button too.
299 if (hides(trimmedField, s)) {
300 addHiddenPassword(frame->layout(), trimmedField, displayValue, lineStyle);
301 } else {
302 SecretTextBrowser *browser =
303 valueBrowser(trimmedField, displayValue, lineStyle);
304 connect(browser, &SecretTextBrowser::copied, this,
305 &PasswordDisplayPanel::textCopied);
306 frame->layout()->addWidget(browser);
307 }
308 // Plain text before the name is set: a field name comes from the entry,
309 // and as rich text <img src="/dev/zero"> has Qt read that file without end.
310 auto *label = new QLabel();
311 label->setTextFormat(Qt::PlainText);
312 label->setText(trimmedField);
313 addRow(position, label, frame);
314}

367// One label plus one frame, like every row, so grid rows match
368// displayFields()' position counter. hidePassword does not apply (the code
369// expires in seconds, next to a visible countdown), and there is no QR
370// button: a QR of the configuration would show the shared secret.
371void PasswordDisplayPanel::addOtpField(int position, const QString &otpConfig,
372 const AppSettings &s) {
373 const std::optional<Totp::Settings> settings = Totp::parse(otpConfig);
374 if (!settings.has_value()) {
375 // Report the problem without echoing what the user stored: the value is
376 // still a would-be secret, so it gets neither a copy nor a QR button.
377 AppSettings inert = s;
378 inert.clipBoardType = Enums::CLIPBOARD_NEVER;
379 inert.useQrencode = false;
380 inert.hidePassword = false; // a message, never behind the eye button
381 addField(position, QObject::tr("OTP code"),
382 QObject::tr("No OTP code found in this password entry"), inert);
383 return;
384 }
385
386 auto *frame = createFieldFrame();
387 auto *otpWidget =
388 new OtpCodeWidget(*settings, s.clipBoardType != Enums::CLIPBOARD_NEVER,
389 s.useMonospace, frame);
390 connect(otpWidget, &OtpCodeWidget::copyRequested, this,
391 &PasswordDisplayPanel::copyRequested);
392 frame->layout()->addWidget(otpWidget);
393
394 addRow(position, new QLabel(QObject::tr("OTP code")), frame);
395}
