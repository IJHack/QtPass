64// The helpers are shared by the owning thread and the worker; thread affinity
65// tells them apart. Only the worker touches its QProcess, so a cancel can
66// never hit an exited process or a reused pid.
67auto ImitatePass::execBlocking(const QString &app, const QStringList &args,
68 const QString &input, QString *process_out,
69 QString *process_err) -> int {
70 // In the store's environment: the git ceiling QtPass's own does not
71 // carry.
72 QProcess process;
73 if (QThread::currentThread() == thread()) {
74 process.setProcessEnvironment(storeEnvironment());
75 return Executor::executeBlocking(process, app, args, input, process_out,
76 process_err);
77 }
78 process.setProcessEnvironment(withCurrentGitPin(m_reencryptEnv));
79 return Executor::executeBlocking(process, app, args, input, process_out,
80 process_err, &m_reencryptCancel);
81}

201void ImitatePass::Insert(QString file, QString newValue, bool overwrite) {
202 // The dialog names a new entry relative to the store; gpg used to resolve
203 // that in its working directory. Everything below works on the one path.
204 file =
205 QDir::cleanPath(QDir::isAbsolutePath(file)
206 ? file + ".gpg"
207 : QDir(m_settings.passStore).filePath(file) + ".gpg");
208 if (refuseLinkedPath(file)) {
209 return;
210 }
211 if (refuseUnfinishedGitOperation()) {
212 return;
213 }
214 // gpg used to refuse an existing output without --yes; it writes
215 // elsewhere now, so the check is ours here, and again at the rename.
216 const QFileInfo target(file);
217 if (!overwrite && (target.exists() || target.isSymLink())) {
218 emit critical(tr("Cannot add"), tr("%1 already exists.").arg(file));
219 return;
220 }
221 QStringList recipients;
222 if (!recipientsForEntry(file, &recipients)) {
223 return;
224 }
225 // gpg never opens a store path for output: a co-writer could make it a link
226 // after refuseLinkedPath(). gpg writes into a 0700 directory of our own and
227 // placeEncryptedFile() brings the bytes in via an open handle and a rename.
228 auto scratch = std::make_shared<QTemporaryDir>();
229 if (!scratch->isValid()) {
230 emit critical(tr("Cannot write"),
231 tr("Cannot create a temporary directory: %1")
232 .arg(scratch->errorString()));
233 return;
234 }
235 const QString output = scratch->filePath(QStringLiteral("entry.gpg"));
236 TransactionHelper trans(&m_transaction, PASS_INSERT);
237 const QStringList args =
238 encryptArgs({"--batch", "--status-fd", "2"}, pgpg(output), recipients);
239 m_pendingInserts.enqueue({std::move(scratch), output, file, overwrite});
240 // Past every refusal: this insert is now queued and will report once.
241 queueInsert();
242 executeGpg(PASS_INSERT, args, newValue);
243 if (!gitReady()) {
244 return;
245 }
246 if (!overwrite) {
247 executeGit(GIT_ADD, {"add", "--", pgit(file)});
248 }
249 QString path = QDir(m_settings.passStore).relativeFilePath(file);
250 path.replace(Util::endsWithGpg(), "");
251 gitCommit(file, QString(overwrite ? "Edit" : "Add") + " for " + path +
252 " using QtPass.");
253}

277void ImitatePass::Remove(QString file, bool isDir) {
278 // No trailing separator: "link/" makes git look for what is behind the
279 // link ("pathspec did not match") and rm follow it.
280 file = QDir::cleanPath(m_settings.passStore + file);
281 if (!isDir) {
282 file += ".gpg";
283 }
284 // A link itself may go; anything behind one is not the store's to delete.
285 if (refuseLinkedPath(file, LinkScope::ParentsOnly)) {
286 return;
287 }
288 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
289 return;
290 }
291 TransactionHelper trans(&m_transaction, PASS_REMOVE);
292 QString path = QDir(m_settings.passStore).relativeFilePath(file);
293 path.replace(Util::endsWithGpg(), "");
294 if (Util::isLinkedFolder(file)) {
295 // Unlink it here, never through a recursive rm: the link goes, what it
296 // points to stays. Git then only has to forget it, if it knew it.
297 if (!Util::removeTree(file)) {
298 emit critical(tr("Delete failed"),
299 tr("Could not remove the link %1.").arg(file));
300 return;
301 }
302 // Only when git knew it: a commit matching nothing (a link never
303 // committed) exits 1 and reads as a failed removal. ls-files reads the
304 // index, so it still answers now that the link is gone.
305 if (gitReady() && gitTracks(file)) {
306 executeGit(GIT_RM, {"rm", "-q", "--cached", "--", pgit(file)});
307 gitCommit(file, "Remove for " + path + " using QtPass.");
308 }
309 return;
310 }
311 if (gitReady()) {
312 executeGit(GIT_RM, {"rm", (isDir ? "-rf" : "-f"), "--", pgit(file)});
313 gitCommit(file, "Remove for " + path + " using QtPass.");
314 } else {
315 if (isDir) {
316 // Never QDir::removeRecursively(): it follows a junction inside the
317 // folder and empties whatever that points to.
318 Util::removeTree(file);
319 } else {
320 QFile(file).remove();
321 }
322 }

334auto ImitatePass::addGenerationHeader(const QString &gpgIdFile,
335 const GpgIdSigner &signer,
336 QByteArray *contents) -> bool {
337 const std::optional<QString> folder =
338 GpgIdGeneration::folderOf(gpgIdFile, m_settings.passStore);
339 if (!folder) {
340 emit critical(tr("Cannot update"),
341 tr("%1 is not inside the password store, or a folder "
342 "name on the way holds a line break or another control "
343 "character, which a signed list cannot record.")
344 .arg(gpgIdFile));
345 return false;
346 }
347 // The list on disk counts only if it is this folder's own, verified list:
348 // a signature vouches for the bytes, the folder line for the place.
349 std::optional<qint64> verifiedOnDisk;
350 std::optional<QString> storeOnDisk;
351 QByteArray current;
352 if (signer.verifyFile(gpgIdFile, &current)) {
353 const auto header = GpgIdGeneration::parse(current);
354 if (header && (!header->folder || *header->folder == *folder)) {
355 verifiedOnDisk = header->generation;
356 storeOnDisk = header->store;
357 }
358 }
359 QString why;
360 // The store's id: the one this device knows, else the one this store's
361 // lists already carry (#2050).
362 const std::optional<QString> store =
363 GpgIdGeneration::storeIdFor(m_settings.passStore, storeOnDisk, &why);
364 if (!store) {
365 emit critical(tr("Cannot update"), why);
366 return false;
367 }
368 const std::optional<qint64> generation =
369 GpgIdGeneration::reserveNext(gpgIdFile, verifiedOnDisk, &why);
370 if (!generation) {
371 emit critical(tr("Cannot update"), why);
372 return false;
373 }
374 *contents =
375 GpgIdGeneration::withHeader(*generation, *folder, *contents, *store);
376 return true;
377}

379auto ImitatePass::writeGpgIdFile(const QString &gpgIdFile,
380 const QList<UserInfo> &users,
381 QByteArray *written) -> bool {
382 QByteArray contents;
383 bool secret_selected = false;
384 for (const UserInfo &user : users) {
385 if (user.enabled) {
386 contents += (user.key_id + "\n").toUtf8();
387 secret_selected |= user.have_secret;
388 }
389 }
390 // Without a signing key the plain list stays (GpgIdGeneration).
391 const GpgIdSigner signer = gpgIdSigner();
392 if (signer.enabled() && !addGenerationHeader(gpgIdFile, signer, &contents)) {
393 return false;
394 }
395 // Whole or not at all, owner-only (it names the store's keys); a link
396 // planted since Init's check is replaced, not written through.
397 QString writeError;
398 if (!Util::writeFileReplacing(gpgIdFile, contents, true, &writeError)) {
399 emit critical(tr("Cannot update"), writeError);
400 return false;
401 }
402 if (written != nullptr) {
403 *written = contents;
404 }
405 if (signer.enabled()) {
406 // From now on these bytes are this generation. The list is signed either
407 // way (unsigned is refused everywhere); if the record could not take them
408 // (another writer, record busy), the user is told to save once more.
409 QString why;
410 if (!GpgIdGeneration::recordWritten(gpgIdFile, contents, &why)) {
411 qCWarning(lcQtPass) << "Could not record the written .gpg-id:" << why;
412 emit critical(
413 tr("Recipient list written, but not recorded"),
414 tr("%1 Save the recipients once more to get through.").arg(why));
415 }
416 }
417 if (!secret_selected) {
418 emit critical(
419 tr("Check selected users!"),
420 tr("None of the selected keys have a secret key available.\n"
421 "You will not be able to decrypt any newly added passwords!"));
422 }
423 return true;
424}

426auto ImitatePass::signGpgIdFile(const QString &gpgIdFile,
427 const QByteArray &contents) -> bool {
428 const GpgIdSigner signer = gpgIdSigner();
429 QString why;
430 if (!signer.sign(gpgIdFile, contents, &why)) {
431 emit critical(
432 tr("GPG signing failed!"),
433 why.trimmed().isEmpty()
434 ? tr("Failed to sign %1.").arg(gpgIdFile)
435 : tr("Failed to sign %1: %2").arg(gpgIdFile, why.trimmed()));
436 return false;
437 }
438 QByteArray signedBytes;
439 if (!signer.verifyFile(gpgIdFile, &signedBytes)) {
440 emit critical(tr("Check .gpg-id file signature!"),
441 tr("Signature for %1 is invalid.").arg(gpgIdFile));
442 return false;
443 }
444 return true;
445}

447// Synchronous on purpose: queued on the async executor, the add/commit raced
448// reencryptPath()'s blocking backup commit for index.lock and either left the
449// .gpg-id untracked or found nothing to commit.
450auto ImitatePass::gitAddGpgId(const QString &gpgIdFile,
451 const QString &gpgIdSigFile, QString *out,
452 QString *err) -> int {
453 const QString git = m_settings.gitExecutable;
454 const QString store = pgit(m_settings.passStore);
455 auto run = [&](const QStringList &args) -> int {
456 QString runOut;
457 QString runErr;
458 const int rc =
459 execBlocking(git, QStringList{"-C", store} + args, &runOut, &runErr);
460 if (out != nullptr) {
461 out->append(runOut);
462 }
463 if (err != nullptr) {
464 err->append(runErr);
465 }
466 return rc;
467 };
468 QStringList paths{pgit(gpgIdFile)};
469 if (!gpgIdSigFile.isEmpty()) {
470 paths << pgit(gpgIdSigFile);
471 }
472 // add stages new and modified files alike; already-clean ones are a no-op.
473 int rc = run(QStringList{"add", "--"} + paths);
474 if (rc != 0) {
475 return rc;
476 }
477 // Re-initialising with the same recipients changes nothing; that is not a
478 // failure, and `git commit` would make it one. diff --quiet: 0 nothing
479 // staged, 1 something staged, anything else is git failing.
480 rc = run(QStringList{"diff", "--cached", "--quiet", "--"} + paths);
481 if (rc == 0) {
482 return 0;
483 }
484 if (rc != 1) {
485 return rc;
486 }
487 QString commitPath = gpgIdFile;
488 commitPath.replace(Util::endsWithGpg(), "");
489 return run(commitArgs() +
490 QStringList{"-m", "Added " + commitPath + " using QtPass.", "--"} +
491 paths);
492}

500auto ImitatePass::refuseLinkedGpgIdFolder(const QString &path) -> bool {
501 // A link planted later is replaced, not written through: the writes stage
502 // and rename.
503 const QString folder = QDir::cleanPath(path);
504 // Init() places a folder spelled through a linked root's target itself.
505 const auto refusePlaced = [this](const QString &placed) {
506 return refuseLinkedPath(placed, LinkScope::EntryAndParents,
507 PathSpelling::PlacedByCaller);
508 };
509 return refusePlaced(path) ||
510 refusePlaced(folder + QStringLiteral("/.gpg-id")) ||
511 refusePlaced(folder + QStringLiteral("/.gpg-id.sig"));

514auto ImitatePass::settleSignature(const QString &gpgIdFile,
515 const QByteArray &written,
516 const GpgIdSigner &signer, bool useGit,
517 QString *sigToCommit) -> bool {
518 const QString gpgIdSigFile = gpgIdFile + ".sig";
519 if (signer.enabled()) {
520 if (!signGpgIdFile(gpgIdFile, written)) {
521 return false;
522 }
523 *sigToCommit = gpgIdSigFile;
524 return true;
525 }
526 if (!QFile::exists(gpgIdSigFile)) {
527 return true;
528 }
529 // Signing was switched off; the removal goes into the same commit.
530 const bool tracked = useGit && gitTracks(gpgIdSigFile);
531 if (!QFile::remove(gpgIdSigFile)) {
532 emit critical(
533 tr("Cannot update"),
534 tr("Failed to remove the old signature %1.").arg(gpgIdSigFile));
535 return false;
536 }
537 if (tracked) {
538 *sigToCommit = gpgIdSigFile;
539 }
540 return true;
541}

543void ImitatePass::Init(QString path, const QList<UserInfo> &users) {
544 // The .gpg-id is written as path + ".gpg-id": without the trailing
545 // separator (the context menu hands over a cleaned path) that would be a
546 // file beside the folder, not the folder's own list. A relative path names
547 // a folder in the store, as it does for pass, not one in the working
548 // directory.
549 if (!QDir::isAbsolutePath(path)) {
550 path = QDir(m_settings.passStore).filePath(path);
551 }
552 // Cleaned before any check: the checks clean what they look at, and the
553 // write must go to the same place. Left as /store/link/../team/, the
554 // system resolves link first and writes beside wherever it points.
555 path = Util::normalizeFolderPath(QDir::cleanPath(path));
556 QString inStore;
557 if (refuseLinkedGpgIdFolder(path) ||
558 refuseFolderOutsideStore(path, &inStore)) {
559 return;
560 }
561 if (refuseUnfinishedGitOperation()) {
562 return;
563 }
564 // From here on in the store's own spelling, the same folder: one spelled
565 // through the target of a linked store root is not under the store as
566 // named, so each entry's nearest list was looked up past the new one and
567 // the entries were encrypted to the store root's.
568 path = Util::normalizeFolderPath(
569 QDir::cleanPath(QDir(m_settings.passStore).filePath(inStore)));
570 // A running re-encryption cannot take a second folder: the new list would
571 // be written and its entries left encrypted to the old recipients, which
572 // for a removed person means they can still read them. Nothing is written.
573 if (m_reencryptActive) {
574 emit critical(tr("Recipients not changed"),
575 tr("A re-encryption is still running. Wait until it has "
576 "finished, then save the recipients again."));
577 return;
578 }
579 const GpgIdSigner signer = gpgIdSigner();
580 if (signer.enabled() && !signer.haveSecretKey()) {
581 emit critical(tr("No signing key!"),
582 tr("None of the secret signing keys is available.\n"
583 "You will not be able to change the user list!"));
584 return;
585 }
586
587 const bool useGit = gitReady();
588 const QString gpgIdFile = path + ".gpg-id";
589 QByteArray written;
590 QString sigToCommit;
591 if (!writeGpgIdFile(gpgIdFile, users, &written) ||
592 !settleSignature(gpgIdFile, written, signer, useGit, &sigToCommit)) {
593 return;
594 }
595
596 QString gitOut;
597 QString gitErr;
598 if (useGit && m_settings.addGPGId) {
599 // Commit the .gpg-id (and .sig) before re-encrypting to it: the backup
600 // commit only takes tracked files (#1685) and addFolder does not stage
601 // it, so entries were pushed without their recipients file (#1682).
602 const int gitExit = gitAddGpgId(gpgIdFile, sigToCommit, &gitOut, &gitErr);
603 if (gitExit != 0) {
604 Pass::finished(PASS_INIT, gitExit, gitOut, gitErr);
605 // The new list is on disk either way. Stopping here left every entry
606 // readable by whoever was just removed from it, so re-encrypt without
607 // Git and say the change is not committed. Putting the old list back
608 // is no way out: with signing the generation record has moved on and
609 // would refuse it as a rollback.
610 const QString why = gitErr.trimmed();
611 reencryptWithoutGit(
612 path,
613 why.isEmpty() ? tr("Git exited with code %1.").arg(gitExit) : why);
614 return;
615 }
616 }
617 reencryptPath(path);
618 // finishedInit once the .gpg-id landed in git, as the async path did.
619 if (useGit) {
620 Pass::finished(PASS_INIT, 0, gitOut, gitErr);
621 }
622}

624auto ImitatePass::loadVerifiedRecipients(const QString &gpgIdFile,
625 QStringList *recipients, QString *why)
626 -> bool {
627 recipients->clear();
628 if (why != nullptr) {
629 *why = tr("Signature for %1 is invalid.").arg(gpgIdFile);
630 }
631 // A link under the .gpg-id name is not the store's list: with signing
632 // off it reads as missing, with signing on verifyFile() refuses it.
633 if (Util::isLinkedFolder(gpgIdFile)) {
634 return !gpgIdSigner().enabled();
635 }
636 QByteArray contents;
637 const GpgIdSigner signer = gpgIdSigner();
638 if (signer.enabled()) {
639 if (!verifySignedList(signer, gpgIdFile, &contents, why)) {
640 return false;
641 }
642 } else {
643 if (const QFileInfo info(gpgIdFile); !info.exists() && !info.isSymLink()) {
644 // No file is not a bad signature; the empty list says it all.
645 return true;
646 }
647 // Bounded, and only a regular file: see Util::readRegularFile().
648 std::optional<QByteArray> bytes =
649 Util::readRegularFile(gpgIdFile, Util::kMaxGpgIdBytes);
650 if (!bytes) {
651 if (why != nullptr) {
652 *why = tr("%1 cannot be read as a recipient list (a regular file "
653 "of at most %2).")
654 .arg(gpgIdFile,
655 QLocale().formattedDataSize(Util::kMaxGpgIdBytes));
656 }
657 return false;
658 }
659 contents = *std::move(bytes);
660 }
661 *recipients = Pass::parseRecipients(contents, gpgIdFile);
662 return true;
663}

665void ImitatePass::reportUnrestorable(const QString &path) {
666 emit critical(tr("Leftover from an earlier re-encryption"),
667 tr("%1 is not a regular file and was not restored. Look "
668 "at it and remove it, then re-encrypt again.")

684auto ImitatePass::restoreBackup(const QString &path) -> bool {
685 // Belt and braces: the walker only lists regular files.
686 const QFileInfo backup(path);
687 if (backup.isSymLink() || !backup.isFile()) {
688 reportUnrestorable(path);
689 return false;
690 }
691 const QString original =
692 path.chopped(QStringLiteral(".reencrypt.bak").size());
693 if (QFileInfo::exists(original)) {
694 emit critical(tr("Leftover from an earlier re-encryption"),
695 tr("%1 exists next to %2. Both are encrypted copies of the "
696 "entry; check which one you want and delete the other, "
697 "then re-encrypt again.")
698 .arg(path, original));
699 return false;
700 }
701 if (!QFile::rename(path, original)) {
702 emit critical(tr("Leftover from an earlier re-encryption"),
703 tr("%1 is missing and its backup %2 could not be renamed "
704 "back. Rename it by hand, then re-encrypt again.")
705 .arg(original, path));
706 return false;
707 }
708 qCWarning(lcQtPass) << "Restored" << original << "from" << path;
709 emit statusMsg(tr("Restored %1 from the backup an interrupted "
710 "re-encryption left behind.")
711 .arg(original),
712 5000);
713 return true;

716auto ImitatePass::recoverReencryptLeftovers(const QString &dir) -> bool {
717 // .qtpass-XXXXXX.tmp is today's staging name; 1.8.x replaced entries via
718 // X.gpg.reencrypt.tmp and .bak in two renames, builds up to 2.0 wrote
719 // X.gpg.XXXXXX.tmp. A .tmp is never a source of truth; a .bak is the only
720 // copy when X.gpg is missing, and one of two valid ones when it is not.
721 //
722 // Regular files only: renaming a link under a leftover name into an entry's
723 // place would re-encrypt whatever it points to. Links come back in
724 // `skipped`; linked directories are left to reencryptFiles() to mention.
725 const QStringList leftoverNames{QStringLiteral(".qtpass-??????.tmp"),
726 QStringLiteral("*.gpg.reencrypt.tmp"),
727 QStringLiteral("*.gpg.??????.tmp"),
728 QStringLiteral("*.gpg.reencrypt.bak")};
729 const auto isTemporary = [](const QString &path) {
730 return path.endsWith(QStringLiteral(".tmp"));
731 };
732 QStringList skipped;
733 // Hidden files included: the staged name starts with a dot.
734 const QStringList leftovers = Util::regularFilesUnder(
735 QDir::cleanPath(dir), leftoverNames, &skipped, true);
736 bool clean = true;
737 for (const QString &path : skipped) {
738 if (!QDir::match(leftoverNames, QFileInfo(path).fileName())) {
739 continue;
740 }
741 // Removing a link removes the link, never what it points to; a junction
742 // or a directory symlink on Windows is a directory entry and goes with
743 // rmdir.
744 if (isTemporary(path)) {
745 if (!QFile::remove(path) && !QDir().rmdir(path)) {
746 qCWarning(lcQtPass) << "Could not remove stale temporary" << path;
747 }
748 continue;
749 }
750 reportUnrestorable(path);
751 clean = false;
752 }
753 for (const QString &path : leftovers) {
754 if (isTemporary(path)) {
755 dropStaleTemporary(path);
756 } else {
757 clean = restoreBackup(path) && clean;
758 }
759 }
760 return clean;
761}

763auto ImitatePass::verifyGpgIdForDir(const QString &file,
764 QHash<QString, QStringList> &verified,
765 QStringList &gpgId) -> bool {
766 // Before the cache: two folders can resolve to the same list, and only
767 // one of them may have had its own.
768 if (refuseSkippedGpgId(file, gpgIdSigner().enabled())) {
769 return false;
770 }
771 const QString gpgIdPath = Pass::getGpgIdPath(file, m_settings.passStore);
772 // Keyed by .gpg-id, so every file under it gets exactly the list its
773 // signature covered.
774 const auto cached = verified.constFind(gpgIdPath);
775 if (cached != verified.constEnd()) {
776 gpgId = cached.value();
777 return true;
778 }
779 QStringList recipients;
780 QString why;
781 if (!loadVerifiedRecipients(gpgIdPath, &recipients, &why)) {
782 // An interrupted gpg is a cancel, not a bad signature.
783 if (!m_reencryptCancel.load())
784 emit critical(tr("Check .gpg-id file signature!"), why);
785 return false;
786 }
787 recipients.sort();
788 verified.insert(gpgIdPath, recipients);
789 gpgId = recipients;
790 return true;

794 QStringList args = {
795 "-v", "--no-secmem-warning", "--no-permission-warning",
796 "--list-only", "--keyid-format=long", "--",
797 pgpg(fileName)};
798 QString keys;
799 QString err;
800 const int result = execBlocking(m_settings.gpgExecutable, args, &keys, &err);
801 if (result != 0) {
802 return {};
803 }
804 QStringList actualKeys;
805 keys += err;

807 QListIterator<QString> itr(key);
808 while (itr.hasNext()) {
809 QString current = itr.next();
810 QStringList cur = current.split(" ");
811 if (cur.length() > 4) {
812 QString actualKey = cur.takeAt(4);
813 if (actualKey.length() == 16) {
814 actualKeys << actualKey;
815 }
816 }
817 }
818 actualKeys.sort();
819 return actualKeys;

870auto ImitatePass::commitReencrypted(const QString &fileName) -> bool {
871 if (!reencryptUsesGit()) {
872 return true;
873 }
874 // -C the store so git runs there rather than in QtPass's launch directory
875 // (executeBlocking sets no working directory).
876 const QString store = pgit(m_settings.passStore);
877 if (execBlocking(m_settings.gitExecutable,
878 {"-C", store, "add", "--", pgit(fileName)}) != 0) {
879 // The file on disk is re-encrypted correctly; only the repository is
880 // now behind. Report it so the caller counts this file as failed and
881 // the run is not pushed.
882 qCDebug(lcQtPass) << "git add failed after re-encrypting:" << fileName;
883 return false;
884 }
885 QString path = QDir(m_settings.passStore).relativeFilePath(fileName);
886 path.replace(Util::endsWithGpg(), "");
887 if (execBlocking(m_settings.gitExecutable,
888 QStringList{"-C", store} + commitArgs() +
889 QStringList{"-m",
890 "Re-encrypt for " + path + " using QtPass.",
891 "--", pgit(fileName)}) != 0) {
892 qCDebug(lcQtPass) << "git commit failed after re-encrypting:" << fileName;
893 return false;
894 }
895 return true;

898auto ImitatePass::reencryptSingleFile(const QString &fileName,
899 const QStringList &recipients,
900 QString *why) -> bool {
901 qCDebug(lcQtPass) << "reencrypt" << fileName << "for" << recipients.size()
902 << "recipients";
903 if (recipients.isEmpty()) {
904 emit critical(tr("Can not edit"),
905 tr("Could not read encryption key to use, .gpg-id "
906 "file missing or invalid."));
907 return false;
908 }
909 QString plaintext;
910 if (!decryptEntry(fileName, &plaintext)) {
911 return false;
912 }
913
914 // gpg writes outside the store, into a 0700 directory with an unguessable
915 // name no co-writer can pre-create or link; placeEncryptedFile() then
916 // replaces the entry in one rename, as in Insert().
917 QTemporaryDir scratch;
918 if (!scratch.isValid()) {
919 qCDebug(lcQtPass) << "Cannot create a scratch directory for re-encrypting"
920 << fileName;
921 return false;
922 }
923 const QString tempPath = scratch.filePath(QStringLiteral("reencrypted.tmp"));
924 if (!encryptFor(tempPath, recipients, plaintext) ||
925 !ciphertextHolds(tempPath, plaintext)) {
926 return false;
927 }
928
929 // Another client may have removed the entry while gpg ran; the rename
930 // would bring it back under its old name. Best effort: the check and the
931 // replace are two steps, as pass's own write is.
932 const QFileInfo still(fileName);
933 if (!still.exists() && !still.isSymLink()) {
934 qCDebug(lcQtPass) << "Entry vanished before it could be replaced:"
935 << fileName;
936 return false;
937 }
938 if (!placeEncryptedFile(tempPath, fileName, true, why)) {
939 // Reported once, in the run's summary: an unwritable folder fails every
940 // entry alike. The old ciphertext stays unless the post-rename check
941 // failed, and then `why` says so.
942 if (why != nullptr && !why->contains(fileName)) {
943 // Every reason but the copy's names the entry; that one names the
944 // file gpg wrote, in a scratch directory the user never sees.
945 *why = tr("%1 could not be re-encrypted: %2").arg(fileName, *why);
946 }
947 return false;
948 }
949 return commitReencrypted(fileName);
950}

978auto ImitatePass::createBackupCommit() -> bool {
979 if (!reencryptUsesGit()) {
980 return true;
981 }
982 emit statusMsg(tr("Creating backup commit"), 2000);
983 const QString git = m_settings.gitExecutable;
984 // -C the store: executeBlocking sets no working directory, so git would
985 // run in QtPass's launch directory, possibly an unrelated repository.
986 const QString store = pgit(m_settings.passStore);
987 // Tracked files only: an untracked plaintext export or swap file must not
988 // be swept into a commit that autoPush sends to the shared remote.
989 QString statusOut;
990 if (execBlocking(
991 git, {"-C", store, "status", "--porcelain", "--untracked-files=no"},
992 &statusOut) != 0) {
993 // An interrupted git is a cancel, not a failure worth a dialog.
994 if (!m_reencryptCancel.load())
995 emit critical(
996 tr("Backup commit failed"),
997 tr("Could not inspect git status. Re-encryption was aborted."));
998 return false;
999 }
1000 // An unresolved conflict needs no merge or rebase in progress: a
1001 // `git stash pop` that conflicted leaves none of the markers the earlier
1002 // checks look for. `add -u` would record the work tree, conflict markers
1003 // and all, as the resolution, and the backup commit would be pushed.
1004 // The repository-wide status, not `ls-files` (which stops at the store's
1005 // folder): `add -u` stages a repository above the store as a whole.
1006 if (listsUnmergedPath(statusOut)) {
1007 if (!m_reencryptCancel.load())
1008 emit critical(tr("Unfinished Git operation"),
1009 tr("The password store's Git repository has unmerged "
1010 "files. Resolve the conflict with Git before "
1011 "re-encrypting."));
1012 return false;
1013 }
1014 if (!statusOut.trimmed().isEmpty()) {
1015 if (execBlocking(git, {"-C", store, "add", "-u"}) != 0 ||
1016 execBlocking(
1017 git, QStringList{"-C", store} + commitArgs() +
1018 QStringList{"-m", "Backup before re-encryption"}) != 0) {
1019 if (!m_reencryptCancel.load())
1020 emit critical(tr("Backup commit failed"),
1021 tr("Re-encryption was aborted because a git backup "
1022 "could not be created."));
1023 return false;
1024 }
1025 }
1026 return true;
1027}

1099// Init(), Move() and Copy() queue async git commands right before
1100// reencryptPath(); the worker's blocking git would fight them for the index
1101// lock, so wait for the queue to drain.
1102void ImitatePass::startReencryptWorker(const QString &dir) {
1103 if (m_reencryptCancel.load()) {
1104 ReencryptResult result;
1105 result.cancelled = true;
1106 finishReencrypt(result);
1107 return;
1108 }
1109 if (!exec.isIdle()) {
1110 QTimer::singleShot(kReencryptRetryMs, this,
1111 [this, dir]() { startReencryptWorker(dir); });
1112 return;
1113 }
1114 // Re-check now the queue is idle: a pull queued earlier through the
1115 // asynchronous GitPull() may have finished during the wait and left a
1116 // conflict. reencryptPath() checked before scheduling, but that was before
1117 // such a pull could run; starting the worker now would let its backup commit
1118 // conclude the stopped operation. The refusal already reported it, so finish
1119 // as an aborted run (clears m_reencryptActive, emits endReencryptPath).
1120 if (refuseUnfinishedGitOperation()) {
1121 ReencryptResult result;
1122 result.aborted = true;
1123 finishReencrypt(result);
1124 return;
1125 }
1126
1127 // The worker calls member functions, so `this` must outlive it: the
1128 // destructor cancels and joins m_reencryptThread. `self` only guards the
1129 // queued completion, which may run after the thread object is gone.
1130 QPointer<ImitatePass> self(this);
1131 // The worker's own copy: env is the owning thread's to change meanwhile.
1132 updateEnv();
1133 m_reencryptEnv = storeEnvironment();
1134 QThread *thread = QThread::create([this, self, dir]() {
1135 ReencryptResult result = reencryptFiles(dir);
1136 QMetaObject::invokeMethod(
1137 self,
1138 [self, result = std::move(result)]() {
1139 if (self)
1140 self->finishReencrypt(result);
1141 },
1142 Qt::QueuedConnection);
1143 });
1144 m_reencryptThread = thread;
1145 connect(thread, &QThread::finished, this, [this, thread]() {
1146 if (m_reencryptThread == thread)
1147 m_reencryptThread = nullptr;
1148 });
1149 connect(thread, &QThread::finished, thread, &QObject::deleteLater);
1150 thread->start();
1151}

1158auto ImitatePass::pullBeforeReencrypt() -> bool {
1159 if (!m_settings.autoPull || !reencryptUsesGit()) {
1160 return true;
1161 }
1162 emit statusMsg(tr("Updating password-store"), 2000);
1163 if (execBlocking(m_settings.gitExecutable,
1164 {"-C", pgit(m_settings.passStore), "pull"}) == 0) {

1168 // that stopped in a merge leaves conflict markers and an unmerged index,
1169 // and re-encrypting on top of that would commit the mess.

1176 tr("Could not check the store for unmerged files after the "
1177 "pull. Check it with git before re-encrypting."));
1178 return false;

1180 if (!unmerged.trimmed().isEmpty()) {
1181 emit critical(tr("Git pull failed"),
1182 tr("The pull left the store with unmerged files. Resolve "
1183 "the conflict before re-encrypting."));
1184 return false;
1185 }
1186 emit statusMsg(tr("Git pull failed, re-encrypting the store as it is"), 5000);
1187 return true;

1226// Worker thread: only blocking helpers, nothing touching `exec` or the
1227// transaction state; signals reach the GUI queued. A helper failing while the
1228// cancel flag is set was interrupted: its file is neither checked nor failed.
1229auto ImitatePass::reencryptFiles(const QString &dir) -> ReencryptResult {
1230 ReencryptResult result;
1231 if (!pullBeforeReencrypt()) {
1232 result.aborted = true;
1233 return result;
1234 }
1235 // The GUI-thread checks ran before this worker's own pull.
1236 // pullBeforeReencrypt refuses an unmerged index, but a pull that stopped in a
1237 // rebase can leave a rebase-merge/rebase-apply state with no unmerged entry
1238 // yet; the backup commit's `git add -u` would then conclude it. Re-check
1239 // here, before any write, with the non-emitting detection (the refusal is
1240 // reported once, as an aborted run).
1241 if (const QString midOp = unfinishedGitOperationReason(); !midOp.isEmpty()) {
1242 if (!m_reencryptCancel.load()) {
1243 emit critical(tr("Unfinished Git operation"), midOp);
1244 }
1245 result.aborted = true;
1246 return result;
1247 }
1248 // reencryptPath() checked the folder before the pull; a pulled commit can
1249 // have made it, or a folder above it, a link since. Recovery and the walk
1250 // below would then rename and remove files beyond the store.
1251 if (Util::isUnderLink(dir, m_settings.passStore)) {
1252 emit critical(tr("Not part of the store"),
1253 tr("%1 is, or lies behind, a symbolic link or junction "
1254 "after pulling. What that points to is not part of the "
1255 "password store and is left alone.")
1256 .arg(QDir::toNativeSeparators(QDir::cleanPath(dir))));
1257 result.aborted = true;
1258 return result;
1259 }
1260
1261 // Leftovers of an interrupted run first: a restored entry then goes into
1262 // the backup commit like everything else, and a stale temporary does not.
1263 if (!recoverReencryptLeftovers(dir)) {
1264 result.aborted = true;
1265 return result;
1266 }
1267
1268 if (!createBackupCommit()) {
1269 if (m_reencryptCancel.load()) {
1270 result.cancelled = true;
1271 } else {
1272 result.aborted = true;
1273 }
1274 return result;
1275 }
1276
1277 const QStringList files = entriesToReencrypt(dir);
1278 result.total = files.size();
1279 emit reencryptProgress(0, result.total);
1280
1281 QString currentDir;
1282 QHash<QString, QStringList> gpgIdFilesVerified;
1283 QStringList gpgId;
1284 for (const QString &fileName : std::as_const(files)) {
1285 if (m_reencryptCancel.load()) {
1286 result.cancelled = true;
1287 break;
1288 }
1289 const QString fileDir = QFileInfo(fileName).path();
1290 if (fileDir != currentDir) {
1291 if (!recipientsForDir(fileName, gpgIdFilesVerified, gpgId, result)) {
1292 return result;
1293 }
1294 currentDir = fileDir;
1295 }
1296 if (getKeysFromFile(fileName) != gpgId) {
1297 QString why;
1298 if (reencryptSingleFile(fileName, gpgId, &why)) {
1299 result.reencrypted++;
1300 } else if (m_reencryptCancel.load()) {
1301 // Interrupted by the cancel: the file is untouched, not failed.
1302 result.cancelled = true;
1303 break;
1304 } else {
1305 // The reason names the entry itself when there is one to give.
1306 result.failed << (why.isEmpty() ? fileName : why);
1307 }
1308 }
1309 result.checked++;
1310 emit reencryptProgress(result.checked, result.total);
1311 }
1312 return result;
1313}

1315// No push after a cancel, abort or per-file failure: a partially re-encrypted
1316// store must not reach the remote before the user inspects it.
1317void ImitatePass::finishReencrypt(const ReencryptResult &result) {
1318 if (!result.failed.isEmpty()) {
1319 QStringList listed = result.failed.mid(0, kReencryptMaxListedFailures);
1320 const int more = result.failed.size() - listed.size();
1321 if (more > 0) {

1416auto ImitatePass::resolveMoveDestination(const QString &src,
1417 const QString &dest, bool force)
1418 -> QString {
1419 QFileInfo srcFileInfo(src);
1420 QFileInfo destFileInfo(dest);
1421 QString destFile;
1422 QString srcFileBaseName = srcFileInfo.fileName();
1423
1424 if (srcFileInfo.isFile()) {
1425 if (destFileInfo.isFile()) {
1426 if (!force) {
1427 qCDebug(lcQtPass) << "Destination file already exists";
1428 return {};
1429 }
1430 destFile = dest;
1431 } else if (destFileInfo.isDir()) {
1432 destFile = QDir(dest).filePath(srcFileBaseName);
1433 } else {
1434 destFile = dest;
1435 }
1436
1437 // Case-sensitive: only an actual ".gpg" entry already carries the suffix.
1438 // A neighbouring file like "login.GPG" is a different name, so it must not
1439 // be chopped to "login" and re-suffixed to the real "login.gpg" entry,
1440 // which the forced move would then overwrite.
1441 if (destFile.endsWith(QLatin1String(".gpg"))) {
1442 destFile.chop(4);
1443 }
1444 destFile.append(".gpg");
1445 } else if (srcFileInfo.isDir()) {
1446 if (destFileInfo.isDir()) {
1447 destFile = QDir(dest).filePath(srcFileBaseName);
1448 } else if (destFileInfo.isFile()) {
1449 qCDebug(lcQtPass) << "Destination is a file";
1450 return {};
1451 } else {
1452 destFile = dest;
1453 }
1454 } else {
1455 qCDebug(lcQtPass) << "Source file does not exist";
1456 return {};
1457 }
1458 return destFile;
1459}

1511auto ImitatePass::copyDestination(const QString &src, const QString &dest,
1512 bool force) -> QString {
1513 const auto refuse = [this, &src](const QString &to) {
1514 emit critical(tr("Copy failed"),
1515 tr("Could not copy %1 to %2.").arg(src, to));
1516 return QString();
1517 };
1518 // Like `pass cp`, dest may be an existing folder (a drag-and-drop copy hands
1519 // over the folder, not the new file name). Resolve the real target the same
1520 // way Move does: into the folder, .gpg appended, no clobbering without force.
1521 const QString destFile = resolveMoveDestination(src, dest, force);
1522 if (destFile.isEmpty()) {
1523 return refuse(dest);
1524 }
1525 // A link planted under that name (dangling ones pass exists()) is not an
1526 // entry to write.
1527 if (refuseLinkedPath(destFile)) {
1528 return {};
1529 }
1530 const QFileInfo destFileInfo(destFile);
1531 // A folder destination that is the source's own folder resolves to the
1532 // source itself; with force that would replace the only copy with itself.
1533 // And resolveMoveDestination only sees a clash when dest names the file.
1534 if (QFileInfo(src) == destFileInfo || (!force && destFileInfo.exists())) {
1535 return refuse(destFile);
1536 }
1537 return destFile;
1538}

1540void ImitatePass::Copy(const QString src, const QString dest,
1541 const bool force) {
1542 // QFile::copy reads through a link: the target's bytes would become an
1543 // entry of the store.
1544 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1545 return;
1546 }
1547 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
1548 return;
1549 }
1550 TransactionHelper trans(&m_transaction, PASS_COPY);
1551 const QString destFile = copyDestination(src, dest, force);
1552 if (destFile.isEmpty()) {
1553 return;

1573 reencryptPath(written.absoluteFilePath());
1574 } else if (written.isFile()) {
1575 reencryptPath(written.dir().path());

1609// Only PASS_* processes reach Pass::finished, so to the interface this looks
1610// the same as RealPass.
1611void ImitatePass::finished(int id, int exitCode, const QString &out,
1612 const QString &err) {
1613 qCDebug(lcQtPass) << "Imitate Pass";
1614 QString error = err;
1615 if (id == PASS_INSERT && !m_pendingInserts.isEmpty()) {
1616 // Insert()'s gpg step: place the ciphertext before the queued git steps
1617 // run (the executor starts the next item only after this returns). On
1618 // failure it fails like a gpg error: git steps cancelled below.
1619 // No dialog here: it would spin the event loop and run git steps first.
1620 const PendingInsert pending = m_pendingInserts.dequeue();
1621 // The flag answers for this insert, the one whose gpg step just ended, not
1622 // for whichever Insert() was called last: with two inserts queued, a gpg
1623 // failure of the second must report written=false even though the first
1624 // was placed. Cleared here, before placement, and set only on success.
1625 m_insertPlaced = false;
1626 if (exitCode == 0) {
1627 // Insert() checked the repository when the save was requested, but a
1628 // pull queued before it (the asynchronous Update button) may have run
1629 // and stopped in a conflict while this gpg step encrypted. Placing the
1630 // entry and committing it now would land it in the unfinished operation,
1631 // where `git rebase --abort` would drop it. Re-check and, if so, fail the
1632 // save instead of placing it; the queued git steps are cancelled below.
1633 const QString midOp = unfinishedGitOperationReason();
1634 if (!midOp.isEmpty()) {
1635 error = midOp;
1636 exitCode = 1;
1637 } else {
1638 m_insertPlaced = placeEncryptedFile(pending.output, pending.file,
1639 pending.overwrite, &error);
1640 if (!m_insertPlaced) {
1641 exitCode = 1;
1642 }
1643 }
1644 }
1645 }
1646 PROCESS pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
1647 m_transactionOutput.append(out);
1648
1649 if (exitCode == 0) {
1650 if (pid == INVALID) {
1651 return;
1652 }
1653 } else {
1654 while (pid == INVALID) {
1655 id = exec.cancelNext();
1656 if (id == -1) {
1657 // this is probably irrecoverable and shall not happen
1658 qCDebug(lcQtPass) << "No such transaction!";
1659 return;
1660 }
1661 pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
1662 }
1663 }
1664 Pass::finished(pid, exitCode, m_transactionOutput, error);
1665 m_transactionOutput.clear();
1666}
