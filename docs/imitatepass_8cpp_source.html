68auto ImitatePass::prepareBlocking(QProcess &process)
69 -> const std::atomic_bool * {
70 // In the store's environment: the git ceiling QtPass's own does not
71 // carry.
72 if (QThread::currentThread() == thread()) {
73 process.setProcessEnvironment(storeEnvironment());
74 return nullptr;
75 }
76 process.setProcessEnvironment(withCurrentGitPin(m_reencryptEnv));
77 return &m_reencryptCancel;

188void ImitatePass::startMovedReencrypts() {
189 // A run refused at its start (reencryptPath() said why) leaves the next.
190 while (!m_reencryptActive && !m_movedToReencrypt.isEmpty()) {
191 const QString scope = m_movedToReencrypt.takeFirst();
192 reencryptPath(scope);
193 if (!m_reencryptActive) {
194 // What moved is still encrypted for the recipients it came from: the
195 // automatic push after the move waits, as after a run that did not
196 // finish. A push by hand still goes.
197 holdAutoPush(scope);
198 }
199 }
200 sendDeferredPush();
201}

263void ImitatePass::Insert(QString file, QString newValue, bool overwrite) {
264 // The dialog names a new entry relative to the store; gpg used to resolve
265 // that in its working directory. Everything below works on the one path.
266 file =
267 QDir::cleanPath(QDir::isAbsolutePath(file)
268 ? file + ".gpg"
269 : QDir(m_settings.passStore).filePath(file) + ".gpg");
270 if (refuseLinkedPath(file)) {
271 return;
272 }
273 if (refuseUnfinishedGitOperation()) {
274 return;
275 }
276 // gpg used to refuse an existing output without --yes; it writes
277 // elsewhere now, so the check is ours here, and again at the rename.
278 const QFileInfo target(file);
279 if (!overwrite && (target.exists() || target.isSymLink())) {
280 emit critical(tr("Cannot add"), tr("%1 already exists.").arg(file));
281 return;
282 }
283 QStringList recipients;
284 if (!recipientsForEntry(file, &recipients)) {
285 return;
286 }
287 // gpg never opens a store path for output: a co-writer could make it a link
288 // after refuseLinkedPath(). gpg writes into a 0700 directory of our own and
289 // placeEncryptedFile() brings the bytes in via an open handle and a rename.
290 auto scratch = std::make_shared<QTemporaryDir>();
291 if (!scratch->isValid()) {
292 emit critical(tr("Cannot write"),
293 tr("Cannot create a temporary directory: %1")
294 .arg(scratch->errorString()));
295 return;
296 }
297 const QString output = scratch->filePath(QStringLiteral("entry.gpg"));
298 TransactionHelper trans(&m_transaction, PASS_INSERT);
299 const QStringList args =
300 encryptArgs({"--batch", "--status-fd", "2"}, pgpg(output), recipients);
301 m_pendingInserts.enqueue({std::move(scratch), output, file, overwrite});
302 // Past every refusal: this insert is now queued and will report once.
303 queueInsert();
304 executeGpg(PASS_INSERT, args, newValue);
305 if (!gitReady()) {
306 return;
307 }
308 if (!overwrite) {
309 executeGit(GIT_ADD, {"add", "--", pgit(file)});
310 }
311 gitCommit(file, QString(overwrite ? "Edit" : "Add") + " for " +
312 commitName(file) + " using QtPass.");
313}

340void ImitatePass::Remove(QString file, bool isDir) {
341 // No trailing separator: "link/" makes git look for what is behind the
342 // link ("pathspec did not match") and rm follow it.
343 file = QDir::cleanPath(m_settings.passStore + file);
344 if (!isDir) {
345 file += ".gpg";
346 }
347 // A link itself may go; anything behind one is not the store's to delete.
348 if (refuseLinkedPath(file, LinkScope::ParentsOnly)) {
349 return;
350 }
351 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
352 return;
353 }
354 // As in Move(): no message box inside the transaction, so it holds only
355 // the Git commands.
356 const QString path = commitName(file);
357 if (Util::isLinkedFolder(file)) {
358 // Unlink it here, never through a recursive rm: the link goes, what it
359 // points to stays. Git then only has to forget it, if it knew it.
360 if (!Util::removeTree(file)) {
361 emit critical(tr("Delete failed"),
362 tr("Could not remove the link %1.").arg(file));
363 return;
364 }
365 // Only when git knew it: a commit matching nothing (a link never
366 // committed) exits 1 and reads as a failed removal. ls-files reads the
367 // index, so it still answers now that the link is gone.
368 if (gitReady() && gitTracks(file)) {
369 TransactionHelper trans(&m_transaction, PASS_REMOVE);
370 executeGit(GIT_RM, {"rm", "-q", "--cached", "--", pgit(file)});
371 gitCommit(file, "Remove for " + path + " using QtPass.");
372 }
373 return;
374 }
375 if (gitReady()) {
376 TransactionHelper trans(&m_transaction, PASS_REMOVE);
377 executeGit(GIT_RM, {"rm", (isDir ? "-rf" : "-f"), "--", pgit(file)});
378 gitCommit(file, "Remove for " + path + " using QtPass.");
379 } else {
380 // Without Git nothing else reports a failure: git rm would print its own.
381 if (isDir) {
382 // Never QDir::removeRecursively(): it follows a junction inside the
383 // folder and empties whatever that points to.
384 if (!Util::removeTree(file)) {
385 // As rm -rf, it went on past what it could not remove: the folder
386 // is still there, without the entries that could go.
387 emit critical(tr("Delete failed"),
388 tr("Could not remove all of %1. What could be removed "
389 "is gone; check what is left in it.")
390 .arg(file));
391 }
392 } else {
393 QFile entry(file);
394 if (!entry.remove()) {
395 emit critical(
396 tr("Delete failed"),
397 tr("Could not remove %1: %2").arg(file, entry.errorString()));
398 }
399 }
400 }

412auto ImitatePass::addGenerationHeader(const QString &gpgIdFile,
413 const GpgIdSigner &signer,
414 QByteArray *contents) -> bool {
415 const std::optional<QString> folder =
416 GpgIdGeneration::folderOf(gpgIdFile, m_settings.passStore);
417 if (!folder) {
418 emit critical(tr("Cannot update"),
419 tr("%1 is not inside the password store, or a folder "
420 "name on the way holds a line break or another control "
421 "character, which a signed list cannot record.")
422 .arg(gpgIdFile));
423 return false;
424 }
425 // The list on disk counts only if it is this folder's own, verified list:
426 // a signature vouches for the bytes, the folder line for the place.
427 std::optional<qint64> verifiedOnDisk;
428 std::optional<QString> storeOnDisk;
429 QByteArray current;
430 if (signer.verifyFile(gpgIdFile, &current)) {
431 const auto header = GpgIdGeneration::parse(current);
432 if (header && (!header->folder || *header->folder == *folder)) {
433 verifiedOnDisk = header->generation;
434 // The store's id comes only from a list this store accepts. A refused
435 // one (a rollback, a conflict) may be another store's, signed with
436 // the same key and copied in: taking its id over would make the list
437 // saved now one of that store too. Its generation still counts, so the
438 // next one is above it. Without the record, which it is cannot be
439 // told: a store id made up now could split the store from devices
440 // that know its own.
441 QString unavailable;
442 const GpgIdGeneration::Verdict verdict = GpgIdGeneration::accept(
443 gpgIdFile, current, m_settings.passStore, &unavailable);
444 if (verdict == GpgIdGeneration::Verdict::RecordUnavailable) {
445 emit critical(tr("Cannot update"), unavailable);
446 return false;
447 }
448 if (verdict == GpgIdGeneration::Verdict::Accepted) {
449 storeOnDisk = header->store;
450 }
451 }
452 }
453 QString why;
454 // The store's id: the one this device knows, else the one this store's
455 // lists already carry (#2050).
456 const std::optional<QString> store =
457 GpgIdGeneration::storeIdFor(m_settings.passStore, storeOnDisk, &why);
458 if (!store) {
459 emit critical(tr("Cannot update"), why);
460 return false;
461 }
462 const std::optional<qint64> generation =
463 GpgIdGeneration::reserveNext(gpgIdFile, verifiedOnDisk, &why);
464 if (!generation) {
465 emit critical(tr("Cannot update"), why);
466 return false;
467 }
468 *contents =
469 GpgIdGeneration::withHeader(*generation, *folder, *contents, *store);
470 return true;
471}

473auto ImitatePass::writeGpgIdFile(const QString &gpgIdFile,
474 const QList<UserInfo> &users,
475 QByteArray *written) -> bool {
476 QByteArray contents;
477 bool secret_selected = false;
478 for (const UserInfo &user : users) {
479 if (user.enabled) {
480 contents += (user.key_id + "\n").toUtf8();
481 secret_selected |= user.have_secret;
482 }
483 }
484 // Without a signing key the plain list stays (GpgIdGeneration).
485 const GpgIdSigner signer = gpgIdSigner();
486 if (signer.enabled() && !addGenerationHeader(gpgIdFile, signer, &contents)) {
487 return false;
488 }
489 // Whole or not at all, owner-only (it names the store's keys); a link
490 // planted since Init's check is replaced, not written through.
491 QString writeError;
492 if (!Util::writeFileReplacing(gpgIdFile, contents, true, &writeError)) {
493 emit critical(tr("Cannot update"), writeError);
494 return false;
495 }
496 if (written != nullptr) {
497 *written = contents;
498 }
499 if (signer.enabled()) {
500 // From now on these bytes are this generation. The list is signed either
501 // way (unsigned is refused everywhere); if the record could not take them
502 // (another writer, record busy), the user is told to save once more.
503 QString why;
504 if (!GpgIdGeneration::recordWritten(gpgIdFile, contents, &why)) {
505 qCWarning(lcQtPass) << "Could not record the written .gpg-id:" << why;
506 emit critical(
507 tr("Recipient list written, but not recorded"),
508 tr("%1 Save the recipients once more to get through.").arg(why));
509 }
510 }
511 if (!secret_selected) {
512 emit critical(
513 tr("Check selected users!"),
514 tr("None of the selected keys have a secret key available.\n"
515 "You will not be able to decrypt any newly added passwords!"));
516 }
517 return true;
518}

520auto ImitatePass::signGpgIdFile(const QString &gpgIdFile,
521 const QByteArray &contents) -> bool {
522 const GpgIdSigner signer = gpgIdSigner();
523 QString why;
524 if (!signer.sign(gpgIdFile, contents, &why)) {
525 emit critical(
526 tr("GPG signing failed!"),
527 why.trimmed().isEmpty()
528 ? tr("Failed to sign %1.").arg(gpgIdFile)
529 : tr("Failed to sign %1: %2").arg(gpgIdFile, why.trimmed()));
530 return false;
531 }
532 QByteArray signedBytes;
533 if (!signer.verifyFile(gpgIdFile, &signedBytes)) {
534 emit critical(tr("Check .gpg-id file signature!"),
535 tr("Signature for %1 is invalid.").arg(gpgIdFile));
536 return false;
537 }
538 return true;
539}

541// Synchronous on purpose: queued on the async executor, the add/commit raced
542// reencryptPath()'s blocking backup commit for index.lock and either left the
543// .gpg-id untracked or found nothing to commit.
544auto ImitatePass::gitAddGpgId(const QString &gpgIdFile,
545 const QString &gpgIdSigFile, QString *out,
546 QString *err) -> int {
547 const QString git = m_settings.gitExecutable;
548 const QString store = pgit(m_settings.passStore);
549 auto run = [&](const QStringList &args) -> int {
550 QString runOut;
551 QString runErr;
552 const int rc =
553 execBlocking(git, QStringList{"-C", store} + args, &runOut, &runErr);
554 if (out != nullptr) {
555 out->append(runOut);
556 }
557 if (err != nullptr) {
558 err->append(runErr);
559 }
560 return rc;
561 };
562 QStringList paths{pgit(gpgIdFile)};
563 if (!gpgIdSigFile.isEmpty()) {
564 paths << pgit(gpgIdSigFile);
565 }
566 // add stages new and modified files alike; already-clean ones are a no-op.
567 int rc = run(QStringList{"add", "--"} + paths);
568 if (rc != 0) {
569 return rc;
570 }
571 // Re-initialising with the same recipients changes nothing; that is not a
572 // failure, and `git commit` would make it one. diff --quiet: 0 nothing
573 // staged, 1 something staged, anything else is git failing.
574 rc = run(QStringList{"diff", "--cached", "--quiet", "--"} + paths);
575 if (rc == 0) {
576 return 0;
577 }
578 if (rc != 1) {
579 return rc;
580 }
581 return run(commitArgs() +
582 QStringList{"-m",
583 "Added " + commitName(gpgIdFile) + " using QtPass.",
584 "--"} +
585 paths);
586}

595 // A link planted later is replaced, not written through: the writes stage

608auto ImitatePass::settleSignature(const QString &gpgIdFile,
609 const QByteArray &written,
610 const GpgIdSigner &signer, bool useGit,
611 QString *sigToCommit) -> bool {
612 const QString gpgIdSigFile = gpgIdFile + ".sig";
613 if (signer.enabled()) {
614 if (!signGpgIdFile(gpgIdFile, written)) {
615 return false;
616 }
617 *sigToCommit = gpgIdSigFile;
618 return true;
619 }
620 if (!QFile::exists(gpgIdSigFile)) {
621 return true;
622 }
623 // Signing was switched off; the removal goes into the same commit.
624 const bool tracked = useGit && gitTracks(gpgIdSigFile);
625 if (!QFile::remove(gpgIdSigFile)) {
626 emit critical(
627 tr("Cannot update"),
628 tr("Failed to remove the old signature %1.").arg(gpgIdSigFile));
629 return false;
630 }
631 if (tracked) {
632 *sigToCommit = gpgIdSigFile;
633 }
634 return true;
635}

637void ImitatePass::Init(QString path, const QList<UserInfo> &users) {
638 // The .gpg-id is written as path + ".gpg-id": without the trailing
639 // separator (the context menu hands over a cleaned path) that would be a
640 // file beside the folder, not the folder's own list. A relative path names
641 // a folder in the store, as it does for pass, not one in the working
642 // directory.
643 if (!QDir::isAbsolutePath(path)) {
644 path = QDir(m_settings.passStore).filePath(path);
645 }
646 // Cleaned before any check: the checks clean what they look at, and the
647 // write must go to the same place. Left as /store/link/../team/, the
648 // system resolves link first and writes beside wherever it points.
649 path = Util::normalizeFolderPath(QDir::cleanPath(path));
650 QString inStore;
651 if (refuseLinkedGpgIdFolder(path) ||
652 refuseFolderOutsideStore(path, &inStore)) {
653 return;
654 }
655 if (refuseUnfinishedGitOperation()) {
656 return;
657 }
658 // From here on in the store's own spelling, the same folder: one spelled
659 // through the target of a linked store root is not under the store as
660 // named, so each entry's nearest list was looked up past the new one and
661 // the entries were encrypted to the store root's.
662 path = Util::normalizeFolderPath(
663 QDir::cleanPath(QDir(m_settings.passStore).filePath(inStore)));
664 // A running re-encryption cannot take a second folder: the new list would
665 // be written and its entries left encrypted to the old recipients, which
666 // for a removed person means they can still read them. Nothing is written.
667 if (m_reencryptActive) {
668 emit critical(tr("Recipients not changed"),
669 tr("A re-encryption is still running. Wait until it has "
670 "finished, then save the recipients again."));
671 return;
672 }
673 const GpgIdSigner signer = gpgIdSigner();
674 if (signer.enabled() && !signer.haveSecretKey()) {
675 emit critical(tr("No signing key!"),
676 tr("None of the secret signing keys is available.\n"
677 "You will not be able to change the user list!"));
678 return;
679 }
680
681 const bool useGit = gitReady();
682 const QString gpgIdFile = path + ".gpg-id";
683 // Each name free or a regular file, before anything is written: a folder
684 // named .gpg-id.sig stopped the change after the new list was on disk,
685 // which was then neither committed nor were the entries re-encrypted to
686 // it, while the dialog had closed as if done. Links were refused above.
687 for (const QString &name : {gpgIdFile, gpgIdFile + QStringLiteral(".sig")}) {
688 if (const QFileInfo info(name); info.exists() && !info.isFile()) {
689 emit critical(tr("Recipients not changed"),
690 tr("%1 is not a regular file, so the recipients were not "
691 "changed. Move it out of the way, then save the "
692 "recipients again.")
693 .arg(QDir::toNativeSeparators(name)));
694 return;
695 }
696 }
697 QByteArray written;
698 QString sigToCommit;
699 if (!writeGpgIdFile(gpgIdFile, users, &written)) {
700 return;
701 }
702 // The new list is on disk: held from here, so a stop before the run (a
703 // signature that fails, a refused start) leaves it out of a later backup
704 // commit's push; a clean run of this folder releases it.
705 holdAutoPush(path);
706 if (!settleSignature(gpgIdFile, written, signer, useGit, &sigToCommit)) {
707 return;
708 }
709
710 QString gitOut;
711 QString gitErr;
712 if (useGit && m_settings.addGPGId) {
713 // Commit the .gpg-id (and .sig) before re-encrypting to it: the backup
714 // commit only takes tracked files (#1685) and addFolder does not stage
715 // it, so entries were pushed without their recipients file (#1682).
716 const int gitExit = gitAddGpgId(gpgIdFile, sigToCommit, &gitOut, &gitErr);
717 if (gitExit != 0) {
718 Pass::finished(PASS_INIT, gitExit, gitOut, gitErr);
719 // The new list is on disk either way. Stopping here left every entry
720 // readable by whoever was just removed from it, so re-encrypt without
721 // Git and say the change is not committed. Putting the old list back
722 // is no way out: with signing the generation record has moved on and
723 // would refuse it as a rollback.
724 const QString why = gitErr.trimmed();
725 reencryptWithoutGit(
726 path,
727 why.isEmpty() ? tr("Git exited with code %1.").arg(gitExit) : why);
728 return;
729 }
730 }
731 m_writtenGpgId = WrittenGpgId{gpgIdFile, written};
732 reencryptPath(path);
733 if (!m_reencryptActive) {
734 // Refused before it started; reencryptPath() said why.
735 m_writtenGpgId.reset();
736 }
737 // finishedInit once the .gpg-id landed in git, as the async path did.
738 if (useGit) {
739 Pass::finished(PASS_INIT, 0, gitOut, gitErr);
740 }
741}

743auto ImitatePass::loadVerifiedRecipients(const QString &gpgIdFile,
744 QStringList *recipients, QString *why)
745 -> bool {
746 recipients->clear();
747 if (why != nullptr) {
748 *why = tr("Signature for %1 is invalid.").arg(gpgIdFile);
749 }
750 // A link under the .gpg-id name is not the store's list: with signing
751 // off it reads as missing, with signing on verifyFile() refuses it.
752 if (Util::isLinkedFolder(gpgIdFile)) {
753 return !gpgIdSigner().enabled();
754 }
755 QByteArray contents;
756 const GpgIdSigner signer = gpgIdSigner();
757 if (signer.enabled()) {
758 if (!verifySignedList(signer, gpgIdFile, &contents, why)) {
759 return false;
760 }
761 } else {
762 if (const QFileInfo info(gpgIdFile); !info.exists() && !info.isSymLink()) {
763 // No file is not a bad signature; the empty list says it all.
764 return true;
765 }
766 // Bounded, and only a regular file: see Util::readRegularFile().
767 std::optional<QByteArray> bytes =
768 Util::readRegularFile(gpgIdFile, Util::kMaxGpgIdBytes);
769 if (!bytes) {
770 if (why != nullptr) {
771 *why = tr("%1 cannot be read as a recipient list (a regular file "
772 "of at most %2).")
773 .arg(gpgIdFile,
774 QLocale().formattedDataSize(Util::kMaxGpgIdBytes));
775 }
776 return false;
777 }
778 contents = *std::move(bytes);
779 }
780 *recipients = Pass::parseRecipients(contents, gpgIdFile);
781 return true;
782}

834auto ImitatePass::recoverReencryptLeftovers(const QString &dir) -> bool {
835 // .qtpass-XXXXXX.tmp is today's staging name; 1.8.x replaced entries via
836 // X.gpg.reencrypt.tmp and .bak in two renames, builds up to 2.0 wrote
837 // X.gpg.XXXXXX.tmp. A .tmp is never a source of truth; a .bak is the only
838 // copy when X.gpg is missing, and one of two valid ones when it is not.
839 //
840 // Regular files only: renaming a link under a leftover name into an entry's
841 // place would re-encrypt whatever it points to. Links come back in
842 // `skipped`; linked directories are left to reencryptFiles() to mention.
843 const QStringList leftoverNames{QStringLiteral(".qtpass-??????.tmp"),
844 QStringLiteral("*.gpg.reencrypt.tmp"),
845 QStringLiteral("*.gpg.??????.tmp"),
846 QStringLiteral("*.gpg.reencrypt.bak")};
847 const auto isTemporary = [](const QString &path) {
848 return path.endsWith(QStringLiteral(".tmp"));
849 };
850 QStringList skipped;
851 // Hidden files included: the staged name starts with a dot.
852 const QStringList leftovers = Util::regularFilesUnder(
853 QDir::cleanPath(dir), leftoverNames, &skipped, true);
854 // One report for all of them: a box per leftover opened each from inside
855 // the one before (a modal box runs the next queued critical()), and a
856 // store with thousands of them overflowed the stack.
857 QStringList problems;
858 for (const QString &path : skipped) {
859 if (!QDir::match(leftoverNames, QFileInfo(path).fileName())) {
860 continue;
861 }
862 // Removing a link removes the link, never what it points to; a junction
863 // or a directory symlink on Windows is a directory entry and goes with
864 // rmdir.
865 if (isTemporary(path)) {
866 if (!QFile::remove(path) && !QDir().rmdir(path)) {
867 qCWarning(lcQtPass) << "Could not remove stale temporary" << path;
868 }
869 continue;
870 }
871 reportUnrestorable(path, problems);
872 }
873 for (const QString &path : leftovers) {
874 if (isTemporary(path)) {
875 dropStaleTemporary(path);
876 } else {
877 restoreBackup(path, problems);
878 }
879 }
880 if (problems.isEmpty()) {
881 return true;
882 }
883 const qsizetype count = problems.size();
884 if (count == 1) {
885 emit critical(tr("Leftover from an earlier re-encryption"),
886 problems.first());
887 return false;
888 }
889 QStringList listed = problems.mid(0, kReencryptMaxListedFailures);
890 if (count > listed.size()) {
891 listed << tr("... and %n more", nullptr,
892 static_cast<int>(count - listed.size()));
893 }
894 emit critical(tr("Leftovers from an earlier re-encryption"),
895 tr("%n leftover(s) of an earlier re-encryption need a look "
896 "before the run can start:",
897 nullptr, static_cast<int>(count)) +
898 "\n\n" + listed.join("\n\n"));
899 return false;
900}

902auto ImitatePass::verifyGpgIdForDir(const QString &file,
903 QHash<QString, QStringList> &verified,
904 QStringList &gpgId) -> bool {
905 // Before the cache: two folders can resolve to the same list, and only
906 // one of them may have had its own.
907 if (refuseSkippedGpgId(file, gpgIdSigner().enabled())) {
908 return false;
909 }
910 const QString gpgIdPath = Pass::getGpgIdPath(file, m_settings.passStore);
911 // Keyed by .gpg-id, so every file under it gets exactly the list its
912 // signature covered.
913 const auto cached = verified.constFind(gpgIdPath);
914 if (cached != verified.constEnd()) {
915 gpgId = cached.value();
916 return true;
917 }
918 QStringList recipients;
919 QString why;
920 if (!loadVerifiedRecipients(gpgIdPath, &recipients, &why)) {
921 // An interrupted gpg is a cancel, not a bad signature.
922 if (!m_reencryptCancel.load())
923 emit critical(tr("Check .gpg-id file signature!"), why);
924 return false;
925 }
926 recipients.sort();
927 verified.insert(gpgIdPath, recipients);
928 gpgId = recipients;
929 return true;
930}

961auto ImitatePass::decryptEntry(const QString &fileName, QByteArray *plaintext,
962 QString *why) -> bool {
963 // Read here, without following a link, and handed to gpg on stdin: the
964 // name was listed as a regular file before the run, and a pull since may
965 // have put a link there. gpg given the name would follow it, and the
966 // file it points to would be re-encrypted for this folder's recipients.
967 // The cap is on what decrypts; the ciphertext of an entry just under it
968 // is a little larger (packet headers, one session key per recipient).
969 constexpr qint64 kCiphertextMargin = 1024LL * 1024;
970 // That read refuses a link only as the last part of the name; a folder
971 // on the way that became a link is checked here.
972 const std::optional<QByteArray> ciphertext =
973 Util::isUnderLink(fileName, m_settings.passStore)
974 ? std::nullopt
975 : Util::readRegularFile(fileName,
976 kMaxPlaintextBytes + kCiphertextMargin);
977 if (!ciphertext) {
978 qCDebug(lcQtPass) << "Not a regular file any more, or too large:"
979 << fileName;
980 if (why != nullptr) {
981 *why = tr("%1 is no longer a regular file, or is too large, and was "
982 "not re-encrypted.")
983 .arg(fileName);
984 }
985 return false;
986 }
987 const QStringList args =
988 QStringList{"-d", "--quiet", "--yes", "--no-encrypt-to",
989 "--batch", "--use-agent"} +
990 maxOutputArgs();
991 if (execBlockingRaw(m_settings.gpgExecutable, args, *ciphertext, plaintext) !=
992 0 ||
993 plaintext->isEmpty()) {
994 qCDebug(lcQtPass) << "Decrypt error on re-encrypt for:" << fileName;
995 return false;
996 }
997 // No newline added: an entry is re-encrypted as it was, as pass(1) does
998 // (gpg -d | gpg -e); a binary key without one would have grown a byte.
999 return true;
1000}

1015auto ImitatePass::ciphertextHolds(const QString &ciphertext,
1016 const QByteArray &plaintext) -> bool {
1017 QByteArray decrypted;
1018 const QStringList args =
1019 QStringList{"-d", "--quiet", "--batch", "--use-agent"} + maxOutputArgs() +
1020 QStringList{QStringLiteral("--"), pgpg(ciphertext)};
1021 if (execBlockingRaw(m_settings.gpgExecutable, args, QByteArray(),
1022 &decrypted) != 0 ||
1023 decrypted.isEmpty()) {
1024 qCDebug(lcQtPass) << "Verification failed for:" << ciphertext;
1025 return false;
1026 }

1028 // Exact: gpg returns the bytes it was given. Compared as text, the check
1029 // passed for an entry that had lost bytes, as both sides lost the same.
1030 if (decrypted != plaintext) {
1031 qCDebug(lcQtPass) << "Verification content mismatch for:" << ciphertext;
1032 return false;
1033 }
1034 return true;

1038 if (!reencryptUsesGit()) {
1039 return true;
1040 }

1042 // (executeBlocking sets no working directory).

1045 {"-C", store, "add", "--", pgit(fileName)}) != 0) {

1047 // now behind. Report it so the caller counts this file as failed and
1048 // the run is not pushed.
1049 qCDebug(lcQtPass) << "git add failed after re-encrypting:" << fileName;
1050 return false;
1051 }
1052 const QString path = commitName(fileName);
1053 if (execBlocking(m_settings.gitExecutable,
1054 QStringList{"-C", store} + commitArgs() +
1055 QStringList{"-m",
1056 "Re-encrypt for " + path + " using QtPass.",
1057 "--", pgit(fileName)}) != 0) {
1058 qCDebug(lcQtPass) << "git commit failed after re-encrypting:" << fileName;
1059 return false;

1064auto ImitatePass::reencryptSingleFile(const QString &fileName,
1065 const QStringList &recipients,
1066 QString *why) -> bool {
1067 qCDebug(lcQtPass) << "reencrypt" << fileName << "for" << recipients.size()
1068 << "recipients";
1069 if (recipients.isEmpty()) {
1070 emit critical(tr("Can not edit"),
1071 tr("Could not read encryption key to use, .gpg-id "
1072 "file missing or invalid."));
1073 return false;
1074 }
1075 QByteArray plaintext;
1076 if (!decryptEntry(fileName, &plaintext, why)) {
1077 return false;
1078 }
1079
1080 // gpg writes outside the store, into a 0700 directory with an unguessable
1081 // name no co-writer can pre-create or link; placeEncryptedFile() then
1082 // replaces the entry in one rename, as in Insert().
1083 QTemporaryDir scratch;
1084 if (!scratch.isValid()) {
1085 qCDebug(lcQtPass) << "Cannot create a scratch directory for re-encrypting"
1086 << fileName;
1087 return false;
1088 }
1089 const QString tempPath = scratch.filePath(QStringLiteral("reencrypted.tmp"));
1090 if (!encryptFor(tempPath, recipients, plaintext) ||
1091 !ciphertextHolds(tempPath, plaintext)) {
1092 return false;
1093 }
1094
1095 // Another client may have removed the entry while gpg ran; the rename
1096 // would bring it back under its old name. Best effort: the check and the
1097 // replace are two steps, as pass's own write is.
1098 const QFileInfo still(fileName);
1099 if (!still.exists() && !still.isSymLink()) {
1100 qCDebug(lcQtPass) << "Entry vanished before it could be replaced:"
1101 << fileName;
1102 return false;
1103 }
1104 if (!placeEncryptedFile(tempPath, fileName, true, why)) {
1105 // Reported once, in the run's summary: an unwritable folder fails every
1106 // entry alike. The old ciphertext stays unless the post-rename check
1107 // failed, and then `why` says so.
1108 if (why != nullptr && !why->contains(fileName)) {
1109 // Every reason but the copy's names the entry; that one names the
1110 // file gpg wrote, in a scratch directory the user never sees.
1111 *why = tr("%1 could not be re-encrypted: %2").arg(fileName, *why);
1112 }
1113 return false;
1114 }
1115 return commitReencrypted(fileName);
1116}

1144auto ImitatePass::createBackupCommit() -> bool {
1145 if (!reencryptUsesGit()) {
1146 return true;
1147 }
1148 emit statusMsg(tr("Creating backup commit"), 2000);
1149 const QString git = m_settings.gitExecutable;
1150 // -C the store: executeBlocking sets no working directory, so git would
1151 // run in QtPass's launch directory, possibly an unrelated repository.
1152 const QString store = pgit(m_settings.passStore);
1153 // Tracked files only: an untracked plaintext export or swap file must not
1154 // be swept into a commit that autoPush sends to the shared remote.
1155 QString statusOut;
1156 if (execBlocking(
1157 git, {"-C", store, "status", "--porcelain", "--untracked-files=no"},
1158 &statusOut) != 0) {
1159 // An interrupted git is a cancel, not a failure worth a dialog.
1160 if (!m_reencryptCancel.load())
1161 emit critical(
1162 tr("Backup commit failed"),
1163 tr("Could not inspect git status. Re-encryption was aborted."));
1164 return false;
1165 }
1166 // An unresolved conflict needs no merge or rebase in progress: a
1167 // `git stash pop` that conflicted leaves none of the markers the earlier
1168 // checks look for. `add -u` would record the work tree, conflict markers
1169 // and all, as the resolution, and the backup commit would be pushed.
1170 // The repository-wide status, not `ls-files` (which stops at the store's
1171 // folder): `add -u` stages a repository above the store as a whole.
1172 if (listsUnmergedPath(statusOut)) {
1173 if (!m_reencryptCancel.load())
1174 emit critical(tr("Unfinished Git operation"),
1175 tr("The password store's Git repository has unmerged "
1176 "files. Resolve the conflict with Git before "
1177 "re-encrypting."));
1178 return false;
1179 }
1180 // The store only: in a repository above it, a modified tracked file
1181 // outside the store is not the store's to back up, and autoPush would
1182 // publish it. `.` is the store, as git runs with -C the store.
1183 QString storeStatus;
1184 if (execBlocking(git,
1185 {"-C", store, "status", "--porcelain",
1186 "--untracked-files=no", "--", "."},
1187 &storeStatus) != 0) {
1188 if (!m_reencryptCancel.load())
1189 emit critical(
1190 tr("Backup commit failed"),
1191 tr("Could not inspect git status. Re-encryption was aborted."));
1192 return false;
1193 }
1194 if (!storeStatus.trimmed().isEmpty()) {
1195 if (execBlocking(git, {"-C", store, "add", "-u", "--", "."}) != 0 ||
1196 execBlocking(git, QStringList{"-C", store} + commitArgs() +
1197 QStringList{"-m", "Backup before re-encryption",
1198 "--", "."}) != 0) {
1199 if (!m_reencryptCancel.load())
1200 emit critical(tr("Backup commit failed"),
1201 tr("Re-encryption was aborted because a git backup "
1202 "could not be created."));
1203 return false;
1204 }
1205 }
1206 return true;
1207}

1273// The worker's kill() after the grace period is needed: terminate() is only
1274// SIGTERM, or WM_CLOSE on Windows, which a console gpg ignores.
1275void ImitatePass::cancelReencryptPath() {
1276 if (!m_reencryptActive)
1277 return;
1278 m_reencryptCancel.store(true);
1279}
1280
1281// Init(), Move() and Copy() queue async git commands right before
1282// reencryptPath(); the worker's blocking git would fight them for the index
1283// lock, so wait for the queue to drain. The other backend's too: with pass
1284// as the backend an Update's `pass git pull` runs on RealPass's queue, and
1285// the run listed the store while that pull could still change it.
1286void ImitatePass::startReencryptWorker(const QString &dir) {
1287 if (m_reencryptCancel.load()) {
1288 ReencryptResult result;
1289 result.cancelled = true;
1290 finishReencrypt(result);
1291 return;
1292 }
1293 if (!exec.isIdle() || !PassBackendFactory::isIdle()) {
1294 QTimer::singleShot(kReencryptRetryMs, this,
1295 [this, dir]() { startReencryptWorker(dir); });
1296 return;
1297 }
1298 // Re-check now the queue is idle: a pull queued earlier through the
1299 // asynchronous GitPull() may have finished during the wait and left a
1300 // conflict. reencryptPath() checked before scheduling, but that was before
1301 // such a pull could run; starting the worker now would let its backup commit
1302 // conclude the stopped operation. The refusal already reported it, so finish
1303 // as an aborted run (clears m_reencryptActive, emits endReencryptPath).
1304 if (refuseUnfinishedGitOperation()) {
1305 ReencryptResult result;
1306 result.aborted = true;
1307 finishReencrypt(result);
1308 return;
1309 }
1310
1311 // The worker calls member functions, so `this` must outlive it: the
1312 // destructor cancels and joins m_reencryptThread. `self` only guards the
1313 // queued completion, which may run after the thread object is gone.
1314 QPointer<ImitatePass> self(this);
1315 // The worker's own copy: env is the owning thread's to change meanwhile.
1316 updateEnv();
1317 m_reencryptEnv = storeEnvironment();
1318 QThread *thread = QThread::create([this, self, dir]() {
1319 ReencryptResult result = reencryptFiles(dir);
1320 QMetaObject::invokeMethod(
1321 self,
1322 [self, result = std::move(result)]() {
1323 if (self)
1324 self->finishReencrypt(result);
1325 },
1326 Qt::QueuedConnection);
1327 });
1328 m_reencryptThread = thread;
1329 connect(thread, &QThread::finished, this, [this, thread]() {
1330 if (m_reencryptThread == thread)
1331 m_reencryptThread = nullptr;
1332 });
1333 connect(thread, &QThread::finished, thread, &QObject::deleteLater);
1334 thread->start();
1335}

1342auto ImitatePass::pullBeforeReencrypt() -> bool {
1343 if (!m_settings.autoPull || !reencryptUsesGit()) {
1344 return true;
1345 }
1346 emit statusMsg(tr("Updating password-store"), 2000);
1347 if (execBlocking(m_settings.gitExecutable,
1348 {"-C", pgit(m_settings.passStore), "pull"}) == 0) {
1349 return true;
1350 }
1351 // A pull that could not reach the remote leaves the store as it was; one
1352 // that stopped in a merge leaves conflict markers and an unmerged index,
1353 // and re-encrypting on top of that would commit the mess.
1354 QString unmerged;
1355 if (execBlocking(m_settings.gitExecutable,
1356 {"-C", pgit(m_settings.passStore), "ls-files", "--unmerged"},
1357 &unmerged) != 0) {
1358 // Unknown is not clean: the pull may have stopped in a merge.
1359 emit critical(tr("Git pull failed"),
1360 tr("Could not check the store for unmerged files after the "
1361 "pull. Check it with git before re-encrypting."));
1362 return false;
1363 }
1364 if (!unmerged.trimmed().isEmpty()) {
1365 emit critical(tr("Git pull failed"),
1366 tr("The pull left the store with unmerged files. Resolve "
1367 "the conflict before re-encrypting."));
1368 return false;
1369 }
1370 // A third kind: the fetch got the commits and the checkout then failed at
1371 // one path (a link git could not create, say). Git leaves HEAD and the
1372 // index where they were, but the pulled versions of the other files are
1373 // already in the work tree, and the backup commit would take them as the
1374 // user's own, signed with the user's key. HEAD still behind the upstream
1375 // after a failed pull means the pull may have got that far. No upstream
1376 // (rev-list fails) means there was nothing to merge.
1377 QString behind;
1378 if (execBlocking(m_settings.gitExecutable,
1379 {"-C", pgit(m_settings.passStore), "rev-list", "--count",
1380 "HEAD..@{upstream}"},
1381 &behind) == 0 &&
1382 behind.trimmed().toInt() > 0) {
1383 emit critical(tr("Git pull failed"),
1384 tr("The pull stopped before the store was up to date with "
1385 "its remote, and may have left files of the pulled "
1386 "commits in it. Check the store with git before "
1387 "re-encrypting."));
1388 return false;
1389 }
1390 emit statusMsg(tr("Git pull failed, re-encrypting the store as it is"), 5000);
1391 return true;
1392}

1394auto ImitatePass::entriesToReencrypt(const QString &dir, QStringList *failed)
1395 -> QStringList {
1396 // Regular files only: a symlink or junction is not a password entry, and
1397 // following one would decrypt and rewrite something outside the store.
1398 // One entry (what a move moved): itself, when it is one.
1399 if (const QFileInfo one(dir); !one.isDir()) {
1400 if (one.isFile() && !Util::isLinkedFolder(dir) &&
1401 one.fileName().endsWith(QLatin1String(".gpg"))) {
1402 return {one.filePath()};
1403 }
1404 return {};
1405 }
1406 QStringList skipped;
1407 QStringList unlisted;
1408 // A dot-named entry (.env.gpg) too, as with `pass init`: left out, it
1409 // stayed encrypted to a recipient just removed while the run reported
1410 // success. A hidden folder (.git, a sync tool's .stversions) stays closed.
1411 const QStringList files = Util::regularFilesUnder(
1412 dir, QStringList() << "*.gpg", &skipped, true, &unlisted);
1413 // A folder that cannot be read looked empty, and the run reported success
1414 // over it with its entries still encrypted to the old recipients (a key
1415 // just removed among them). Each is a failure: the dialog names it and
1416 // nothing is pushed.
1417 for (const QString &folder : std::as_const(unlisted)) {
1418 *failed << tr("%1 could not be read: its entries were not re-encrypted")
1419 .arg(QDir::toNativeSeparators(folder));
1420 }
1421 if (!skipped.isEmpty()) {
1422 emit statusMsg(tr("%n entr(y/ies) skipped: a symlink, junction or special "
1423 "file is not part of the store.",
1424 "", static_cast<int>(skipped.size())),
1425 5000);
1426 }
1427 return files;
1428}

1431 QHash<QString, QStringList> &verified,
1432 QStringList &gpgId, ReencryptResult &result)
1433 -> bool {

1436 result.cancelled = true;
1437 } else {
1438 result.aborted = true;

1440 return false;

1442 if (gpgId.isEmpty() && !verified.isEmpty()) {
1443 emit critical(tr("GPG ID verification failed"),
1444 tr("Could not verify .gpg-id for directory."));

1446 return false;
1447 }

1451// Worker thread: only blocking helpers, nothing touching `exec` or the
1452// transaction state; signals reach the GUI queued. A helper failing while the
1453// cancel flag is set was interrupted: its file is neither checked nor failed.
1454auto ImitatePass::reencryptFiles(const QString &dir) -> ReencryptResult {
1455 ReencryptResult result;
1456 if (!pullBeforeReencrypt()) {
1457 result.aborted = true;
1458 return result;
1459 }
1460 // The GUI-thread checks ran before this worker's own pull.
1461 // pullBeforeReencrypt refuses an unmerged index, but a pull that stopped in a
1462 // rebase can leave a rebase-merge/rebase-apply state with no unmerged entry
1463 // yet; the backup commit's `git add -u` would then conclude it. Re-check
1464 // here, before any write, with the non-emitting detection (the refusal is
1465 // reported once, as an aborted run).
1466 if (const QString midOp = unfinishedGitOperationReason(); !midOp.isEmpty()) {
1467 if (!m_reencryptCancel.load()) {
1468 emit critical(tr("Unfinished Git operation"), midOp);
1469 }
1470 result.aborted = true;
1471 return result;
1472 }
1473 // reencryptPath() checked the folder before the pull; a pulled commit can
1474 // have made it, or a folder above it, a link since. Recovery and the walk
1475 // below would then rename and remove files beyond the store.
1476 if (Util::isUnderLink(dir, m_settings.passStore)) {
1477 emit critical(tr("Not part of the store"),
1478 tr("%1 is, or lies behind, a symbolic link or junction "
1479 "after pulling. What that points to is not part of the "
1480 "password store and is left alone.")
1481 .arg(QDir::toNativeSeparators(QDir::cleanPath(dir))));
1482 result.aborted = true;
1483 return result;
1484 }
1485
1486 // Init() wrote the recipients the user ticked and committed them. A pull
1487 // that merged another change to the same list cleanly (a line removed
1488 // here, one added there) would have the run encrypt to keys nobody chose
1489 // in the dialog, and without a signing key nothing else notices.
1490 if (m_writtenGpgId) {
1491 const std::optional<QByteArray> now =
1492 Util::readRegularFile(m_writtenGpgId->file, Util::kMaxGpgIdBytes);
1493 if (!now || *now != m_writtenGpgId->bytes) {
1494 if (!m_reencryptCancel.load()) {
1495 emit critical(
1496 tr("Recipients changed by the pull"),
1497 tr("The pull changed the recipients of %1 after you saved "
1498 "them, so nothing was re-encrypted: entries in it may still "
1499 "be encrypted to the previous recipients, and anyone you "
1500 "just removed may still be able to read them. Open the users "
1501 "dialog for that folder again, check the recipients, and "
1502 "save.")
1503 .arg(QDir::toNativeSeparators(
1504 QFileInfo(m_writtenGpgId->file).path())));
1505 }
1506 result.aborted = true;
1507 return result;
1508 }
1509 }
1510
1511 // Leftovers of an interrupted run first: a restored entry then goes into
1512 // the backup commit like everything else, and a stale temporary does not.
1513 if (!recoverReencryptLeftovers(dir)) {
1514 result.aborted = true;
1515 return result;
1516 }
1517
1518 if (!createBackupCommit()) {
1519 if (m_reencryptCancel.load()) {
1520 result.cancelled = true;
1521 } else {
1522 result.aborted = true;
1523 }
1524 return result;
1525 }
1526
1527 const QStringList files = entriesToReencrypt(dir, &result.failed);
1528 result.total = files.size();
1529 emit reencryptProgress(0, result.total);
1530
1531 QString currentDir;
1532 QHash<QString, QStringList> gpgIdFilesVerified;
1533 QStringList gpgId;
1534 for (const QString &fileName : std::as_const(files)) {
1535 if (m_reencryptCancel.load()) {
1536 result.cancelled = true;
1537 break;
1538 }
1539 const QString fileDir = QFileInfo(fileName).path();
1540 if (fileDir != currentDir) {
1541 if (!recipientsForDir(fileName, gpgIdFilesVerified, gpgId, result)) {
1542 return result;
1543 }
1544 currentDir = fileDir;
1545 }
1546 // A pull since the listing may have put a link under the name, or in
1547 // place of a folder on the way to it. gpg given the name reads the
1548 // target, and one already encrypted for these recipients would pass the
1549 // comparison without a word.
1550 const QFileInfo entry(fileName);
1551 if (entry.isSymLink() || !entry.isFile() ||
1552 Util::isUnderLink(fileName, m_settings.passStore)) {
1553 result.failed << tr("%1 is no longer a regular file, or lies behind a "
1554 "link, and was not re-encrypted.")
1555 .arg(fileName);
1556 } else if (getKeysFromFile(fileName) != gpgId) {
1557 QString why;
1558 if (reencryptSingleFile(fileName, gpgId, &why)) {
1559 result.reencrypted++;
1560 } else if (m_reencryptCancel.load()) {
1561 // Interrupted by the cancel: the file is untouched, not failed.
1562 result.cancelled = true;
1563 break;
1564 } else {
1565 // The reason names the entry itself when there is one to give.
1566 result.failed << (why.isEmpty() ? fileName : why);
1567 }
1568 }
1569 result.checked++;
1570 emit reencryptProgress(result.checked, result.total);
1571 }
1572 return result;
1573}

1575// No push after a cancel, abort or per-file failure: a partially re-encrypted
1576// store must not reach the remote before the user inspects it.
1577void ImitatePass::finishReencrypt(const ReencryptResult &result) {
1578 if (!result.failed.isEmpty()) {
1579 QStringList listed = result.failed.mid(0, kReencryptMaxListedFailures);
1580 const int more = result.failed.size() - listed.size();
1581 if (more > 0) {
1582 listed << tr("... and %n more", nullptr, more);
1583 }
1584 emit critical(tr("Re-encryption failed"),
1585 tr("%n file(s) could not be re-encrypted:", nullptr,
1586 result.failed.size()) +
1587 "\n\n" + listed.join('\n'));
1588 }
1589
1590 // The recipients change is committed already: a run that did not finish
1591 // leaves entries for the previous recipients, which the next automatic
1592 // push (after any edit) would publish with it. A clean run releases only
1593 // the holds of what it covered. Also without Git: a recipients change
1594 // left uncommitted now is swept into a later commit (the next run's
1595 // backup commit) once Git is on, and pushed with it.
1596 if (result.cancelled || result.aborted || !result.failed.isEmpty()) {
1597 holdAutoPush(m_reencryptScope);
1598 } else {
1599 releaseAutoPush(m_reencryptScope);
1600 }
1601
1602 if (result.cancelled) {
1603 emit statusMsg(tr("Re-encryption cancelled: %1 of %2 files checked, "
1604 "%3 re-encrypted, %4 failed")
1605 .arg(result.checked)
1606 .arg(result.total)
1607 .arg(result.reencrypted)
1608 .arg(result.failed.size()),
1609 5000);
1610 } else if (!result.aborted) {
1611 if (!result.failed.isEmpty()) {
1612 emit statusMsg(tr("Re-encryption completed: %1 succeeded, %2 failed")
1613 .arg(result.reencrypted)
1614 .arg(result.failed.size()),
1615 5000);
1616 } else {
1617 emit statusMsg(tr("Re-encryption completed: %1 files re-encrypted")
1618 .arg(result.reencrypted),
1619 3000);
1620 }
1621 // With a moved entry's run still to come, that one pushes at its end.
1622 if (m_movedToReencrypt.isEmpty() &&
1623 (m_settings.autoPush || m_pushAfterReencrypt) && reencryptUsesGit()) {
1624 m_pushAfterReencrypt = false;
1625 m_deferredPushByHand = false;
1626 if (result.failed.isEmpty() && autoPushHeld()) {
1627 // Clean here, but another folder's run did not finish.
1628 emit statusMsg(tr("Not pushing: an earlier re-encryption in this "
1629 "store did not finish"),
1630 5000);
1631 } else if (result.failed.isEmpty()) {
1632 emit statusMsg(tr("Updating password-store"), 2000);
1633 m_reencryptPushing = true;
1634 GitPush();
1635 m_reencryptPushing = false;
1636 } else {
1637 emit statusMsg(tr("Not pushing: %n file(s) failed to re-encrypt",
1638 nullptr, result.failed.size()),
1639 5000);
1640 }
1641 }
1642 }
1643 if (m_uncommittedGpgId) {
1644 const UncommittedGpgId uncommitted = *m_uncommittedGpgId;
1645 m_uncommittedGpgId.reset();
1646 reportUncommittedGpgId(uncommitted.folder, uncommitted.gitError,
1647 !result.cancelled && !result.aborted &&
1648 result.failed.isEmpty());
1649 }
1650 m_writtenGpgId.reset();
1651 // No push after a cancel or abort; otherwise one asked for meanwhile is
1652 // still to make (by a run to come, or once a pending move settles).
1653 if (result.cancelled || result.aborted) {
1654 m_pushAfterReencrypt = false;
1655 m_deferredPushByHand = false;
1656 }
1657 m_reencryptActive = false;
1658 emit endReencryptPath();
1659 startMovedReencrypts();
1660}

1707auto ImitatePass::resolveMoveDestination(const QString &src,
1708 const QString &dest, bool force)
1709 -> QString {
1710 QFileInfo srcFileInfo(src);
1711 QFileInfo destFileInfo(dest);
1712 QString destFile;
1713 QString srcFileBaseName = srcFileInfo.fileName();
1714
1715 if (srcFileInfo.isFile()) {
1716 if (destFileInfo.isFile()) {
1717 if (!force) {
1718 qCDebug(lcQtPass) << "Destination file already exists";
1719 return {};
1720 }
1721 destFile = dest;
1722 } else if (destFileInfo.isDir()) {
1723 destFile = QDir(dest).filePath(srcFileBaseName);
1724 } else {
1725 destFile = dest;
1726 }
1727
1728 // Case-sensitive: only an actual ".gpg" entry already carries the suffix.
1729 // A neighbouring file like "login.GPG" is a different name, so it must not
1730 // be chopped to "login" and re-suffixed to the real "login.gpg" entry,
1731 // which the forced move would then overwrite.
1732 if (destFile.endsWith(QLatin1String(".gpg"))) {
1733 destFile.chop(4);
1734 }
1735 destFile.append(".gpg");
1736 } else if (srcFileInfo.isDir()) {
1737 if (destFileInfo.isDir()) {
1738 destFile = QDir(dest).filePath(srcFileBaseName);
1739 } else if (destFileInfo.isFile()) {
1740 qCDebug(lcQtPass) << "Destination is a file";
1741 return {};
1742 } else {
1743 destFile = dest;
1744 }
1745 } else {
1746 qCDebug(lcQtPass) << "Source file does not exist";
1747 return {};
1748 }
1749 return destFile;
1750}

1782void ImitatePass::Move(const QString src, const QString dest,
1783 const bool force) {
1784 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1785 return;
1786 }
1787 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
1788 return;
1789 }
1790 // No message box opens inside the transaction: a process queued while one
1791 // is open (an entry opened from a search another instance forwarded) would
1792 // be booked as a step of the move, and its decrypted output reported as
1793 // the move's, which is not secret. So every check runs first, and the
1794 // transaction holds only the Git commands.
1795 const QString destFile = resolveMoveDestination(src, dest, force);
1796 if (destFile.isEmpty()) {
1797 return;
1798 }
1799 if (refuseLinkedPath(destFile)) {
1800 return;
1801 }
1802
1803 if (refuseOverlappingMove(src, destFile)) {
1804 return;
1805 }
1806
1807 qCDebug(lcQtPass) << "Move Source: " << src;
1808 qCDebug(lcQtPass) << "Move Destination: " << destFile;
1809
1810 const bool folder = QFileInfo(src).isDir();
1811 PendingMove move{src, destFile,
1812 QDir::cleanPath(getGpgIdPath(src, m_settings.passStore)),
1813 std::nullopt};
1814
1815 if (gitReady()) {
1816 // Settled in finished() once the move's commands have run: only then
1817 // is it known whether git mv moved anything, and which list applies
1818 // where it landed (a move queued before this one may create that
1819 // folder). Every Move() through Git ends in one PASS_MOVE.
1820 m_pendingMoves.enqueue(move);
1821 TransactionHelper trans(&m_transaction, PASS_MOVE);
1822 executeMoveGit(src, destFile, force);
1823 return;
1824 }
1825 // A forced move replaces the destination entry in the one rename, as
1826 // git mv -f does: removing it first lost it when the rename then failed
1827 // (and lost the entry itself, moved onto its own name). A folder is never
1828 // replaced, forced or not. Without Git nothing else reports a failure.
1829 const bool moved = force && !folder ? Util::replaceFile(src, destFile, true)
1830 : QDir().rename(src, destFile);
1831 if (!moved) {
1832 emit critical(tr("Move failed"),
1833 tr("Could not move %1 to %2.").arg(src, destFile));
1834 return;
1835 }
1836 move.landed = true;
1837 reencryptMoved(move);
1838 startMovedReencrypts();
1839}

1841auto ImitatePass::refuseOverlappingMove(const QString &src,
1842 const QString &destFile) -> bool {
1843#ifdef Q_OS_WIN
1844 constexpr auto cs = Qt::CaseInsensitive;
1845#else
1846 constexpr auto cs = Qt::CaseSensitive;
1847#endif
1848 // The same path, or one inside the other.
1849 const auto overlap = [](const QString &a, const QString &b) {
1850 const QString x = QDir::cleanPath(a);
1851 const QString y = QDir::cleanPath(b);
1852 return x.compare(y, cs) == 0 || x.startsWith(y + QLatin1Char('/'), cs) ||
1853 y.startsWith(x + QLatin1Char('/'), cs);
1854 };
1855 QStringList busy = m_movedToReencrypt;
1856 for (const PendingMove &move : std::as_const(m_pendingMoves)) {
1857 busy << move.src << move.destFile;
1858 }
1859 if (m_reencryptActive) {
1860 busy << m_reencryptScope;
1861 }
1862 for (const QString &path : std::as_const(busy)) {
1863 if (overlap(path, src) || overlap(path, destFile)) {
1864 emit critical(tr("Move not started"),
1865 tr("%1 is still being moved or re-encrypted. Move it "
1866 "again once that has finished.")
1867 .arg(QDir::toNativeSeparators(QDir::cleanPath(path))));
1868 return true;
1869 }
1870 }
1871 return false;
1872}

1874void ImitatePass::reencryptMoved(const PendingMove &move) {
1875 // Only a move whose git mv (or rename) succeeded: one that failed, or
1876 // never ran, moved nothing, and re-encrypting the destination would
1877 // rewrite the entry that was there. Not the source name: something may
1878 // be written under it again before the move settles.
1879 if (move.landed != true) {
1880 return;
1881 }
1882 const QFileInfo landed(move.destFile);
1883 if (!landed.exists() || Util::isLinkedFolder(move.destFile)) {
1884 return;
1885 }
1886 // As `pass mv`, what moved is encrypted for the recipients where it
1887 // landed. Not needed when the same list applies there (a rename, a move
1888 // within its folder's reach) or a folder took its own list along.
1889 const QString toList =
1890 QDir::cleanPath(getGpgIdPath(move.destFile, m_settings.passStore));
1891 if (toList == move.fromList ||
1892 (landed.isDir() &&
1893 toList == QDir::cleanPath(QDir(move.destFile).filePath(".gpg-id")))) {
1894 return;
1895 }
1896 // The entry alone, or the folder: not its neighbours, which this move
1897 // did not touch. One run at a time: behind the one in flight, if any.
1898 const QString scope = QDir::cleanPath(move.destFile);
1899 if (!m_movedToReencrypt.contains(scope)) {
1900 m_movedToReencrypt << scope;
1901 }
1902}

1904auto ImitatePass::copyDestination(const QString &src, const QString &dest,
1905 bool force) -> QString {
1906 const auto refuse = [this, &src](const QString &to) {
1907 emit critical(tr("Copy failed"),
1908 tr("Could not copy %1 to %2.").arg(src, to));
1909 return QString();
1910 };
1911 // Like `pass cp`, dest may be an existing folder (a drag-and-drop copy hands
1912 // over the folder, not the new file name). Resolve the real target the same
1913 // way Move does: into the folder, .gpg appended, no clobbering without force.
1914 const QString destFile = resolveMoveDestination(src, dest, force);
1915 if (destFile.isEmpty()) {
1916 return refuse(dest);
1917 }
1918 // A link planted under that name (dangling ones pass exists()) is not an
1919 // entry to write.
1920 if (refuseLinkedPath(destFile)) {
1921 return {};
1922 }
1923 const QFileInfo destFileInfo(destFile);
1924 // A folder destination that is the source's own folder resolves to the
1925 // source itself; with force that would replace the only copy with itself.
1926 // And resolveMoveDestination only sees a clash when dest names the file.
1927 if (QFileInfo(src) == destFileInfo || (!force && destFileInfo.exists())) {
1928 return refuse(destFile);
1929 }
1930 return destFile;
1931}

1933void ImitatePass::Copy(const QString src, const QString dest,
1934 const bool force) {
1935 // QFile::copy reads through a link: the target's bytes would become an
1936 // entry of the store.
1937 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1938 return;
1939 }
1940 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
1941 return;
1942 }
1943 // As in Move(): no message box inside the transaction.
1944 const QString destFile = copyDestination(src, dest, force);
1945 if (destFile.isEmpty()) {
1946 return;
1947 }
1948 // git has no "cp": copy on disk in both modes, then stage. Synchronous and
1949 // atomic (Util::copyFileReplacing follows no link), so it exists before the
1950 // re-encryption below and a forced overwrite survives a half-way failure;
1951 // without force nothing is replaced, not even what appeared since the check.
1952 QString why;
1953 if (!Util::copyFileReplacing(src, destFile, force, &why)) {
1954 emit critical(tr("Copy failed"),
1955 tr("Could not copy %1 to %2.").arg(src, destFile) + "\n" +
1956 why);
1957 return;
1958 }
1959 if (gitReady()) {
1960 TransactionHelper trans(&m_transaction, PASS_COPY);
1961 executeGit(GIT_COPY, {"add", "--", pgit(destFile)});
1962 gitCommit(destFile, QString("Copied from %1 to %2 using QtPass.")
1963 .arg(commitName(src), commitName(destFile)));
1964 }
1965 const QFileInfo written(destFile);
1966 if (written.isDir()) {
1967 reencryptPath(written.absoluteFilePath());
1968 } else if (written.isFile()) {
1969 reencryptPath(written.dir().path());
1970 }
1971}

2003// Only PASS_* processes reach Pass::finished, so to the interface this looks
2004// the same as RealPass.
2005void ImitatePass::finished(int id, int exitCode, const QString &out,
2006 const QString &err) {
2007 qCDebug(lcQtPass) << "Imitate Pass";
2008 QString error = err;
2009 // A move's git mv runs first in its transaction, so in order: the first
2010 // pending move without an answer is this one's.
2011 if (id == GIT_MOVE) {
2012 for (PendingMove &move : m_pendingMoves) {
2013 if (!move.landed) {
2014 move.landed = exitCode == 0;
2015 break;
2016 }
2017 }
2018 }
2019 if (id == PASS_INSERT && !m_pendingInserts.isEmpty()) {
2020 // Insert()'s gpg step: place the ciphertext before the queued git steps
2021 // run (the executor starts the next item only after this returns). On
2022 // failure it fails like a gpg error: git steps cancelled below.
2023 // No dialog here: it would spin the event loop and run git steps first.
2024 const PendingInsert pending = m_pendingInserts.dequeue();
2025 // The flag answers for this insert, the one whose gpg step just ended, not
2026 // for whichever Insert() was called last: with two inserts queued, a gpg
2027 // failure of the second must report written=false even though the first
2028 // was placed. Cleared here, before placement, and set only on success.
2029 m_insertPlaced = false;
2030 if (exitCode == 0) {
2031 // Insert() checked the repository when the save was requested, but a
2032 // pull queued before it (the asynchronous Update button) may have run
2033 // and stopped in a conflict while this gpg step encrypted. Placing the
2034 // entry and committing it now would land it in the unfinished operation,
2035 // where `git rebase --abort` would drop it. Re-check and, if so, fail the
2036 // save instead of placing it; the queued git steps are cancelled below.
2037 const QString midOp = unfinishedGitOperationReason();
2038 if (!midOp.isEmpty()) {
2039 error = midOp;
2040 exitCode = 1;
2041 } else {
2042 m_insertPlaced = placeEncryptedFile(pending.output, pending.file,
2043 pending.overwrite, &error);
2044 if (!m_insertPlaced) {
2045 exitCode = 1;
2046 }
2047 }
2048 }
2049 }
2050 PROCESS pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
2051 m_transactionOutput.append(out);
2052
2053 if (exitCode == 0) {
2054 if (pid == INVALID) {
2055 return;
2056 }
2057 } else {
2058 while (pid == INVALID) {
2059 id = exec.cancelNext();
2060 if (id == -1) {
2061 // this is probably irrecoverable and shall not happen
2062 qCDebug(lcQtPass) << "No such transaction!";
2063 return;
2064 }
2065 pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
2066 }
2067 }
2068 Pass::finished(pid, exitCode, m_transactionOutput, error);
2069 m_transactionOutput.clear();
2070 if (pid == PASS_MOVE && !m_pendingMoves.isEmpty()) {
2071 reencryptMoved(m_pendingMoves.dequeue());
2072 startMovedReencrypts();
2073 }
2074}
