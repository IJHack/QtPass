67// tells them apart. Only the worker touches its QProcess, so a cancel can

69auto ImitatePass::prepareBlocking(QProcess &process)
70 -> const std::atomic_bool * {
71 // In the store's environment: the git ceiling QtPass's own does not
72 // carry.
73 if (QThread::currentThread() == thread()) {
74 process.setProcessEnvironment(storeEnvironment());
75 return nullptr;
76 }
77 process.setProcessEnvironment(withCurrentGitPin(m_reencryptEnv));
78 return &m_reencryptCancel;
79}

141void ImitatePass::GitPull_b(int timeoutMs) {
142 if (!gitReady())
143 return;
144 // -C the store: executeBlocking sets no working directory, so without it
145 // git would run in QtPass's launch directory and pull an unrelated
146 // repository, or fail with "not a git repository".
147 QString err;
148 QProcess process;
149 const std::atomic_bool *cancel = prepareBlocking(process);
150 QElapsedTimer timer;
151 timer.start();
152 const int rc =
153 Executor::executeBlocking(process, m_settings.gitExecutable,
154 {"-C", pgit(m_settings.passStore), "pull"},
155 QString(), nullptr, &err, cancel, timeoutMs);
156 if (rc != 0 && timer.elapsed() >= timeoutMs) {
157 emit statusMsg(pullTimedOut(timeoutMs), 5000);
158 } else if (rc != 0) {
159 emit statusMsg(tr("Git pull failed: %1").arg(err.trimmed()), 5000);

180 if (!m_pushAfterReencrypt || m_reencryptActive || !m_pendingMoves.isEmpty() ||
181 !m_movedToReencrypt.isEmpty()) {
182 return;

196void ImitatePass::startMovedReencrypts() {
197 // A run refused at its start (reencryptPath() said why) leaves the next.
198 while (!m_reencryptActive && !m_movedToReencrypt.isEmpty()) {
199 const QString scope = m_movedToReencrypt.takeFirst();
200 reencryptPath(scope);
201 if (!m_reencryptActive) {
202 // What moved is still encrypted for the recipients it came from: the
203 // automatic push after the move waits, as after a run that did not
204 // finish. A push by hand still goes.
205 holdAutoPush(scope);
206 }
207 }
208 sendDeferredPush();
209}

271void ImitatePass::Insert(QString file, QString newValue, bool overwrite) {
272 // The dialog names a new entry relative to the store; gpg used to resolve
273 // that in its working directory. Everything below works on the one path.
274 file =
275 QDir::cleanPath(QDir::isAbsolutePath(file)
276 ? file + ".gpg"
277 : QDir(m_settings.passStore).filePath(file) + ".gpg");
278 if (refuseLinkedPath(file)) {
279 return;
280 }
281 if (refuseUnfinishedGitOperation()) {
282 return;
283 }
284 // gpg used to refuse an existing output without --yes; it writes
285 // elsewhere now, so the check is ours here, and again at the rename.
286 const QFileInfo target(file);
287 if (!overwrite && (target.exists() || target.isSymLink())) {
288 emit critical(tr("Cannot add"), tr("%1 already exists.").arg(file));
289 return;
290 }
291 QStringList recipients;
292 if (!recipientsForEntry(file, &recipients)) {
293 return;
294 }
295 // gpg never opens a store path for output: a co-writer could make it a link
296 // after refuseLinkedPath(). gpg writes into a 0700 directory of our own and
297 // placeEncryptedFile() brings the bytes in via an open handle and a rename.
298 auto scratch = std::make_shared<QTemporaryDir>();
299 if (!scratch->isValid()) {
300 emit critical(tr("Cannot write"),
301 tr("Cannot create a temporary directory: %1")
302 .arg(scratch->errorString()));
303 return;
304 }
305 const QString output = scratch->filePath(QStringLiteral("entry.gpg"));
306 TransactionHelper trans(&m_transaction, PASS_INSERT);
307 const QStringList args =
308 encryptArgs({"--batch", "--status-fd", "2"}, pgpg(output), recipients);
309 m_pendingInserts.enqueue({std::move(scratch), output, file, overwrite});
310 // Past every refusal: this insert is now queued and will report once.
311 queueInsert();
312 executeGpg(PASS_INSERT, args, newValue);
313 if (!gitReady()) {
314 return;
315 }
316 if (!overwrite) {
317 executeGit(GIT_ADD, {"add", "--", pgit(file)});
318 }
319 gitCommit(file, QString(overwrite ? "Edit" : "Add") + " for " +
320 commitName(file) + " using QtPass.");
321}

348void ImitatePass::Remove(QString file, bool isDir) {
349 // No trailing separator: "link/" makes git look for what is behind the
350 // link ("pathspec did not match") and rm follow it.
351 file = QDir::cleanPath(m_settings.passStore + file);
352 if (!isDir) {
353 file += ".gpg";
354 }
355 // A link itself may go; anything behind one is not the store's to delete.
356 if (refuseLinkedPath(file, LinkScope::ParentsOnly)) {
357 return;
358 }
359 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
360 return;
361 }
362 // As in Move(): no message box inside the transaction, so it holds only
363 // the Git commands.
364 const QString path = commitName(file);
365 if (Util::isLinkedFolder(file)) {
366 // Unlink it here, never through a recursive rm: the link goes, what it
367 // points to stays. Git then only has to forget it, if it knew it.
368 if (!Util::removeTree(file)) {
369 emit critical(tr("Delete failed"),
370 tr("Could not remove the link %1.").arg(file));
371 return;
372 }
373 // Only when git knew it: a commit matching nothing (a link never
374 // committed) exits 1 and reads as a failed removal. ls-files reads the
375 // index, so it still answers now that the link is gone.
376 if (gitReady() && gitTracks(file)) {
377 TransactionHelper trans(&m_transaction, PASS_REMOVE);
378 executeGit(GIT_RM, {"rm", "-q", "--cached", "--", pgit(file)});
379 gitCommit(file, "Remove for " + path + " using QtPass.");
380 }
381 return;
382 }
383 if (gitReady()) {
384 TransactionHelper trans(&m_transaction, PASS_REMOVE);
385 executeGit(GIT_RM, {"rm", (isDir ? "-rf" : "-f"), "--", pgit(file)});
386 gitCommit(file, "Remove for " + path + " using QtPass.");
387 } else {
388 // Without Git nothing else reports a failure: git rm would print its own.
389 if (isDir) {
390 // Never QDir::removeRecursively(): it follows a junction inside the
391 // folder and empties whatever that points to.
392 if (!Util::removeTree(file)) {
393 // As rm -rf, it went on past what it could not remove: the folder
394 // is still there, without the entries that could go.
395 emit critical(tr("Delete failed"),
396 tr("Could not remove all of %1. What could be removed "
397 "is gone; check what is left in it.")
398 .arg(file));
399 }
400 } else {
401 QFile entry(file);
402 if (!entry.remove()) {
403 emit critical(
404 tr("Delete failed"),
405 tr("Could not remove %1: %2").arg(file, entry.errorString()));
406 }
407 }
408 }

420auto ImitatePass::addGenerationHeader(const QString &gpgIdFile,
421 const GpgIdSigner &signer,
422 QByteArray *contents) -> bool {
423 const std::optional<QString> folder =
424 GpgIdGeneration::folderOf(gpgIdFile, m_settings.passStore);
425 if (!folder) {
426 emit critical(tr("Cannot update"),
427 tr("%1 is not inside the password store, or a folder "
428 "name on the way holds a line break or another control "
429 "character, which a signed list cannot record.")
430 .arg(gpgIdFile));
431 return false;
432 }
433 // The list on disk counts only if it is this folder's own, verified list:
434 // a signature vouches for the bytes, the folder line for the place.
435 std::optional<qint64> verifiedOnDisk;
436 std::optional<QString> storeOnDisk;
437 QByteArray current;
438 if (signer.verifyFile(gpgIdFile, &current)) {
439 const auto header = GpgIdGeneration::parse(current);
440 if (header && (!header->folder || *header->folder == *folder)) {
441 verifiedOnDisk = header->generation;
442 // The store's id comes only from a list this store accepts. A refused
443 // one (a rollback, a conflict) may be another store's, signed with
444 // the same key and copied in: taking its id over would make the list
445 // saved now one of that store too. Its generation still counts, so the
446 // next one is above it. Without the record, which it is cannot be
447 // told: a store id made up now could split the store from devices
448 // that know its own.
449 QString unavailable;
450 const GpgIdGeneration::Verdict verdict = GpgIdGeneration::accept(
451 gpgIdFile, current, m_settings.passStore, &unavailable);
452 if (verdict == GpgIdGeneration::Verdict::RecordUnavailable) {
453 emit critical(tr("Cannot update"), unavailable);
454 return false;
455 }
456 if (verdict == GpgIdGeneration::Verdict::Accepted) {
457 storeOnDisk = header->store;
458 }
459 }
460 }
461 QString why;
462 // The store's id: the one this device knows, else the one this store's
463 // lists already carry (#2050).
464 const std::optional<QString> store =
465 GpgIdGeneration::storeIdFor(m_settings.passStore, storeOnDisk, &why);
466 if (!store) {
467 emit critical(tr("Cannot update"), why);
468 return false;
469 }
470 const std::optional<qint64> generation =
471 GpgIdGeneration::reserveNext(gpgIdFile, verifiedOnDisk, &why);
472 if (!generation) {
473 emit critical(tr("Cannot update"), why);
474 return false;
475 }
476 *contents =
477 GpgIdGeneration::withHeader(*generation, *folder, *contents, *store);
478 return true;
479}

481auto ImitatePass::writeGpgIdFile(const QString &gpgIdFile,
482 const QList<UserInfo> &users,
483 QByteArray *written) -> bool {
484 QByteArray contents;
485 bool secret_selected = false;
486 for (const UserInfo &user : users) {
487 if (user.enabled) {
488 contents += (user.key_id + "\n").toUtf8();
489 secret_selected |= user.have_secret;
490 }
491 }
492 // Without a signing key the plain list stays (GpgIdGeneration).
493 const GpgIdSigner signer = gpgIdSigner();
494 if (signer.enabled() && !addGenerationHeader(gpgIdFile, signer, &contents)) {
495 return false;
496 }
497 // Whole or not at all, owner-only (it names the store's keys); a link
498 // planted since Init's check is replaced, not written through.
499 QString writeError;
500 if (!Util::writeFileReplacing(gpgIdFile, contents, true, &writeError)) {
501 emit critical(tr("Cannot update"), writeError);
502 return false;
503 }
504 if (written != nullptr) {
505 *written = contents;
506 }
507 if (signer.enabled()) {
508 // From now on these bytes are this generation. The list is signed either
509 // way (unsigned is refused everywhere); if the record could not take them
510 // (another writer, record busy), the user is told to save once more.
511 QString why;
512 if (!GpgIdGeneration::recordWritten(gpgIdFile, contents, &why)) {
513 qCWarning(lcQtPass) << "Could not record the written .gpg-id:" << why;
514 emit critical(
515 tr("Recipient list written, but not recorded"),
516 tr("%1 Save the recipients once more to get through.").arg(why));
517 }
518 }
519 if (!secret_selected) {
520 emit critical(
521 tr("Check selected users!"),
522 tr("None of the selected keys have a secret key available.\n"
523 "You will not be able to decrypt any newly added passwords!"));
524 }
525 return true;
526}

528auto ImitatePass::signGpgIdFile(const QString &gpgIdFile,
529 const QByteArray &contents) -> bool {
530 const GpgIdSigner signer = gpgIdSigner();
531 QString why;
532 if (!signer.sign(gpgIdFile, contents, &why)) {
533 emit critical(
534 tr("GPG signing failed!"),
535 why.trimmed().isEmpty()
536 ? tr("Failed to sign %1.").arg(gpgIdFile)
537 : tr("Failed to sign %1: %2").arg(gpgIdFile, why.trimmed()));
538 return false;
539 }
540 QByteArray signedBytes;
541 if (!signer.verifyFile(gpgIdFile, &signedBytes)) {
542 emit critical(tr("Check .gpg-id file signature!"),
543 tr("Signature for %1 is invalid.").arg(gpgIdFile));
544 return false;
545 }
546 return true;
547}

549// Synchronous on purpose: queued on the async executor, the add/commit raced
550// reencryptPath()'s blocking backup commit for index.lock and either left the
551// .gpg-id untracked or found nothing to commit.
552auto ImitatePass::gitAddGpgId(const QString &gpgIdFile,
553 const QString &gpgIdSigFile, QString *out,
554 QString *err) -> int {
555 const QString git = m_settings.gitExecutable;
556 const QString store = pgit(m_settings.passStore);
557 auto run = [&](const QStringList &args) -> int {
558 QString runOut;
559 QString runErr;
560 const int rc =
561 execBlocking(git, QStringList{"-C", store} + args, &runOut, &runErr);
562 if (out != nullptr) {
563 out->append(runOut);
564 }
565 if (err != nullptr) {
566 err->append(runErr);
567 }
568 return rc;
569 };
570 QStringList paths{pgit(gpgIdFile)};
571 if (!gpgIdSigFile.isEmpty()) {
572 paths << pgit(gpgIdSigFile);
573 }
574 // add stages new and modified files alike; already-clean ones are a no-op.
575 int rc = run(QStringList{"add", "--"} + paths);
576 if (rc != 0) {
577 return rc;
578 }
579 // Re-initialising with the same recipients changes nothing; that is not a
580 // failure, and `git commit` would make it one. diff --quiet: 0 nothing
581 // staged, 1 something staged, anything else is git failing.
582 rc = run(QStringList{"diff", "--cached", "--quiet", "--"} + paths);
583 if (rc == 0) {
584 return 0;
585 }
586 if (rc != 1) {
587 return rc;
588 }
589 return run(commitArgs() +
590 QStringList{"-m",
591 "Added " + commitName(gpgIdFile) + " using QtPass.",
592 "--"} +
593 paths);
594}

602auto ImitatePass::refuseLinkedGpgIdFolder(const QString &path) -> bool {
603 // A link planted later is replaced, not written through: the writes stage
604 // and rename.
605 const QString folder = QDir::cleanPath(path);
606 // Init() places a folder spelled through a linked root's target itself.
607 const auto refusePlaced = [this](const QString &placed) {
608 return refuseLinkedPath(placed, LinkScope::EntryAndParents,
609 PathSpelling::PlacedByCaller);
610 };
611 return refusePlaced(path) ||
612 refusePlaced(folder + QStringLiteral("/.gpg-id")) ||
613 refusePlaced(folder + QStringLiteral("/.gpg-id.sig"));

616auto ImitatePass::settleSignature(const QString &gpgIdFile,
617 const QByteArray &written,
618 const GpgIdSigner &signer, bool useGit,
619 QString *sigToCommit) -> bool {
620 const QString gpgIdSigFile = gpgIdFile + ".sig";
621 if (signer.enabled()) {
622 if (!signGpgIdFile(gpgIdFile, written)) {
623 return false;
624 }
625 *sigToCommit = gpgIdSigFile;
626 return true;
627 }
628 if (!QFile::exists(gpgIdSigFile)) {
629 return true;
630 }
631 // Signing was switched off; the removal goes into the same commit.
632 const bool tracked = useGit && gitTracks(gpgIdSigFile);
633 if (!QFile::remove(gpgIdSigFile)) {
634 emit critical(
635 tr("Cannot update"),
636 tr("Failed to remove the old signature %1.").arg(gpgIdSigFile));
637 return false;
638 }
639 if (tracked) {
640 *sigToCommit = gpgIdSigFile;
641 }
642 return true;
643}

645void ImitatePass::Init(QString path, const QList<UserInfo> &users) {
646 // The .gpg-id is written as path + ".gpg-id": without the trailing
647 // separator (the context menu hands over a cleaned path) that would be a
648 // file beside the folder, not the folder's own list. A relative path names
649 // a folder in the store, as it does for pass, not one in the working
650 // directory.
651 if (!QDir::isAbsolutePath(path)) {
652 path = QDir(m_settings.passStore).filePath(path);
653 }
654 // Cleaned before any check: the checks clean what they look at, and the
655 // write must go to the same place. Left as /store/link/../team/, the
656 // system resolves link first and writes beside wherever it points.
657 path = Util::normalizeFolderPath(QDir::cleanPath(path));
658 QString inStore;
659 if (refuseLinkedGpgIdFolder(path) ||
660 refuseFolderOutsideStore(path, &inStore)) {
661 return;
662 }
663 if (refuseUnfinishedGitOperation()) {
664 return;
665 }
666 // From here on in the store's own spelling, the same folder: one spelled
667 // through the target of a linked store root is not under the store as
668 // named, so each entry's nearest list was looked up past the new one and
669 // the entries were encrypted to the store root's.
670 path = Util::normalizeFolderPath(
671 QDir::cleanPath(QDir(m_settings.passStore).filePath(inStore)));
672 // A running re-encryption cannot take a second folder: the new list would
673 // be written and its entries left encrypted to the old recipients, which
674 // for a removed person means they can still read them. Nothing is written.
675 if (m_reencryptActive) {
676 emit critical(tr("Recipients not changed"),
677 tr("A re-encryption is still running. Wait until it has "
678 "finished, then save the recipients again."));
679 return;
680 }
681 const GpgIdSigner signer = gpgIdSigner();
682 if (signer.enabled() && !signer.haveSecretKey()) {
683 emit critical(tr("No signing key!"),
684 tr("None of the secret signing keys is available.\n"
685 "You will not be able to change the user list!"));
686 return;
687 }
688
689 const bool useGit = gitReady();
690 const QString gpgIdFile = path + ".gpg-id";
691 // Each name free or a regular file, before anything is written: a folder
692 // named .gpg-id.sig stopped the change after the new list was on disk,
693 // which was then neither committed nor were the entries re-encrypted to
694 // it, while the dialog had closed as if done. Links were refused above.
695 for (const QString &name : {gpgIdFile, gpgIdFile + QStringLiteral(".sig")}) {
696 if (const QFileInfo info(name); info.exists() && !info.isFile()) {
697 emit critical(tr("Recipients not changed"),
698 tr("%1 is not a regular file, so the recipients were not "
699 "changed. Move it out of the way, then save the "
700 "recipients again.")
701 .arg(QDir::toNativeSeparators(name)));
702 return;
703 }
704 }
705 QByteArray written;
706 QString sigToCommit;
707 if (!writeGpgIdFile(gpgIdFile, users, &written)) {
708 return;
709 }
710 // The new list is on disk: held from here, so a stop before the run (a
711 // signature that fails, a refused start) leaves it out of a later backup
712 // commit's push; a clean run of this folder releases it.
713 holdAutoPush(path);
714 if (!settleSignature(gpgIdFile, written, signer, useGit, &sigToCommit)) {
715 return;
716 }
717
718 QString gitOut;
719 QString gitErr;
720 if (useGit && m_settings.addGPGId) {
721 // Commit the .gpg-id (and .sig) before re-encrypting to it: the backup
722 // commit only takes tracked files (#1685) and addFolder does not stage
723 // it, so entries were pushed without their recipients file (#1682).
724 const int gitExit = gitAddGpgId(gpgIdFile, sigToCommit, &gitOut, &gitErr);
725 if (gitExit != 0) {
726 Pass::finished(PASS_INIT, gitExit, gitOut, gitErr);
727 // The new list is on disk either way. Stopping here left every entry
728 // readable by whoever was just removed from it, so re-encrypt without
729 // Git and say the change is not committed. Putting the old list back
730 // is no way out: with signing the generation record has moved on and
731 // would refuse it as a rollback.
732 const QString why = gitErr.trimmed();
733 reencryptWithoutGit(
734 path,
735 why.isEmpty() ? tr("Git exited with code %1.").arg(gitExit) : why);
736 return;
737 }
738 }
739 m_writtenGpgId = WrittenGpgId{gpgIdFile, written};
740 reencryptPath(path);
741 if (!m_reencryptActive) {
742 // Refused before it started; reencryptPath() said why.
743 m_writtenGpgId.reset();
744 }
745 // finishedInit once the .gpg-id landed in git, as the async path did.
746 if (useGit) {
747 Pass::finished(PASS_INIT, 0, gitOut, gitErr);
748 }
749}

751auto ImitatePass::loadVerifiedRecipients(const QString &gpgIdFile,
752 QStringList *recipients, QString *why)
753 -> bool {
754 recipients->clear();
755 if (why != nullptr) {
756 *why = tr("Signature for %1 is invalid.").arg(gpgIdFile);
757 }
758 // A link under the .gpg-id name is not the store's list: with signing
759 // off it reads as missing, with signing on verifyFile() refuses it.
760 if (Util::isLinkedFolder(gpgIdFile)) {
761 return !gpgIdSigner().enabled();
762 }
763 QByteArray contents;
764 const GpgIdSigner signer = gpgIdSigner();
765 if (signer.enabled()) {
766 if (!verifySignedList(signer, gpgIdFile, &contents, why)) {
767 return false;
768 }
769 } else {
770 if (const QFileInfo info(gpgIdFile); !info.exists() && !info.isSymLink()) {
771 // No file is not a bad signature; the empty list says it all.
772 return true;
773 }
774 // Bounded, and only a regular file: see Util::readRegularFile().
775 std::optional<QByteArray> bytes =
776 Util::readRegularFile(gpgIdFile, Util::kMaxGpgIdBytes);
777 if (!bytes) {
778 if (why != nullptr) {
779 *why = tr("%1 cannot be read as a recipient list (a regular file "
780 "of at most %2).")
781 .arg(gpgIdFile,
782 QLocale().formattedDataSize(Util::kMaxGpgIdBytes));
783 }
784 return false;
785 }
786 contents = *std::move(bytes);
787 }
788 *recipients = Pass::parseRecipients(contents, gpgIdFile);
789 return true;
790}

842auto ImitatePass::recoverReencryptLeftovers(const QString &dir) -> bool {
843 // .qtpass-XXXXXX.tmp is today's staging name; 1.8.x replaced entries via
844 // X.gpg.reencrypt.tmp and .bak in two renames, builds up to 2.0 wrote
845 // X.gpg.XXXXXX.tmp. A .tmp is never a source of truth; a .bak is the only
846 // copy when X.gpg is missing, and one of two valid ones when it is not.
847 //
848 // Regular files only: renaming a link under a leftover name into an entry's
849 // place would re-encrypt whatever it points to. Links come back in
850 // `skipped`; linked directories are left to reencryptFiles() to mention.
851 const QStringList leftoverNames{QStringLiteral(".qtpass-??????.tmp"),
852 QStringLiteral("*.gpg.reencrypt.tmp"),
853 QStringLiteral("*.gpg.??????.tmp"),
854 QStringLiteral("*.gpg.reencrypt.bak")};
855 const auto isTemporary = [](const QString &path) {
856 return path.endsWith(QStringLiteral(".tmp"));
857 };
858 QStringList skipped;
859 // Hidden files included: the staged name starts with a dot.
860 const QStringList leftovers = Util::regularFilesUnder(
861 QDir::cleanPath(dir), leftoverNames, &skipped, true);
862 // One report for all of them: a box per leftover opened each from inside
863 // the one before (a modal box runs the next queued critical()), and a
864 // store with thousands of them overflowed the stack.
865 QStringList problems;
866 for (const QString &path : skipped) {
867 if (!QDir::match(leftoverNames, QFileInfo(path).fileName())) {
868 continue;
869 }
870 // Removing a link removes the link, never what it points to; a junction
871 // or a directory symlink on Windows is a directory entry and goes with
872 // rmdir.
873 if (isTemporary(path)) {
874 if (!QFile::remove(path) && !QDir().rmdir(path)) {
875 qCWarning(lcQtPass) << "Could not remove stale temporary" << path;
876 }
877 continue;
878 }
879 reportUnrestorable(path, problems);
880 }
881 for (const QString &path : leftovers) {
882 if (isTemporary(path)) {
883 dropStaleTemporary(path);
884 } else {
885 restoreBackup(path, problems);
886 }
887 }
888 if (problems.isEmpty()) {
889 return true;
890 }
891 const qsizetype count = problems.size();
892 if (count == 1) {
893 emit critical(tr("Leftover from an earlier re-encryption"),
894 problems.first());
895 return false;
896 }
897 QStringList listed = problems.mid(0, kReencryptMaxListedFailures);
898 if (count > listed.size()) {
899 listed << tr("... and %n more", nullptr,
900 static_cast<int>(count - listed.size()));
901 }
902 emit critical(tr("Leftovers from an earlier re-encryption"),
903 tr("%n leftover(s) of an earlier re-encryption need a look "
904 "before the run can start:",
905 nullptr, static_cast<int>(count)) +
906 "\n\n" + listed.join("\n\n"));
907 return false;
908}

910auto ImitatePass::verifyGpgIdForDir(const QString &file,
911 QHash<QString, QStringList> &verified,
912 QStringList &gpgId) -> bool {
913 // Before the cache: two folders can resolve to the same list, and only
914 // one of them may have had its own.
915 if (refuseSkippedGpgId(file, gpgIdSigner().enabled())) {
916 return false;
917 }
918 const QString gpgIdPath = Pass::getGpgIdPath(file, m_settings.passStore);
919 // Keyed by .gpg-id, so every file under it gets exactly the list its
920 // signature covered.
921 const auto cached = verified.constFind(gpgIdPath);
922 if (cached != verified.constEnd()) {
923 gpgId = cached.value();
924 return true;
925 }
926 QStringList recipients;
927 QString why;
928 if (!loadVerifiedRecipients(gpgIdPath, &recipients, &why)) {
929 // An interrupted gpg is a cancel, not a bad signature.
930 if (!m_reencryptCancel.load())
931 emit critical(tr("Check .gpg-id file signature!"), why);
932 return false;
933 }
934 recipients.sort();
935 verified.insert(gpgIdPath, recipients);
936 gpgId = recipients;
937 return true;
938}

941 QStringList args = {
942 "-v", "--no-secmem-warning", "--no-permission-warning",

961 actualKeys << actualKey;

963 }
964 }
965 actualKeys.sort();
966 return actualKeys;

969auto ImitatePass::decryptEntry(const QString &fileName, QByteArray *plaintext,
970 QString *why) -> bool {
971 // Read here, without following a link, and handed to gpg on stdin: the
972 // name was listed as a regular file before the run, and a pull since may
973 // have put a link there. gpg given the name would follow it, and the
974 // file it points to would be re-encrypted for this folder's recipients.
975 // The cap is on what decrypts; the ciphertext of an entry just under it
976 // is a little larger (packet headers, one session key per recipient).
977 constexpr qint64 kCiphertextMargin = 1024LL * 1024;
978 // That read refuses a link only as the last part of the name; a folder
979 // on the way that became a link is checked here.
980 const std::optional<QByteArray> ciphertext =
981 Util::isUnderLink(fileName, m_settings.passStore)
982 ? std::nullopt
983 : Util::readRegularFile(fileName,
984 kMaxPlaintextBytes + kCiphertextMargin);
985 if (!ciphertext) {
986 qCDebug(lcQtPass) << "Not a regular file any more, or too large:"
987 << fileName;
988 if (why != nullptr) {
989 *why = tr("%1 is no longer a regular file, or is too large, and was "
990 "not re-encrypted.")
991 .arg(fileName);
992 }
993 return false;
994 }
995 const QStringList args =
996 QStringList{"-d", "--quiet", "--yes", "--no-encrypt-to",
997 "--batch", "--use-agent"} +
998 maxOutputArgs();
999 if (execBlockingRaw(m_settings.gpgExecutable, args, *ciphertext, plaintext) !=
1000 0 ||
1001 plaintext->isEmpty()) {
1002 qCDebug(lcQtPass) << "Decrypt error on re-encrypt for:" << fileName;
1003 return false;
1004 }
1005 // No newline added: an entry is re-encrypted as it was, as pass(1) does
1006 // (gpg -d | gpg -e); a binary key without one would have grown a byte.
1007 return true;
1008}

1023auto ImitatePass::ciphertextHolds(const QString &ciphertext,
1024 const QByteArray &plaintext) -> bool {
1025 QByteArray decrypted;
1026 const QStringList args =
1027 QStringList{"-d", "--quiet", "--batch", "--use-agent"} + maxOutputArgs() +
1028 QStringList{QStringLiteral("--"), pgpg(ciphertext)};
1029 if (execBlockingRaw(m_settings.gpgExecutable, args, QByteArray(),
1030 &decrypted) != 0 ||
1031 decrypted.isEmpty()) {
1032 qCDebug(lcQtPass) << "Verification failed for:" << ciphertext;
1033 return false;
1034 }

1036 // Exact: gpg returns the bytes it was given. Compared as text, the check
1037 // passed for an entry that had lost bytes, as both sides lost the same.
1038 if (decrypted != plaintext) {
1039 qCDebug(lcQtPass) << "Verification content mismatch for:" << ciphertext;
1040 return false;
1041 }
1042 return true;

1066 qCDebug(lcQtPass) << "git commit failed after re-encrypting:" << fileName;

1072auto ImitatePass::reencryptSingleFile(const QString &fileName,
1073 const QStringList &recipients,
1074 QString *why) -> bool {
1075 qCDebug(lcQtPass) << "reencrypt" << fileName << "for" << recipients.size()
1076 << "recipients";
1077 if (recipients.isEmpty()) {
1078 emit critical(tr("Can not edit"),
1079 tr("Could not read encryption key to use, .gpg-id "
1080 "file missing or invalid."));
1081 return false;
1082 }
1083 QByteArray plaintext;
1084 if (!decryptEntry(fileName, &plaintext, why)) {
1085 return false;
1086 }
1087
1088 // gpg writes outside the store, into a 0700 directory with an unguessable
1089 // name no co-writer can pre-create or link; placeEncryptedFile() then
1090 // replaces the entry in one rename, as in Insert().
1091 QTemporaryDir scratch;
1092 if (!scratch.isValid()) {
1093 qCDebug(lcQtPass) << "Cannot create a scratch directory for re-encrypting"
1094 << fileName;
1095 return false;
1096 }
1097 const QString tempPath = scratch.filePath(QStringLiteral("reencrypted.tmp"));
1098 if (!encryptFor(tempPath, recipients, plaintext) ||
1099 !ciphertextHolds(tempPath, plaintext)) {
1100 return false;
1101 }
1102
1103 // Another client may have removed the entry while gpg ran; the rename
1104 // would bring it back under its old name. Best effort: the check and the
1105 // replace are two steps, as pass's own write is.
1106 const QFileInfo still(fileName);
1107 if (!still.exists() && !still.isSymLink()) {
1108 qCDebug(lcQtPass) << "Entry vanished before it could be replaced:"
1109 << fileName;
1110 return false;
1111 }
1112 if (!placeEncryptedFile(tempPath, fileName, true, why)) {
1113 // Reported once, in the run's summary: an unwritable folder fails every
1114 // entry alike. The old ciphertext stays unless the post-rename check
1115 // failed, and then `why` says so.
1116 if (why != nullptr && !why->contains(fileName)) {
1117 // Every reason but the copy's names the entry; that one names the
1118 // file gpg wrote, in a scratch directory the user never sees.
1119 *why = tr("%1 could not be re-encrypted: %2").arg(fileName, *why);
1120 }
1121 return false;
1122 }
1123 return commitReencrypted(fileName);
1124}

1152auto ImitatePass::createBackupCommit() -> bool {
1153 if (!reencryptUsesGit()) {
1154 return true;
1155 }
1156 emit statusMsg(tr("Creating backup commit"), 2000);
1157 const QString git = m_settings.gitExecutable;
1158 // -C the store: executeBlocking sets no working directory, so git would
1159 // run in QtPass's launch directory, possibly an unrelated repository.
1160 const QString store = pgit(m_settings.passStore);
1161 // Tracked files only: an untracked plaintext export or swap file must not
1162 // be swept into a commit that autoPush sends to the shared remote.
1163 QString statusOut;
1164 if (execBlocking(
1165 git, {"-C", store, "status", "--porcelain", "--untracked-files=no"},
1166 &statusOut) != 0) {
1167 // An interrupted git is a cancel, not a failure worth a dialog.
1168 if (!m_reencryptCancel.load())
1169 emit critical(
1170 tr("Backup commit failed"),
1171 tr("Could not inspect git status. Re-encryption was aborted."));
1172 return false;
1173 }
1174 // An unresolved conflict needs no merge or rebase in progress: a
1175 // `git stash pop` that conflicted leaves none of the markers the earlier
1176 // checks look for. `add -u` would record the work tree, conflict markers
1177 // and all, as the resolution, and the backup commit would be pushed.
1178 // The repository-wide status, not `ls-files` (which stops at the store's
1179 // folder): `add -u` stages a repository above the store as a whole.
1180 if (listsUnmergedPath(statusOut)) {
1181 if (!m_reencryptCancel.load())
1182 emit critical(tr("Unfinished Git operation"),
1183 tr("The password store's Git repository has unmerged "
1184 "files. Resolve the conflict with Git before "
1185 "re-encrypting."));
1186 return false;
1187 }
1188 // The store only: in a repository above it, a modified tracked file
1189 // outside the store is not the store's to back up, and autoPush would
1190 // publish it. `.` is the store, as git runs with -C the store.
1191 QString storeStatus;
1192 if (execBlocking(git,
1193 {"-C", store, "status", "--porcelain",
1194 "--untracked-files=no", "--", "."},
1195 &storeStatus) != 0) {
1196 if (!m_reencryptCancel.load())
1197 emit critical(
1198 tr("Backup commit failed"),
1199 tr("Could not inspect git status. Re-encryption was aborted."));
1200 return false;
1201 }
1202 if (!storeStatus.trimmed().isEmpty()) {
1203 if (execBlocking(git, {"-C", store, "add", "-u", "--", "."}) != 0 ||
1204 execBlocking(git, QStringList{"-C", store} + commitArgs() +
1205 QStringList{"-m", "Backup before re-encryption",
1206 "--", "."}) != 0) {
1207 if (!m_reencryptCancel.load())
1208 emit critical(tr("Backup commit failed"),
1209 tr("Re-encryption was aborted because a git backup "
1210 "could not be created."));
1211 return false;
1212 }
1213 }
1214 return true;
1215}

1281// The worker's kill() after the grace period is needed: terminate() is only
1282// SIGTERM, or WM_CLOSE on Windows, which a console gpg ignores.

1285 return;

1288
1289// Init(), Move() and Copy() queue async git commands right before
1290// reencryptPath(); the worker's blocking git would fight them for the index
1291// lock, so wait for the queue to drain. The other backend's too: with pass
1292// as the backend an Update's `pass git pull` runs on RealPass's queue, and
1293// the run listed the store while that pull could still change it.
1294void ImitatePass::startReencryptWorker(const QString &dir) {
1295 if (m_reencryptCancel.load()) {
1296 ReencryptResult result;
1297 result.cancelled = true;
1298 finishReencrypt(result);
1299 return;
1300 }
1301 if (!exec.isIdle() || !PassBackendFactory::isIdle()) {
1302 QTimer::singleShot(kReencryptRetryMs, this,
1303 [this, dir]() { startReencryptWorker(dir); });
1304 return;
1305 }
1306 // Re-check now the queue is idle: a pull queued earlier through the
1307 // asynchronous GitPull() may have finished during the wait and left a
1308 // conflict. reencryptPath() checked before scheduling, but that was before
1309 // such a pull could run; starting the worker now would let its backup commit
1310 // conclude the stopped operation. The refusal already reported it, so finish
1311 // as an aborted run (clears m_reencryptActive, emits endReencryptPath).
1312 if (refuseUnfinishedGitOperation()) {
1313 ReencryptResult result;
1314 result.aborted = true;
1315 finishReencrypt(result);
1316 return;
1317 }
1318
1319 // The worker calls member functions, so `this` must outlive it: the
1320 // destructor cancels and joins m_reencryptThread. `self` only guards the
1321 // queued completion, which may run after the thread object is gone.
1322 QPointer<ImitatePass> self(this);
1323 // The worker's own copy: env is the owning thread's to change meanwhile.
1324 updateEnv();
1325 m_reencryptEnv = storeEnvironment();
1326 QThread *thread = QThread::create([this, self, dir]() {
1327 ReencryptResult result = reencryptFiles(dir);
1328 QMetaObject::invokeMethod(
1329 self,
1330 [self, result = std::move(result)]() {
1331 if (self)
1332 self->finishReencrypt(result);
1333 },
1334 Qt::QueuedConnection);
1335 });
1336 m_reencryptThread = thread;
1337 connect(thread, &QThread::finished, this, [this, thread]() {
1338 if (m_reencryptThread == thread)
1339 m_reencryptThread = nullptr;
1340 });
1341 connect(thread, &QThread::finished, thread, &QObject::deleteLater);
1342 thread->start();
1343}

1350auto ImitatePass::pullBeforeReencrypt() -> bool {
1351 if (!m_settings.autoPull || !reencryptUsesGit()) {
1352 return true;
1353 }
1354 emit statusMsg(tr("Updating password-store"), 2000);
1355 if (execBlocking(m_settings.gitExecutable,
1356 {"-C", pgit(m_settings.passStore), "pull"}) == 0) {
1357 return true;
1358 }
1359 // A pull that could not reach the remote leaves the store as it was; one
1360 // that stopped in a merge leaves conflict markers and an unmerged index,
1361 // and re-encrypting on top of that would commit the mess.
1362 QString unmerged;
1363 if (execBlocking(m_settings.gitExecutable,
1364 {"-C", pgit(m_settings.passStore), "ls-files", "--unmerged"},
1365 &unmerged) != 0) {
1366 // Unknown is not clean: the pull may have stopped in a merge.
1367 emit critical(tr("Git pull failed"),
1368 tr("Could not check the store for unmerged files after the "
1369 "pull. Check it with git before re-encrypting."));
1370 return false;
1371 }
1372 if (!unmerged.trimmed().isEmpty()) {
1373 emit critical(tr("Git pull failed"),
1374 tr("The pull left the store with unmerged files. Resolve "
1375 "the conflict before re-encrypting."));
1376 return false;
1377 }
1378 // A third kind: the fetch got the commits and the checkout then failed at
1379 // one path (a link git could not create, say). Git leaves HEAD and the
1380 // index where they were, but the pulled versions of the other files are
1381 // already in the work tree, and the backup commit would take them as the
1382 // user's own, signed with the user's key. HEAD still behind the upstream
1383 // after a failed pull means the pull may have got that far. No upstream
1384 // (rev-list fails) means there was nothing to merge.
1385 QString behind;
1386 if (execBlocking(m_settings.gitExecutable,
1387 {"-C", pgit(m_settings.passStore), "rev-list", "--count",
1388 "HEAD..@{upstream}"},
1389 &behind) == 0 &&
1390 behind.trimmed().toInt() > 0) {
1391 emit critical(tr("Git pull failed"),
1392 tr("The pull stopped before the store was up to date with "
1393 "its remote, and may have left files of the pulled "
1394 "commits in it. Check the store with git before "
1395 "re-encrypting."));
1396 return false;
1397 }
1398 emit statusMsg(tr("Git pull failed, re-encrypting the store as it is"), 5000);
1399 return true;
1400}

1402auto ImitatePass::entriesToReencrypt(const QString &dir, QStringList *failed)
1403 -> QStringList {
1404 // Regular files only: a symlink or junction is not a password entry, and
1405 // following one would decrypt and rewrite something outside the store.
1406 // One entry (what a move moved): itself, when it is one.
1407 if (const QFileInfo one(dir); !one.isDir()) {
1408 if (one.isFile() && !Util::isLinkedFolder(dir) &&
1409 one.fileName().endsWith(QLatin1String(".gpg"))) {
1410 return {one.filePath()};
1411 }
1412 return {};
1413 }
1414 QStringList skipped;
1415 QStringList unlisted;
1416 // A dot-named entry (.env.gpg) too, as with `pass init`: left out, it
1417 // stayed encrypted to a recipient just removed while the run reported
1418 // success. A hidden folder (.git, a sync tool's .stversions) stays closed.
1419 const QStringList files = Util::regularFilesUnder(
1420 dir, QStringList() << "*.gpg", &skipped, true, &unlisted);
1421 // A folder that cannot be read looked empty, and the run reported success
1422 // over it with its entries still encrypted to the old recipients (a key
1423 // just removed among them). Each is a failure: the dialog names it and
1424 // nothing is pushed.
1425 for (const QString &folder : std::as_const(unlisted)) {
1426 *failed << tr("%1 could not be read: its entries were not re-encrypted")
1427 .arg(QDir::toNativeSeparators(folder));
1428 }
1429 if (!skipped.isEmpty()) {
1430 emit statusMsg(tr("%n entr(y/ies) skipped: a symlink, junction or special "
1431 "file is not part of the store.",
1432 "", static_cast<int>(skipped.size())),
1433 5000);
1434 }
1435 return files;
1436}

1439 QHash<QString, QStringList> &verified,
1440 QStringList &gpgId, ReencryptResult &result)
1441 -> bool {

1443 if (m_reencryptCancel.load()) {
1444 result.cancelled = true;
1445 } else {
1446 result.aborted = true;
1447 }
1448 return false;
1449 }
1450 if (gpgId.isEmpty() && !verified.isEmpty()) {
1451 emit critical(tr("GPG ID verification failed"),
1452 tr("Could not verify .gpg-id for directory."));
1453 result.aborted = true;
1454 return false;
1455 }
1456 return true;

1459// Worker thread: only blocking helpers, nothing touching `exec` or the
1460// transaction state; signals reach the GUI queued. A helper failing while the
1461// cancel flag is set was interrupted: its file is neither checked nor failed.
1462auto ImitatePass::reencryptFiles(const QString &dir) -> ReencryptResult {
1463 ReencryptResult result;
1464 if (!pullBeforeReencrypt()) {
1465 result.aborted = true;
1466 return result;
1467 }
1468 // The GUI-thread checks ran before this worker's own pull.
1469 // pullBeforeReencrypt refuses an unmerged index, but a pull that stopped in a
1470 // rebase can leave a rebase-merge/rebase-apply state with no unmerged entry
1471 // yet; the backup commit's `git add -u` would then conclude it. Re-check
1472 // here, before any write, with the non-emitting detection (the refusal is
1473 // reported once, as an aborted run).
1474 if (const QString midOp = unfinishedGitOperationReason(); !midOp.isEmpty()) {
1475 if (!m_reencryptCancel.load()) {
1476 emit critical(tr("Unfinished Git operation"), midOp);
1477 }
1478 result.aborted = true;
1479 return result;
1480 }
1481 // reencryptPath() checked the folder before the pull; a pulled commit can
1482 // have made it, or a folder above it, a link since. Recovery and the walk
1483 // below would then rename and remove files beyond the store.
1484 if (Util::isUnderLink(dir, m_settings.passStore)) {
1485 emit critical(tr("Not part of the store"),
1486 tr("%1 is, or lies behind, a symbolic link or junction "
1487 "after pulling. What that points to is not part of the "
1488 "password store and is left alone.")
1489 .arg(QDir::toNativeSeparators(QDir::cleanPath(dir))));
1490 result.aborted = true;
1491 return result;
1492 }
1493
1494 // Init() wrote the recipients the user ticked and committed them. A pull
1495 // that merged another change to the same list cleanly (a line removed
1496 // here, one added there) would have the run encrypt to keys nobody chose
1497 // in the dialog, and without a signing key nothing else notices.
1498 if (m_writtenGpgId) {
1499 const std::optional<QByteArray> now =
1500 Util::readRegularFile(m_writtenGpgId->file, Util::kMaxGpgIdBytes);
1501 if (!now || *now != m_writtenGpgId->bytes) {
1502 if (!m_reencryptCancel.load()) {
1503 emit critical(
1504 tr("Recipients changed by the pull"),
1505 tr("The pull changed the recipients of %1 after you saved "
1506 "them, so nothing was re-encrypted: entries in it may still "
1507 "be encrypted to the previous recipients, and anyone you "
1508 "just removed may still be able to read them. Open the users "
1509 "dialog for that folder again, check the recipients, and "
1510 "save.")
1511 .arg(QDir::toNativeSeparators(
1512 QFileInfo(m_writtenGpgId->file).path())));
1513 }
1514 result.aborted = true;
1515 return result;
1516 }
1517 }
1518
1519 // Leftovers of an interrupted run first: a restored entry then goes into
1520 // the backup commit like everything else, and a stale temporary does not.
1521 if (!recoverReencryptLeftovers(dir)) {
1522 result.aborted = true;
1523 return result;
1524 }
1525
1526 if (!createBackupCommit()) {
1527 if (m_reencryptCancel.load()) {
1528 result.cancelled = true;
1529 } else {
1530 result.aborted = true;
1531 }
1532 return result;
1533 }
1534
1535 const QStringList files = entriesToReencrypt(dir, &result.failed);
1536 result.total = files.size();
1537 emit reencryptProgress(0, result.total);
1538
1539 QString currentDir;
1540 QHash<QString, QStringList> gpgIdFilesVerified;
1541 QStringList gpgId;
1542 for (const QString &fileName : std::as_const(files)) {
1543 if (m_reencryptCancel.load()) {
1544 result.cancelled = true;
1545 break;
1546 }
1547 const QString fileDir = QFileInfo(fileName).path();
1548 if (fileDir != currentDir) {
1549 if (!recipientsForDir(fileName, gpgIdFilesVerified, gpgId, result)) {
1550 return result;
1551 }
1552 currentDir = fileDir;
1553 }
1554 // A pull since the listing may have put a link under the name, or in
1555 // place of a folder on the way to it. gpg given the name reads the
1556 // target, and one already encrypted for these recipients would pass the
1557 // comparison without a word.
1558 const QFileInfo entry(fileName);
1559 if (entry.isSymLink() || !entry.isFile() ||
1560 Util::isUnderLink(fileName, m_settings.passStore)) {
1561 result.failed << tr("%1 is no longer a regular file, or lies behind a "
1562 "link, and was not re-encrypted.")
1563 .arg(fileName);
1564 } else if (getKeysFromFile(fileName) != gpgId) {
1565 QString why;
1566 if (reencryptSingleFile(fileName, gpgId, &why)) {
1567 result.reencrypted++;
1568 } else if (m_reencryptCancel.load()) {
1569 // Interrupted by the cancel: the file is untouched, not failed.
1570 result.cancelled = true;
1571 break;
1572 } else {
1573 // The reason names the entry itself when there is one to give.
1574 result.failed << (why.isEmpty() ? fileName : why);
1575 }
1576 }
1577 result.checked++;
1578 emit reencryptProgress(result.checked, result.total);
1579 }
1580 return result;
1581}

1583// No push after a cancel, abort or per-file failure: a partially re-encrypted
1584// store must not reach the remote before the user inspects it.
1585void ImitatePass::finishReencrypt(const ReencryptResult &result) {
1586 if (!result.failed.isEmpty()) {
1587 QStringList listed = result.failed.mid(0, kReencryptMaxListedFailures);
1588 const int more = result.failed.size() - listed.size();
1589 if (more > 0) {
1590 listed << tr("... and %n more", nullptr, more);
1591 }
1592 emit critical(tr("Re-encryption failed"),
1593 tr("%n file(s) could not be re-encrypted:", nullptr,
1594 result.failed.size()) +
1595 "\n\n" + listed.join('\n'));
1596 }
1597
1598 // The recipients change is committed already: a run that did not finish
1599 // leaves entries for the previous recipients, which the next automatic
1600 // push (after any edit) would publish with it. A clean run releases only
1601 // the holds of what it covered. Also without Git: a recipients change
1602 // left uncommitted now is swept into a later commit (the next run's
1603 // backup commit) once Git is on, and pushed with it.
1604 if (result.cancelled || result.aborted || !result.failed.isEmpty()) {
1605 holdAutoPush(m_reencryptScope);
1606 } else {
1607 releaseAutoPush(m_reencryptScope);
1608 }
1609
1610 if (result.cancelled) {
1611 emit statusMsg(tr("Re-encryption cancelled: %1 of %2 files checked, "
1612 "%3 re-encrypted, %4 failed")
1613 .arg(result.checked)
1614 .arg(result.total)
1615 .arg(result.reencrypted)
1616 .arg(result.failed.size()),
1617 5000);
1618 } else if (!result.aborted) {
1619 if (!result.failed.isEmpty()) {
1620 emit statusMsg(tr("Re-encryption completed: %1 succeeded, %2 failed")
1621 .arg(result.reencrypted)
1622 .arg(result.failed.size()),
1623 5000);
1624 } else {
1625 emit statusMsg(tr("Re-encryption completed: %1 files re-encrypted")
1626 .arg(result.reencrypted),
1627 3000);
1628 }
1629 // With a moved entry's run still to come, that one pushes at its end.
1630 if (m_movedToReencrypt.isEmpty() &&
1631 (m_settings.autoPush || m_pushAfterReencrypt) && reencryptUsesGit()) {
1632 m_pushAfterReencrypt = false;
1633 m_deferredPushByHand = false;
1634 if (result.failed.isEmpty() && autoPushHeld()) {
1635 // Clean here, but another folder's run did not finish.
1636 emit statusMsg(tr("Not pushing: an earlier re-encryption in this "
1637 "store did not finish"),
1638 5000);
1639 } else if (result.failed.isEmpty()) {
1640 emit statusMsg(tr("Updating password-store"), 2000);
1641 m_reencryptPushing = true;
1642 GitPush();
1643 m_reencryptPushing = false;
1644 } else {
1645 emit statusMsg(tr("Not pushing: %n file(s) failed to re-encrypt",
1646 nullptr, result.failed.size()),
1647 5000);
1648 }
1649 }
1650 }
1651 if (m_uncommittedGpgId) {
1652 const UncommittedGpgId uncommitted = *m_uncommittedGpgId;
1653 m_uncommittedGpgId.reset();
1654 reportUncommittedGpgId(uncommitted.folder, uncommitted.gitError,
1655 !result.cancelled && !result.aborted &&
1656 result.failed.isEmpty());
1657 }
1658 m_writtenGpgId.reset();
1659 // No push after a cancel or abort; otherwise one asked for meanwhile is
1660 // still to make (by a run to come, or once a pending move settles).
1661 if (result.cancelled || result.aborted) {
1662 m_pushAfterReencrypt = false;
1663 m_deferredPushByHand = false;
1664 }
1665 m_reencryptActive = false;
1666 emit endReencryptPath();
1667 startMovedReencrypts();
1668}

1715auto ImitatePass::resolveMoveDestination(const QString &src,
1716 const QString &dest, bool force)
1717 -> QString {
1718 QFileInfo srcFileInfo(src);
1719 QFileInfo destFileInfo(dest);
1720 QString destFile;
1721 QString srcFileBaseName = srcFileInfo.fileName();
1722
1723 if (srcFileInfo.isFile()) {
1724 if (destFileInfo.isFile()) {
1725 if (!force) {
1726 qCDebug(lcQtPass) << "Destination file already exists";
1727 return {};
1728 }
1729 destFile = dest;
1730 } else if (destFileInfo.isDir()) {
1731 destFile = QDir(dest).filePath(srcFileBaseName);
1732 } else {
1733 destFile = dest;
1734 }
1735
1736 // Case-sensitive: only an actual ".gpg" entry already carries the suffix.
1737 // A neighbouring file like "login.GPG" is a different name, so it must not
1738 // be chopped to "login" and re-suffixed to the real "login.gpg" entry,
1739 // which the forced move would then overwrite.
1740 if (destFile.endsWith(QLatin1String(".gpg"))) {
1741 destFile.chop(4);
1742 }
1743 destFile.append(".gpg");
1744 } else if (srcFileInfo.isDir()) {
1745 if (destFileInfo.isDir()) {
1746 destFile = QDir(dest).filePath(srcFileBaseName);
1747 } else if (destFileInfo.isFile()) {
1748 qCDebug(lcQtPass) << "Destination is a file";
1749 return {};
1750 } else {
1751 destFile = dest;
1752 }
1753 } else {
1754 qCDebug(lcQtPass) << "Source file does not exist";
1755 return {};
1756 }
1757 return destFile;
1758}

1790void ImitatePass::Move(const QString src, const QString dest,
1791 const bool force) {
1792 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1793 return;
1794 }
1795 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
1796 return;
1797 }
1798 // No message box opens inside the transaction: a process queued while one
1799 // is open (an entry opened from a search another instance forwarded) would
1800 // be booked as a step of the move, and its decrypted output reported as
1801 // the move's, which is not secret. So every check runs first, and the
1802 // transaction holds only the Git commands.
1803 const QString destFile = resolveMoveDestination(src, dest, force);
1804 if (destFile.isEmpty()) {
1805 return;
1806 }
1807 if (refuseLinkedPath(destFile)) {
1808 return;
1809 }
1810
1811 if (refuseOverlappingMove(src, destFile)) {
1812 return;
1813 }
1814
1815 qCDebug(lcQtPass) << "Move Source: " << src;
1816 qCDebug(lcQtPass) << "Move Destination: " << destFile;
1817
1818 const bool folder = QFileInfo(src).isDir();
1819 PendingMove move{src, destFile,
1820 QDir::cleanPath(getGpgIdPath(src, m_settings.passStore)),
1821 std::nullopt};
1822
1823 if (gitReady()) {
1824 // Settled in finished() once the move's commands have run: only then
1825 // is it known whether git mv moved anything, and which list applies
1826 // where it landed (a move queued before this one may create that
1827 // folder). Every Move() through Git ends in one PASS_MOVE.
1828 m_pendingMoves.enqueue(move);
1829 TransactionHelper trans(&m_transaction, PASS_MOVE);
1830 executeMoveGit(src, destFile, force);
1831 return;
1832 }
1833 // A forced move replaces the destination entry in the one rename, as
1834 // git mv -f does: removing it first lost it when the rename then failed
1835 // (and lost the entry itself, moved onto its own name). A folder is never
1836 // replaced, forced or not. Without Git nothing else reports a failure.
1837 const bool moved = force && !folder ? Util::replaceFile(src, destFile, true)
1838 : QDir().rename(src, destFile);
1839 if (!moved) {
1840 emit critical(tr("Move failed"),
1841 tr("Could not move %1 to %2.").arg(src, destFile));
1842 return;
1843 }
1844 move.landed = true;
1845 reencryptMoved(move);
1846 startMovedReencrypts();
1847}

1849auto ImitatePass::refuseOverlappingMove(const QString &src,
1850 const QString &destFile) -> bool {
1851#ifdef Q_OS_WIN
1852 constexpr auto cs = Qt::CaseInsensitive;
1853#else
1854 constexpr auto cs = Qt::CaseSensitive;
1855#endif
1856 // The same path, or one inside the other.
1857 const auto overlap = [](const QString &a, const QString &b) {
1858 const QString x = QDir::cleanPath(a);
1859 const QString y = QDir::cleanPath(b);
1860 return x.compare(y, cs) == 0 || x.startsWith(y + QLatin1Char('/'), cs) ||
1861 y.startsWith(x + QLatin1Char('/'), cs);
1862 };
1863 QStringList busy = m_movedToReencrypt;
1864 for (const PendingMove &move : std::as_const(m_pendingMoves)) {
1865 busy << move.src << move.destFile;
1866 }
1867 if (m_reencryptActive) {
1868 busy << m_reencryptScope;
1869 }
1870 for (const QString &path : std::as_const(busy)) {
1871 if (overlap(path, src) || overlap(path, destFile)) {
1872 emit critical(tr("Move not started"),
1873 tr("%1 is still being moved or re-encrypted. Move it "
1874 "again once that has finished.")
1875 .arg(QDir::toNativeSeparators(QDir::cleanPath(path))));
1876 return true;
1877 }
1878 }
1879 return false;
1880}
1881
1882void ImitatePass::reencryptMoved(const PendingMove &move) {
1883 // Only a move whose git mv (or rename) succeeded: one that failed, or
1884 // never ran, moved nothing, and re-encrypting the destination would
1885 // rewrite the entry that was there. Not the source name: something may
1886 // be written under it again before the move settles.
1887 if (move.landed != true) {
1888 return;
1889 }
1890 const QFileInfo landed(move.destFile);
1891 if (!landed.exists() || Util::isLinkedFolder(move.destFile)) {
1892 return;
1893 }
1894 // As `pass mv`, what moved is encrypted for the recipients where it
1895 // landed. Not needed when the same list applies there (a rename, a move
1896 // within its folder's reach) or a folder took its own list along.
1897 const QString toList =
1898 QDir::cleanPath(getGpgIdPath(move.destFile, m_settings.passStore));
1899 if (toList == move.fromList ||
1900 (landed.isDir() &&
1901 toList == QDir::cleanPath(QDir(move.destFile).filePath(".gpg-id")))) {
1902 return;
1903 }
1904 // The entry alone, or the folder: not its neighbours, which this move

1912auto ImitatePass::copyDestination(const QString &src, const QString &dest,
1913 bool force) -> QString {
1914 const auto refuse = [this, &src](const QString &to) {
1915 emit critical(tr("Copy failed"),
1916 tr("Could not copy %1 to %2.").arg(src, to));
1917 return QString();
1918 };
1919 // Like `pass cp`, dest may be an existing folder (a drag-and-drop copy hands
1920 // over the folder, not the new file name). Resolve the real target the same
1921 // way Move does: into the folder, .gpg appended, no clobbering without force.
1922 const QString destFile = resolveMoveDestination(src, dest, force);
1923 if (destFile.isEmpty()) {
1924 return refuse(dest);
1925 }
1926 // A link planted under that name (dangling ones pass exists()) is not an
1927 // entry to write.
1928 if (refuseLinkedPath(destFile)) {
1929 return {};
1930 }
1931 const QFileInfo destFileInfo(destFile);
1932 // A folder destination that is the source's own folder resolves to the
1933 // source itself; with force that would replace the only copy with itself.
1934 // And resolveMoveDestination only sees a clash when dest names the file.
1935 if (QFileInfo(src) == destFileInfo || (!force && destFileInfo.exists())) {
1936 return refuse(destFile);
1937 }
1938 return destFile;
1939}

1941void ImitatePass::Copy(const QString src, const QString dest,
1942 const bool force) {
1943 // QFile::copy reads through a link: the target's bytes would become an
1944 // entry of the store.
1945 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1946 return;
1947 }
1948 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
1949 return;
1950 }
1951 // As in Move(): no message box inside the transaction.
1952 const QString destFile = copyDestination(src, dest, force);
1953 if (destFile.isEmpty()) {
1954 return;
1955 }
1956 // git has no "cp": copy on disk in both modes, then stage. Synchronous and
1957 // atomic (Util::copyFileReplacing follows no link), so it exists before the
1958 // re-encryption below and a forced overwrite survives a half-way failure;
1959 // without force nothing is replaced, not even what appeared since the check.
1960 QString why;
1961 if (!Util::copyFileReplacing(src, destFile, force, &why)) {
1962 emit critical(tr("Copy failed"),
1963 tr("Could not copy %1 to %2.").arg(src, destFile) + "\n" +
1964 why);
1965 return;
1966 }
1967 if (gitReady()) {
1968 TransactionHelper trans(&m_transaction, PASS_COPY);
1969 executeGit(GIT_COPY, {"add", "--", pgit(destFile)});
1970 gitCommit(destFile, QString("Copied from %1 to %2 using QtPass.")
1971 .arg(commitName(src), commitName(destFile)));
1972 }
1973 const QFileInfo written(destFile);
1974 if (written.isDir()) {
1975 reencryptPath(written.absoluteFilePath());
1976 } else if (written.isFile()) {
1977 reencryptPath(written.dir().path());
1978 }
1979}

2011// Only PASS_* processes reach Pass::finished, so to the interface this looks
2012// the same as RealPass.
2013void ImitatePass::finished(int id, int exitCode, const QString &out,
2014 const QString &err) {
2015 qCDebug(lcQtPass) << "Imitate Pass";
2016 QString error = err;
2017 // A move's git mv runs first in its transaction, so in order: the first
2018 // pending move without an answer is this one's.
2019 if (id == GIT_MOVE) {
2020 for (PendingMove &move : m_pendingMoves) {
2021 if (!move.landed) {
2022 move.landed = exitCode == 0;
2023 break;
2024 }
2025 }
2026 }
2027 if (id == PASS_INSERT && !m_pendingInserts.isEmpty()) {
2028 // Insert()'s gpg step: place the ciphertext before the queued git steps
2029 // run (the executor starts the next item only after this returns). On
2030 // failure it fails like a gpg error: git steps cancelled below.
2031 // No dialog here: it would spin the event loop and run git steps first.
2032 const PendingInsert pending = m_pendingInserts.dequeue();
2033 // The flag answers for this insert, the one whose gpg step just ended, not
2034 // for whichever Insert() was called last: with two inserts queued, a gpg
2035 // failure of the second must report written=false even though the first
2036 // was placed. Cleared here, before placement, and set only on success.
2037 m_insertPlaced = false;
2038 if (exitCode == 0) {
2039 // Insert() checked the repository when the save was requested, but a
2040 // pull queued before it (the asynchronous Update button) may have run
2041 // and stopped in a conflict while this gpg step encrypted. Placing the
2042 // entry and committing it now would land it in the unfinished operation,
2043 // where `git rebase --abort` would drop it. Re-check and, if so, fail the
2044 // save instead of placing it; the queued git steps are cancelled below.
2045 const QString midOp = unfinishedGitOperationReason();
2046 if (!midOp.isEmpty()) {
2047 error = midOp;
2048 exitCode = 1;
2049 } else {
2050 m_insertPlaced = placeEncryptedFile(pending.output, pending.file,
2051 pending.overwrite, &error);
2052 if (!m_insertPlaced) {
2053 exitCode = 1;
2054 }
2055 }
2056 }
2057 }
2058 PROCESS pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
2059 m_transactionOutput.append(out);
2060
2061 if (exitCode == 0) {
2062 if (pid == INVALID) {
2063 return;
2064 }
2065 } else {
2066 while (pid == INVALID) {
2067 id = exec.cancelNext();
2068 if (id == -1) {
2069 // this is probably irrecoverable and shall not happen
2070 qCDebug(lcQtPass) << "No such transaction!";
2071 return;
2072 }
2073 pid = m_transaction.transactionIsOver(static_cast<PROCESS>(id));
2074 }
2075 }
2076 Pass::finished(pid, exitCode, m_transactionOutput, error);
2077 m_transactionOutput.clear();
2078 if (pid == PASS_MOVE && !m_pendingMoves.isEmpty()) {
2079 reencryptMoved(m_pendingMoves.dequeue());
2080 startMovedReencrypts();
2081 }
2082}
