27// Locale-independent: gpg's machine-readable status output via --status-fd 1.
28// See doc/DETAILS in the GnuPG source for IMPORT_OK / IMPORTED grammar.
29// Doxygen reads `Type name(args);` as a function, so these three constants
30// are hidden from it and described here instead.
32// All three are anchored at the start of the line: gpg echoes the key's
33// user ID, which whoever made the key chooses, on the IMPORTED line and in
34// the human line, so "[GNUPG:] IMPORT_OK 1 <hex>" inside a user ID must not
35// pass for gpg's own status line and name the fingerprint to verify.
36// `[GNUPG:] IMPORT_OK <reason> <fingerprint>`: captures the full fingerprint
37// (40 hex, or 64 for v5/v6 keys), the preferred identifier, which ends the
38// line.
39static const QRegularExpression IMPORT_OK_RE(QStringLiteral(
40 R"(^\‍[GNUPG:\‍] IMPORT_OK \d+ ([0-9A-Fa-f]{40}|[0-9A-Fa-f]{64})$)"));
41// `[GNUPG:] IMPORTED <keyid> <userid>`: captures the 16-hex long key id, tried
42// after IMPORT_OK_RE.
43static const QRegularExpression IMPORTED_RE(
44 QStringLiteral(R"(^\‍[GNUPG:\‍] IMPORTED ([0-9A-Fa-f]{16})(?: |$))"));
45// Fallback for the human-readable (English-locale) `gpg: key <id>: ...
46// imported` line, last resort when no status line was seen.
47static const QRegularExpression KEY_IMPORTED_FALLBACK(QStringLiteral(
48 R"(^gpg: key ([0-9A-Fa-f]{40}|[0-9A-Fa-f]{16}):.*imported)"));

65// Every character the editor starts a new block (or line) at counts, as
66// QTextCursor::insertText() splits: LF, a CR alone or before an LF (one
67// separator), the paragraph and line separators and Qt's frame markers. A
68// CR ignored, a file of CRs only made millions of blocks all the same.
69auto ImportKeyDialog::fitsTheEditor(QStringView text) -> bool {
70 qsizetype lines = 1;
71 qsizetype run = 0;
72 for (qsizetype i = 0; i < text.size(); ++i) {
73 switch (text.at(i).unicode()) {
74 case u'\r':
75 if (i + 1 < text.size() && text.at(i + 1) == QLatin1Char('\n')) {
76 ++i;
77 }
78 [[fallthrough]];
79 case u'\n':
80 case QChar::LineSeparator:
81 case QChar::ParagraphSeparator:
82 case 0xFDD0: // QTextBeginningOfFrame
83 case 0xFDD1: // QTextEndOfFrame
84 if (++lines > kMaxKeyLines) {
85 return false;
86 }
87 run = 0;
88 break;
89 default:
90 if (++run > kMaxKeyLine) {
91 return false;
92 }
93 }
94 }
95 return true;
96}

103static auto notLaidOutLikeAKey(const QString &what) -> QString {
104 return QCoreApplication::translate(

112void ImportKeyDialog::on_fileButton_clicked() {
113 const QString fileName = QFileDialog::getOpenFileName(
114 this, tr("Import GPG key"), QString(),
115 tr("ASCII-armored GPG key") + " (*.asc);;" + tr("All files") + " (*)");
116
117 if (fileName.isEmpty()) {
118 return;
119 }
120
121 // A key file, so a regular one of sensible size: a FIFO would block the
122 // GUI and /dev/zero (or a sparse file) would be read without end.
123 constexpr qint64 kMaxKeyFileBytes = 16LL * 1024 * 1024;
124 // A link is followed, deliberately: the user picked this file, unlike a
125 // name found in the store. Its target is then opened as the regular file
126 // it is at the moment of the open, without blocking: a FIFO swapped in
127 // after the pick would otherwise hang the GUI waiting for a writer.
128 QFile file;
129 const QString target = QFileInfo(fileName).canonicalFilePath();
130 if (target.isEmpty() || !Util::openRegularFile(target, file)) {
131 PlainMessageBox::warning(this, tr("Import key"),
132 tr("Could not open file: %1").arg(fileName));
133 return;
134 }
135
136 // The size first, so a large file is not read at all; the read limit
137 // still guards one that grows meanwhile.
138 const bool tooLarge = file.size() > kMaxKeyFileBytes;
139 const QByteArray bytes =
140 tooLarge ? QByteArray() : file.read(kMaxKeyFileBytes + 1);
141 file.close();
142 if (tooLarge || bytes.size() > kMaxKeyFileBytes) {
143 PlainMessageBox::warning(
144 this, tr("Import key"),
145 tr("%1 is larger than a key file can be.").arg(fileName));
146 return;
147 }
148
149 // Only ASCII-armored content is shown in the text edit; binary keyrings
150 // would lose bytes through UTF-8 conversion. Reject anything that doesn't
151 // start with the PGP armor header.
152 if (!bytes.trimmed().startsWith("-----BEGIN PGP")) {
153 // Message body is rich text (uses <code>/<b>); escape the path so any
154 // characters in it cannot reach the HTML subset Qt renders.
155 QMessageBox::warning(
156 this, tr("Import key"),
157 tr("%1 does not look like an ASCII-armored GPG key. Convert it with "
158 "<code>gpg --armor --export</code> first, or paste the armored "
159 "block via <b>From clipboard</b>.")
160 .arg(fileName.toHtmlEscaped()));
161 return;
162 }
163
164 const QString text = QString::fromUtf8(bytes);
165 if (!fitsTheEditor(text)) {
166 PlainMessageBox::warning(this, tr("Import key"),
167 notLaidOutLikeAKey(fileName));
168 return;
169 }
170 ui->inputTextEdit->setPlainText(text);
171}

205bool ImportKeyDialog::importFromString(const QString &input) {
206 QElapsedTimer timer;
207 timer.start();
208
209 // gpg imports every key in the block, and all the user learns is the one
210 // fingerprint named below. List the block first, writing nothing, so a
211 // second key or a private key cannot come along unmentioned. Both runs
212 // skip gpg.conf, so they judge the block alike: a list-filter there hides
213 // keys from this listing only, and allow-non-selfsigned-uid lets the
214 // import take a key the listing would have skipped.
215 QString listing;
216 QString listErr;
217 if (!runGpg({QStringLiteral("--no-options"), QStringLiteral("--batch"),
218 QStringLiteral("--with-colons"),
219 QStringLiteral("--import-options"), QStringLiteral("show-only"),
220 QStringLiteral("--import")},
221 input, &listing, &listErr, timer)) {
222 return false;
223 }
224 const BlockKeys keys = keysInBlock(listing);
225 if (keys.secretKeys > 0) {
226 showError(tr("This text holds a private key. QtPass imports public keys "
227 "only, so nothing was imported. Import the public key: "
228 "gpg --armor --export writes it."));
229 return false;
230 }
231 if (keys.publicKeys == 0) {
232 showError(tr("gpg found no key in this text, so nothing was imported."));
233 return false;
234 }
235 if (keys.publicKeys > 1) {
236 showError(tr("This text holds %n keys. QtPass imports one key at a time "
237 "and names it, so you can check its fingerprint: nothing "
238 "was imported. Import each key on its own.",
239 nullptr, keys.publicKeys));
240 return false;
241 }
242
243 QString stdOut;
244 QString stdErr;
245 if (!runGpg({QStringLiteral("--no-options"), QStringLiteral("--status-fd"),
246 QStringLiteral("1"), QStringLiteral("--import"),
247 QStringLiteral("--batch"), QStringLiteral("--yes")},
248 input, &stdOut, &stdErr, timer)) {
249 return false;
250 }
251
252 QString keyId = parseGpgImportOutput(stdOut);
253 if (keyId.isEmpty()) {
254 keyId = parseGpgImportOutput(stdErr);
255 }
256
257 if (keyId.isEmpty()) {
258 showError(tr("Could not parse imported key id from GPG output."));
259 return false;
260 }
261
262 m_importedKeyId = keyId;
263 showSuccess(keyId);
264 return true;
265}

267auto ImportKeyDialog::runGpg(const QStringList &args, const QString &input,
268 QString *stdOut, QString *stdErr,
269 const QElapsedTimer &timer) -> bool {
270 const QString gpgExe = m_gpgExe.isEmpty() ? QStringLiteral("gpg") : m_gpgExe;
271 const qint64 left = m_timeoutMs - timer.elapsed();
272 int result = -1;
273 if (left > 0) {
274 QProcess gpg;
275 result = Executor::executeBlocking(gpg, gpgExe, args, input, stdOut, stdErr,
276 nullptr, static_cast<int>(left));
277 }
278 if (result != 0 && timer.elapsed() >= m_timeoutMs) {
279 const int seconds = (m_timeoutMs + 999) / 1000;
280 showError(tr("gpg did not finish the import within %n second(s) and was "
281 "stopped. Part of the key may be in your keyring already: "
282 "check it with gpg --list-keys before you use it.",
283 nullptr, seconds));
284 return false;
285 }
286 if (result != 0) {
287 showError(tr("GPG import failed:\n%1").arg(*stdErr));
288 return false;
289 }
290 return true;
291}

293auto ImportKeyDialog::keysInBlock(const QString &listing) -> BlockKeys {
294 BlockKeys keys;
295 const QStringList lines = listing.split(
296 QRegularExpression(QStringLiteral("\\r?\\n")), Qt::SkipEmptyParts);
297 for (const QString &line : lines) {
298 if (line.startsWith(QLatin1String("pub:"))) {
299 ++keys.publicKeys;
300 } else if (line.startsWith(QLatin1String("sec:"))) {
301 ++keys.secretKeys;
302 }
303 }
304 return keys;
305}

307auto ImportKeyDialog::parseGpgImportOutput(const QString &output) -> QString {
308 // Line by line, a CRLF end included (gpg on Windows): the patterns are
309 // anchored at both ends of the line.
310 const QStringList lines = output.split(
311 QRegularExpression(QStringLiteral("\\r?\\n")), Qt::SkipEmptyParts);
312 // The order here is the priority order: prefer the locale-independent
313 // status lines over the English human-readable line, and within the
314 // status lines prefer IMPORT_OK (full fingerprint) over IMPORTED
315 // (long key id). Each regex is scanned across every line before we
316 // fall through to the next regex; the per-line ordering of gpg's
317 // output (which emits IMPORTED before IMPORT_OK) must not pick the
318 // weaker identifier.
319 for (const QString &line : lines) {
320 const QRegularExpressionMatch match = IMPORT_OK_RE.match(line);
321 if (match.hasMatch()) {
322 return match.captured(1);
323 }
324 }
325 for (const QString &line : lines) {
326 const QRegularExpressionMatch match = IMPORTED_RE.match(line);
327 if (match.hasMatch()) {
328 return match.captured(1);
329 }
330 }
331 for (const QString &line : lines) {
332 const QRegularExpressionMatch match = KEY_IMPORTED_FALLBACK.match(line);
333 if (match.hasMatch()) {
334 return match.captured(1);
335 }
336 }
337 return {};
338}
