25// Locale-independent: gpg's machine-readable status output via --status-fd 1.
26// See doc/DETAILS in the GnuPG source for IMPORT_OK / IMPORTED grammar.
27// Doxygen reads `Type name(args);` as a function, so these three constants
28// are hidden from it and described here instead.
30// `[GNUPG:] IMPORT_OK <reason> <fingerprint>`: captures the full fingerprint,
31// the preferred identifier.
32static const QRegularExpression
33 IMPORT_OK_RE(QStringLiteral(R"(\‍[GNUPG:\‍] IMPORT_OK \d+ ([0-9A-Fa-f]+))"));
34// `[GNUPG:] IMPORTED <keyid> <userid>`: captures the 16-hex long key id, tried
35// after IMPORT_OK_RE.
36static const QRegularExpression
37 IMPORTED_RE(QStringLiteral(R"(\‍[GNUPG:\‍] IMPORTED ([0-9A-Fa-f]{16}))"));
38// Fallback for the human-readable (English-locale) `gpg: key <id>: ...
39// imported` line, last resort when no status line was seen.
40static const QRegularExpression KEY_IMPORTED_FALLBACK(
41 QStringLiteral(R"(gpg: key ([0-9A-Fa-f]{40}|[0-9A-Fa-f]{16}):.*imported)"));

56void ImportKeyDialog::on_fileButton_clicked() {
57 const QString fileName = QFileDialog::getOpenFileName(
58 this, tr("Import GPG key"), QString(),
59 tr("ASCII-armored GPG key") + " (*.asc);;" + tr("All files") + " (*)");
60
61 if (fileName.isEmpty()) {
62 return;
63 }
64
65 // A key file, so a regular one of sensible size: a FIFO would block the
66 // GUI and /dev/zero (or a sparse file) would be read without end.
67 constexpr qint64 kMaxKeyFileBytes = 16LL * 1024 * 1024;
68 // A link is followed, deliberately: the user picked this file, unlike a
69 // name found in the store. Its target is then opened as the regular file
70 // it is at the moment of the open, without blocking: a FIFO swapped in
71 // after the pick would otherwise hang the GUI waiting for a writer.
72 QFile file;
73 const QString target = QFileInfo(fileName).canonicalFilePath();
74 if (target.isEmpty() || !Util::openRegularFile(target, file)) {
75 PlainMessageBox::warning(this, tr("Import key"),
76 tr("Could not open file: %1").arg(fileName));
77 return;
78 }
79
80 // The size first, so a large file is not read at all; the read limit
81 // still guards one that grows meanwhile.
82 const bool tooLarge = file.size() > kMaxKeyFileBytes;
83 const QByteArray bytes =
84 tooLarge ? QByteArray() : file.read(kMaxKeyFileBytes + 1);
85 file.close();
86 if (tooLarge || bytes.size() > kMaxKeyFileBytes) {
87 PlainMessageBox::warning(
88 this, tr("Import key"),
89 tr("%1 is larger than a key file can be.").arg(fileName));
90 return;
91 }
92
93 // Only ASCII-armored content is shown in the text edit; binary keyrings
94 // would lose bytes through UTF-8 conversion. Reject anything that doesn't
95 // start with the PGP armor header.
96 if (!bytes.trimmed().startsWith("-----BEGIN PGP")) {
97 // Message body is rich text (uses <code>/<b>); escape the path so any
98 // characters in it cannot reach the HTML subset Qt renders.
99 QMessageBox::warning(
100 this, tr("Import key"),
101 tr("%1 does not look like an ASCII-armored GPG key. Convert it with "
102 "<code>gpg --armor --export</code> first, or paste the armored "
103 "block via <b>From clipboard</b>.")
104 .arg(fileName.toHtmlEscaped()));
105 return;
106 }
107
108 ui->inputTextEdit->setPlainText(QString::fromUtf8(bytes));
109}

138bool ImportKeyDialog::importFromString(const QString &input) {
139 QString gpgExe = m_gpgExe.isEmpty() ? QStringLiteral("gpg") : m_gpgExe;
140 QStringList args = {"--status-fd", "1", "--import", "--batch", "--yes"};
141
142 QString stdOut;
143 QString stdErr;
144
145 int result = Executor::executeBlocking(gpgExe, args, input, &stdOut, &stdErr);
146
147 if (result != 0) {
148 showError(tr("GPG import failed:\n%1").arg(stdErr));
149 return false;
150 }
151
152 QString keyId = parseGpgImportOutput(stdOut);
153 if (keyId.isEmpty()) {
154 keyId = parseGpgImportOutput(stdErr);
155 }
156
157 if (keyId.isEmpty()) {
158 showError(tr("Could not parse imported key id from GPG output."));
159 return false;
160 }
161
162 m_importedKeyId = keyId;
163 showSuccess(keyId);
164 return true;
165}

167auto ImportKeyDialog::parseGpgImportOutput(const QString &output) -> QString {
168 const QStringList lines = output.split('\n', Qt::SkipEmptyParts);
169 // The order here is the priority order: prefer the locale-independent
170 // status lines over the English human-readable line, and within the
171 // status lines prefer IMPORT_OK (full fingerprint) over IMPORTED
172 // (long key id). Each regex is scanned across every line before we
173 // fall through to the next regex; the per-line ordering of gpg's
174 // output (which emits IMPORTED before IMPORT_OK) must not pick the
175 // weaker identifier.
176 for (const QString &line : lines) {
177 const QRegularExpressionMatch match = IMPORT_OK_RE.match(line);
178 if (match.hasMatch()) {
179 return match.captured(1);
180 }
181 }
182 for (const QString &line : lines) {
183 const QRegularExpressionMatch match = IMPORTED_RE.match(line);
184 if (match.hasMatch()) {
185 return match.captured(1);
186 }
187 }
188 for (const QString &line : lines) {
189 const QRegularExpressionMatch match = KEY_IMPORTED_FALLBACK.match(line);
190 if (match.hasMatch()) {
191 return match.captured(1);
192 }
193 }
194 return {};
195}
