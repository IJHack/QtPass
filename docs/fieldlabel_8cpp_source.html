42void FieldLabel::startEdit() {
43 if (m_editor != nullptr) {
44 m_editor->setFocus();
45 return;
46 }
47 m_editor = new QLineEdit(text(), parentWidget());
48 m_editor->setObjectName(QStringLiteral("fieldNameEditor"));
49 m_editor->installEventFilter(this);
50 connect(m_editor, &QLineEdit::editingFinished, this,
51 [this] { finishEdit(true); });
52 // Take the label's cell in the form, so the editor gets the label column
53 // (which widens for it) instead of floating over the value next to it.
54 auto *form = qobject_cast<QFormLayout *>(parentWidget()->layout());
55 int row = -1;
56 QFormLayout::ItemRole role{};
57 if (form != nullptr) {
58 form->getWidgetPosition(this, &row, &role);
59 }
60 if (row >= 0) {
61 form->removeWidget(this);
62 hide();
63 form->setWidget(row, role, m_editor);
64 } else {
65 m_editor->setGeometry(geometry());
66 }
67 m_editor->show();
68 m_editor->selectAll();
69 m_editor->setFocus();
70}

72auto FieldLabel::eventFilter(QObject *watched, QEvent *event) -> bool {
73 if (watched != m_editor || event->type() != QEvent::KeyPress) {
74 return QLabel::eventFilter(watched, event);
75 }
76 switch (static_cast<QKeyEvent *>(event)->key()) {
77 case Qt::Key_Escape:
78 finishEdit(false);
79 return true;
80 case Qt::Key_Return:
81 case Qt::Key_Enter:
82 // Enter confirms the name only; left alone it would also reach the
83 // dialog's default button and save and close the whole entry.
84 finishEdit(true);
85 return true;
86 default:
87 return QLabel::eventFilter(watched, event);
88 }
89}

91void FieldLabel::finishEdit(bool commit) {
92 if (m_editor == nullptr) {
93 return;
94 }
95 // editingFinished fires again when the editor loses focus on hide; take
96 // the pointer first so the second call finds nothing to do.
97 QLineEdit *editor = m_editor;
98 m_editor = nullptr;
99 // A colon would start the value early when the line is written back.
100 const QString name = editor->text().remove(u':').trimmed();
101 auto *form = qobject_cast<QFormLayout *>(parentWidget()->layout());
102 int row = -1;
103 QFormLayout::ItemRole role{};
104 if (form != nullptr) {
105 form->getWidgetPosition(editor, &row, &role);
106 }
107 editor->hide();
108 if (row >= 0) {
109 form->removeWidget(editor);
110 form->setWidget(row, role, this);
111 show();
112 }
113 editor->deleteLater();
114 if (commit && !name.isEmpty() && name != text()) {
115 emit renamed(text(), name);
116 }
117}
