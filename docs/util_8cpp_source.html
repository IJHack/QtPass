127auto Util::passUmask(const QString &value) -> int {
128 // Taken as it is, as bash's umask builtin (which pass hands it to) takes
129 // it: " 022" is no umask there either. pass then keeps the session's;
130 // QtPass falls back to 077.
131 // anchoredPattern(), not ^...$: $ also matches before a final newline.
132 static const QRegularExpression octal(
133 QRegularExpression::anchoredPattern(QStringLiteral("[0-7]{1,4}")));
134 if (!octal.match(value).hasMatch()) {
135 return 077;
136 }
137 return value.toInt(nullptr, 8);
138}

164 const int fd = ::open(QFile::encodeName(path).constData(),

180auto Util::repositoryPassClimbsTo(const QString &store) -> QString {
181#ifdef Q_OS_WIN
182 Q_UNUSED(store);
183 return {};
184#else
185 if (!store.contains(QLatin1Char(':'))) {
186 return {};
187 }
188 // git walks up from the real folder, so the real path is the one to walk,
189 // and for a store that does not exist yet the one it will have: its
190 // nearest existing folder resolved, the rest appended. By the raw string:
191 // QDir and QFileInfo clean a path by name, and "link/.." is the parent
192 // of the link's target, not the folder the link is in. On macOS the
193 // temporary folder is under /var, a link to /private/var.
194 // Absolute as git means it, a leading "/" (Qt would take ":name" for a
195 // resource path).
196 QString path =
197 store.startsWith(u'/') ? store : QDir::currentPath() + u'/' + store;
198 while (path.size() > 1 && path.endsWith(u'/')) {
199 path.chop(1);
200 }
201 QString rest;
202 while (!QFileInfo::exists(path)) {
203 const qsizetype slash = path.lastIndexOf(u'/');
204 if (slash < 0) {
205 return {};
206 }
207 const QString name = path.mid(slash + 1);
208 rest = rest.isEmpty() ? name : name + u'/' + rest;
209 path = slash == 0 ? QStringLiteral("/") : path.left(slash);
210 }
211 const QString real = QFileInfo(path).canonicalFilePath();
212 QString start = real.isEmpty() ? QDir::cleanPath(path) : real;
213 if (!rest.isEmpty()) {
214 start = QDir::cleanPath(start + u'/' + rest);
215 }
216 // The pieces of pass's ceiling as git takes them (setup.c,
217 // canonicalize_ceiling_entry): relative ones are dropped, the rest
218 // resolved, one that cannot be resolved is dropped; after an empty piece
219 // ("::") they are kept exactly as written. A piece can still be a real
220 // ceiling: "/repo/:vault/.." holds "/repo/".
221 QString ceiling;
222 bool afterEmpty = false;
223 for (const QString &piece :
224 (store + QStringLiteral("/..")).split(QLatin1Char(':'))) {
225 if (piece.isEmpty()) {
226 afterEmpty = true;
227 continue;
228 }
229 if (!piece.startsWith(u'/')) {
230 continue;
231 }
232 const QString entry =
233 afterEmpty ? piece : QFileInfo(piece).canonicalFilePath();
234 // git stops below the longest entry the store lies under
235 // (longest_ancestor_length: "/" is above everything).
236 const QString under = entry == u'/' ? entry : entry + u'/';
237 if (!entry.isEmpty() && start.startsWith(under) &&
238 entry.size() > ceiling.size()) {
239 ceiling = entry;
240 }
241 }
242 // What git takes for a repository (setup.c, is_git_directory()): HEAD
243 // naming a ref or an object, objects/ and refs/. An empty or broken .git
244 // is passed by, as git passes it by.
245 const auto isGitDirectory = [](const QString &candidate) {
246 const QDir dir(candidate);
247 if (!QFileInfo(dir.filePath(QStringLiteral("objects"))).isDir() ||
248 !QFileInfo(dir.filePath(QStringLiteral("refs"))).isDir()) {
249 return false;
250 }
251 // As git (setup.c, validate_headref()): a HEAD that is a link counts
252 // when the link text starts with "refs/", without reading what it
253 // names, and not otherwise. Anything else is read only when it is a
254 // regular file, without waiting: a HEAD that is a FIFO (a pushed commit
255 // can carry a link to one, or to a device) blocked the open and the
256 // read on the GUI thread when Settings opened.
257 const QString headPath = dir.filePath(QStringLiteral("HEAD"));
258 if (const QFileInfo headInfo(headPath); headInfo.isSymLink()) {
259 // The link text as stored (Qt 6.6), not the path it resolves to.
260 return headInfo.readSymLink().startsWith(QLatin1String("refs/"));
261 }
262 QFile head;
263 if (!openFollowedRegularFile(headPath, head)) {
264 return false;
265 }
266 static const QRegularExpression valid(
267 QStringLiteral("^(?:ref: refs/|[0-9a-f]{40}\\s*$|[0-9a-f]{64}\\s*$)"));
268 return valid.match(QString::fromUtf8(head.read(256))).hasMatch();
269 };
270 // At each folder, as git: a .git folder that is a repository, or a .git
271 // file naming one ("gitdir: ..."), is the repository. Any other .git file
272 // stops git with an error, and pass then commits nothing; a .git folder
273 // that is no repository is passed by.
274 enum class Found { Nothing, Repository, Refused };
275 const auto repositoryAt = [&isGitDirectory](const QString &dir) {
276 const QString dotGit = QDir(dir).filePath(QStringLiteral(".git"));
277 const QFileInfo info(dotGit);
278 if (info.isDir()) {
279 return isGitDirectory(dotGit) ? Found::Repository : Found::Nothing;
280 }
281 if (!info.exists()) {
282 return Found::Nothing;
283 }
284 // The same for a .git file (or a link to one): a FIFO there blocked in
285 // open().
286 QFile file;
287 if (!openFollowedRegularFile(dotGit, file)) {
288 return Found::Refused;
289 }
290 const QString line = QString::fromUtf8(file.readLine(4096)).trimmed();
291 if (!line.startsWith(QLatin1String("gitdir: "))) {
292 return Found::Refused;
293 }
294 const QString target = line.mid(8);
295 return isGitDirectory(target.startsWith(u'/') ? target
296 : QDir(dir).filePath(target))
297 ? Found::Repository
298 : Found::Refused;
299 };
300 // The store's own repository comes first; only without one does git look
301 // further up, and never into the ceiling. By name from here: the real
302 // path holds no link or "..".
303 for (QString dir = start;;) {
304 switch (repositoryAt(dir)) {
305 case Found::Repository:
306 return dir == start ? QString() : dir;
307 case Found::Refused:
308 return {};
309 case Found::Nothing:
310 break;
311 }
312 // A bare repository: git stops there, and pass, finding no work tree,
313 // commits nothing.
314 if (isGitDirectory(dir)) {
315 return {};
316 }
317 const QString up = QFileInfo(dir).absolutePath();
318 if (up == dir || up == ceiling) {
319 return {};
320 }
321 dir = up;
322 }
323#endif
324}

360auto isSafeDirectory(const QString &root, const QString &git) -> bool {
361 // Outside any repository, as git reads its protected config: no
362 // repository's own config, and no includeIf "gitdir:" or "onbranch:"
363 // that matches (a GIT_DIR, even one that cannot exist, would be matched
364 // against them, and one that cannot exist makes git config fail). From
365 // "/", which a user cannot make a repository; a local scope is dropped
366 // below in any case.
367 QProcessEnvironment env = QProcessEnvironment::systemEnvironment();
368 env.remove(QStringLiteral("GIT_DIR"));
369 env.remove(QStringLiteral("GIT_WORK_TREE"));
370 env.remove(QStringLiteral("GIT_COMMON_DIR"));
371 // git config alone reads GIT_CONFIG in place of the system and global
372 // configuration (and reports it as "command"); the ownership check never
373 // reads it.
374 env.remove(QStringLiteral("GIT_CONFIG"));
375 QProcess process;
376 process.setProcessEnvironment(env);
377 process.setWorkingDirectory(QStringLiteral("/"));
378 // Bounded: Git calls wait for this, and a git that never answers trusts
379 // nothing.
380 constexpr int kQueryTimeoutMs = 5000;
381 QString out;
382 if (Executor::executeBlocking(
383 process, git.isEmpty() ? QStringLiteral("git") : git,
384 {QStringLiteral("config"), QStringLiteral("--show-scope"),
385 QStringLiteral("--type=path"), QStringLiteral("--get-all"),
386 QStringLiteral("safe.directory")},
387 QString(), &out, nullptr, nullptr, kQueryTimeoutMs) != 0) {
388 return false;
389 }
390 bool safe = false;
391 const QStringList lines = out.split(QLatin1Char('\n'));
392 for (const QString &line : lines) {
393 const qsizetype tab = line.indexOf(QLatin1Char('\t'));
394 if (tab < 0) {
395 continue;
396 }
397 const QString scope = line.left(tab);
398 if (scope != QLatin1String("system") && scope != QLatin1String("global") &&
399 scope != QLatin1String("command")) {
400 continue;
401 }
402 const QString value = line.mid(tab + 1);
403 if (value.isEmpty()) {
404 safe = false;
405 } else if (value == QLatin1String("*")) {
406 safe = true;
407 } else if (!QDir::isAbsolutePath(value)) {
408 continue;
409 } else if (value.endsWith(QLatin1String("/*"))) {
410 const QString base = value.chopped(2);
411 QString prefix = QFileInfo(base.isEmpty() ? QStringLiteral("/") : base)
412 .canonicalFilePath();
413 if (prefix.isEmpty()) {
414 continue;
415 }
416 if (!prefix.endsWith(QLatin1Char('/'))) {
417 prefix += QLatin1Char('/');
418 }
419 safe = safe || root.startsWith(prefix);
420 } else {
421 const QString named = QFileInfo(value).canonicalFilePath();
422 safe = safe || (!named.isEmpty() && named == root);
423 }
424 }
425 return safe;
426}

430auto Util::gitStoreVariables(const QString &store, const QString &git,
431 std::optional<uint> asUser)
432 -> QList<std::pair<QString, QString>> {
433 const QString ceiling = gitCeiling(store);
434 if (ceiling.isEmpty()) {
435 return {};
436 }
437 QList<std::pair<QString, QString>> vars{
438 {QStringLiteral("GIT_CEILING_DIRECTORIES"), ceiling}};
439#ifdef Q_OS_WIN
440 const QChar separator = QLatin1Char(';');
441#else
442 const QChar separator = QLatin1Char(':');
443#endif
444 // git splits the ceiling list on the separator: "/home/a:b" would be
445 // "/home/a" and "b", neither of them the parent, and git would climb.
446 if (ceiling.contains(separator)) {
447 QString root = QFileInfo(store).canonicalFilePath();
448 if (root.isEmpty()) {
449 root = QDir::cleanPath(QDir(store).absolutePath());
450 }
451 QString gitDir = QDir(root).filePath(QStringLiteral(".git"));
452#ifdef Q_OS_UNIX
453 // What discovery's ownership check would have looked at: the work tree,
454 // the .git, and the directory a gitfile names. Not the user's: no
455 // repository at all, under a path that cannot be created (its parent
456 // is a device), rather than another user's hooks run as this one.
457 const uint me = asUser.value_or(
458 s_gitOwnerForTests.value_or(static_cast<uint>(::geteuid())));
459 const auto ownedOrMissing = [me](const QString &path) {
460 const QFileInfo info(path);
461 return !info.exists() || info.ownerId() == me;
462 };
463 bool owned = ownedOrMissing(root) && ownedOrMissing(gitDir);
464 if (owned && QFileInfo(gitDir).isFile()) {
465 QFile gitFile(gitDir);
466 if (gitFile.open(QIODevice::ReadOnly)) {
467 const QString line =
468 QString::fromUtf8(gitFile.readLine(4096)).trimmed();
469 if (line.startsWith(QLatin1String("gitdir:"))) {
470 const QString target = line.mid(7).trimmed();
471 owned = ownedOrMissing(QDir(root).absoluteFilePath(target));
472 }
473 }
474 }
475 // Or trusted as git would trust it in place of ownership.
476 if (!owned && !isSafeDirectory(root, git)) {
477 gitDir = QStringLiteral("/dev/null/.git");
478 }
479#else
480 // Git for Windows runs the same ownership check during discovery, on
481 // SIDs, with its own rules for the Administrators group; this pin skips
482 // it there and does not repeat it (a known gap, #2089).
483 Q_UNUSED(git);
484 Q_UNUSED(asUser);
485#endif
486 vars.append({QStringLiteral("GIT_DIR"), gitDir});
487 vars.append({QStringLiteral("GIT_WORK_TREE"), root});
488 }
489 return vars;
490}

492auto Util::mkpathForStore(const QString &path) -> bool {
493 const QString clean = QDir::cleanPath(QDir(path).absolutePath());
494 if (QFileInfo(clean).isDir()) {
495 return true;
496 }
497 const QString parent = QFileInfo(clean).absolutePath();
498 if (parent != clean && !mkpathForStore(parent)) {
499 return false;
500 }
501#ifdef Q_OS_UNIX
502 // The store's umask, as git and gpg get it: a store shared with a group
503 // (PASSWORD_STORE_UMASK=007) gets group folders, the default owner-only.
504 const int umask = Executor::childUmask() >= 0 ? Executor::childUmask() : 077;
505 QFile::Permissions permissions;
506 const QFile::Permission bits[] = {
507 QFile::ReadOwner, QFile::WriteOwner, QFile::ExeOwner,
508 QFile::ReadGroup, QFile::WriteGroup, QFile::ExeGroup,
509 QFile::ReadOther, QFile::WriteOther, QFile::ExeOther};
510 for (int i = 0; i < 9; ++i) {
511 if ((umask & (0400 >> i)) == 0) {
512 permissions |= bits[i];
513 }
514 }
515 // mkdir() still takes QtPass's own umask off (a session's 022 would drop
516 // the group's write bit); set the mode exactly afterwards. That only ever
517 // widens it to what was asked, never past.
518 const bool made = QDir().mkdir(clean, permissions) &&
519 QFile::setPermissions(clean, permissions);
520#else
521 // Windows: permissions here would become an ACL of their own, replacing
522 // the inherited one; the folder takes its parent's, as before.
523 const bool made = QDir().mkdir(clean);
524#endif
525 return made || QFileInfo(clean).isDir();
526}

528auto Util::findBinaryInPath(const QString &binary) -> QString {
529 if (binary.isEmpty()) {
530 return {};
531 }
532
533 initialiseEnvironment();
534
535 const QStringList dirs =
536 _env.value(QStringLiteral("PATH"))
537 .split(QDir::listSeparator(), Qt::SkipEmptyParts);
538 QString ret;
539 if (QDir::fromNativeSeparators(binary).contains(u'/')) {
540 // An explicit path is not a PATH search: an absolute path is checked
541 // as-is, a relative one is resolved against the PATH directories. The
542 // directory-list overload refuses such names, so handle them here.
543 if (QDir::isAbsolutePath(binary)) {
544 ret = QStandardPaths::findExecutable(binary);
545 } else if (!dirs.isEmpty()) {
546 ret = QStandardPaths::findExecutable(binary, dirs);
547 }
548 } else {
549 ret = findBinaryInPath(binary, dirs);
550 }
551#ifdef Q_OS_WIN
552 if (ret.isEmpty()) {
553 // Cache per-binary WSL lookup result — the wsl --version probe is a
554 // blocking subprocess that can run several times per session for
555 // missing binaries; once decided, the answer doesn't change at runtime.
556 static QHash<QString, QString> wslBinaryCache;
557 const bool hasWhitespace =
558 std::any_of(binary.cbegin(), binary.cend(),
559 [](const QChar ch) { return ch.isSpace(); });
560 if (!hasWhitespace) {
561 auto cached = wslBinaryCache.constFind(binary);
562 if (cached != wslBinaryCache.constEnd()) {
563 ret = cached.value();
564 } else {
565 QString wslCommand = QStringLiteral("wsl ") + binary;
566 qCDebug(lcQtPass)
567 << "Util::findBinaryInPath(): falling back to WSL for binary"
568 << binary;
569 QString out, err;
570 QString cachedResult;
571 if (Executor::executeBlocking(wslCommand, {"--version"}, &out, &err) ==
572 0 &&
573 !out.isEmpty() && err.isEmpty()) {
574 qCDebug(lcQtPass)
575 << "Util::findBinaryInPath(): using WSL binary" << wslCommand;
576 cachedResult = wslCommand;
577 }
578 wslBinaryCache.insert(binary, cachedResult);
579 ret = cachedResult;
580 }
581 }
582 }
583#endif
584
585 return ret;
586}

588// Bare names only: QStandardPaths::findExecutable() returns an absolute binary
589// without consulting searchPaths, and a relative one with ".." could escape
590// them.
591auto Util::findBinaryInPath(const QString &binary,
592 const QStringList &searchPaths) -> QString {
593 if (binary.isEmpty() || QDir::fromNativeSeparators(binary).contains(u'/')) {
594 return {};
595 }
596 QStringList dirs;
597 dirs.reserve(searchPaths.size());
598 for (const QString &dir : searchPaths) {
599 if (!dir.isEmpty()) {
600 dirs.append(dir);
601 }
602 }
603 if (dirs.isEmpty()) {
604 // QStandardPaths::findExecutable() treats an empty list as "use PATH".
605 return {};
606 }
607 return QStandardPaths::findExecutable(binary, dirs);
608}

663auto authorityLength(const QString &url) -> qsizetype {
664 const auto endsPart = [](QChar c) {
665 return c == QLatin1Char('/') || c == QLatin1Char('?') ||
666 c == QLatin1Char('#');
667 };
668 const auto isAsciiLetter = [](QChar c) {
669 return (c >= QLatin1Char('a') && c <= QLatin1Char('z')) ||
670 (c >= QLatin1Char('A') && c <= QLatin1Char('Z'));
671 };
672 qsizetype colon = -1;
673 for (qsizetype i = 0; i < url.size() && !endsPart(url.at(i)); ++i) {
674 if (url.at(i) == QLatin1Char(':')) {
675 colon = i;
676 break;
677 }
678 }
679 bool scheme = colon > 0 && isAsciiLetter(url.at(0));
680 for (qsizetype i = 1; scheme && i < colon; ++i) {
681 const QChar c = url.at(i);
682 scheme =
683 isAsciiLetter(c) || (c >= QLatin1Char('0') && c <= QLatin1Char('9')) ||
684 c == QLatin1Char('+') || c == QLatin1Char('-') || c == QLatin1Char('.');
685 }
686 const qsizetype hier = scheme ? colon + 1 : 0;
687 if (url.mid(hier, 2) != QLatin1String("//")) {
688 return -1;
689 }
690 qsizetype end = hier + 2;
691 while (end < url.size() && !endsPart(url.at(end))) {
692 ++end;
693 }
694 return end - (hier + 2);
695}

698auto Util::isLaunchableWebUrl(const QString &value) -> bool {
699 const QString trimmed = value.trimmed();
700 if (trimmed.isEmpty()) {
701 return false;
702 }
703 // Reject control characters first, before QUrl normalisation can hide a
704 // CR/LF/NUL injection into the OS URL handler.
705 for (const QChar &c : trimmed) {
706 if (c == QLatin1Char('\r') || c == QLatin1Char('\n') ||
707 c == QChar(QChar::Null)) {
708 return false;
709 }
710 }
711 // QUrl runs the host through Unicode normalisation, which takes time in
712 // the square of a run of combining marks out of canonical order: one line
713 // of an entry kept the window busy for minutes. A host has at most 253
714 // characters and a port six more, so a longer authority is no web
715 // address anyway.
716 constexpr qsizetype kMaxAuthority = 259;
717 if (authorityLength(trimmed) > kMaxAuthority) {
718 return false;
719 }
720 const QUrl url(trimmed, QUrl::StrictMode);
721 if (!url.isValid()) {
722 return false;
723 }
724 const QString scheme = url.scheme().toLower();
725 if (scheme != QLatin1String("http") && scheme != QLatin1String("https")) {
726 return false;
727 }
728 if (url.host().isEmpty()) {
729 return false;
730 }
731 // Embedded userinfo (user:pass@host) would leak into browser history.
732 if (!url.userName().isEmpty() || !url.password().isEmpty()) {
733 return false;
734 }
735 return true;
736}

738auto Util::displayable(const QString &text, bool lineBreaks) -> QString {
739 QString shown = text;
740 for (qsizetype i = 0; i < shown.size(); ++i) {
741 const char16_t u = shown.at(i).unicode();
742 // A carriage return before a line feed is a Windows line end (git or
743 // gpg output quoted in a message), not a character of its own.
744 const bool lineEnd =
745 lineBreaks && (u == u'\n' || (u == u'\r' && i + 1 < shown.size() &&
746 shown.at(i + 1) == u'\n'));
747 if ((u < 0x20 && u != u'\t' && !lineEnd) || (u >= 0x7f && u <= 0x9f) ||
748 u == 0x2028 || u == 0x2029 || (u >= 0x202a && u <= 0x202e) ||
749 (u >= 0x2066 && u <= 0x2069)) {
750 shown[i] = QChar(QChar::ReplacementCharacter);
751 }
752 }
753 return shown;
754}

756auto Util::linkifyUrls(const QString &text, bool *linked) -> QString {
757 if (linked != nullptr) {
758 *linked = false;
759 }
760 QString html;
761 html.reserve(text.size());
762 qsizetype lastIndex = 0;
763 QRegularExpressionMatchIterator it = protocolRegex().globalMatch(text);
764 while (it.hasNext()) {
765 const QRegularExpressionMatch match = it.next();
766 const QString url = match.captured(0);
767 // Only plain ASCII becomes a link: the link text is the URL as stored,
768 // and a right-to-left letter, a bidi control or a look-alike host in it
769 // makes it read as another address than the one it opens. Such a URL
770 // stays plain text; the open button still takes it.
771 const bool plain = std::all_of(url.cbegin(), url.cend(), [](QChar c) {
772 return c.unicode() > 0x20 && c.unicode() < 0x7f;
773 });
774 if (!plain || !isLaunchableWebUrl(url)) {
775 // Not a web URL (or it carries credentials): leave it in the escaped
776 // plain-text run instead of making it clickable.
777 continue;
778 }
779 const qsizetype start = match.capturedStart(0);
780 html += text.mid(lastIndex, start - lastIndex).toHtmlEscaped();
781 const QString escapedUrl = url.toHtmlEscaped();
782 html += QStringLiteral("<a href=\"%1\">%1</a>").arg(escapedUrl);
783 lastIndex = match.capturedEnd(0);
784 if (linked != nullptr) {
785 *linked = true;
786 }
787 }
788 html += text.mid(lastIndex).toHtmlEscaped();
789 // With a link in it, no bidi embedding, override or isolate control is
790 // drawn: an override earlier in the entry (U+202E) drew the link text
791 // reversed, so it read as another address. Shown as U+FFFD, so the user
792 // sees that something was there; the copy button still copies the value.
793 if (linked != nullptr ? *linked : html.contains(QLatin1String("<a href="))) {
794 for (QChar &c : html) {
795 const char16_t u = c.unicode();
796 if ((u >= 0x202A && u <= 0x202E) || (u >= 0x2066 && u <= 0x2069)) {
797 c = QChar(QChar::ReplacementCharacter);
798 }
799 }
800 }
801 return html;
802}

842auto Util::shownName(const QString &name, qsizetype max) -> QString {
843 QString line = displayable(name, false);
844 line.replace(QLatin1Char('\t'), QLatin1Char(' '));
845 if (line.size() <= max) {
846 return line;
847 }
848 // Too short for an ellipsis between two ends: the head alone.
849 if (max < 3) {
850 qsizetype keep = std::max<qsizetype>(max, 0);
851 if (keep > 0 && line.at(keep - 1).isHighSurrogate()) {
852 --keep;
853 }
854 return line.left(keep);
855 }
856 qsizetype head = max / 3;
857 if (line.at(head - 1).isHighSurrogate()) {
858 --head;
859 }
860 qsizetype tail = max - max / 3 - 1;
861 if (line.at(line.size() - tail).isLowSurrogate()) {
862 --tail;
863 }
864 return line.left(head) + QChar(0x2026) + line.right(tail);
865}

867// One pass, no regex: a pattern such as \s{1001,} retries from every
868// character of a shorter run, which is quadratic on an entry made of them.
869// A tab or a line feed ends a run in Qt's layout as well, so it ends one
870// here; \v and U+0085 do not, so they count.
871auto Util::boundSpaceRuns(const QString &text) -> QString {
872 QString bounded;
873 // Text before this index is in bounded already, or dropped.
874 qsizetype from = 0;
875 qsizetype run = 0;
876 for (qsizetype i = 0; i < text.size(); ++i) {
877 const QChar ch = text.at(i);
878 if (!ch.isSpace() || ch == QLatin1Char('\t') || ch == QLatin1Char('\n')) {
879 run = 0;
880 continue;
881 }
882 if (++run <= kMaxSpaceRun) {
883 continue;
884 }
885 if (run == kMaxSpaceRun + 1) {
886 bounded += QStringView(text).sliced(from, i - from);
887 }
888 from = i + 1;
889 }
890 if (from == 0) {
891 return text;
892 }
893 bounded += QStringView(text).sliced(from);
894 return bounded;
895}

937template <typename Visit>
938void walkStore(const QString &dir, Visit visit,
939 QStringList *unlisted = nullptr) {
940 // Absolute, so the results are whatever the caller's cwd; not canonical,
941 // so a linked root keeps the name it was configured under.
942 QStringList pending{QDir(QDir::cleanPath(dir)).absolutePath()};
943 while (!pending.isEmpty()) {
944 const QDir current(pending.takeLast());
945 if (unlisted != nullptr && current.exists() && !canList(current.path())) {
946 qCWarning(lcQtPass) << "Cannot list" << current.path();
947 *unlisted << current.path();
948 continue;
949 }
950 // Every entry once, links and hidden entries included, so the decision
951 // what to do with each is taken here, before any recursion. QDir::System
952 // keeps dangling links in the listing.
953 const QFileInfoList entries =
954 current.entryInfoList(QDir::Dirs | QDir::Files | QDir::Hidden |
955 QDir::System | QDir::NoDotAndDotDot,
956 QDir::Name);
957 QStringList subdirs;
958 for (const QFileInfo &entry : entries) {
959 if (visit(entry)) {
960 subdirs << entry.filePath();
961 }
962 }
963 // Pushed last-to-first so the next one taken is the first by name.
964 for (auto it = subdirs.crbegin(); it != subdirs.crend(); ++it) {
965 pending << *it;
966 }
967 }
968}

1087 }
1088 }

1140auto placeWithoutReplacing(const QByteArray &source, const QByteArray &target)
1141 -> bool {
1142 // linkat() without AT_SYMLINK_FOLLOW follows nothing; link() would, on
1143 // macOS and the BSDs, hard-link whatever a symlink planted under the
1144 // source's name points at.
1145 if (::linkat(AT_FDCWD, source.constData(), AT_FDCWD, target.constData(), 0) ==
1146 0) {
1147 ::unlink(source.constData());
1148 return true;
1149 }
1150 if (!hasNoHardLinks(errno)) {
1151 return false;
1152 }
1153 if (const std::optional<bool> renamed = renameExclusive(source, target)) {
1154 return *renamed;
1155 }
1156 struct stat st{};
1157 if (::lstat(target.constData(), &st) == 0 || errno != ENOENT) {
1158 errno = EEXIST;
1159 return false;
1160 }
1161 return ::rename(source.constData(), target.constData()) == 0;
1162}

1207auto Util::openRegularFile(const QString &path, QFile &file) -> bool {
1208#ifdef Q_OS_WIN
1209 // FILE_FLAG_OPEN_REPARSE_POINT opens a symbolic link or junction itself
1210 // rather than its target, so the handle's attributes say what the name
1211 // was at the moment of the open.
1212 const std::wstring native =
1213 QDir::toNativeSeparators(QFileInfo(path).absoluteFilePath())
1214 .toStdWString();
1215 HANDLE handle = CreateFileW(
1216 native.c_str(), GENERIC_READ,
1217 FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE, nullptr,
1218 OPEN_EXISTING, FILE_FLAG_OPEN_REPARSE_POINT, nullptr);
1219 if (handle == INVALID_HANDLE_VALUE) {
1220 return false;
1221 }
1222 BY_HANDLE_FILE_INFORMATION info{};
1223 if (!GetFileInformationByHandle(handle, &info) ||
1224 (info.dwFileAttributes &
1225 (FILE_ATTRIBUTE_REPARSE_POINT | FILE_ATTRIBUTE_DIRECTORY |
1226 FILE_ATTRIBUTE_DEVICE)) != 0 ||
1227 GetFileType(handle) != FILE_TYPE_DISK) {
1228 CloseHandle(handle);
1229 return false;
1230 }
1231 const int fd = _open_osfhandle(reinterpret_cast<intptr_t>(handle),
1232 _O_RDONLY | _O_BINARY);
1233 if (fd < 0) {
1234 CloseHandle(handle);
1235 return false;
1236 }
1237 if (!file.open(fd, QIODevice::ReadOnly, QFileDevice::AutoCloseHandle)) {
1238 _close(fd);
1239 return false;
1240 }
1241 return true;
1242#else
1243 // O_NOFOLLOW fails with ELOOP on a symbolic link; O_NONBLOCK keeps a FIFO
1244 // from blocking the open until fstat() can refuse it.
1245 const int fd = ::open(QFile::encodeName(path).constData(),
1246 O_RDONLY | O_NOFOLLOW | O_NONBLOCK | O_CLOEXEC);
1247 if (fd < 0) {
1248 return false;
1249 }
1250 struct stat st{};
1251 if (::fstat(fd, &st) != 0 || !S_ISREG(st.st_mode)) {
1252 ::close(fd);
1253 return false;
1254 }
1255 // The descriptor, not the name, is what QFile opens here: the object
1256 // fstat() judged is the object read. CodeQL's check-then-use pattern
1257 // matcher sees an open after a check and cannot tell.
1258 if (!file.open(fd, QIODevice::ReadOnly, // codeql[cpp/toctou-race-condition]
1259 QFileDevice::AutoCloseHandle)) {
1260 ::close(fd);
1261 return false;
1262 }
1263 return true;
1264#endif
1265}

1298auto Util::readRegularFileUnder(const QString &root, const QString &path,
1299 qint64 maxBytes) -> std::optional<QByteArray> {
1300 const QString cleanRoot = QDir::cleanPath(root);
1301 const QString relative =
1302 QDir(cleanRoot).relativeFilePath(QDir::cleanPath(path));
1303 const QStringList parts = relative.split(QLatin1Char('/'));
1304 if (relative.isEmpty() || QDir::isAbsolutePath(relative) ||
1305 parts.contains(QStringLiteral("..")) ||
1306 parts.contains(QStringLiteral(".")) || parts.contains(QString())) {
1307 return std::nullopt;
1308 }
1309#ifdef Q_OS_WIN
1310 // No openat() here: each folder from the root down is opened by name and
1311 // held open, without FILE_SHARE_DELETE, until the file is read. A folder
1312 // held that way cannot be renamed, removed or replaced by a junction
1313 // meanwhile, so the file is opened through exactly the folders judged.
1314 // The root itself is followed (a linked store is the user's setup).
1315 std::vector<HANDLE> held;
1316 const auto release = qScopeGuard([&held] {
1317 for (HANDLE handle : held) {
1318 CloseHandle(handle);
1319 }
1320 });
1321 QString folder = cleanRoot;
1322 for (qsizetype i = 0; i < parts.size(); ++i) {
1323 const bool isRoot = i == 0;
1324 if (!isRoot) {
1325 folder += QLatin1Char('/') + parts.at(i - 1);
1326 }
1327 const std::wstring native = QDir::toNativeSeparators(folder).toStdWString();
1328 // GENERIC_READ (listing the folder) is data access, so the share mode
1329 // is enforced against a later open for DELETE; attribute access alone
1330 // would not be.
1331 HANDLE handle =
1332 CreateFileW(native.c_str(), GENERIC_READ,
1333 FILE_SHARE_READ | FILE_SHARE_WRITE, nullptr, OPEN_EXISTING,
1334 FILE_FLAG_BACKUP_SEMANTICS |
1335 (isRoot ? 0 : FILE_FLAG_OPEN_REPARSE_POINT),
1336 nullptr);
1337 if (handle == INVALID_HANDLE_VALUE) {
1338 return std::nullopt;
1339 }
1340 held.push_back(handle);
1341 // A link is what the walk (isLink(): QFileInfo::isSymLink() or
1342 // isJunction()) leaves out: a symbolic link or a junction. A folder with
1343 // another reparse tag, such as a cloud provider's placeholder, is listed
1344 // by the walk and is read through here too.
1345 FILE_ATTRIBUTE_TAG_INFO info{};
1346 if (!GetFileInformationByHandleEx(handle, FileAttributeTagInfo, &info,
1347 sizeof(info)) ||
1348 (info.FileAttributes & FILE_ATTRIBUTE_DIRECTORY) == 0) {
1349 return std::nullopt;
1350 }
1351 const bool link =
1352 (info.FileAttributes & FILE_ATTRIBUTE_REPARSE_POINT) != 0 &&
1353 (info.ReparseTag == IO_REPARSE_TAG_SYMLINK ||
1354 info.ReparseTag == IO_REPARSE_TAG_MOUNT_POINT);
1355 if (!isRoot && link) {
1356 return std::nullopt;
1357 }
1358 }
1359 // openRegularFile() refuses a reparse point as the file itself.
1360 return readRegularFile(path, maxBytes);
1361#else
1362 // The root by name, followed if it is a link (the user's setup); below it
1363 // nothing is.
1364 int dir = ::open(QFile::encodeName(cleanRoot).constData(),
1365 O_RDONLY | O_DIRECTORY | O_CLOEXEC);
1366 if (dir < 0) {
1367 return std::nullopt;
1368 }
1369 for (qsizetype i = 0; i + 1 < parts.size(); ++i) {
1370 const int next = ::openat(dir, QFile::encodeName(parts.at(i)).constData(),
1371 O_RDONLY | O_DIRECTORY | O_NOFOLLOW | O_CLOEXEC);

1429auto identityOf(const QFileDevice &file) -> FileIdentity {
1430#ifdef Q_OS_WIN
1431 const HANDLE handle = reinterpret_cast<HANDLE>(_get_osfhandle(file.handle()));
1432 BY_HANDLE_FILE_INFORMATION info;
1433 if (handle == INVALID_HANDLE_VALUE ||
1434 GetFileInformationByHandle(handle, &info) == 0) {
1435 return {};
1436 }
1437 // On FAT/exFAT the file ID is the directory entry's offset, which a rename
1438 // can move. They have no hard links either (the swap this catches needs
1439 // one), so the identity is left unknown there.
1440 DWORD flags = 0;
1441 if (GetVolumeInformationByHandleW(handle, nullptr, 0, nullptr, nullptr,
1442 &flags, nullptr, 0) == 0 ||
1443 (flags & FILE_SUPPORTS_HARD_LINKS) == 0) {
1444 return {};
1445 }
1446 return {info.dwVolumeSerialNumber,
1447 (static_cast<quint64>(info.nFileIndexHigh) << 32) |
1448 info.nFileIndexLow,
1449 true};
1450#else
1451 struct stat st{};
1452 if (::fstat(file.handle(), &st) != 0) {
1453 return {};
1454 }
1455 return {static_cast<quint64>(st.st_dev), static_cast<quint64>(st.st_ino),
1456 true};

1489 file.setAutoRemove(false);

1495 }

1498 // entry is an entry. QTemporaryFile creates 0600 already; say so for

1502 if (why->isEmpty() && !Util::syncToDisk(file)) {
1503 *why = QCoreApplication::translate("Util", "Cannot write %1: %2")
1504 .arg(path, file.errorString());
1505 }

1507#ifndef Q_OS_WIN
1508 // Closed and unlinked by a co-writer before the rename, the temporary's
1509 // inode number would be free, and ext4 hands it to the next file made:
1510 // the very file swapped in would then pass the check below. Windows
1511 // cannot rename an open file, and NTFS does not reuse a file index so.
1512 if (const int fd = ::dup(file.handle()); fd >= 0) {
1513 staged.pin =
1514 std::shared_ptr<void>(nullptr, [fd](void *) { ::close(fd); });
1515 }
1516#endif
1517 }
1518 if (!why->isEmpty()) {
1519 QFile::remove(staged.path);
1520 return std::nullopt;
1521 }
