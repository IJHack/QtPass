126auto Util::passUmask(const QString &value) -> int {
127 // Taken as it is, as bash's umask builtin (which pass hands it to) takes
128 // it: " 022" is no umask there either. pass then keeps the session's;
129 // QtPass falls back to 077.
130 // anchoredPattern(), not ^...$: $ also matches before a final newline.
131 static const QRegularExpression octal(
132 QRegularExpression::anchoredPattern(QStringLiteral("[0-7]{1,4}")));
133 if (!octal.match(value).hasMatch()) {
134 return 077;
135 }
136 return value.toInt(nullptr, 8);
137}

163 const int fd = ::open(QFile::encodeName(path).constData(),
164 O_RDONLY | O_NONBLOCK | O_CLOEXEC);

179auto Util::repositoryPassClimbsTo(const QString &store) -> QString {
180#ifdef Q_OS_WIN
181 Q_UNUSED(store);
182 return {};
183#else
184 if (!store.contains(QLatin1Char(':'))) {
185 return {};
186 }
187 // git walks up from the real folder, so the real path is the one to walk,
188 // and for a store that does not exist yet the one it will have: its
189 // nearest existing folder resolved, the rest appended. By the raw string:
190 // QDir and QFileInfo clean a path by name, and "link/.." is the parent
191 // of the link's target, not the folder the link is in. On macOS the
192 // temporary folder is under /var, a link to /private/var.
193 // Absolute as git means it, a leading "/" (Qt would take ":name" for a
194 // resource path).
195 QString path =
196 store.startsWith(u'/') ? store : QDir::currentPath() + u'/' + store;
197 while (path.size() > 1 && path.endsWith(u'/')) {
198 path.chop(1);
199 }
200 QString rest;
201 while (!QFileInfo::exists(path)) {
202 const qsizetype slash = path.lastIndexOf(u'/');
203 if (slash < 0) {
204 return {};
205 }
206 const QString name = path.mid(slash + 1);
207 rest = rest.isEmpty() ? name : name + u'/' + rest;
208 path = slash == 0 ? QStringLiteral("/") : path.left(slash);
209 }
210 const QString real = QFileInfo(path).canonicalFilePath();
211 QString start = real.isEmpty() ? QDir::cleanPath(path) : real;
212 if (!rest.isEmpty()) {
213 start = QDir::cleanPath(start + u'/' + rest);
214 }
215 // The pieces of pass's ceiling as git takes them (setup.c,
216 // canonicalize_ceiling_entry): relative ones are dropped, the rest
217 // resolved, one that cannot be resolved is dropped; after an empty piece
218 // ("::") they are kept exactly as written. A piece can still be a real
219 // ceiling: "/repo/:vault/.." holds "/repo/".
220 QString ceiling;
221 bool afterEmpty = false;
222 for (const QString &piece :
223 (store + QStringLiteral("/..")).split(QLatin1Char(':'))) {
224 if (piece.isEmpty()) {
225 afterEmpty = true;
226 continue;
227 }
228 if (!piece.startsWith(u'/')) {
229 continue;
230 }
231 const QString entry =
232 afterEmpty ? piece : QFileInfo(piece).canonicalFilePath();
233 // git stops below the longest entry the store lies under
234 // (longest_ancestor_length: "/" is above everything).
235 const QString under = entry == u'/' ? entry : entry + u'/';
236 if (!entry.isEmpty() && start.startsWith(under) &&
237 entry.size() > ceiling.size()) {
238 ceiling = entry;
239 }
240 }
241 // What git takes for a repository (setup.c, is_git_directory()): HEAD
242 // naming a ref or an object, objects/ and refs/. An empty or broken .git
243 // is passed by, as git passes it by.
244 const auto isGitDirectory = [](const QString &candidate) {
245 const QDir dir(candidate);
246 if (!QFileInfo(dir.filePath(QStringLiteral("objects"))).isDir() ||
247 !QFileInfo(dir.filePath(QStringLiteral("refs"))).isDir()) {
248 return false;
249 }
250 // As git (setup.c, validate_headref()): a HEAD that is a link counts
251 // when the link text starts with "refs/", without reading what it
252 // names, and not otherwise. Anything else is read only when it is a
253 // regular file, without waiting: a HEAD that is a FIFO (a pushed commit
254 // can carry a link to one, or to a device) blocked the open and the
255 // read on the GUI thread when Settings opened.
256 const QString headPath = dir.filePath(QStringLiteral("HEAD"));
257 if (const QFileInfo headInfo(headPath); headInfo.isSymLink()) {
258 // The link text as stored (Qt 6.6), not the path it resolves to.
259 return headInfo.readSymLink().startsWith(QLatin1String("refs/"));
260 }
261 QFile head;
262 if (!openFollowedRegularFile(headPath, head)) {
263 return false;
264 }
265 static const QRegularExpression valid(
266 QStringLiteral("^(?:ref: refs/|[0-9a-f]{40}\\s*$|[0-9a-f]{64}\\s*$)"));
267 return valid.match(QString::fromUtf8(head.read(256))).hasMatch();
268 };
269 // At each folder, as git: a .git folder that is a repository, or a .git
270 // file naming one ("gitdir: ..."), is the repository. Any other .git file
271 // stops git with an error, and pass then commits nothing; a .git folder
272 // that is no repository is passed by.
273 enum class Found { Nothing, Repository, Refused };
274 const auto repositoryAt = [&isGitDirectory](const QString &dir) {
275 const QString dotGit = QDir(dir).filePath(QStringLiteral(".git"));
276 const QFileInfo info(dotGit);
277 if (info.isDir()) {
278 return isGitDirectory(dotGit) ? Found::Repository : Found::Nothing;
279 }
280 if (!info.exists()) {
281 return Found::Nothing;
282 }
283 // The same for a .git file (or a link to one): a FIFO there blocked in
284 // open().
285 QFile file;
286 if (!openFollowedRegularFile(dotGit, file)) {
287 return Found::Refused;
288 }
289 const QString line = QString::fromUtf8(file.readLine(4096)).trimmed();
290 if (!line.startsWith(QLatin1String("gitdir: "))) {
291 return Found::Refused;
292 }
293 const QString target = line.mid(8);
294 return isGitDirectory(target.startsWith(u'/') ? target
295 : QDir(dir).filePath(target))
296 ? Found::Repository
297 : Found::Refused;
298 };
299 // The store's own repository comes first; only without one does git look
300 // further up, and never into the ceiling. By name from here: the real
301 // path holds no link or "..".
302 for (QString dir = start;;) {
303 switch (repositoryAt(dir)) {
304 case Found::Repository:
305 return dir == start ? QString() : dir;
306 case Found::Refused:
307 return {};
308 case Found::Nothing:
309 break;
310 }
311 // A bare repository: git stops there, and pass, finding no work tree,
312 // commits nothing.
313 if (isGitDirectory(dir)) {
314 return {};
315 }
316 const QString up = QFileInfo(dir).absolutePath();
317 if (up == dir || up == ceiling) {
318 return {};
319 }
320 dir = up;
321 }
322#endif
323}

359auto isSafeDirectory(const QString &root, const QString &git) -> bool {
360 // Outside any repository, as git reads its protected config: no
361 // repository's own config, and no includeIf "gitdir:" or "onbranch:"
362 // that matches (a GIT_DIR, even one that cannot exist, would be matched
363 // against them, and one that cannot exist makes git config fail). From
364 // "/", which a user cannot make a repository; a local scope is dropped
365 // below in any case.
366 QProcessEnvironment env = QProcessEnvironment::systemEnvironment();
367 env.remove(QStringLiteral("GIT_DIR"));
368 env.remove(QStringLiteral("GIT_WORK_TREE"));
369 env.remove(QStringLiteral("GIT_COMMON_DIR"));
370 // git config alone reads GIT_CONFIG in place of the system and global
371 // configuration (and reports it as "command"); the ownership check never
372 // reads it.
373 env.remove(QStringLiteral("GIT_CONFIG"));
374 QProcess process;
375 process.setProcessEnvironment(env);
376 process.setWorkingDirectory(QStringLiteral("/"));
377 // Bounded: Git calls wait for this, and a git that never answers trusts
378 // nothing.
379 constexpr int kQueryTimeoutMs = 5000;
380 QString out;
381 if (Executor::executeBlocking(
382 process, git.isEmpty() ? QStringLiteral("git") : git,
383 {QStringLiteral("config"), QStringLiteral("--show-scope"),
384 QStringLiteral("--type=path"), QStringLiteral("--get-all"),
385 QStringLiteral("safe.directory")},
386 QString(), &out, nullptr, nullptr, kQueryTimeoutMs) != 0) {
387 return false;
388 }
389 bool safe = false;
390 const QStringList lines = out.split(QLatin1Char('\n'));
391 for (const QString &line : lines) {
392 const qsizetype tab = line.indexOf(QLatin1Char('\t'));
393 if (tab < 0) {
394 continue;
395 }
396 const QString scope = line.left(tab);
397 if (scope != QLatin1String("system") && scope != QLatin1String("global") &&
398 scope != QLatin1String("command")) {
399 continue;
400 }
401 const QString value = line.mid(tab + 1);
402 if (value.isEmpty()) {
403 safe = false;
404 } else if (value == QLatin1String("*")) {
405 safe = true;
406 } else if (!QDir::isAbsolutePath(value)) {
407 continue;
408 } else if (value.endsWith(QLatin1String("/*"))) {
409 const QString base = value.chopped(2);
410 QString prefix = QFileInfo(base.isEmpty() ? QStringLiteral("/") : base)
411 .canonicalFilePath();
412 if (prefix.isEmpty()) {
413 continue;
414 }
415 if (!prefix.endsWith(QLatin1Char('/'))) {
416 prefix += QLatin1Char('/');
417 }
418 safe = safe || root.startsWith(prefix);
419 } else {
420 const QString named = QFileInfo(value).canonicalFilePath();
421 safe = safe || (!named.isEmpty() && named == root);
422 }
423 }
424 return safe;
425}

429auto Util::gitStoreVariables(const QString &store, const QString &git,
430 std::optional<uint> asUser)
431 -> QList<std::pair<QString, QString>> {
432 const QString ceiling = gitCeiling(store);
433 if (ceiling.isEmpty()) {
434 return {};
435 }
436 QList<std::pair<QString, QString>> vars{
437 {QStringLiteral("GIT_CEILING_DIRECTORIES"), ceiling}};
438#ifdef Q_OS_WIN
439 const QChar separator = QLatin1Char(';');
440#else
441 const QChar separator = QLatin1Char(':');
442#endif
443 // git splits the ceiling list on the separator: "/home/a:b" would be
444 // "/home/a" and "b", neither of them the parent, and git would climb.
445 if (ceiling.contains(separator)) {
446 QString root = QFileInfo(store).canonicalFilePath();
447 if (root.isEmpty()) {
448 root = QDir::cleanPath(QDir(store).absolutePath());
449 }
450 QString gitDir = QDir(root).filePath(QStringLiteral(".git"));
451#ifdef Q_OS_UNIX
452 // What discovery's ownership check would have looked at: the work tree,
453 // the .git, and the directory a gitfile names. Not the user's: no
454 // repository at all, under a path that cannot be created (its parent
455 // is a device), rather than another user's hooks run as this one.
456 const uint me = asUser.value_or(
457 s_gitOwnerForTests.value_or(static_cast<uint>(::geteuid())));
458 const auto ownedOrMissing = [me](const QString &path) {
459 const QFileInfo info(path);
460 return !info.exists() || info.ownerId() == me;
461 };
462 bool owned = ownedOrMissing(root) && ownedOrMissing(gitDir);
463 if (owned && QFileInfo(gitDir).isFile()) {
464 QFile gitFile(gitDir);
465 if (gitFile.open(QIODevice::ReadOnly)) {
466 const QString line =
467 QString::fromUtf8(gitFile.readLine(4096)).trimmed();
468 if (line.startsWith(QLatin1String("gitdir:"))) {
469 const QString target = line.mid(7).trimmed();
470 owned = ownedOrMissing(QDir(root).absoluteFilePath(target));
471 }
472 }
473 }
474 // Or trusted as git would trust it in place of ownership.
475 if (!owned && !isSafeDirectory(root, git)) {
476 gitDir = QStringLiteral("/dev/null/.git");
477 }
478#else
479 // Git for Windows runs the same ownership check during discovery, on
480 // SIDs, with its own rules for the Administrators group; this pin skips
481 // it there and does not repeat it (a known gap, #2089).
482 Q_UNUSED(git);
483 Q_UNUSED(asUser);
484#endif
485 vars.append({QStringLiteral("GIT_DIR"), gitDir});
486 vars.append({QStringLiteral("GIT_WORK_TREE"), root});
487 }
488 return vars;
489}

491auto Util::mkpathForStore(const QString &path) -> bool {
492 const QString clean = QDir::cleanPath(QDir(path).absolutePath());
493 if (QFileInfo(clean).isDir()) {
494 return true;
495 }
496 const QString parent = QFileInfo(clean).absolutePath();
497 if (parent != clean && !mkpathForStore(parent)) {
498 return false;
499 }
500#ifdef Q_OS_UNIX
501 // The store's umask, as git and gpg get it: a store shared with a group
502 // (PASSWORD_STORE_UMASK=007) gets group folders, the default owner-only.
503 const int umask = Executor::childUmask() >= 0 ? Executor::childUmask() : 077;
504 QFile::Permissions permissions;
505 const QFile::Permission bits[] = {
506 QFile::ReadOwner, QFile::WriteOwner, QFile::ExeOwner,
507 QFile::ReadGroup, QFile::WriteGroup, QFile::ExeGroup,
508 QFile::ReadOther, QFile::WriteOther, QFile::ExeOther};
509 for (int i = 0; i < 9; ++i) {
510 if ((umask & (0400 >> i)) == 0) {
511 permissions |= bits[i];
512 }
513 }
514 // mkdir() still takes QtPass's own umask off (a session's 022 would drop
515 // the group's write bit); set the mode exactly afterwards. That only ever
516 // widens it to what was asked, never past.
517 const bool made = QDir().mkdir(clean, permissions) &&
518 QFile::setPermissions(clean, permissions);
519#else
520 // Windows: permissions here would become an ACL of their own, replacing
521 // the inherited one; the folder takes its parent's, as before.
522 const bool made = QDir().mkdir(clean);
523#endif
524 return made || QFileInfo(clean).isDir();
525}

527auto Util::findBinaryInPath(const QString &binary) -> QString {
528 if (binary.isEmpty()) {
529 return {};
530 }
531
532 initialiseEnvironment();
533
534 const QStringList dirs =
535 _env.value(QStringLiteral("PATH"))
536 .split(QDir::listSeparator(), Qt::SkipEmptyParts);
537 QString ret;
538 if (QDir::fromNativeSeparators(binary).contains(u'/')) {
539 // An explicit path is not a PATH search: an absolute path is checked
540 // as-is, a relative one is resolved against the PATH directories. The
541 // directory-list overload refuses such names, so handle them here.
542 if (QDir::isAbsolutePath(binary)) {
543 ret = QStandardPaths::findExecutable(binary);
544 } else if (!dirs.isEmpty()) {
545 ret = QStandardPaths::findExecutable(binary, dirs);
546 }
547 } else {
548 ret = findBinaryInPath(binary, dirs);
549 }
550#ifdef Q_OS_WIN
551 if (ret.isEmpty()) {
552 // Cache per-binary WSL lookup result — the wsl --version probe is a
553 // blocking subprocess that can run several times per session for
554 // missing binaries; once decided, the answer doesn't change at runtime.
555 static QHash<QString, QString> wslBinaryCache;
556 const bool hasWhitespace =
557 std::any_of(binary.cbegin(), binary.cend(),
558 [](const QChar ch) { return ch.isSpace(); });
559 if (!hasWhitespace) {
560 auto cached = wslBinaryCache.constFind(binary);
561 if (cached != wslBinaryCache.constEnd()) {
562 ret = cached.value();
563 } else {
564 QString wslCommand = QStringLiteral("wsl ") + binary;
565 qCDebug(lcQtPass)
566 << "Util::findBinaryInPath(): falling back to WSL for binary"
567 << binary;
568 QString out, err;
569 QString cachedResult;
570 if (Executor::executeBlocking(wslCommand, {"--version"}, &out, &err) ==
571 0 &&
572 !out.isEmpty() && err.isEmpty()) {
573 qCDebug(lcQtPass)
574 << "Util::findBinaryInPath(): using WSL binary" << wslCommand;
575 cachedResult = wslCommand;
576 }
577 wslBinaryCache.insert(binary, cachedResult);
578 ret = cachedResult;
579 }
580 }
581 }
582#endif
583
584 return ret;
585}

587// Bare names only: QStandardPaths::findExecutable() returns an absolute binary
588// without consulting searchPaths, and a relative one with ".." could escape

590auto Util::findBinaryInPath(const QString &binary,
591 const QStringList &searchPaths) -> QString {
592 if (binary.isEmpty() || QDir::fromNativeSeparators(binary).contains(u'/')) {
593 return {};
594 }
595 QStringList dirs;
596 dirs.reserve(searchPaths.size());
597 for (const QString &dir : searchPaths) {

603 // QStandardPaths::findExecutable() treats an empty list as "use PATH".

605 }
606 return QStandardPaths::findExecutable(binary, dirs);

662auto authorityLength(const QString &url) -> qsizetype {
663 const auto endsPart = [](QChar c) {
664 return c == QLatin1Char('/') || c == QLatin1Char('?') ||
665 c == QLatin1Char('#');
666 };
667 const auto isAsciiLetter = [](QChar c) {
668 return (c >= QLatin1Char('a') && c <= QLatin1Char('z')) ||
669 (c >= QLatin1Char('A') && c <= QLatin1Char('Z'));
670 };
671 qsizetype colon = -1;
672 for (qsizetype i = 0; i < url.size() && !endsPart(url.at(i)); ++i) {
673 if (url.at(i) == QLatin1Char(':')) {
674 colon = i;
675 break;
676 }
677 }
678 bool scheme = colon > 0 && isAsciiLetter(url.at(0));
679 for (qsizetype i = 1; scheme && i < colon; ++i) {
680 const QChar c = url.at(i);
681 scheme =
682 isAsciiLetter(c) || (c >= QLatin1Char('0') && c <= QLatin1Char('9')) ||
683 c == QLatin1Char('+') || c == QLatin1Char('-') || c == QLatin1Char('.');
684 }
685 const qsizetype hier = scheme ? colon + 1 : 0;
686 if (url.mid(hier, 2) != QLatin1String("//")) {
687 return -1;
688 }
689 qsizetype end = hier + 2;
690 while (end < url.size() && !endsPart(url.at(end))) {
691 ++end;
692 }
693 return end - (hier + 2);
694}

697auto Util::isLaunchableWebUrl(const QString &value) -> bool {
698 const QString trimmed = value.trimmed();
699 if (trimmed.isEmpty()) {
700 return false;
701 }
702 // Reject control characters first, before QUrl normalisation can hide a
703 // CR/LF/NUL injection into the OS URL handler.
704 for (const QChar &c : trimmed) {
705 if (c == QLatin1Char('\r') || c == QLatin1Char('\n') ||
706 c == QChar(QChar::Null)) {
707 return false;
708 }
709 }
710 // QUrl runs the host through Unicode normalisation, which takes time in
711 // the square of a run of combining marks out of canonical order: one line
712 // of an entry kept the window busy for minutes. A host has at most 253
713 // characters and a port six more, so a longer authority is no web
714 // address anyway.
715 constexpr qsizetype kMaxAuthority = 259;
716 if (authorityLength(trimmed) > kMaxAuthority) {
717 return false;
718 }
719 const QUrl url(trimmed, QUrl::StrictMode);
720 if (!url.isValid()) {
721 return false;
722 }
723 const QString scheme = url.scheme().toLower();
724 if (scheme != QLatin1String("http") && scheme != QLatin1String("https")) {
725 return false;
726 }
727 if (url.host().isEmpty()) {
728 return false;
729 }
730 // Embedded userinfo (user:pass@host) would leak into browser history.
731 if (!url.userName().isEmpty() || !url.password().isEmpty()) {
732 return false;
733 }
734 return true;
735}

737auto Util::displayable(const QString &text, bool lineBreaks) -> QString {
738 QString shown = text;
739 for (qsizetype i = 0; i < shown.size(); ++i) {
740 const char16_t u = shown.at(i).unicode();
741 // A carriage return before a line feed is a Windows line end (git or
742 // gpg output quoted in a message), not a character of its own.
743 const bool lineEnd =
744 lineBreaks && (u == u'\n' || (u == u'\r' && i + 1 < shown.size() &&
745 shown.at(i + 1) == u'\n'));
746 if ((u < 0x20 && u != u'\t' && !lineEnd) || (u >= 0x7f && u <= 0x9f) ||
747 u == 0x2028 || u == 0x2029 || (u >= 0x202a && u <= 0x202e) ||
748 (u >= 0x2066 && u <= 0x2069)) {
749 shown[i] = QChar(QChar::ReplacementCharacter);
750 }
751 }
752 return shown;
753}

755auto Util::linkifyUrls(const QString &text, bool *linked) -> QString {
756 if (linked != nullptr) {
757 *linked = false;
758 }
759 QString html;
760 html.reserve(text.size());
761 qsizetype lastIndex = 0;
762 QRegularExpressionMatchIterator it = protocolRegex().globalMatch(text);
763 while (it.hasNext()) {
764 const QRegularExpressionMatch match = it.next();
765 const QString url = match.captured(0);
766 // Only plain ASCII becomes a link: the link text is the URL as stored,
767 // and a right-to-left letter, a bidi control or a look-alike host in it
768 // makes it read as another address than the one it opens. Such a URL
769 // stays plain text; the open button still takes it.
770 const bool plain = std::all_of(url.cbegin(), url.cend(), [](QChar c) {
771 return c.unicode() > 0x20 && c.unicode() < 0x7f;
772 });
773 if (!plain || !isLaunchableWebUrl(url)) {
774 // Not a web URL (or it carries credentials): leave it in the escaped
775 // plain-text run instead of making it clickable.
776 continue;
777 }
778 const qsizetype start = match.capturedStart(0);
779 html += text.mid(lastIndex, start - lastIndex).toHtmlEscaped();
780 const QString escapedUrl = url.toHtmlEscaped();
781 html += QStringLiteral("<a href=\"%1\">%1</a>").arg(escapedUrl);
782 lastIndex = match.capturedEnd(0);
783 if (linked != nullptr) {
784 *linked = true;
785 }
786 }
787 html += text.mid(lastIndex).toHtmlEscaped();
788 // With a link in it, no bidi embedding, override or isolate control is
789 // drawn: an override earlier in the entry (U+202E) drew the link text
790 // reversed, so it read as another address. Shown as U+FFFD, so the user
791 // sees that something was there; the copy button still copies the value.
792 if (linked != nullptr ? *linked : html.contains(QLatin1String("<a href="))) {
793 for (QChar &c : html) {
794 const char16_t u = c.unicode();
795 if ((u >= 0x202A && u <= 0x202E) || (u >= 0x2066 && u <= 0x2069)) {
796 c = QChar(QChar::ReplacementCharacter);
797 }
798 }
799 }
800 return html;
801}

817auto Util::hasLongUnbrokenRun(QStringView text) -> bool {
818 qsizetype run = 0;
819 for (const QChar c : text) {
820 // A no-break space (U+00A0, U+2007, U+202F) is white space to
821 // isSpace(), but Qt does not wrap at it.
822 const char16_t u = c.unicode();
823 const bool breaks =
824 c.isSpace() && u != 0x00A0 && u != 0x2007 && u != 0x202F;
825 run = breaks ? 0 : run + 1;
826 if (run > kMaxUnbrokenRun) {
827 return true;
828 }
829 }
830 return false;
831}

833// One pass, no regex: a pattern such as \s{1001,} retries from every
834// character of a shorter run, which is quadratic on an entry made of them.
835// A tab or a line feed ends a run in Qt's layout as well, so it ends one
836// here; \v and U+0085 do not, so they count.
837auto Util::boundSpaceRuns(const QString &text) -> QString {
838 QString bounded;
839 // Text before this index is in bounded already, or dropped.
840 qsizetype from = 0;
841 qsizetype run = 0;
842 for (qsizetype i = 0; i < text.size(); ++i) {
843 const QChar ch = text.at(i);
844 if (!ch.isSpace() || ch == QLatin1Char('\t') || ch == QLatin1Char('\n')) {
845 run = 0;
846 continue;
847 }
848 if (++run <= kMaxSpaceRun) {
849 continue;
850 }
851 if (run == kMaxSpaceRun + 1) {
852 bounded += QStringView(text).sliced(from, i - from);
853 }
854 from = i + 1;
855 }
856 if (from == 0) {
857 return text;
858 }
859 bounded += QStringView(text).sliced(from);
860 return bounded;
861}

903template <typename Visit>
904void walkStore(const QString &dir, Visit visit,
905 QStringList *unlisted = nullptr) {
906 // Absolute, so the results are whatever the caller's cwd; not canonical,
907 // so a linked root keeps the name it was configured under.
908 QStringList pending{QDir(QDir::cleanPath(dir)).absolutePath()};
909 while (!pending.isEmpty()) {
910 const QDir current(pending.takeLast());
911 if (unlisted != nullptr && current.exists() && !canList(current.path())) {
912 qCWarning(lcQtPass) << "Cannot list" << current.path();
913 *unlisted << current.path();
914 continue;
915 }
916 // Every entry once, links and hidden entries included, so the decision
917 // what to do with each is taken here, before any recursion. QDir::System
918 // keeps dangling links in the listing.
919 const QFileInfoList entries =
920 current.entryInfoList(QDir::Dirs | QDir::Files | QDir::Hidden |
921 QDir::System | QDir::NoDotAndDotDot,
922 QDir::Name);
923 QStringList subdirs;
924 for (const QFileInfo &entry : entries) {
925 if (visit(entry)) {
926 subdirs << entry.filePath();
927 }
928 }
929 // Pushed last-to-first so the next one taken is the first by name.
930 for (auto it = subdirs.crbegin(); it != subdirs.crend(); ++it) {
931 pending << *it;
932 }
933 }
934}

957auto Util::regularFilesUnder(const QString &dir, const QStringList &nameFilters,
958 QStringList *skipped, bool hiddenFiles,
959 QStringList *unlisted) -> QStringList {
960 QStringList files;
961 walkStore(
962 dir,
963 [&](const QFileInfo &entry) {
964 if (isStoreDirectory(entry)) {
965 return true;
966 }
967 const bool named = QDir::match(nameFilters, entry.fileName());
968 if (isLink(entry) || (!entry.isDir() && !entry.isFile())) {
969 // A linked directory hides everything behind it; a linked file, or a
970 // FIFO, socket or device, is only of interest under a name the caller
971 // asked for.
972 if (skipped != nullptr && (entry.isDir() || named)) {
973 qCWarning(lcQtPass) << "Skipping" << entry.filePath()
974 << ": not a regular file or directory";
975 *skipped << entry.filePath();
976 }
977 return false;
978 }
979 if (entry.isFile() && named && (hiddenFiles || !isHiddenEntry(entry))) {
980 files << entry.filePath();
981 }
982 return false;
983 },
984 unlisted);
985 return files;
986}

1025auto Util::isUnderLink(const QString &path, const QString &storeRoot,
1026 bool includeSelf) -> bool {
1027 // "C:/Store" and "c:/store" are one directory on Windows; a path spelled
1028 // the other way must not skip the walk (as in Pass::getGpgIdPath).
1029#ifdef Q_OS_WIN
1030 constexpr auto cs = Qt::CaseInsensitive;
1031#else
1032 constexpr auto cs = Qt::CaseSensitive;
1033#endif
1034 const QString root = QDir::cleanPath(storeRoot);
1035 // A root of "/" or "C:/" already ends in the separator.
1036 const QString prefix =
1037 root.endsWith(QLatin1Char('/')) ? root : root + QLatin1Char('/');
1038 const auto isRoot = [&](const QString &p) {
1039 return p.compare(root, cs) == 0;
1040 };
1041 QString current = QDir::cleanPath(path);
1042 if (!current.startsWith(prefix, cs)) {
1043 // Not under the store as named: only the entry itself can be judged.
1044 return includeSelf && !isRoot(current) && isLinkedFolder(current);
1045 }
1046 if (!includeSelf) {
1047 current = QFileInfo(current).path();
1048 }
1049 for (; !isRoot(current) && current.startsWith(prefix, cs);
1050 current = QFileInfo(current).path()) {
1051 if (isLinkedFolder(current)) {
1052 return true;
1053 }
1054 }
1055 return false;
1056}

1106auto placeWithoutReplacing(const QByteArray &source, const QByteArray &target)
1107 -> bool {
1108 // linkat() without AT_SYMLINK_FOLLOW follows nothing; link() would, on
1109 // macOS and the BSDs, hard-link whatever a symlink planted under the
1110 // source's name points at.
1111 if (::linkat(AT_FDCWD, source.constData(), AT_FDCWD, target.constData(), 0) ==
1112 0) {
1113 ::unlink(source.constData());
1114 return true;
1115 }
1116 if (!hasNoHardLinks(errno)) {
1117 return false;
1118 }
1119 if (const std::optional<bool> renamed = renameExclusive(source, target)) {
1120 return *renamed;
1121 }
1122 struct stat st{};
1123 if (::lstat(target.constData(), &st) == 0 || errno != ENOENT) {
1124 errno = EEXIST;
1125 return false;
1126 }
1127 return ::rename(source.constData(), target.constData()) == 0;
1128}

1133auto Util::replaceFile(const QString &from, const QString &to, bool replace)
1134 -> bool {
1135#ifdef Q_OS_WIN
1136 const std::wstring source =
1137 QDir::toNativeSeparators(QFileInfo(from).absoluteFilePath())
1138 .toStdWString();
1139 const std::wstring target =
1140 QDir::toNativeSeparators(QFileInfo(to).absoluteFilePath()).toStdWString();
1141 // WRITE_THROUGH: the new entry is on the device when this returns.
1142 return MoveFileExW(source.c_str(), target.c_str(),
1143 (replace ? MOVEFILE_REPLACE_EXISTING : 0) |
1144 MOVEFILE_WRITE_THROUGH) != 0;
1145#else
1146 const QByteArray source = QFile::encodeName(from);
1147 const QByteArray target = QFile::encodeName(to);
1148 if (replace) {
1149 if (::rename(source.constData(), target.constData()) != 0) {
1150 return false;
1151 }
1152 } else {
1153 if (!placeWithoutReplacing(source, target)) {
1154 return false;
1155 }
1156 }
1157 // Sync the directory entry so a crash does not lose the new name. Best
1158 // effort: the bytes are already synced, and some network filesystems
1159 // cannot sync a directory.
1160 const int dir = ::open(QFile::encodeName(QFileInfo(to).path()).constData(),
1161 O_RDONLY | O_DIRECTORY | O_CLOEXEC);
1162 if (dir >= 0) {
1163 int rc;
1164 do {
1165 rc = ::fsync(dir);
1166 } while (rc != 0 && errno == EINTR);
1167 ::close(dir);
1168 }
1169 return true;
1170#endif
1171}

1173auto Util::openRegularFile(const QString &path, QFile &file) -> bool {
1174#ifdef Q_OS_WIN
1175 // FILE_FLAG_OPEN_REPARSE_POINT opens a symbolic link or junction itself
1176 // rather than its target, so the handle's attributes say what the name
1177 // was at the moment of the open.
1178 const std::wstring native =
1179 QDir::toNativeSeparators(QFileInfo(path).absoluteFilePath())
1180 .toStdWString();
1181 HANDLE handle = CreateFileW(
1182 native.c_str(), GENERIC_READ,
1183 FILE_SHARE_READ | FILE_SHARE_WRITE | FILE_SHARE_DELETE, nullptr,
1184 OPEN_EXISTING, FILE_FLAG_OPEN_REPARSE_POINT, nullptr);
1185 if (handle == INVALID_HANDLE_VALUE) {
1186 return false;
1187 }
1188 BY_HANDLE_FILE_INFORMATION info{};
1189 if (!GetFileInformationByHandle(handle, &info) ||
1190 (info.dwFileAttributes &
1191 (FILE_ATTRIBUTE_REPARSE_POINT | FILE_ATTRIBUTE_DIRECTORY |
1192 FILE_ATTRIBUTE_DEVICE)) != 0 ||
1193 GetFileType(handle) != FILE_TYPE_DISK) {
1194 CloseHandle(handle);
1195 return false;
1196 }
1197 const int fd = _open_osfhandle(reinterpret_cast<intptr_t>(handle),
1198 _O_RDONLY | _O_BINARY);
1199 if (fd < 0) {
1200 CloseHandle(handle);
1201 return false;
1202 }
1203 if (!file.open(fd, QIODevice::ReadOnly, QFileDevice::AutoCloseHandle)) {
1204 _close(fd);
1205 return false;
1206 }
1207 return true;
1208#else
1209 // O_NOFOLLOW fails with ELOOP on a symbolic link; O_NONBLOCK keeps a FIFO
1210 // from blocking the open until fstat() can refuse it.
1211 const int fd = ::open(QFile::encodeName(path).constData(),
1212 O_RDONLY | O_NOFOLLOW | O_NONBLOCK | O_CLOEXEC);
1213 if (fd < 0) {
1214 return false;
1215 }
1216 struct stat st{};
1217 if (::fstat(fd, &st) != 0 || !S_ISREG(st.st_mode)) {
1218 ::close(fd);
1219 return false;
1220 }
1221 // The descriptor, not the name, is what QFile opens here: the object
1222 // fstat() judged is the object read. CodeQL's check-then-use pattern
1223 // matcher sees an open after a check and cannot tell.
1224 if (!file.open(fd, QIODevice::ReadOnly, // codeql[cpp/toctou-race-condition]
1225 QFileDevice::AutoCloseHandle)) {
1226 ::close(fd);
1227 return false;
1228 }
1229 return true;
1230#endif
1231}

1281auto identityOf(const QFileDevice &file) -> FileIdentity {
1282#ifdef Q_OS_WIN
1283 const HANDLE handle = reinterpret_cast<HANDLE>(_get_osfhandle(file.handle()));
1284 BY_HANDLE_FILE_INFORMATION info;

1286 GetFileInformationByHandle(handle, &info) == 0) {
1287 return {};
1288 }

1290 // can move. They have no hard links either (the swap this catches needs
1291 // one), so the identity is left unknown there.

1294 &flags, nullptr, 0) == 0 ||
1295 (flags & FILE_SUPPORTS_HARD_LINKS) == 0) {
1296 return {};

1299 (static_cast<quint64>(info.nFileIndexHigh) << 32) |
1300 info.nFileIndexLow,
1301 true};
1302#else
1303 struct stat st{};
1304 if (::fstat(file.handle(), &st) != 0) {
1305 return {};
1306 }
1307 return {static_cast<quint64>(st.st_dev), static_cast<quint64>(st.st_ino),
1308 true};
1309#endif

1335auto stageNextTo(const QString &path, const Util::Filler &fill, QString *why)
1336 -> std::optional<Staged> {
1337 Staged staged;
1338 {
1339 QTemporaryFile file(QFileInfo(path).path() +
1340 QStringLiteral("/.qtpass-XXXXXX.tmp"));
1341 file.setAutoRemove(false);
1342 if (!file.open()) {
1343 *why = QCoreApplication::translate(
1344 "Util", "Cannot create a temporary file next to %1: %2")
1345 .arg(path, file.errorString());
1346 return std::nullopt;
1347 }
1348 staged.path = file.fileName();
1349 // Owner-only: a .gpg-id names the keys a store is encrypted to, an
1350 // entry is an entry. QTemporaryFile creates 0600 already; say so for
1351 // platforms where it may not.
1352 file.setPermissions(QFile::ReadOwner | QFile::WriteOwner);
1353 *why = fill(file);
1354 if (why->isEmpty() && !Util::syncToDisk(file)) {
1355 *why = QCoreApplication::translate("Util", "Cannot write %1: %2")
1356 .arg(path, file.errorString());
1357 }
1358 staged.identity = identityOf(file);
1359#ifndef Q_OS_WIN
1360 // Closed and unlinked by a co-writer before the rename, the temporary's
1361 // inode number would be free, and ext4 hands it to the next file made:
1362 // the very file swapped in would then pass the check below. Windows
1363 // cannot rename an open file, and NTFS does not reuse a file index so.
1364 if (const int fd = ::dup(file.handle()); fd >= 0) {
1365 staged.pin =
1366 std::shared_ptr<void>(nullptr, [fd](void *) { ::close(fd); });
1367 }
1368#endif
1369 }
1370 if (!why->isEmpty()) {
1371 QFile::remove(staged.path);
1372 return std::nullopt;
1373 }
1374 return staged;
1375}

1515auto Util::removeTree(const QString &dir) -> bool {

1525 return false;

1529 QDir(path).entryInfoList(QDir::Dirs | QDir::Files | QDir::Hidden |
1530 QDir::System | QDir::NoDotAndDotDot,
1531 QDir::Name);
