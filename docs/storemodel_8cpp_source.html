57auto StoreModel::showThis(const QModelIndex &index) const -> bool {
58 if (fs == nullptr) {
59 return false;
60 }
61 // Through filePath() a filter pass cost entries times sibling folders,
62 // again on every key typed in the search.
63 const QString filePath = pathOf(index);
64 QString path = QDir(store).relativeFilePath(filePath);
65 // Only what lies in the current store. The file system model keeps the
66 // rows of a store shown earlier (a profile switch only re-roots it), and
67 // a search would otherwise match those, open one and save it encrypted
68 // to this store's recipients. Folders above the store still lead there.
69 const bool outside = path == QLatin1String("..") ||
70 path.startsWith(QLatin1String("../")) ||
71 QDir::isAbsolutePath(path);
72 // A link inside the store is neither an entry nor a folder (SECURITY.md):
73 // it is listed by its name, so it can be deleted, but never entered. A
74 // planted `x -> .` was an endlessly expandable folder, and the search's
75 // expandAll() loaded it level after level until memory ran out.
76 const bool inside = !outside && path != QLatin1String(".");
77 if (inside && Util::isUnderLink(filePath, store, false)) {
78 return false;
79 }
80 const bool link = inside && Util::isLinkedFolder(filePath);
81 if (!link && sourceModel()->rowCount(index) > 0) {
82 for (int nChild = 0; nChild < sourceModel()->rowCount(index); ++nChild) {
83 const QModelIndex childIndex = sourceModel()->index(nChild, 0, index);
84 if (!childIndex.isValid()) {
85 break;
86 }
87 if (showThis(childIndex)) {
88 return true;
89 }
90 }
91 return false;
92 }
93 if (outside || path.startsWith(".git")) {
94 return false;
95 }
96 path.replace(Util::endsWithGpg(), "");
97 return path.contains(filterRegularExpression());
98}

100auto StoreModel::wordsFilter(const QString &query) -> QRegularExpression {
101 // Regular expressions belong to the content search; here a typed "[" or
102 // "(" must not switch filtering off, so each word is escaped.
103 QString pattern;
104 for (const QString &word :
105 query.split(QLatin1Char(' '), Qt::SkipEmptyParts)) {
106 pattern += QStringLiteral("(?>.*?") + QRegularExpression::escape(word) +
107 QLatin1Char(')');
108 }
109 if (pattern.isEmpty()) {
110 return QRegularExpression();
111 }
112 return QRegularExpression(QLatin1Char('^') + pattern,
113 QRegularExpression::CaseInsensitiveOption |
114 QRegularExpression::DotMatchesEverythingOption);
115}

117auto StoreModel::isLinkInStore(const QModelIndex &proxyIndex) const -> bool {
118 if (fs == nullptr || !proxyIndex.isValid()) {
119 return false;
120 }
121 // The same path showThis() lists the row by, or a link it lists would
122 // still get an expand arrow on Windows.
123 const QString filePath = pathOf(mapToSource(proxyIndex));
124 const QString path = QDir(store).relativeFilePath(filePath);
125 if (path == QLatin1String(".") || path == QLatin1String("..") ||
126 path.startsWith(QLatin1String("../")) || QDir::isAbsolutePath(path)) {
127 return false;
128 }
129 return Util::isLinkedFolder(filePath);
130}

219auto StoreModel::mimeData(const QModelIndexList &indexes) const -> QMimeData * {
220 StoreDragItem info;
221
222 if (indexes.isEmpty())
223 return nullptr;
224 QByteArray encodedData;
225 // only use the first, otherwise we should enable multiselection
226 QModelIndex index = indexes.at(0);
227 if (index.isValid()) {
228 QModelIndex useIndex = mapToSource(index);
229 const QFileInfo fileInfo = fs->fileInfo(useIndex);
230
231 if (fileInfo.isDir()) {
232 info.kind = StoreDragItem::ItemKind::Directory;
233 } else if (fileInfo.isFile()) {
234 info.kind = StoreDragItem::ItemKind::File;
235 }
236 info.path = fileInfo.absoluteFilePath();
237 QDataStream stream(&encodedData, QIODevice::WriteOnly);
238 stream << info;
239 }
240
241 auto *mimeData = new QMimeData();
242 mimeData->setData(kStoreDragMimeType, encodedData);
243 return mimeData;
244}

246auto StoreModel::canDropMimeData(const QMimeData *data, Qt::DropAction action,
247 int row, int column,
248 const QModelIndex &parent) const -> bool {
249 qCDebug(lcQtPass) << "canDropMimeData" << action << row;
250
251 const auto parsed = parseDropData(data);
252 if (!parsed) {
253 return false;
254 }
255 const StoreDragItem &info = *parsed;
256
257 QModelIndex useIndex =
258 this->index(parent.row(), parent.column(), parent.parent());
259
260 if (column > 0) {
261 return false;
262 }
263
264 using IK = StoreDragItem::ItemKind;
265 // you can drop a folder on a folder
266 if (fs->fileInfo(mapToSource(useIndex)).isDir() &&
267 info.kind == IK::Directory) {
268 return true;
269 }
270 // you can drop a file on a folder
271 if (fs->fileInfo(mapToSource(useIndex)).isDir() && info.kind == IK::File) {
272 return true;
273 }
274 // you can drop a file on a file
275 if (fs->fileInfo(mapToSource(useIndex)).isFile() && info.kind == IK::File) {
276 return true;
277 }
278
279 return false;
280}

282auto StoreModel::dropMimeData(const QMimeData *data, Qt::DropAction action,
283 int row, int column, const QModelIndex &parent)
284 -> bool {
285 if (!canDropMimeData(data, action, row, column, parent)) {
286 return false;
287 }
288
289 if (action == Qt::IgnoreAction) {
290 return true;
291 }
292
293 if (action != Qt::MoveAction && action != Qt::CopyAction) {
294 return false;
295 }
296
297 const auto info = parseDropData(data);
298 if (!info) {
299 return false;
300 }
301
302 return executeDropAction(*info, action, parent);
303}

305auto StoreModel::parseDropData(const QMimeData *data)
306 -> std::optional<StoreDragItem> {
307 if (data == nullptr || !data->hasFormat(kStoreDragMimeType)) {
308 return std::nullopt;
309 }
310 QByteArray encodedData = data->data(kStoreDragMimeType);
311 if (encodedData.isEmpty()) {
312 return std::nullopt;
313 }
314 QDataStream stream(&encodedData, QIODevice::ReadOnly);
315 StoreDragItem info;
316 stream >> info;
317 if (stream.status() != QDataStream::Ok) {
318 return std::nullopt;
319 }
320 return info;
321}

323auto StoreModel::executeDropAction(const StoreDragItem &info,
324 Qt::DropAction action,
325 const QModelIndex &parent) -> bool {
326 QModelIndex destIndex =
327 this->index(parent.row(), parent.column(), parent.parent());
328 QFileInfo destFileinfo = fs->fileInfo(mapToSource(destIndex));
329 QFileInfo srcFileInfo = QFileInfo(info.path);
330
331 QString cleanedSrc = QDir::cleanPath(srcFileInfo.absoluteFilePath());
332 QString cleanedDest = QDir::cleanPath(destFileinfo.absoluteFilePath());
333
334 // Drop data could be crafted: both endpoints must resolve inside the store
335 // after symlink resolution (e.g. no in-store symlink pointing at /etc).
336 if (!PathValidator::isPathInStore(store, cleanedSrc) ||
337 !PathValidator::isPathInStore(store, cleanedDest)) {
338 qCWarning(lcQtPass)
339 << "executeDropAction: rejecting drop that escapes the store"
340 << "(src=" << cleanedSrc << "dest=" << cleanedDest << ")";
341 return false;
342 }
343
344 switch (info.kind) {
345 case StoreDragItem::ItemKind::Directory: {
346 // Dropping a folder onto a folder: move/copy it *into* the target.
347 if (!destFileinfo.isDir()) {
348 return false;
349 }
350 const QString destDir =
351 QDir::cleanPath(QDir(cleanedDest).filePath(srcFileInfo.fileName()));
352 return performDrop(cleanedSrc, destDir, action, false);
353 }
354 case StoreDragItem::ItemKind::File:
355 // File onto a folder drops into it (no clash); file onto an existing
356 // file asks before overwriting.
357 if (destFileinfo.isDir()) {
358 return performDrop(cleanedSrc, cleanedDest, action, false);
359 }
360 return performDrop(cleanedSrc, cleanedDest, action,
361 PlainMessageBox::askYes(
362 qobject_cast<QWidget *>(QObject::parent()),
363 tr("Force overwrite?"),
364 tr("overwrite %1 with %2?")
365 .arg(Util::displayable(cleanedDest, false),
366 Util::displayable(cleanedSrc, false))));
367 default:
368 qCWarning(lcQtPass)
369 << "executeDropAction: unexpected ItemKind, ignoring drop";
370 return false;
371 }
372}

374auto StoreModel::performDrop(const QString &cleanedSrc,
375 const QString &cleanedDest, Qt::DropAction action,
376 bool force) -> bool {
377 if (!m_pass) {
378 return false;
379 }
380 if (action == Qt::MoveAction) {
381 m_pass->Move(cleanedSrc, cleanedDest, force);
382 } else if (action == Qt::CopyAction) {
383 m_pass->Copy(cleanedSrc, cleanedDest, force);
384 }
385 return true;
