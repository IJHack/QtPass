56auto StoreModel::showThis(const QModelIndex &index) const -> bool {
57 bool retVal = false;
58 if (fs == nullptr) {
59 return retVal;
60 }
61 if (sourceModel()->rowCount(index) > 0) {
62 for (int nChild = 0; nChild < sourceModel()->rowCount(index); ++nChild) {
63 QModelIndex childIndex = sourceModel()->index(nChild, 0, index);
64 if (!childIndex.isValid()) {
65 break;
66 }
67 retVal = showThis(childIndex);
68 if (retVal) {
69 break;
70 }
71 }
72 } else {
73 QModelIndex useIndex = sourceModel()->index(index.row(), 0, index.parent());
74 QString path = fs->filePath(useIndex);
75 path = QDir(store).relativeFilePath(path);
76 if (path.startsWith(".git")) {
77 return false;
78 }
79 path.replace(Util::endsWithGpg(), "");
80 retVal = path.contains(filterRegularExpression());
81 }
82 return retVal;
83}

151auto StoreModel::mimeData(const QModelIndexList &indexes) const -> QMimeData * {
152 StoreDragItem info;
153
154 if (indexes.isEmpty())
155 return nullptr;
156 QByteArray encodedData;
157 // only use the first, otherwise we should enable multiselection
158 QModelIndex index = indexes.at(0);
159 if (index.isValid()) {
160 QModelIndex useIndex = mapToSource(index);
161 const QFileInfo fileInfo = fs->fileInfo(useIndex);
162
163 if (fileInfo.isDir()) {
164 info.kind = StoreDragItem::ItemKind::Directory;
165 } else if (fileInfo.isFile()) {
166 info.kind = StoreDragItem::ItemKind::File;
167 }
168 info.path = fileInfo.absoluteFilePath();
169 QDataStream stream(&encodedData, QIODevice::WriteOnly);
170 stream << info;
171 }
172
173 auto *mimeData = new QMimeData();
174 mimeData->setData(kStoreDragMimeType, encodedData);
175 return mimeData;
176}

178auto StoreModel::canDropMimeData(const QMimeData *data, Qt::DropAction action,
179 int row, int column,
180 const QModelIndex &parent) const -> bool {
181 qCDebug(lcQtPass) << "canDropMimeData" << action << row;
182
183 const auto parsed = parseDropData(data);
184 if (!parsed) {
185 return false;
186 }
187 const StoreDragItem &info = *parsed;
188
189 QModelIndex useIndex =
190 this->index(parent.row(), parent.column(), parent.parent());
191
192 if (column > 0) {
193 return false;
194 }
195
196 using IK = StoreDragItem::ItemKind;
197 // you can drop a folder on a folder
198 if (fs->fileInfo(mapToSource(useIndex)).isDir() &&
199 info.kind == IK::Directory) {
200 return true;
201 }
202 // you can drop a file on a folder
203 if (fs->fileInfo(mapToSource(useIndex)).isDir() && info.kind == IK::File) {
204 return true;
205 }
206 // you can drop a file on a file
207 if (fs->fileInfo(mapToSource(useIndex)).isFile() && info.kind == IK::File) {
208 return true;
209 }
210
211 return false;
212}

214auto StoreModel::dropMimeData(const QMimeData *data, Qt::DropAction action,
215 int row, int column, const QModelIndex &parent)
216 -> bool {
217 if (!canDropMimeData(data, action, row, column, parent)) {
218 return false;
219 }
220
221 if (action == Qt::IgnoreAction) {
222 return true;
223 }
224
225 if (action != Qt::MoveAction && action != Qt::CopyAction) {
226 return false;
227 }
228
229 const auto info = parseDropData(data);
230 if (!info) {
231 return false;
232 }
233
234 return executeDropAction(*info, action, parent);

237auto StoreModel::parseDropData(const QMimeData *data)
238 -> std::optional<StoreDragItem> {
239 if (data == nullptr || !data->hasFormat(kStoreDragMimeType)) {
240 return std::nullopt;
241 }
242 QByteArray encodedData = data->data(kStoreDragMimeType);
243 if (encodedData.isEmpty()) {
244 return std::nullopt;
245 }
246 QDataStream stream(&encodedData, QIODevice::ReadOnly);
247 StoreDragItem info;
248 stream >> info;
249 if (stream.status() != QDataStream::Ok) {
250 return std::nullopt;
251 }
252 return info;
253}

255auto StoreModel::executeDropAction(const StoreDragItem &info,
256 Qt::DropAction action,
257 const QModelIndex &parent) -> bool {
258 QModelIndex destIndex =
259 this->index(parent.row(), parent.column(), parent.parent());
260 QFileInfo destFileinfo = fs->fileInfo(mapToSource(destIndex));
261 QFileInfo srcFileInfo = QFileInfo(info.path);
262
263 QString cleanedSrc = QDir::cleanPath(srcFileInfo.absoluteFilePath());
264 QString cleanedDest = QDir::cleanPath(destFileinfo.absoluteFilePath());
265
266 // Drop data could be crafted: both endpoints must resolve inside the store
267 // after symlink resolution (e.g. no in-store symlink pointing at /etc).
268 if (!PathValidator::isPathInStore(store, cleanedSrc) ||
269 !PathValidator::isPathInStore(store, cleanedDest)) {
270 qCWarning(lcQtPass)
271 << "executeDropAction: rejecting drop that escapes the store"
272 << "(src=" << cleanedSrc << "dest=" << cleanedDest << ")";
273 return false;
274 }
275
276 switch (info.kind) {
277 case StoreDragItem::ItemKind::Directory: {
278 // Dropping a folder onto a folder: move/copy it *into* the target.
279 if (!destFileinfo.isDir()) {
280 return false;
281 }
282 const QString destDir =
283 QDir::cleanPath(QDir(cleanedDest).filePath(srcFileInfo.fileName()));
284 return performDrop(cleanedSrc, destDir, action, false);
285 }
286 case StoreDragItem::ItemKind::File:
287 // File onto a folder drops into it (no clash); file onto an existing
288 // file asks before overwriting.
289 if (destFileinfo.isDir()) {
290 return performDrop(cleanedSrc, cleanedDest, action, false);
291 }
292 return performDrop(
293 cleanedSrc, cleanedDest, action,
294 PlainMessageBox::askYes(
295 qobject_cast<QWidget *>(QObject::parent()), tr("Force overwrite?"),
296 tr("overwrite %1 with %2?").arg(cleanedDest, cleanedSrc)));
297 default:
298 qCWarning(lcQtPass)
299 << "executeDropAction: unexpected ItemKind, ignoring drop";
300 return false;
301 }
302}

304auto StoreModel::performDrop(const QString &cleanedSrc,
305 const QString &cleanedDest, Qt::DropAction action,
306 bool force) -> bool {
307 if (!m_pass) {
308 return false;
309 }
310 if (action == Qt::MoveAction) {
311 m_pass->Move(cleanedSrc, cleanedDest, force);
312 } else if (action == Qt::CopyAction) {
313 m_pass->Copy(cleanedSrc, cleanedDest, force);
314 }
315 return true;
316}

318auto StoreModel::lessThan(const QModelIndex &source_left,

320/* matches logic in QFileSystemModelSorter::compareNodes() */
321#ifndef Q_OS_MAC

323 bool leftD = fs->isDir(source_left);
324 bool rightD = fs->isDir(source_right);
325
326 if (leftD ^ rightD) {
