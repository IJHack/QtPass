33 connect(&m_process, &QProcess::finished, this, &Executor::onProcessFinished);
34 connect(&m_process, &QProcess::started, this, &Executor::starting);

40 if (m_discardStdout) {
41 m_process.readAllStandardOutput();

53auto utf8SequenceLength(unsigned char lead) -> int {
54 if (lead < 0x80) {
55 return 1;
56 }
57 if ((lead & 0xC0) == 0x80) {
58 return 0;
59 }
60 if ((lead & 0xE0) == 0xC0) {
61 return 2;
62 }
63 if ((lead & 0xF0) == 0xE0) {
64 return 3;
65 }
66 return 4;
67}

71void Executor::CappedOutput::countStatus(const QByteArray &line) {
72 static const QByteArray prefix = QByteArrayLiteral("[GNUPG:] ");
73 if (!line.startsWith(prefix)) {
74 return;
75 }
76 QByteArray keyword = line.mid(prefix.size());
77 keyword.truncate(keyword.indexOf(' ') < 0 ? keyword.size()
78 : keyword.indexOf(' '));
79 if (keyword.endsWith('\r')) {
80 keyword.chop(1);
81 }
82 const QString key = QString::fromLatin1(keyword);
83 if (m_status.contains(key) || m_status.size() < kMaxStatusKeywords) {
84 ++m_status[key];
85 }
86}

88void Executor::CappedOutput::append(const QByteArray &chunk) {
89 // Status records, whole lines at a time, before the text is cut.
90 qsizetype start = 0;
91 for (qsizetype end = chunk.indexOf('\n'); end >= 0;
92 start = end + 1, end = chunk.indexOf('\n', start)) {
93 if (!m_lineTooLong &&
94 m_line.size() + (end - start) <= kMaxStatusLineBytes) {
95 m_line.append(chunk.constData() + start, end - start);
96 countStatus(m_line);
97 }
98 m_line.clear();
99 m_lineTooLong = false;
100 }
101 if (!m_lineTooLong) {
102 if (m_line.size() + (chunk.size() - start) > kMaxStatusLineBytes) {
103 m_line.clear();
104 m_lineTooLong = true;
105 } else {
106 m_line.append(chunk.constData() + start, chunk.size() - start);
107 }
108 }
109
110 constexpr qsizetype half = kMaxStderrBytes / 2;
111 qsizetype from = 0;
112 if (m_head.size() < half) {
113 from = qMin(half - m_head.size(), chunk.size());
114 m_head.append(chunk.constData(), from);
115 }
116 if (from == chunk.size()) {
117 return;
118 }
119 m_tail.append(chunk.constData() + from, chunk.size() - from);
120 if (m_tail.size() > half) {
121 const qsizetype excess = m_tail.size() - half;
122 m_dropped += excess;
123 m_tail.remove(0, excess);
124 }
125}

127// When bytes were left out, the head may end and the tail begin in the
128// middle of a character: both are cut back to whole ones, so the text
129// still decodes as UTF-8 rather than falling back to another encoding.
130auto Executor::CappedOutput::take() -> QByteArray {
131 QByteArray all;
132 if (m_dropped == 0) {
133 all = m_head + m_tail;
134 } else {
135 QByteArray head = m_head;
136 qsizetype chopped = 0;
137 for (qsizetype back = 1; back <= qMin<qsizetype>(4, head.size()); ++back) {
138 const int length = utf8SequenceLength(
139 static_cast<unsigned char>(head.at(head.size() - back)));
140 if (length != 0) {
141 if (length > back) {
142 chopped = back;
143 head.chop(back);
144 }
145 break;
146 }
147 }
148 qsizetype start = 0;
149 while (start < m_tail.size() &&
150 utf8SequenceLength(static_cast<unsigned char>(m_tail.at(start))) ==
151 0) {
152 ++start;
153 }
154 all = head + QByteArrayLiteral("\n[... ") +
155 QByteArray::number(m_dropped + chopped + start) +
156 QByteArrayLiteral(" bytes left out ...]\n") + m_tail.mid(start);
157 }
158 m_head.clear();
159 m_tail.clear();
160 m_dropped = 0;
161 // A last line without a line end is a record too.
162 if (!m_lineTooLong && !m_line.isEmpty()) {
163 countStatus(m_line);
164 }
165 m_line.clear();
166 m_lineTooLong = false;
167 return all;
168}

172void Executor::detachFromTerminal(QProcess &process) {
173#if defined(Q_OS_UNIX) && QT_VERSION >= QT_VERSION_CHECK(6, 7, 0)
174 // A child inherits QtPass's controlling terminal when QtPass was started
175 // from one. git (HTTPS credentials) and ssh (key passphrase, host key) then
176 // prompt on that /dev/tty, and as the queue runs one command at a time,
177 // every decrypt and save waits behind a prompt nobody sees: QtPass looks
178 // frozen. In a session of its own the child has no terminal, so git and
179 // ssh use SSH_ASKPASS/GIT_ASKPASS when set and otherwise fail with an
180 // error QtPass shows. GIT_TERMINAL_PROMPT=0 is not set on top: without a
181 // tty git has nothing to prompt on, and some credential helpers read it as
182 // "no interaction at all" and would skip their own dialogs.
183 //
184 // gpg is unaffected: pinentry is started by gpg-agent, which finds the
185 // terminal through GPG_TTY (set by the user's shell) or the tty on gpg's
186 // stdin, here always a pipe. pass is a shell script running that gpg and
187 // git, so the same holds for it.
188 //
189 // Signal dispositions are left alone (no ResetSignalHandlers): exec()
190 // already resets caught signals, and QtPass sets no SIG_IGN of its own, so
191 // that flag would only undo what QtPass's own parent chose (nohup, say).
192 // Terminal signals (Ctrl+C, the SIGHUP of a closed terminal) now reach
193 // QtPass but not the child; the children are short-lived and, with no
194 // terminal, cannot sit waiting for input.
195 QProcess::UnixProcessParameters params = process.unixProcessParameters();
196 params.flags |= QProcess::UnixProcessFlag::CreateNewSession;
197 process.setUnixProcessParameters(params);
198#else
199 Q_UNUSED(process);
200#endif
201}

207void Executor::startProcess(QProcess &process, const QString &app,
208 const QStringList &args) {
209 detachFromTerminal(process);
210#ifdef Q_OS_UNIX
211 // In the child, between fork and exec: umask() is async-signal-safe.
212 if (const int mask = g_childUmask.load(); mask >= 0) {
213 process.setChildProcessModifier(
214 [mask] { ::umask(static_cast<mode_t>(mask)); });
215 }
216#endif
217 if (const auto wsl = parseWslCommand(app)) {
218 process.start(wsl->launcher, wsl->argv(args));
219 } else {
220 process.start(resolveExecutable(app), args);
221 }

224auto Executor::resolveExecutable(const QString &app) -> QString {
225 if (app.isEmpty() || QDir::isAbsolutePath(app)) {
226 return app;
227 }
228 const QDir appDir(QCoreApplication::applicationDirPath());
229 QStringList candidates{QDir::cleanPath(appDir.absoluteFilePath(app))};
230#ifdef Q_OS_WIN
231 candidates << QDir::cleanPath(
232 appDir.absoluteFilePath(app + QStringLiteral(".exe")));
233#endif
234 for (const QString &candidate : std::as_const(candidates)) {
235 // A stray non-executable file of that name must not mask PATH.
236 const QFileInfo info(candidate);
237 if (info.isFile() && info.isExecutable()) {
238 return candidate;
239 }
240 }
241 return app;
242}

252auto Executor::parseWslCommand(const QString &app)
253 -> std::optional<WslCommand> {
254 const QStringList parts = QProcess::splitCommand(app);
255 if (parts.size() < 2) {
256 return std::nullopt;
257 }
258 // The file name after the last separator of either kind: a Windows path
259 // has to be recognised as such wherever the parser runs.
260 const QString &launcher = parts.first();
261 const qsizetype separator = std::max(launcher.lastIndexOf(QLatin1Char('/')),
262 launcher.lastIndexOf(QLatin1Char('\\')));
263 const QString launcherName = launcher.mid(separator + 1);
264 const bool isWsl =
265 launcherName.compare(QLatin1String("wsl"), Qt::CaseInsensitive) == 0 ||
266 launcherName.compare(QLatin1String("wsl.exe"), Qt::CaseInsensitive) == 0;
267 if (!isWsl) {
268 return std::nullopt;
269 }
270 WslCommand wsl;
271#ifdef Q_OS_WIN
272 // A bare wsl or wsl.exe in any case is looked up on PATH as `wsl`; a path
273 // stays as is.
274 wsl.launcher = separator < 0 ? QStringLiteral("wsl") : launcher;
275#else
276 // QtPass running inside WSL reaches wsl.exe through interop under exactly
277 // that name; nothing says `wsl` resolves too, so start what was written.
278 wsl.launcher = launcher;
279#endif
280 // wsl.exe options that take a value; anything else starting with `-` is a
281 // flag. A user-written -e/--exec is dropped, argv() always adds one.
282 static const QStringList valued{
283 QStringLiteral("-d"), QStringLiteral("--distribution"),
284 QStringLiteral("-u"), QStringLiteral("--user"),
285 QStringLiteral("--cd"), QStringLiteral("--shell-type")};
286 qsizetype i = 1;
287 for (; i < parts.size() && parts.at(i).startsWith(QLatin1Char('-')); ++i) {
288 const QString &option = parts.at(i);
289 if (option == QLatin1String("-e") || option == QLatin1String("--exec")) {
290 continue;
291 }
292 if (valued.contains(option)) {
293 if (i + 1 >= parts.size()) {
294 return std::nullopt;
295 }
296 wsl.options << option << parts.at(++i);
297 } else {
298 wsl.options << option;
299 }
300 }
301 // Exactly one program. More is a shell command line (`sh -c "..."`),
302 // which is what --exec exists to keep out.
303 if (i != parts.size() - 1) {
304 return std::nullopt;
305 }
306 wsl.command = parts.at(i);
307 return wsl;
308}

315auto Executor::translatePathForWsl(const QString &path, const QString &exe)
316 -> QString {
317 QString normalizedPath = QDir::cleanPath(path);
318 const auto wsl = parseWslCommand(exe);
319 if (!wsl) {
320 return normalizedPath;
321 }
322 QString wslPath;
323 const auto wslpath = wsl->with(QStringLiteral("wslpath"));
324 const int rc = executeBlocking(wslpath.launcher,
325 wslpath.argv({normalizedPath}), &wslPath);
326 // Only the line end wslpath prints: white space at the end of the path is
327 // the name's own ("team " beside "team", a folder named " "), and trimmed
328 // it made git remove or move the neighbour, or the whole store.
329 QString translated = wslPath;
330 if (translated.endsWith(QLatin1Char('\n'))) {
331 translated.chop(1);
332 }
333 if (translated.endsWith(QLatin1Char('\r'))) {
334 translated.chop(1);
335 }
336 return (rc == 0 && !translated.isEmpty()) ? translated : normalizedPath;
337}

339void Executor::executeNext() {
340 // Nothing starts while a failed start's error() is still on its way: every
341 // other failure is emitted before the next item starts (onProcessFinished),
342 // and a caller that drops the rest of its work on a failure (ImitatePass's
343 // insert-then-commit) relies on that order. The posted error continues the
344 // queue once it is out.
345 if (running || m_pendingErrors > 0 || m_execQueue.isEmpty()) {
346 return;
347 }
348 // A copy, and marked running, before anyone hears of it: a slot on
349 // dequeuing() that queues more cannot invalidate it or start it a second
350 // time, and one that calls cancelNext() cannot drop it.
351 const ExecQueueItem i = m_execQueue.head();
352 running = true;
353 emit dequeuing(i.id, true);
354
355 // An empty executable never emits finished(); dropping it wedged the queue
356 // (#1682). Fail it like a failed-to-start process so the queue drains.
357 if (i.app.isEmpty()) {
358 qCDebug(lcQtPass) << "No executable set for:" << i.id;
359 const int failedId = i.id;
360 running = false;
361 m_execQueue.dequeue();
362 ++m_pendingErrors;
363 QMetaObject::invokeMethod(
364 this,
365 [this, failedId]() {
366 emit error(failedId, -1, QString(),
367 tr("No executable configured for this command"));
368 --m_pendingErrors;
369 executeNext();
370 postDrainedIfIdle();
371 },
372 Qt::QueuedConnection);
373 return;
374 }
375
376 // Always set it: an item without a directory runs in the application's
377 // cwd, not wherever the previous item happened to run.
378 m_process.setWorkingDirectory(i.workingDir);
379 m_stderr.clear();
380 m_discardStdout = !i.readStdout;
381 startProcess(m_process, i.app, i.args);
382
383 // A failed start never emits finished(), so `running` would stall the queue
384 // and waiters (the keygen dialog) would hang. The emit is queued so it does
385 // not re-enter KeygenDialog::done(), still on the stack; -1 goes through
386 // Pass::finished's non-zero error gate.
387 if (!m_process.waitForStarted(-1)) {
388 qCDebug(lcQtPass) << "Process failed to start:" << i.id << " " << i.app;
389 const int failedId = i.id;
390 const QString failedApp = i.app;
391 m_process.closeWriteChannel();
392 running = false;
393 m_execQueue.dequeue();
394 ++m_pendingErrors;
395 QMetaObject::invokeMethod(
396 this,
397 [this, failedId, failedApp]() {
398 emit error(failedId, -1, QString(),
399 tr("Failed to start %1").arg(failedApp));
400 --m_pendingErrors;
401 executeNext();
402 postDrainedIfIdle();
403 },
404 Qt::QueuedConnection);
405 return;
406 }
407
408 if (!i.input.isEmpty()) {
409 QByteArray data = i.input.toUtf8();
410 if (m_process.write(data) != data.length()) {
411 qCDebug(lcQtPass) << "Not all data written to process:" << i.id << " "
412 << i.app;
413 }
414 }
415 m_process.closeWriteChannel();
416}

434void Executor::execute(int id, const QString &workDir, const QString &app,
435 const QStringList &args, QString input, bool readStdout,
436 bool readStderr) {
437 // Queue even an empty executable: executeNext() reports it (#1682).
438 // startProcess resolves the executable, as on the blocking path.
439 m_execQueue.push_back(
440 {id, app, args, std::move(input), readStdout, readStderr, workDir});
441 executeNext();
442}

444// UTF-8 first; on a decoding error fall back to the system encoding.
445// lossless, when given, says whether the text encodes back to exactly these
446// bytes: what writing it to gpg or pass (toUtf8(), as every writer here
447// does) would store. Not for bytes that are not UTF-8 (each became U+FFFD,
448// or on Windows a code-page character), nor for a leading byte order mark,
449// which the decoder drops without an error. A literal U+FFFD, as older
450// QtPass versions left in such entries, does encode back.
451static auto decodeAssumingUtf8(const QByteArray &in, bool *lossless = nullptr)
452 -> QString {
453 // Stateless: a truncated trailing byte becomes a replacement character
454 // instead of being held back for a continuation that never comes.
455 auto converter =
456 QStringDecoder(QStringDecoder::Utf8, QStringDecoder::Flag::Stateless);
457 QString out = converter(in);
458 if (converter.hasError()) {
459 auto fallback =
460 QStringDecoder(QStringDecoder::System, QStringDecoder::Flag::Stateless);
461 out = fallback(in);
462 }
463 if (lossless != nullptr) {
464 *lossless = out.toUtf8() == in;
465 }
466 return out;
467}

477#ifdef Q_OS_UNIX
485auto waitForGroup(pid_t group, int ms) -> bool {
486 QDeadlineTimer deadline(ms);
487 // kill(-group, 0) fails with ESRCH once no process is left in it; while
488 // one is, no new process can be given that id.
489 while (::kill(-group, 0) == 0) {
490 if (deadline.hasExpired()) {
491 return false;
492 }
493 QThread::msleep(10);
494 }
495 return true;
496}
497#endif

509void stopBlocking(QProcess &process) {
510#ifdef Q_OS_UNIX
511 const auto group = static_cast<pid_t>(process.processId());
512 const auto signalGroup = [group](int signal) {
513 if (group > 0) {
514 ::kill(-group, signal);
515 }
516 };
517 signalGroup(SIGTERM);
518#endif
519 process.terminate();
520 if (!process.waitForFinished(kBlockingKillGraceMs)) {
521#ifdef Q_OS_UNIX
522 signalGroup(SIGKILL);
523#endif
524 process.kill();
525 // SIGKILL cannot be ignored, but a child in uninterruptible sleep (a
526 // hung NFS or FUSE mount) dies only when the kernel lets it. Wait one
527 // more grace period, not forever: the run is reported as failed either
528 // way, and a blocking run then lasts at most its timeout plus twice the
529 // grace.
530 if (!process.waitForFinished(kBlockingKillGraceMs)) {
531 qCWarning(lcQtPass) << "Process still running after kill():"
532 << process.program();
533 }
534 }
535#ifdef Q_OS_UNIX
536 if (group > 0 && !waitForGroup(group, kBlockingKillGraceMs)) {
537 signalGroup(SIGKILL);
538 if (!waitForGroup(group, kBlockingKillGraceMs)) {
539 qCWarning(lcQtPass) << "Processes started by" << process.program()
540 << "still running after kill()";
541 }
542 }
543#endif
544}

550auto msecsLeft(const QDeadlineTimer &deadline) -> int {
551 if (deadline.isForever()) {
552 return -1;
553 }
554 return static_cast<int>(
555 std::clamp<qint64>(deadline.remainingTime(), 0, INT_MAX));
556}

567auto waitOrCancel(QProcess &process, const std::atomic_bool *cancel,
568 const QDeadlineTimer &deadline) -> bool {
569 if (cancel == nullptr) {
570 // waitForFinished() can also fail (a poll error) with the child still
571 // running; that is not completion either. Stop it rather than report
572 // it done.
573 if (!process.waitForFinished(msecsLeft(deadline)) &&
574 process.state() != QProcess::NotRunning) {
575 stopBlocking(process);
576 return false;
577 }
578 return true;
579 }
580 while (!process.waitForFinished(
581 deadline.isForever()
582 ? kBlockingCancelPollMs
583 : qMin(kBlockingCancelPollMs, msecsLeft(deadline)))) {
584 if (process.state() == QProcess::NotRunning) {
585 return true;
586 }
587 if (cancel->load() || deadline.hasExpired()) {
588 stopBlocking(process);
589 return false;
590 }
591 }
592 return true;
593}

597auto Executor::runBlocking(QProcess &process, const QString &app,
598 const QStringList &args, const QString &input,
599 QString *process_out, QString *process_err,
600 const std::atomic_bool *cancel, int timeoutMs)
601 -> int {
602 QByteArray out;
603 const int rc = runBlockingRaw(process, app, args, input.toUtf8(),
604 process_out != nullptr ? &out : nullptr,
605 process_err, cancel, timeoutMs);
606 if (process_out != nullptr) {
607 *process_out = decodeAssumingUtf8(out);
608 }
609 return rc;
610}

612auto Executor::runBlockingRaw(QProcess &process, const QString &app,
613 const QStringList &args, const QByteArray &input,
614 QByteArray *process_out, QString *process_err,
615 const std::atomic_bool *cancel, int timeoutMs)
616 -> int {
617 if (cancel != nullptr && cancel->load())
618 return -1;
619 const QDeadlineTimer deadline = timeoutMs < 0
620 ? QDeadlineTimer(QDeadlineTimer::Forever)
621 : QDeadlineTimer(timeoutMs);
622 // Read as it arrives (QProcess emits these from inside its waitFor*()
623 // calls too), so neither channel piles up in QProcess's buffers: stderr
624 // into the cap, stdout nobody asked for into nothing. Wanted stdout is
625 // kept whole, as a decrypt must be lossless; gpg's --max-output bounds it.
626 CappedOutput err;
627 const QMetaObject::Connection errReader = QObject::connect(
628 &process, &QProcess::readyReadStandardError,
629 [&process, &err]() { err.append(process.readAllStandardError()); });
630 const QMetaObject::Connection outDropper =
631 process_out != nullptr
632 ? QMetaObject::Connection()
633 : QObject::connect(&process, &QProcess::readyReadStandardOutput,
634 [&process]() { process.readAllStandardOutput(); });
635 const auto disconnectReaders = qScopeGuard([&errReader, &outDropper]() {
636 QObject::disconnect(errReader);
637 QObject::disconnect(outDropper);
638 });
639 startProcess(process, app, args);
640 if (!process.waitForStarted(msecsLeft(deadline))) {
641 qCDebug(lcQtPass) << "Process failed to start:" << app;
642 if (process.state() != QProcess::NotRunning) {
643 stopBlocking(process);
644 }
645 return -1;
646 }
647 if (!input.isEmpty()) {
648 if (process.write(input) != input.length()) {
649 qCDebug(lcQtPass) << "Not all input written:" << app;
650 }
651 }
652 // Always close stdin so a child blocking on EOF doesn't hang.
653 process.closeWriteChannel();
654 if (!waitOrCancel(process, cancel, deadline)) {
655 return -1;
656 }
657 // Read before judging the exit: a process that crashed may have said why,
658 // as the queued path (onProcessFinished) keeps it too.
659 if (process_out != nullptr) {
660 *process_out = process.readAllStandardOutput();
661 }
662 if (process_err != nullptr) {
663 err.append(process.readAllStandardError());
664 *process_err = decodeAssumingUtf8(err.take());
665 }
666 return process.exitStatus() == QProcess::NormalExit ? process.exitCode() : -1;
667}

718// Also -1 while the head is already running: that one is not cancelled.
719auto Executor::cancelNext() -> int {
720 if (running || m_execQueue.isEmpty()) {
721 return -1;
722 }
723 const int id = m_execQueue.dequeue().id;
724 // As in executeNext(): while the signal is out, a slot's cancelNext()
725 // drops nothing and its execute() only queues. Nothing is started here
726 // afterwards: the caller (ImitatePass dropping the rest of a failed
727 // transaction from its completion handler) may cancel again, and the
728 // queue moves on when that handler returns (onProcessFinished()).
729 running = true;
730 emit dequeuing(id, false);
731 running = false;
732 return id;
733}

735void Executor::onProcessFinished(int exitCode,
736 QProcess::ExitStatus exitStatus) {
737 ExecQueueItem i = m_execQueue.dequeue();
738 running = false;
739 auto [output, err] = collectOutput(i, exitCode);
740 if (exitStatus == QProcess::NormalExit) {
741 if (exitCode != 0) {
742 qCDebug(lcQtPass) << i.app << "exited with" << exitCode << err;
743 }
744 emit finished(i.id, exitCode, output, err);
745 m_outputLossless = true;
746 m_lastStatus.clear();
747 } else {
748 // A signal-killed process usually leaves stderr empty; without this the
749 // password pane would show nothing at all for the failure.
750 if (err.trimmed().isEmpty()) {
751 err = tr("%1 crashed or was killed").arg(i.app);
752 }
753 // Qt leaves exitCode undefined after a CrashExit (the signal number on
754 // Unix), and Pass::finished gates on it: 0 would pass as success and 1
755 // as grep's "no matches". Report -1, like the failed-to-start path.
756 emit error(i.id, -1, output, err);
757 m_outputLossless = true;
758 m_lastStatus.clear();
759 }
760 executeNext();
761 postDrainedIfIdle();
762}

764void Executor::postDrainedIfIdle() {
765 // One check at a time, run after what is emitted now; it fires only once
766 // nothing runs or waits and every posted error() of a failed start is
767 // out (the last of those posts the check again).
768 if (m_drainCheckPosted) {
769 return;
770 }
771 m_drainCheckPosted = true;
772 QMetaObject::invokeMethod(
773 this,
774 [this]() {
775 m_drainCheckPosted = false;
776 if (isIdle() && m_pendingErrors == 0) {
777 emit drained();
778 }
779 },
780 Qt::QueuedConnection);
781}

783auto Executor::collectOutput(const ExecQueueItem &item, int exitCode)
784 -> std::pair<QString, QString> {
785 QString output;
786 QString err;
787 m_outputLossless = true;
788 if (item.readStdout) {
789 output = decodeAssumingUtf8(m_process.readAllStandardOutput(),
790 &m_outputLossless);
791 }
792 m_stderr.append(m_process.readAllStandardError());
793 const QByteArray errBytes = m_stderr.take();
794 m_lastStatus = m_stderr.statusCounts();
795 if (item.readStderr || exitCode != 0) {
796 err = decodeAssumingUtf8(errBytes);
797 }
798 return {output, err};
799}
