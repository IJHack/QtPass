65 // date with the remote.

121auto RealPass::refuseList(const QString &file, const GpgIdSigner &signer,
122 QSet<QString> &verified, const QString &within,
123 const QString &except, SkippedWalks *walks) -> bool {
124 if (refuseSkippedGpgId(file, true, except, walks)) {
125 return true;
126 }
127 const QString gpgIdFile = Pass::getGpgIdPath(file, m_settings.passStore);
128 const QString absolute =
129 QDir::cleanPath(QFileInfo(gpgIdFile).absoluteFilePath());
130 if (verified.contains(gpgIdFile) ||
131 (!except.isEmpty() &&
132 absolute == QDir::cleanPath(QFileInfo(except).absoluteFilePath())) ||
133 (!within.isEmpty() && !absolute.startsWith(within))) {
134 return false;
135 }
136 QString why =
137 QCoreApplication::translate("ImitatePass", "Signature for %1 is invalid.")
138 .arg(gpgIdFile);
139 QByteArray contents;
140 if (!verifySignedList(signer, gpgIdFile, &contents, &why)) {
141 emit critical(QCoreApplication::translate("ImitatePass",
142 "Check .gpg-id file signature!"),
143 why);
144 return true;
145 }
146 verified.insert(gpgIdFile);
147 return false;
148}

174auto RealPass::refuseMisreadList(const QString &file, const QString &within,
175 const QString &except, QSet<QString> &checked)
176 -> bool {
177 const QString gpgIdFile = Pass::getGpgIdPath(file, m_settings.passStore);
178 const QString absolute =
179 QDir::cleanPath(QFileInfo(gpgIdFile).absoluteFilePath());
180 if (checked.contains(absolute) ||
181 (!except.isEmpty() &&
182 absolute == QDir::cleanPath(QFileInfo(except).absoluteFilePath())) ||
183 (!within.isEmpty() && !absolute.startsWith(within))) {
184 return false;
185 }
186 // A missing list is pass's to report. One that is there but not a
187 // regular file within the limit (too large, a FIFO, a device, unreadable)
188 // pass may still read in full, while the Users dialog shows none of it.
189 // A link was refused before this.
190 const std::optional<QByteArray> bytes =
191 Util::readRegularFile(gpgIdFile, Util::kMaxGpgIdBytes);
192 if (!bytes && QFileInfo::exists(gpgIdFile)) {
193 //: %1 is the list's path, %2 a size such as "1 MiB".
194 emit critical(tr("Recipient list cannot be read"),
195 tr("%1 is not a regular file of at most %2, so QtPass "
196 "cannot show who pass would encrypt to. Nothing was "
197 "encrypted. Check the recipients in Users for that "
198 "folder and press OK to write the list anew.")
199 .arg(QDir::toNativeSeparators(gpgIdFile),
200 QLocale().formattedDataSize(Util::kMaxGpgIdBytes)));
201 return true;
202 }
203 const QString misread =
204 bytes ? Pass::passReadsDifferently(*bytes) : QString();
205 if (misread.isEmpty()) {
206 checked.insert(absolute);
207 return false;
208 }
209 //: %1 is the list's path, %2 says what is wrong with it.
210 emit critical(tr("Recipient list reads differently"),
211 tr("pass reads %1 differently from the Users dialog (%2), so "
212 "it could encrypt to recipients the dialog does not show. "
213 "Nothing was encrypted. Check the recipients in Users for "
214 "that folder and press OK to write the list anew.")
215 .arg(QDir::toNativeSeparators(gpgIdFile), misread));
216 return true;
217}

219auto RealPass::refuseUnverifiedList(const QString &file) -> bool {
220 // pass verifies the signature itself, but knows nothing of the generation
221 // and folder lines: an older authentic list put back, or one signed for
222 // another folder, would get the secret (#2077). The same checks as the
223 // native backend, with gpg in the store's environment. pass reads the list
224 // again afterwards; a swap in between is a concurrent writer, out of
225 // scope (SECURITY.md). A linked list differs from what QtPass shows,
226 // signing key or not.
227 QSet<QString> readAlike;
228 if (refuseLinkedList(file) ||
229 refuseMisreadList(file, QString(), QString(), readAlike)) {
230 return true;
231 }
232 const GpgIdSigner signer = listSigner();
233 if (!signer.enabled()) {
234 return false;
235 }
236 QSet<QString> verified;
237 return refuseList(file, signer, verified);
238}

308auto RealPass::refuseUnverifiedTree(const QString &dir, bool onlyListsWithin,
309 const QString &except) -> bool {
310 // Every folder pass's walk takes. A name Qt cannot read hides what is
311 // behind it, a linked list included, so the walk must be complete first,
312 // signing key or not.
313 QStringList unreadable;
314 const QStringList folders = passFoldersUnder(dir, unreadable);
315 if (!unreadable.isEmpty()) {
316 const QString title = tr("Cannot check the recipient lists");
317 //: %1 is a folder. "UTF-8" is the text encoding; keep it as is.
318 const QString text =
319 tr("%1 holds a file or folder whose name is not valid UTF-8. QtPass "
320 "cannot see what is behind that name, while pass would re-encrypt "
321 "it, so nothing was changed. Rename it, then try again.")
322 .arg(QDir::toNativeSeparators(unreadable.first()));
323 emit critical(title, text);
324 return true;
325 }
326 if (refuseSpecialEntry(firstSpecialEntry(folders))) {
327 return true;
328 }
329 // Absolute, as getGpgIdPath() hands lists back: a relative or other
330 // spelling would match nothing and check nothing.
331 const QString within =
332 onlyListsWithin ? QDir::cleanPath(QFileInfo(dir).absoluteFilePath()) +
333 QLatin1Char('/')
334 : QString();
335 // A linked list differs from what QtPass shows, signing key or not: in
336 // every folder of the walk, and above them unless only the lists within
337 // count (the folder that moves gets the destination's list, the folder
338 // pass init writes its own).
339 QSet<QString> readAlike;
340 for (const QString &folder : folders) {
341 if (refuseLinkedList(folder + QLatin1Char('/'), within, except) ||
342 refuseMisreadList(folder + QLatin1Char('/'), within, except,
343 readAlike)) {
344 return true;
345 }
346 }
347 const GpgIdSigner signer = listSigner();
348 if (!signer.enabled()) {
349 return false;
350 }
351 // One gpg call per list, before pass runs: a store with many lists makes
352 // that a noticeable pause, so say what it is.
353 emit statusMsg(tr("Checking the recipient lists under %1…")
354 .arg(QDir::toNativeSeparators(QDir::cleanPath(dir))),
355 3000);
356 // pass mv and cp re-encrypt every entry of the folder they write to, each
357 // to its nearest list, walking as passFoldersUnder() does. For the folder
358 // that moves, the lists inside it go along: checked where they lie now
359 // (after the move their folder line names the old place until the
360 // recipients are saved there again), while a list above it is replaced
361 // by the destination's, which the caller checks. For entries already in
362 // the destination, their lists stay where they are: all of them count.
363 QSet<QString> verified;
364 // Each folder's skipped-list answer once for the whole tree: asked again
365 // from every folder, a deep chain cost depth squared record lookups.
366 SkippedWalks walks;
367 // Every folder's list counts, also where no entry uses it yet: cp -r
368 // merges an incoming folder into one of the same name, so the entries it
369 // brings get a list already there, and a list it brings replaces the one
370 // there for the entries already in it.
371 for (const QString &folder : folders) {
372 if (refuseList(folder, signer, verified, within, except, &walks)) {
373 return true;
374 }
375 }
376 return false;
377}

407auto RealPass::repositoryOf(const QString &path) -> QString {
408 if (!m_settings.useGit) {
409 return {};
410 }
411 const QString root = QDir::cleanPath(m_settings.passStore);
412 // From the nearest folder that exists, as pass looks for it.
413 QString folder = QDir::cleanPath(path);
414 while (!QFileInfo(folder).isDir() &&
415 folder.startsWith(root + QLatin1Char('/'))) {
416 folder = QFileInfo(folder).path();
417 }
418 const QString relative = QDir(root).relativeFilePath(folder);
419 if (relative == QLatin1String(".") || relative == QLatin1String("..") ||
420 relative.startsWith(QLatin1String("../"))) {
421 return {};
422 }
423 // The folder's place in its repository ("team/" in the store's own,
424 // "" at the top of the one it holds itself).
425 QString prefix;
426 storeGit({QStringLiteral("-C"), relative, QStringLiteral("rev-parse"),
427 QStringLiteral("--show-prefix")},
428 &prefix);
429 if (prefix.endsWith(QLatin1Char('\n'))) {
430 prefix.chop(1);
431 }
432 const QString withSlash = relative + QLatin1Char('/');
433 if (!withSlash.endsWith(prefix)) {
434 return {};
435 }
436 QString repository = withSlash.chopped(prefix.size());
437 if (repository.endsWith(QLatin1Char('/'))) {
438 repository.chop(1);
439 }
440 return repository;
441}

443auto RealPass::statusEntries(const QStringList &args, const QString &repository)
444 -> QStringList {
445 if (!m_settings.useGit) {
446 return {};
447 }
448 QString out;
449 // -z: names as they are, not quoted when they hold a non-ASCII letter.
450 // `pass git` exits 0 whatever git did: the output decides, and a git
451 // that cannot say (no repository) prints nothing to it.
452 storeGit(inRepository(repository, QStringList{QStringLiteral("status"),
453 QStringLiteral("--porcelain"),
454 QStringLiteral("-z")} +
455 args),
456 &out);
457 const QString place =
458 repository.isEmpty() ? QString() : repository + QLatin1Char('/');
459 QStringList entries;
460 const QStringList fields = out.split(QChar(0), Qt::SkipEmptyParts);
461 for (qsizetype i = 0; i < fields.size(); ++i) {
462 const QString &field = fields.at(i);
463 if (field.size() < 4 || field.at(2) != QLatin1Char(' ')) {
464 continue;
465 }
466 entries << field.left(3) + place + field.mid(3);
467 // A rename or copy, in the index or in the work tree (an intent-to-add
468 // file), is followed by its old name.
469 if (field.at(0) == QLatin1Char('R') || field.at(0) == QLatin1Char('C') ||
470 field.at(1) == QLatin1Char('R') || field.at(1) == QLatin1Char('C')) {
471 ++i;
472 }
473 }
474 return entries;
475}

477auto RealPass::unrecorded(const QStringList &paths, bool ignored)
478 -> QStringList {
479 if (paths.isEmpty()) {
480 return {};
481 }
482 const QDir store(QDir::cleanPath(m_settings.passStore));
483 // Each path in the repository pass commits it in: a folder may hold one
484 // of its own, which the store's status does not look into.
485 QMap<QString, QStringList> byRepository;
486 for (const QString &path : paths) {
487 const QString repository = repositoryOf(path);
488 const QDir top(repository.isEmpty() ? store.path()
489 : store.filePath(repository));
490 byRepository[repository] << top.relativeFilePath(QDir::cleanPath(path));
491 }
492 QStringList entries;
493 for (auto it = byRepository.cbegin(); it != byRepository.cend(); ++it) {
494 QStringList args{QStringLiteral("--untracked-files=all")};
495 if (ignored) {
496 args << QStringLiteral("--ignored");
497 }
498 args << QStringLiteral("--");
499 entries += statusEntries(args + it.value(), it.key());
500 }
501 return entries;
502}

564auto RealPass::refuseUnrecordedChanges(const QStringList &paths,
565 const QStringList &committing) -> bool {
566 QStringList changes = unrecorded(paths, false);
567 QStringList repositories;
568 for (const QString &path : committing) {
569 const QString repository = repositoryOf(path);
570 if (!repositories.contains(repository)) {
571 repositories << repository;
572 }
573 }
574 // pass commits the whole index, not only what it staged: anything staged
575 // elsewhere goes along.
576 for (const QString &repository : std::as_const(repositories)) {
577 for (const QString &entry :
578 statusEntries({QStringLiteral("--untracked-files=no")}, repository)) {
579 if (entry.at(0) != QLatin1Char(' ') && !changes.contains(entry)) {
580 changes << entry;
581 }
582 }
583 }
584 if (changes.isEmpty()) {
585 return false;
586 }
587 QString why =
588 tr("The folder holds changes Git has not recorded, and pass would "
589 "commit and push them along with this one:\n\n%1\n\nCommit, "
590 "remove or ignore them with Git first, then try again.")
591 .arg(namedChanges(changes));
592 // These come back in every folder the file manager opens: removing them
593 // does not last, ignoring them once does, in the .gitignore of the
594 // repository that reports them (a folder may hold one of its own).
595 QStringList writtenAt;
596 const QStringList written = fileManagerFiles(changes, &writtenAt);
597 if (!written.isEmpty()) {
598 const QDir store(QDir::cleanPath(m_settings.passStore));
599 // Each .gitignore with the names its repository reports, alone: a
600 // rule for a name that is not there could hide something else.
601 QStringList ignoreFiles;
602 QMap<QString, QStringList> namesFor;
603 for (const QString &path : std::as_const(writtenAt)) {
604 const QString repository = repositoryOf(store.filePath(path));
605 const QString file = repository.isEmpty()
606 ? QStringLiteral(".gitignore")
607 : repository + QStringLiteral("/.gitignore");
608 if (!ignoreFiles.contains(file)) {
609 ignoreFiles << file;
610 }
611 const QString name = QFileInfo(path).fileName();
612 if (!namesFor[file].contains(name)) {
613 namesFor[file] << name;
614 }
615 }
616 QStringList blocks;
617 for (const QString &file : std::as_const(ignoreFiles)) {
618 blocks << file + QStringLiteral(":\n") +
619 namesFor.value(file).join(QLatin1Char('\n'));
620 }
621 why += QStringLiteral("\n\n") +
622 tr("The system writes %1 by itself, in folders a file manager "
623 "opens (Finder on macOS, Explorer on Windows, Dolphin on "
624 "KDE). Git leaves such files alone in every folder once each "
625 ".gitignore in the store below holds its lines, committed:"
626 "\n\n%2")
627 .arg(written.join(QStringLiteral(", ")),
628 blocks.join(QStringLiteral("\n\n")));
629 }
630 // As the other refusals: the interface arms when an operation starts and
631 // is released by processErrorExit(); a refusal starts nothing.
632 emit startingExecuteWrapper();
633 emit critical(tr("Changes Git has not recorded"), why);
634 emit processErrorExit(1, why);
635 return true;
636}

657auto RealPass::refuseIgnoredThatMoves(const QString &src, const QString &dest)
658 -> bool {
659 if (!m_settings.useGit || !QFileInfo(src).isDir()) {
660 return false;
661 }
662 const QDir store(QDir::cleanPath(m_settings.passStore));
663 // pass moves a folder into an existing one, as mv does.
664 const QString landing = QFileInfo(dest).isDir()
665 ? QDir(dest).filePath(QFileInfo(src).fileName())
666 : dest;
667 // Each side in its own repository: where the files lie now, and the one
668 // pass commits them in.
669 const QString fromRepository = repositoryOf(src);
670 const QString toRepository = repositoryOf(landing);
671 const QString from =
672 QDir(fromRepository.isEmpty() ? store.path()
673 : store.filePath(fromRepository))
674 .relativeFilePath(QDir::cleanPath(src));
675 const QString to =
676 QDir(toRepository.isEmpty() ? store.path() : store.filePath(toRepository))
677 .relativeFilePath(QDir::cleanPath(landing));
678 // ls-files names files relative to where git runs, where status names
679 // them from the repository's root; both are that repository here.
680 QString out;
681 storeGit(
682 inRepository(fromRepository,
683 {QStringLiteral("ls-files"), QStringLiteral("-z"),
684 QStringLiteral("--others"), QStringLiteral("--ignored"),
685 QStringLiteral("--exclude-standard"), QStringLiteral("--"),
686 from}),
687 &out);
688 QStringList moving;
689 QStringList landed;
690 for (const QString &file : out.split(QChar(0), Qt::SkipEmptyParts)) {
691 if (!file.startsWith(from + QLatin1Char('/')) ||
692 isStoreFileName(QFileInfo(file).fileName())) {
693 continue;
694 }
695 moving << file;
696 landed << to + file.mid(from.size());
697 }
698 if (landed.isEmpty()) {
699 return false;
700 }
701 // A rule in a .gitignore inside the folder goes with it, so what it
702 // ignores stays ignored where it lands, and decides there: deeper rules
703 // win. Into a folder that is already there, a rule waiting deeper in it
704 // could undo that, so only the lookup there counts.
705 QStringList travelling;
706 QStringList rules;
707 if (!QFileInfo::exists(landing) &&
708 checkIgnore(fromRepository, moving, true, &rules)) {
709 for (qsizetype i = 0; i + 3 < rules.size(); i += 4) {
710 if (rules.at(i).startsWith(from + QLatin1Char('/'))) {
711 travelling << rules.at(i + 3);
712 }
713 }
714 }
715 // A lookup that failed counts nothing as ignored: the move is refused.
716 QStringList ignoredThere;
717 if (!checkIgnore(toRepository, landed, false, &ignoredThere)) {
718 ignoredThere.clear();
719 }
720 QStringList exposed;
721 for (qsizetype i = 0; i < landed.size(); ++i) {
722 if (!travelling.contains(moving.at(i)) &&
723 !ignoredThere.contains(landed.at(i))) {
724 exposed << moving.at(i) + QStringLiteral(" > ") + landed.at(i);
725 }
726 }
727 if (exposed.isEmpty()) {
728 return false;
729 }
730 const QString why =
731 tr("Git ignores these files only where they are now, and pass would "
732 "commit and push them where they land:\n\n%1\n\nMove them out of "
733 "the folder, or have Git ignore them there too, then try again.")
734 .arg(namedChanges(exposed));
735 emit startingExecuteWrapper();
736 emit critical(tr("Ignored files would be committed"), why);
737 emit processErrorExit(1, why);
738 return true;
739}

741auto RealPass::checkIgnore(const QString &repository, const QStringList &paths,
742 bool verbose, QStringList *fields) -> bool {
743 // check-ignore takes -z only with --stdin. Without
744 // GIT_LITERAL_PATHSPECS (below) a name starting with ':' reads as
745 // pathspec magic: "./" keeps every name a name. Git answers with the
746 // name as given.
747 const QString here = QStringLiteral("./");
748 QString input;
749 for (const QString &path : paths) {
750 input += here + path + QChar(0);
751 }
752 QStringList args{QStringLiteral("check-ignore"), QStringLiteral("--stdin"),
753 QStringLiteral("-z")};
754 if (verbose) {
755 args << QStringLiteral("-v");
756 }
757 // check-ignore refuses every path under GIT_LITERAL_PATHSPECS, which
758 // the store's environment sets; it takes its paths as names anyway.
759 QProcessEnvironment environment = storeEnvironment();
760 environment.remove(QStringLiteral("GIT_LITERAL_PATHSPECS"));
761 QProcess lookup;
762 lookup.setProcessEnvironment(environment);
763 QString out;
764 QString err;
765 (void)Executor::executeBlocking(lookup, m_settings.passExecutable,
766 QStringList{QStringLiteral("git")} +
767 inRepository(repository, args),
768 input, &out, &err);
769 // pass git exits 0 whatever git did: a lookup that failed shows only on
770 // stderr.
771 if (!err.trimmed().isEmpty()) {
772 return false;
773 }
774 *fields = out.split(QChar(0));
775 if (!fields->isEmpty() && fields->constLast().isEmpty()) {
776 fields->removeLast();
777 }
778 // The path is every field, or every fourth with -v.
779 const qsizetype step = verbose ? 4 : 1;
780 for (qsizetype i = step - 1; i < fields->size(); i += step) {
781 if ((*fields)[i].startsWith(here)) {
782 (*fields)[i].remove(0, here.size());
783 }
784 }
785 return true;
786}

819void RealPass::onDequeuing(int id, bool completes) {
820 if (id == PASS_INIT || id == PASS_MOVE || id == PASS_COPY) {
821 // Leaves the queue with its command: a dropped one's scope must not
822 // pair with the next completion.
823 const QString scope =
824 m_reencryptScopes.isEmpty() ? QString() : m_reencryptScopes.dequeue();
825 if (!completes) {
826 return;
827 }
828 m_reencryptPending.enqueue(scope);
829 // pass exits 0 when it could not re-encrypt an entry (reencrypt_path
830 // drops the temporary and carries on), so its exit code says nothing.
831 // gpg's status lines do, in every locale: on stderr (fd 1 is the
832 // `gpg -d | gpg -e` pipe), for this one process (Pass resets the
833 // environment before each start), the last --status-fd winning over a
834 // user's own; pass's own `--status-fd=1` for verifying a signed
835 // .gpg-id comes later on its command line and still wins. WSLENV
836 // already forwards PASSWORD_STORE_GPG_OPTS.
837 QProcessEnvironment environment = exec.environment();
838 const QString key = QStringLiteral("PASSWORD_STORE_GPG_OPTS");
839 environment.insert(
840 key,
841 (environment.value(key) + QStringLiteral(" --status-fd 2")).trimmed());
842 exec.setEnvironment(environment);
843 return;
844 }
845 if (id != PASS_INSERT || m_insertQueue.isEmpty()) {
846 return;
847 }
848 const QString target = m_insertQueue.dequeue();
849 if (completes) {
850 m_insertPending.enqueue({target, entryFingerprint(target)});
851 }

854auto RealPass::insertWritten(const QString &err) const -> bool {
855 if (m_insertDone.target.isEmpty()) {
856 return false;
857 }
858 // gpg broke off (a full disk, killed): pass says so, and what gpg left
859 // at the entry's name, if anything, is no entry. The line is pass's own
860 // (`die "Password encryption aborted."` after gpg in cmd_insert and
861 // cmd_edit of password-store.sh), which pass does not translate.
862 if (err.contains(QLatin1String("Password encryption aborted."))) {
863 return false;
864 }
865 // pass writes the new ciphertext to the entry's name before it commits,
866 // and an encryption never gives the same bytes twice (a fresh session
867 // key), so an edited entry has changed as surely as a new one appeared.
868 const QByteArray after = entryFingerprint(m_insertDone.target);
869 return !after.isEmpty() && after != m_insertDone.before;
870}

872auto RealPass::entryFingerprint(const QString &path) -> QByteArray {
873 QFile file;
874 if (!Util::openRegularFile(path, file)) {
875 return {};
876 }
877 // The size and the head only: this runs on the GUI thread before every
878 // insert, and a co-writer could pad an entry to gigabytes. An encryption
879 // never gives the same head twice, as the session key encrypted to each
880 // recipient comes first and is fresh every time.
881 constexpr qint64 kHeadBytes = 64 * 1024;
882 const QByteArray head = file.read(kHeadBytes);
883 if (head.isEmpty() && file.size() > 0) {
884 return {};
885 }
886 QCryptographicHash hash(QCryptographicHash::Sha256);
887 hash.addData(QByteArray::number(file.size()) + '\n');
888 hash.addData(head);
889 return hash.result();
890}

892void RealPass::Remove(QString file, bool isDir) {
893 // Nothing behind a link is the store's to delete.
894 if (refuseLinkedPath(isDir ? file : file + ".gpg", LinkScope::ParentsOnly)) {
895 return;
896 }
897 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
898 return;
899 }
900 // pass rm turns a folder into "<folder>/" and rm -rf then follows a link
901 // and empties its target, so a linked folder is unlinked here and pass's
902 // git is only asked to forget it. pass rm -f on a linked .gpg unlinks.
903 const QString full = QDir::cleanPath(m_settings.passStore + file);
904 if (isDir && Util::isLinkedFolder(full)) {
905 if (!Util::removeTree(full)) {
906 const QString why = tr("Could not remove the link %1.").arg(full);
907 emit critical(tr("Delete failed"), why);
908 emit processErrorExit(1, why);
909 return;
910 }
911 if (m_settings.useGit) {
912 // `pass git` exits 0 whatever git did, so ask by output whether git
913 // knew the link (these quick lookups block; the commit is queued).
914 const QString rel = QDir(m_settings.passStore).relativeFilePath(full);
915 // With the store's environment (PASSWORD_STORE_DIR), as executePass.
916 QString known;
917 QString err;
918 // The link is gone from disk either way. A failed lookup or rm leaves
919 // git tracking it: say so rather than report a clean removal.
920 bool recorded = Executor::executeBlocking(
921 exec.environment(), m_settings.passExecutable,
922 {"git", "ls-files", "--", rel}, &known, &err) == 0;
923 if (recorded && !known.trimmed().isEmpty()) {
924 recorded =
925 Executor::executeBlocking(
926 exec.environment(), m_settings.passExecutable,
927 {"git", "rm", "-q", "--cached", "--", rel}, &known, &err) == 0;
928 if (recorded) {
929 // `pass git commit` is git's commit as it is: pass adds -S only
930 // to the commits it makes itself, so ask as its git_commit() does.
931 QString signs;
932 const bool sign =
933 Executor::executeBlocking(
934 exec.environment(), m_settings.passExecutable,
935 {"git", "config", "--bool", "--get", "pass.signcommits"},
936 &signs, nullptr) == 0 &&
937 signs.trimmed() == QLatin1String("true");
938 QStringList commit{"git", "commit", "-q"};
939 if (sign) {
940 commit << "-S";
941 }
942 // Queued, as every other commit, so a signer or hook that waits
943 // (a pinentry, an agent) does not freeze the window; finished()
944 // judges it by what is left staged, since `pass git` exits 0
945 // whatever git did.
946 m_linkCommits.enqueue(
947 {rel, m_settings.passExecutable, exec.environment()});
948 executePass(
949 GIT_COMMIT,
950 commit + QStringList{"-m", "Remove for " + rel + " using QtPass.",
951 "--", rel});
952 return;
953 }
954 }
955 if (!recorded) {
956 const QString why =
957 tr("The link %1 was removed, but git could not record it: %2")
958 .arg(rel, err.trimmed());
959 emit critical(tr("Delete incomplete"), why);
960 emit processErrorExit(1, why);
961 }
962 }
963 // Nothing ran for pass to finish: the removal is done here, and the
964 // store changed.
965 emit finishedRemove(QString(), QString());
966 return;
967 }
968 executePass(PASS_REMOVE, {"rm", (isDir ? "-rf" : "-f"), file});
969}

971void RealPass::finished(int id, int exitCode, const QString &out,
972 const QString &err) {
973 if (static_cast<PROCESS>(id) == GIT_COMMIT && !m_linkCommits.isEmpty()) {
974 // Whatever the exit code (pass itself failing, or 0 whatever git
975 // did), what is left staged decides.
976 finishLinkCommit(m_linkCommits.dequeue(), out, err);
977 return;
978 }
979 if (static_cast<PROCESS>(id) == PASS_INSERT) {
980 // This completion's own insert, not whichever left the queue last.
981 m_insertDone =
982 m_insertPending.isEmpty() ? InsertRecord{} : m_insertPending.dequeue();
983 }
984 const auto pid = static_cast<PROCESS>(id);
985 if (pid == PASS_INIT || pid == PASS_MOVE || pid == PASS_COPY) {
986 // Without a recorded scope (not queued through Init/Move/Copy), the
987 // whole store: held as a whole, released by nothing short of it.
988 const QString scope =
989 m_reencryptPending.isEmpty() ? QString() : m_reencryptPending.dequeue();
990 if (exitCode == 0 && reencryptFailed(out, err)) {
991 holdAutoPush(scope);
992 sendDeferredPush();
993 reportReencryptFailure(pid, out, err);
994 return;
995 }
996 // A run that failed may have stopped after its commit (pass init commits
997 // the new .gpg-id before re-encrypting; signing it may fail, the process
998 // may be killed): held as well, until a clean init covers it. A move or
999 // copy that failed before re-encrypting anything (the destination
1000 // exists, a mistyped name) changed no recipients: not held.
1001 if (exitCode != 0 && (pid == PASS_INIT ||
1002 out.contains(QLatin1String(": reencrypting to ")))) {
1003 holdAutoPush(scope);
1004 }
1005 // pass init re-encrypts everything under its folder: a clean run
1006 // releases the holds there. A move or copy covers only its entries.
1007 if (pid == PASS_INIT && exitCode == 0 && !scope.isEmpty()) {
1008 releaseAutoPush(scope);
1009 }
1010 if (exitCode == 0 && !scope.isEmpty()) {
1011 reportUnrecorded({scope});
1012 }
1013 sendDeferredPush();
1014 // The status lines are for this check, not for the output panel.
1015 Pass::finished(id, exitCode, out, withoutStatusLines(err));
1016 return;
1017 }
1018 if (pid == PASS_INSERT && exitCode == 0 && !m_insertDone.target.isEmpty()) {
1019 reportUnrecorded({m_insertDone.target});
1020 }
1021 Pass::finished(id, exitCode, out, err);
1022}

1024auto RealPass::reencryptFailed(const QString &out, const QString &err) -> bool {
1025 // pass's own line for each entry it re-encrypts ("name: reencrypting to
1026 // KEY..."), which it does not translate; one per line, counted once
1027 // even when the entry's own name holds the words.
1028 const QStringList outLines = out.split(QLatin1Char('\n'));
1029 const auto attempts = std::count_if(
1030 outLines.cbegin(), outLines.cend(), [](const QString &line) {
1031 return line.contains(QLatin1String(": reencrypting to "));
1032 });
1033 qsizetype encrypted = 0;
1034 bool status = false;
1035 for (QString line : err.split(QLatin1Char('\n'))) {
1036 line.remove(QLatin1Char('\r'));
1037 if (!line.startsWith(QLatin1String("[GNUPG:] "))) {
1038 continue;
1039 }
1040 status = true;
1041 // A recipient gpg cannot encrypt to, or a decrypt that failed (no
1042 // secret key, pinentry cancelled, no smartcard): that entry was left
1043 // as it was. KEYEXPIRED and NO_SECKEY alone are not failures.
1044 if (line.startsWith(QLatin1String("[GNUPG:] INV_RECP ")) ||
1045 line.startsWith(QLatin1String("[GNUPG:] FAILURE ")) ||
1046 line.startsWith(QLatin1String("[GNUPG:] DECRYPTION_FAILED"))) {
1047 return true;
1048 }
1049 if (line.startsWith(QLatin1String("[GNUPG:] END_ENCRYPTION"))) {
1050 ++encrypted;
1051 }
1052 }
1053 // Without any status line the option did not reach gpg (a pass that does
1054 // not pass PASSWORD_STORE_GPG_OPTS on): nothing to judge by.
1055 return status && encrypted < attempts;
1056}

1058void RealPass::reportReencryptFailure(PROCESS pid, const QString &out,
1059 const QString &err) {
1060 QStringList failures;
1061 for (QString line : err.split(QLatin1Char('\n'))) {
1062 line.remove(QLatin1Char('\r'));
1063 // Encrypt-side only: gpgErrorMessage() reads any FAILURE as a key that
1064 // cannot encrypt, wrong advice for a decrypt that failed (no secret
1065 // key, pinentry cancelled), whose reason gpg's own text gives.
1066 if (line.startsWith(QLatin1String("[GNUPG:] INV_RECP ")) ||
1067 line.startsWith(QLatin1String("[GNUPG:] FAILURE encrypt"))) {
1068 failures << line;
1069 }
1070 }
1071 const QString human = withoutStatusLines(err).trimmed();
1072 const QString reason = gpgErrorMessage(failures.join(QLatin1Char('\n')));
1073 QString body =
1074 tr("The recipients were changed, but some entries could not be "
1075 "re-encrypted for them: those are still encrypted to the previous "
1076 "recipients, who can still read them. With Git the change is "
1077 "committed locally and not pushed: QtPass does not push this store "
1078 "automatically until a re-encryption succeeds, and a push by hand "
1079 "sends it.");
1080 for (const QString &part : {reason, human}) {
1081 if (!part.isEmpty()) {
1082 body += QStringLiteral("\n\n") + part;
1083 }
1084 }
1085 // pass's "reencrypting to" lines and gpg's own text, for the output panel.
1086 emit finishedAnyWithPid(out, human, pid);
1087 emit critical(
1088 QCoreApplication::translate("ImitatePass", "Re-encryption failed"), body);
1089 // Not finishedInit/Move/Copy: that would push the store as it now is.
1090 emit processErrorExit(1, body);
1091}

1120void RealPass::Init(QString path, const QList<UserInfo> &users) {
1121 // A relative path names a folder in the store, as --path does for pass:
1122 // resolved once, so the checks below look where pass writes, not in the
1123 // working directory.
1124 const QString folder = QDir::cleanPath(
1125 QDir::isAbsolutePath(path) ? path
1126 : QDir(m_settings.passStore).filePath(path));
1127 // pass init writes .gpg-id with a shell redirection and gpg writes the
1128 // signature with --output; both follow a link planted under those names.
1129 // --path is store-relative, or pass creates passStore/passStore/dir: the
1130 // store check hands it over as it judged it, so a store root reached
1131 // through a link and a folder spelled through its target agree (a lexical
1132 // relativeFilePath() made that ../../target/dir, which pass refuses).
1133 QString dirWithoutPassdir;
1134 const auto refusePlaced = [this](const QString &placed) {
1135 return refuseLinkedPath(placed, LinkScope::EntryAndParents,
1136 PathSpelling::PlacedByCaller);
1137 };
1138 if (refusePlaced(folder) ||
1139 refusePlaced(folder + QStringLiteral("/.gpg-id")) ||
1140 refusePlaced(folder + QStringLiteral("/.gpg-id.sig")) ||
1141 refuseFolderOutsideStore(folder, &dirWithoutPassdir)) {
1142 return;
1143 }
1144 // pass init writes this folder's list and then re-encrypts everything
1145 // under it to each entry's nearest list, a deeper folder's own included:
1146 // those are checked as for a move. This folder's list is the one being
1147 // written, so it is neither verified nor missed. Walked in the store's
1148 // own spelling: a folder spelled through the target of a linked store
1149 // root is not under the store as named, and the lookup of each nearest
1150 // list would stop at once and miss the deeper ones.
1151 const QString inStore =
1152 QDir::cleanPath(QDir(m_settings.passStore).filePath(dirWithoutPassdir));
1153 if (refuseUnverifiedTree(inStore, true,
1154 QDir(inStore).filePath(QStringLiteral(".gpg-id")))) {
1155 return;
1156 }
1157 if (refuseUnfinishedGitOperation() || refusePendingPull() ||
1158 refuseUnrecordedChanges({inStore}, {inStore})) {
1159 return;
1160 }
1161 QStringList args = {"init", "--path=" + dirWithoutPassdir};
1162 for (const UserInfo &user : users) {
1163 if (user.enabled) {
1164 args.append(user.key_id);
1165 }
1166 }
1167 m_reencryptScopes.enqueue(inStore);
1168 executePass(PASS_INIT, args);
1169}

1189void RealPass::passMoveOrCopy(PROCESS id, const QString &subcommand,
1190 const QString &src, const QString &dest,
1191 const bool force) {
1192 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
1193 return;
1194 }
1195 QFileInfo srcFileInfo = QFileInfo(src);
1196 QFileInfo destFileInfo = QFileInfo(dest);
1197 // A drop hands over the folder; pass then writes <folder>/<name>, and cp
1198 // writes through a link planted under that name.
1199 if (destFileInfo.isDir() &&
1200 refuseLinkedPath(QDir(dest).filePath(srcFileInfo.fileName()))) {
1201 return;
1202 }
1203
1204 // pass re-encrypts a moved or copied entry at its new place, and gpg
1205 // never returns from a FIFO named like one.
1206 if (refuseSpecialEntry(srcFileInfo.exists() && !srcFileInfo.isFile() &&
1207 !srcFileInfo.isDir()
1208 ? src
1209 : QString())) {
1210 return;
1211 }
1212
1213 // pass always forces when called from QtPass, so refuse file-onto-file.
1214 if (!force && srcFileInfo.isFile() && destFileInfo.isFile()) {
1215 return;
1216 }
1217
1218 // pass re-encrypts what it moves or copies to the destination's list, a
1219 // folder's entries also to the lists that go along with it, and, when it
1220 // writes into an existing folder, everything already in that folder too
1221 // (reencrypt_path on the whole destination).
1222 if (refuseUnverifiedList(destFileInfo.isDir()
1223 ? QDir(dest).filePath(srcFileInfo.fileName())
1224 : dest) ||
1225 (srcFileInfo.isDir() && refuseUnverifiedTree(src, true)) ||
1226 (destFileInfo.isDir() && refuseUnverifiedTree(dest, false))) {
1227 return;
1228 }
1229 // pass stages what it moved or copied with `git add` on the new path,
1230 // and a folder brings whatever lies in it along. It commits there, and a
1231 // move commits the removal where the original was as well.
1232 const QStringList committing =
1233 id == PASS_MOVE ? QStringList{src, dest} : QStringList{dest};
1234 if (refuseUnfinishedGitOperation() || refusePendingPull() ||
1235 refuseUnrecordedChanges({src, dest}, committing) ||
1236 refuseIgnoredThatMoves(src, dest)) {
1237 return;
1238 }
1239
1240 // pass appends .gpg itself, so an entry named with it would come out as
1241 // "name.gpg.gpg"; the destination usually does not exist yet.
1242 const QString passSrc =
1243 passName(src, srcFileInfo.isFile() &&
1244 srcFileInfo.suffix() == QLatin1String("gpg"));
1245 const QString passDest =
1246 passName(dest, !destFileInfo.isDir() &&
1247 destFileInfo.suffix() == QLatin1String("gpg"));
1248
1249 QStringList args;
1250 args << subcommand;
1251 if (force) {
1252 args << "-f";
1253 }
1254 args << passSrc;
1255 args << passDest;
1256 m_reencryptScopes.enqueue(QDir::cleanPath(QDir(dest).absolutePath()));
1257 executePass(id, args);
1258}
