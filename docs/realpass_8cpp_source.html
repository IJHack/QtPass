77auto RealPass::refuseList(const QString &file, const GpgIdSigner &signer,
78 QSet<QString> &verified, const QString &within,
79 const QString &except) -> bool {
80 if (refuseSkippedGpgId(file, true, except)) {
81 return true;
82 }
83 const QString gpgIdFile = Pass::getGpgIdPath(file, m_settings.passStore);
84 const QString absolute =
85 QDir::cleanPath(QFileInfo(gpgIdFile).absoluteFilePath());
86 if (verified.contains(gpgIdFile) ||
87 (!except.isEmpty() &&
88 absolute == QDir::cleanPath(QFileInfo(except).absoluteFilePath())) ||
89 (!within.isEmpty() && !absolute.startsWith(within))) {
90 return false;
91 }
92 QString why =
93 QCoreApplication::translate("ImitatePass", "Signature for %1 is invalid.")
94 .arg(gpgIdFile);
95 QByteArray contents;
96 if (!verifySignedList(signer, gpgIdFile, &contents, &why)) {
97 emit critical(QCoreApplication::translate("ImitatePass",
98 "Check .gpg-id file signature!"),
99 why);
100 return true;
101 }
102 verified.insert(gpgIdFile);
103 return false;
104}

106auto RealPass::refuseUnverifiedList(const QString &file) -> bool {
107 // pass verifies the signature itself, but knows nothing of the generation
108 // and folder lines: an older authentic list put back, or one signed for
109 // another folder, would get the secret (#2077). The same checks as the
110 // native backend, with gpg in the store's environment. pass reads the list
111 // again afterwards; a swap in between is a concurrent writer, out of
112 // scope (SECURITY.md).
113 const GpgIdSigner signer = listSigner();
114 if (!signer.enabled()) {
115 return false;
116 }
117 QSet<QString> verified;
118 return refuseList(file, signer, verified);
119}

141auto RealPass::refuseUnverifiedTree(const QString &dir, bool onlyListsWithin,
142 const QString &except) -> bool {
143 const GpgIdSigner signer = listSigner();
144 if (!signer.enabled()) {
145 return false;
146 }
147 // One gpg call per list, before pass runs: a store with many lists makes
148 // that a noticeable pause, so say what it is.
149 emit statusMsg(tr("Checking the recipient lists under %1…")
150 .arg(QDir::toNativeSeparators(QDir::cleanPath(dir))),
151 3000);
152 // pass mv and cp re-encrypt every entry of the folder they write to, each
153 // to its nearest list, walking as passFoldersUnder() does. For the folder
154 // that moves, the lists inside it go along: checked where they lie now
155 // (after the move their folder line names the old place until the
156 // recipients are saved there again), while a list above it is replaced
157 // by the destination's, which the caller checks. For entries already in
158 // the destination, their lists stay where they are: all of them count.
159 // Absolute, as getGpgIdPath() hands lists back: a relative or other
160 // spelling would match nothing and check nothing.
161 const QString within =
162 onlyListsWithin ? QDir::cleanPath(QFileInfo(dir).absoluteFilePath()) +
163 QLatin1Char('/')
164 : QString();
165 QSet<QString> verified;
166 // Every folder's list counts, also where no entry uses it yet: cp -r
167 // merges an incoming folder into one of the same name, so the entries it
168 // brings get a list already there, and a list it brings replaces the one
169 // there for the entries already in it.
170 for (const QString &folder : passFoldersUnder(dir)) {
171 if (refuseList(folder, signer, verified, within, except)) {
172 return true;
173 }
174 }
175 return false;
176}

190void RealPass::Insert(QString file, QString newValue, bool overwrite) {
191 if (refuseLinkedPath(file + ".gpg")) {
192 return;
193 }
194 const QString entry =
195 QDir::cleanPath(QDir::isAbsolutePath(file)
196 ? file + ".gpg"
197 : QDir(m_settings.passStore).filePath(file) + ".gpg");
198 if (refuseUnverifiedList(entry)) {
199 return;
200 }
201 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
202 return;
203 }
204 m_insertQueue.enqueue(entry);
205 // Past every refusal: this insert is now queued and will report once.
206 queueInsert();
207 QStringList args = {"insert", "-m"};
208 if (overwrite) {
209 args.append("-f");
210 }
211 args.append(file);
212 executePass(PASS_INSERT, args, newValue);
213}

225auto RealPass::insertWritten(const QString &err) const -> bool {
226 if (m_insertDone.target.isEmpty()) {
227 return false;
228 }
229 // gpg broke off (a full disk, killed): pass says so, and what gpg left
230 // at the entry's name, if anything, is no entry. The line is pass's own
231 // (`die "Password encryption aborted."` after gpg in cmd_insert and
232 // cmd_edit of password-store.sh), which pass does not translate.
233 if (err.contains(QLatin1String("Password encryption aborted."))) {
234 return false;
235 }
236 // pass writes the new ciphertext to the entry's name before it commits,
237 // and an encryption never gives the same bytes twice (a fresh session
238 // key), so an edited entry has changed as surely as a new one appeared.
239 const QByteArray after = entryFingerprint(m_insertDone.target);
240 return !after.isEmpty() && after != m_insertDone.before;
241}

243auto RealPass::entryFingerprint(const QString &path) -> QByteArray {
244 QFile file;
245 if (!Util::openRegularFile(path, file)) {
246 return {};
247 }
248 QCryptographicHash hash(QCryptographicHash::Sha256);
249 if (!hash.addData(&file)) {
250 return {};
251 }
252 return hash.result();
253}

255void RealPass::Remove(QString file, bool isDir) {
256 // Nothing behind a link is the store's to delete.
257 if (refuseLinkedPath(isDir ? file : file + ".gpg", LinkScope::ParentsOnly)) {
258 return;
259 }
260 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
261 return;
262 }
263 // pass rm turns a folder into "<folder>/" and rm -rf then follows a link
264 // and empties its target, so a linked folder is unlinked here and pass's
265 // git is only asked to forget it. pass rm -f on a linked .gpg unlinks.
266 const QString full = QDir::cleanPath(m_settings.passStore + file);
267 if (isDir && Util::isLinkedFolder(full)) {
268 if (!Util::removeTree(full)) {
269 const QString why = tr("Could not remove the link %1.").arg(full);
270 emit critical(tr("Delete failed"), why);
271 emit processErrorExit(1, why);
272 return;
273 }
274 if (m_settings.useGit) {
275 // `pass git` exits 0 whatever git did, so ask by output whether git
276 // knew the link (these quick lookups block; the commit is queued).
277 const QString rel = QDir(m_settings.passStore).relativeFilePath(full);
278 // With the store's environment (PASSWORD_STORE_DIR), as executePass.
279 QString known;
280 QString err;
281 // The link is gone from disk either way. A failed lookup or rm leaves
282 // git tracking it: say so rather than report a clean removal.
283 bool recorded = Executor::executeBlocking(
284 exec.environment(), m_settings.passExecutable,
285 {"git", "ls-files", "--", rel}, &known, &err) == 0;
286 if (recorded && !known.trimmed().isEmpty()) {
287 recorded =
288 Executor::executeBlocking(
289 exec.environment(), m_settings.passExecutable,
290 {"git", "rm", "-q", "--cached", "--", rel}, &known, &err) == 0;
291 if (recorded) {
292 // `pass git commit` is git's commit as it is: pass adds -S only
293 // to the commits it makes itself, so ask as its git_commit() does.
294 QString signs;
295 const bool sign =
296 Executor::executeBlocking(
297 exec.environment(), m_settings.passExecutable,
298 {"git", "config", "--bool", "--get", "pass.signcommits"},
299 &signs, nullptr) == 0 &&
300 signs.trimmed() == QLatin1String("true");
301 QStringList commit{"git", "commit", "-q"};
302 if (sign) {
303 commit << "-S";
304 }
305 // Queued, as every other commit, so a signer or hook that waits
306 // (a pinentry, an agent) does not freeze the window; finished()
307 // judges it by what is left staged, since `pass git` exits 0
308 // whatever git did.
309 m_linkCommits.enqueue(
310 {rel, m_settings.passExecutable, exec.environment()});
311 executePass(
312 GIT_COMMIT,
313 commit + QStringList{"-m", "Remove for " + rel + " using QtPass.",
314 "--", rel});
315 return;
316 }
317 }
318 if (!recorded) {
319 const QString why =
320 tr("The link %1 was removed, but git could not record it: %2")
321 .arg(rel, err.trimmed());
322 emit critical(tr("Delete incomplete"), why);
323 emit processErrorExit(1, why);
324 }
325 }
326 // Nothing ran for pass to finish: the removal is done here, and the
327 // store changed.
328 emit finishedRemove(QString(), QString());
329 return;
330 }
331 executePass(PASS_REMOVE, {"rm", (isDir ? "-rf" : "-f"), file});
332}

334void RealPass::finished(int id, int exitCode, const QString &out,
335 const QString &err) {
336 if (static_cast<PROCESS>(id) == GIT_COMMIT && !m_linkCommits.isEmpty()) {
337 // Whatever the exit code (pass itself failing, or 0 whatever git
338 // did), what is left staged decides.
339 finishLinkCommit(m_linkCommits.dequeue(), out, err);
340 return;
341 }
342 if (static_cast<PROCESS>(id) == PASS_INSERT) {
343 // This completion's own insert, not whichever left the queue last.
344 m_insertDone =
345 m_insertPending.isEmpty() ? InsertRecord{} : m_insertPending.dequeue();
346 }
347 Pass::finished(id, exitCode, out, err);
348}

350void RealPass::finishLinkCommit(const LinkCommit &commit, const QString &out,
351 const QString &err) {
352 // In the store the commit ran in, which a profile switch meanwhile does
353 // not change.
354 const QString &rel = commit.rel;
355 QString staged;
356 const bool committed =
357 Executor::executeBlocking(
358 commit.environment, commit.passExecutable,
359 {"git", "diff", "--cached", "--name-only", "--", rel}, &staged,
360 nullptr) == 0 &&
361 staged.trimmed().isEmpty();
362 if (committed) {
363 // As Pass::finished() would: git's output (a hook's warning, say) for
364 // the output panel, then the removal.
365 emit finishedAnyWithPid(out, err, GIT_COMMIT);
366 emit finishedRemove(out, err);
367 return;
368 }
369 const QString why = tr("The link %1 was removed and git no longer tracks "
370 "it, but the commit failed: %2")
371 .arg(rel, err.trimmed());
372 emit critical(tr("Delete incomplete"), why);
373 emit processErrorExit(1, why);
374 emit finishedRemove(QString(), QString());
375}

377void RealPass::Init(QString path, const QList<UserInfo> &users) {
378 // A relative path names a folder in the store, as --path does for pass:
379 // resolved once, so the checks below look where pass writes, not in the
380 // working directory.
381 const QString folder = QDir::cleanPath(
382 QDir::isAbsolutePath(path) ? path
383 : QDir(m_settings.passStore).filePath(path));
384 // pass init writes .gpg-id with a shell redirection and gpg writes the
385 // signature with --output; both follow a link planted under those names.
386 // --path is store-relative, or pass creates passStore/passStore/dir: the
387 // store check hands it over as it judged it, so a store root reached
388 // through a link and a folder spelled through its target agree (a lexical
389 // relativeFilePath() made that ../../target/dir, which pass refuses).
390 QString dirWithoutPassdir;
391 const auto refusePlaced = [this](const QString &placed) {
392 return refuseLinkedPath(placed, LinkScope::EntryAndParents,
393 PathSpelling::PlacedByCaller);
394 };
395 if (refusePlaced(folder) ||
396 refusePlaced(folder + QStringLiteral("/.gpg-id")) ||
397 refusePlaced(folder + QStringLiteral("/.gpg-id.sig")) ||
398 refuseFolderOutsideStore(folder, &dirWithoutPassdir)) {
399 return;
400 }
401 // pass init writes this folder's list and then re-encrypts everything
402 // under it to each entry's nearest list, a deeper folder's own included:
403 // those are checked as for a move. This folder's list is the one being
404 // written, so it is neither verified nor missed. Walked in the store's
405 // own spelling: a folder spelled through the target of a linked store
406 // root is not under the store as named, and the lookup of each nearest
407 // list would stop at once and miss the deeper ones.
408 const QString inStore =
409 QDir::cleanPath(QDir(m_settings.passStore).filePath(dirWithoutPassdir));
410 if (refuseUnverifiedTree(inStore, true,
411 QDir(inStore).filePath(QStringLiteral(".gpg-id")))) {
412 return;
413 }
414 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
415 return;
416 }
417 QStringList args = {"init", "--path=" + dirWithoutPassdir};
418 for (const UserInfo &user : users) {
419 if (user.enabled) {
420 args.append(user.key_id);
421 }
422 }
423 executePass(PASS_INIT, args);
424}

444void RealPass::passMoveOrCopy(PROCESS id, const QString &subcommand,
445 const QString &src, const QString &dest,
446 const bool force) {
447 if (refuseLinkedPath(src) || refuseLinkedPath(dest)) {
448 return;
449 }
450 QFileInfo srcFileInfo = QFileInfo(src);
451 QFileInfo destFileInfo = QFileInfo(dest);
452 // A drop hands over the folder; pass then writes <folder>/<name>, and cp
453 // writes through a link planted under that name.
454 if (destFileInfo.isDir() &&
455 refuseLinkedPath(QDir(dest).filePath(srcFileInfo.fileName()))) {
456 return;
457 }
458
459 // pass always forces when called from QtPass, so refuse file-onto-file.
460 if (!force && srcFileInfo.isFile() && destFileInfo.isFile()) {
461 return;
462 }
463
464 // pass re-encrypts what it moves or copies to the destination's list, a
465 // folder's entries also to the lists that go along with it, and, when it
466 // writes into an existing folder, everything already in that folder too
467 // (reencrypt_path on the whole destination).
468 if (refuseUnverifiedList(destFileInfo.isDir()
469 ? QDir(dest).filePath(srcFileInfo.fileName())
470 : dest) ||
471 (srcFileInfo.isDir() && refuseUnverifiedTree(src, true)) ||
472 (destFileInfo.isDir() && refuseUnverifiedTree(dest, false))) {
473 return;
474 }
475 if (refuseUnfinishedGitOperation() || refusePendingPull()) {
476 return;
477 }
478
479 // pass appends .gpg itself, so an entry named with it would come out as
480 // "name.gpg.gpg"; the destination usually does not exist yet.
481 const QString passSrc =
482 passName(src, srcFileInfo.isFile() &&
483 srcFileInfo.suffix() == QLatin1String("gpg"));
484 const QString passDest =
485 passName(dest, !destFileInfo.isDir() &&
486 destFileInfo.suffix() == QLatin1String("gpg"));
487
488 QStringList args;
489 args << subcommand;
490 if (force) {
491 args << "-f";
492 }
493 args << passSrc;
494 args << passDest;
495 executePass(id, args);
496}
