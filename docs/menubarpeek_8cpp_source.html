17MenuBarPeek::MenuBarPeek(QMainWindow *window, QAction *toggle)
18 : QObject(window), m_window(window), m_toggle(toggle) {
19 // Key events go to the focus widget, and an open menu is a window of its
20 // own, so the filter has to sit on the application.
21 qApp->installEventFilter(this);
22 connect(m_toggle, &QAction::toggled, this, [this] { m_peeking = false; });
23 for (QAction *entry : m_window->menuBar()->actions()) {
24 if (QMenu *menu = entry->menu()) {
25 connect(menu, &QMenu::aboutToHide, this,
26 &MenuBarPeek::hideWhenNoMenuIsOpen);
27 }
28 }
29}

33auto MenuBarPeek::eventFilter(QObject *watched, QEvent *event) -> bool {
34 switch (event->type()) {
35 case QEvent::KeyPress:
36 onKeyPress(watched, static_cast<QKeyEvent *>(event));
37 break;
38 case QEvent::KeyRelease:
39 onKeyRelease(static_cast<QKeyEvent *>(event));
40 break;
41 case QEvent::MouseButtonPress:
42 if (m_peeking && inWindow(watched) && !onMenuBar(watched)) {
43 hideWhenNoMenuIsOpen();
44 }
45 break;
46 case QEvent::WindowDeactivate:
47 if (watched == m_window) {
48 // Alt+Tab: the release goes to another application.
49 m_altDown = false;
50 hideWhenNoMenuIsOpen();
51 }
52 break;
53 default:
54 break;
55 }
56 return false;
57}

59// A key event reaches the filter once for every widget it propagates to, and
60// again as auto-repeat; m_altDown makes the second sighting a no-op.
61void MenuBarPeek::onKeyPress(QObject *watched, const QKeyEvent *event) {
62 if (event->key() != Qt::Key_Alt) {
63 m_altAlone = false;
64 if (event->key() == Qt::Key_Escape && m_peeking) {
65 hideWhenNoMenuIsOpen();
66 }
67 return;
68 }
69 if (m_altDown || event->isAutoRepeat() || !inWindow(watched)) {
70 return;
71 }
72 m_altDown = true;
73 m_altAlone = true;
74 m_shownByThisPress =
75 !m_peeking && !m_toggle->isChecked() && !m_window->menuBar()->isVisible();
76 if (m_shownByThisPress) {
77 show();
78 }
79}

81void MenuBarPeek::onKeyRelease(const QKeyEvent *event) {
82 if (event->key() != Qt::Key_Alt || !m_altDown || event->isAutoRepeat()) {
83 return;
84 }
85 m_altDown = false;
86 if (!m_peeking) {
87 return;
88 }
89 if (m_altAlone && !m_shownByThisPress) {
90 hide(); // the second tap
91 } else if (!m_altAlone) {
92 hideWhenNoMenuIsOpen(); // Alt+something that was not a menu
93 }
94}

101auto MenuBarPeek::onMenuBar(QObject *watched) const -> bool {
102 const auto *widget = qobject_cast<QWidget *>(watched);
103 const QMenuBar *bar = m_window->menuBar();

119// Deferred: a menu's aboutToHide also fires when the pointer moves on to the
120// next menu, and Escape is seen here before the bar leaves keyboard mode.
121void MenuBarPeek::hideWhenNoMenuIsOpen() {
122 QTimer::singleShot(0, this, [this] {
123 if (m_peeking && QApplication::activePopupWidget() == nullptr &&
124 m_window->menuBar()->activeAction() == nullptr) {
125 hide();
126 }
127 });
