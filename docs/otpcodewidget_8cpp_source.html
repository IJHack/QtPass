32OtpCodeWidget::OtpCodeWidget(Totp::Settings settings, bool withCopyButton,
33 bool monospace, QWidget *parent)
34 : QWidget(parent), m_settings(std::move(settings)) {
35 // Same borderless look as the action buttons PasswordDisplayPanel builds.
36 const QString buttonStyle =
37 "QPushButton { border-style: none; background: transparent; padding: 0; "
38 "margin: 0; icon-size: 16px; color: inherit; }";
39
40 auto *layout = new QHBoxLayout(this);
41 layout->setContentsMargins(0, 0, 0, 0);
42 layout->setSpacing(4);
43
44 if (withCopyButton) {
45 m_copyButton = new QPushButtonWithClipboard(QString(), this);
46 m_copyButton->setStyleSheet(buttonStyle);
47 connect(m_copyButton, &QPushButtonWithClipboard::clicked, this,
48 &OtpCodeWidget::copyRequested);
49 layout->addWidget(m_copyButton);
50 }
51
52 m_codeLabel = new QLabel(this);
53 m_codeLabel->setObjectName(QStringLiteral("OTP code"));
54 m_codeLabel->setTextInteractionFlags(Qt::TextSelectableByMouse);
55 if (monospace) {
56 m_codeLabel->setStyleSheet("QLabel { font-family: monospace; }");
57 }
58 layout->addWidget(m_codeLabel);
59 layout->addStretch();
60
61 m_progress = new QProgressBar(this);
62 m_progress->setObjectName(QStringLiteral("otpCountdown"));
63 m_progress->setRange(0, static_cast<int>(m_settings.step));
64 // Number-only format: no translatable string, and no unit to mistranslate.
65 m_progress->setFormat(QStringLiteral("%v"));
66 m_progress->setFixedWidth(kProgressWidth);
67 m_progress->setMaximumHeight(kProgressHeight);
68 m_progress->setToolTip(tr("Seconds until the OTP code changes"));
69 layout->addWidget(m_progress);
70
71 // The timer is a child of this widget: deleting the row stops it, so a
72 // queued timeout can never reach a destroyed label.
73 auto *timer = new QTimer(this);
74 timer->setInterval(kRefreshIntervalMs);
75 connect(timer, &QTimer::timeout, this,
76 [this]() { refresh(Totp::currentUnixTime()); });
77 timer->start();
78
79 refresh(Totp::currentUnixTime());
80}

82void OtpCodeWidget::refresh(quint64 unixTime) {
83 const quint64 counter = Totp::counter(m_settings, unixTime);
84 if (!m_hasCode || counter != m_counter) {
85 m_counter = counter;
86 m_hasCode = true;
87 m_code = Totp::generate(m_settings, unixTime);
88 m_codeLabel->setText(m_code);
89 if (m_copyButton != nullptr) {
90 m_copyButton->setTextToCopy(m_code);
91 }
92 }
93
94 m_secondsRemaining = Totp::secondsRemaining(m_settings, unixTime);
95 m_progress->setValue(static_cast<int>(m_secondsRemaining));
96}
