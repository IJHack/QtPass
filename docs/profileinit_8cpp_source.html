49auto gitEnvironment(const QString &dir, const AppSettings &s)
50 -> QProcessEnvironment {
51 QProcessEnvironment env =
52 withGpgHome(QProcessEnvironment::systemEnvironment(), s);
53 // As for the active store: the paths handed to git are names, not
54 // patterns (see Pass::updateEnv()).
55 env.insert(QStringLiteral("GIT_LITERAL_PATHSPECS"), QStringLiteral("1"));
56 // The identity check runs git before `git init` made the folder a
57 // repository, so a pushed bare-repository layout there would be read.
58 Util::requireExplicitBareRepository(env);
59 if (!s.gitUseParentRepository) {
60 for (const auto &[name, value] :
61 Util::gitStoreVariables(dir, s.gitExecutable)) {
62 env.insert(name, value);
63 }
64 }
65 return env;
66}

88auto blocksGit(const QString &dir, const QStringList &files) -> QString {
89 QSet<QString> folders{QStringLiteral(".")};
90 for (const QString &file : files) {
91 for (QString folder = QFileInfo(file).path();
92 folder != QLatin1String(".") && !folders.contains(folder);
93 folder = QFileInfo(folder).path()) {
94 folders.insert(folder);
95 }
96 }
97 const QDir base(dir);
98 for (const QString &folder : std::as_const(folders)) {
99 const QDir here(base.filePath(folder));
100 for (const char *name : {".gitattributes", ".gitignore"}) {
101 const QFileInfo info(here.filePath(QLatin1String(name)));
102 if (info.exists() && !info.isFile()) {
103 return info.filePath();
104 }
105 }
106 const QFileInfo nested(here.filePath(QStringLiteral(".git")));
107 if (folder != QLatin1String(".") &&
108 (nested.exists() || nested.isSymLink())) {
109 return nested.filePath();
110 }
111 }
112 return {};
113}

165auto ProfileInit::initialise(const QString &dir, const QList<UserInfo> &users,
166 const AppSettings &s, bool useGit, QString *note)
167 -> bool {
168 QString scratch;
169 QString &out = note != nullptr ? *note : scratch;
170 out.clear();
171
172 const QDir folder(dir);
173 if (!folder.exists() && !Util::mkpathForStore(folder.absolutePath())) {
174 out = tr("Could not create %1.").arg(folder.absolutePath());
175 return false;
176 }
177 // Before anything is written: a refusal later would leave a .gpg-id
178 // behind, and a store that looks finished.
179 if (useGit &&
180 refusedForGit(folder.absolutePath(),
181 {QStringLiteral(".gpg-id"), QStringLiteral(".gpg-id.sig")},
182 &out)) {
183 return false;
184 }
185 const QString gpgIdFile = folder.filePath(QStringLiteral(".gpg-id"));
186 // Before the list is written, as ImitatePass::Init() does: writing it
187 // takes the next generation, and a list that then cannot be signed left
188 // this device's record one ahead of the store's own, refused afterwards
189 // as a rollback.
190 if (!s.passSigningKey.trimmed().isEmpty() && !signerFor(s).haveSecretKey()) {
191 out = tr("The secret key of the signing key %1 is not available, so no "
192 "recipient list was written.")
193 .arg(GpgIdSigner::keysFromSetting(s.passSigningKey).first());
194 // withGpgHome() fell back to the inherited home, as for signing below.
195 if (!s.gpgHome.isEmpty()) {
196 QDir home(s.gpgHome);
197 home.makeAbsolute();
198 if (!home.exists()) {
199 out += QLatin1Char(' ') + tr("The configured GPG home %1 does not "
200 "exist.")
201 .arg(home.path());
202 }
203 }
204 return false;
205 }
206 QByteArray written;
207 if (!writeGpgId(gpgIdFile, users, !s.passSigningKey.trimmed().isEmpty(), &out,
208 &written)) {
209 return false;
210 }
211 QString sigFile;
212 if (!s.passSigningKey.trimmed().isEmpty()) {
213 if (!signGpgId(gpgIdFile, written, s, &out)) {
214 return false;
215 }
216 sigFile = gpgIdFile + QStringLiteral(".sig");
217 }
218 if (useGit &&
219 !commitGpgId(folder.absolutePath(), gpgIdFile, sigFile, s, &out)) {
220 return false;
221 }
222
223 // Entries in sub-folders too: a store taken over whole has most of them
224 // there. Visible regular files only, not through links, .git not walked.
225 // Not those under a sub-folder with a .gpg-id of its own: its recipients
226 // did not change here, and Users on the profile's root leaves it alone,
227 // as pass does.
228 const QString root = QDir::cleanPath(folder.absolutePath());
229 const bool viaPass = s.usePass;
230 const QString rootList =
231 QFileInfo(QDir(root).filePath(QStringLiteral(".gpg-id")))
232 .canonicalFilePath();
233 // What a folder's own .gpg-id decides, if it has one.
234 const auto ownList = [&rootList,
235 viaPass](const QString &dir) -> std::optional<bool> {
236 // A list of its own: a regular file. A linked .gpg-id is no list to
237 // the native backend, as everywhere in QtPass, so its folder's entries
238 // still count; pass follows the link (`-f`) and keeps that folder on
239 // the linked list, so with it they do not, unless the link leads to the
240 // root's own list: that folder follows the root, and Users on it
241 // re-encrypts them.
242 const QFileInfo list(QDir(dir).filePath(QStringLiteral(".gpg-id")));
243 if (viaPass && list.isFile() && !rootList.isEmpty() &&
244 list.canonicalFilePath() == rootList) {
245 return false;
246 }
247 if (list.isFile() &&
248 (viaPass || (!list.isSymLink() && !list.isJunction()))) {
249 return true;
250 }
251 return std::nullopt;
252 };
253 // Each folder's verdict once: climbing from every entry to the root again
254 // cost entries times depth, and a co-writer's deep chain of folders froze
255 // Finish for tens of minutes.
256 QHash<QString, bool> verdictOf;
257 const auto underSubStore = [&root, &ownList,
258 &verdictOf](const QString &entry) {
259 QStringList climbed;
260 QString dir = QFileInfo(entry).absolutePath();
261 bool verdict = false;
262 while (dir != root && dir.startsWith(root + QLatin1Char('/'))) {
263 if (const auto known = verdictOf.constFind(dir);
264 known != verdictOf.constEnd()) {
265 verdict = known.value();
266 break;
267 }
268 climbed << dir;
269 if (const std::optional<bool> own = ownList(dir)) {
270 verdict = *own;
271 break;
272 }
273 dir = QFileInfo(dir).absolutePath();
274 }
275 for (const QString &folder : std::as_const(climbed)) {
276 verdictOf.insert(folder, verdict);
277 }
278 return verdict;
279 };
280 QStringList existing =
281 Util::regularFilesUnder(root, {QStringLiteral("*.gpg")});
282 existing.erase(
283 std::remove_if(existing.begin(), existing.end(), underSubStore),
284 existing.end());
285 if (!existing.isEmpty()) {
286 out = tr("%1 already contains %n encrypted file(s); they were not "
287 "re-encrypted. Switch to the profile and open Users to do that.",
288 nullptr, static_cast<int>(existing.size()))
289 .arg(folder.absolutePath());
290 }
291 return true;
292}

294auto ProfileInit::writeGpgId(const QString &gpgIdFile,
295 const QList<UserInfo> &users, bool signed_,
296 QString *note, QByteArray *written) -> bool {
297 QStringList ids;
298 for (const UserInfo &user : users) {
299 if (user.enabled) {
300 ids << user.key_id;
301 }
302 }
303 if (ids.isEmpty()) {
304 *note = tr("No recipient selected; %1 was not written.").arg(gpgIdFile);
305 return false;
306 }
307 // As ImitatePass::writeGpgIdFile: renamed into place whole, so an
308 // interrupted write leaves no half .gpg-id to sign or use. Owner-only: the
309 // list leaks which keys the store is encrypted to.
310 QByteArray contents =
311 (ids.join(QLatin1Char('\n')) + QLatin1Char('\n')).toUtf8();
312 if (signed_) {
313 // Same rule as ImitatePass::writeGpgIdFile: reserved and recorded
314 // before the write, bound to the store root (GpgIdGeneration). A new
315 // profile has no list on disk to take a generation from.
316 QString why;
317 const std::optional<QString> store = GpgIdGeneration::storeIdFor(
318 QFileInfo(gpgIdFile).absolutePath(), std::nullopt, &why);
319 if (!store) {
320 *note = why;
321 return false;
322 }
323 const std::optional<qint64> generation =
324 GpgIdGeneration::reserveNext(gpgIdFile, std::nullopt, &why);
325 if (!generation) {
326 *note = why;
327 return false;
328 }
329 contents = GpgIdGeneration::withHeader(*generation, QStringLiteral("."),
330 contents, *store);
331 }
332 QString why;
333 if (!Util::writeFileReplacing(gpgIdFile, contents, true, &why)) {
334 *note = why;
335 return false;
336 }
337 if (written != nullptr) {
338 *written = contents;
339 }
340 if (signed_) {
341 // Same as ImitatePass::writeGpgIdFile: the bytes written are the ones
342 // recognised; not fatal, the first read digests them otherwise.
343 QString why;
344 if (!GpgIdGeneration::recordWritten(gpgIdFile, contents, &why)) {
345 qCWarning(lcQtPass) << "Could not record the written .gpg-id:" << why;
346 }
347 }
348 return true;
349}

351auto ProfileInit::signGpgId(const QString &gpgIdFile,
352 const QByteArray &contents, const AppSettings &s,
353 QString *note) -> bool {
354 const GpgIdSigner signer = signerFor(s);
355 QString err;
356 if (!signer.sign(gpgIdFile, contents, &err)) {
357 *note = err.trimmed().isEmpty()
358 ? tr("Could not sign %1 with %2.")
359 .arg(gpgIdFile, signer.keys().first())
360 : tr("Could not sign %1 with %2: %3")
361 .arg(gpgIdFile, signer.keys().first(), err.trimmed());
362 // withGpgHome() fell back to the inherited home: gpg's "no secret key"
363 // would not say that the configured one is missing.
364 if (!s.gpgHome.isEmpty()) {
365 QDir home(s.gpgHome);
366 home.makeAbsolute();
367 if (!home.exists()) {
368 *note +=
369 QLatin1Char(' ') +
370 tr("The configured GPG home %1 does not exist.").arg(home.path());
371 }
372 }
373 return false;
374 }
375 return true;
376}

378auto ProfileInit::runGit(QProcess &git, const AppSettings &s,
379 const QString &dir, const QStringList &args,
380 QString *note) -> bool {
381 const bool bounded = args.first() == QLatin1String("init") ||
382 args.first() == QLatin1String("add");
383 QElapsedTimer clock;
384 clock.start();
385 QString err;
386 const int rc = Executor::executeBlocking(git, s.gitExecutable, args,
387 QString(), nullptr, &err, nullptr,
388 bounded ? kGitSetupTimeoutMs : -1);
389 if (rc == 0) {
390 return true;
391 }
392 // -1 is what a stopped run returns; git's own failures are positive.
393 if (bounded && rc == -1 && clock.elapsed() >= kGitSetupTimeoutMs) {
394 *note = tr("git %1 did not finish within %2 minutes in %3, and was "
395 "stopped.")
396 .arg(args.first())
397 .arg(kGitSetupTimeoutMs / 60000)
398 .arg(dir);
399 } else {
400 *note = tr("git %1 failed in %2: %3").arg(args.first(), dir, err.trimmed());
401 }
402 return false;
403}

405auto ProfileInit::refusedForGit(const QString &dir, const QStringList &files,
406 QString *note) -> bool {
407 const QString path = blocksGit(dir, files);
408 if (path.isEmpty()) {
409 return false;
410 }
411 *note = QFileInfo(path).fileName() == QLatin1String(".git")
412 ? tr("%1 is a Git repository inside the store. Move it out of "
413 "the store, then try again.")
414 .arg(QDir::toNativeSeparators(QFileInfo(path).path()))
415 : tr("%1 is not a regular file, and Git could wait on it for "
416 "good. Remove it or make it a regular file, then try "
417 "again.")
418 .arg(QDir::toNativeSeparators(path));
419 return true;
420}

422auto ProfileInit::commitGpgId(const QString &dir, const QString &gpgIdFile,
423 const QString &sigFile, const AppSettings &s,
424 QString *note) -> bool {
425 // A process of our own in the profile directory: nothing here may touch
426 // the shared Executor or PASSWORD_STORE_DIR of the active store.
427 QProcess git;
428 git.setWorkingDirectory(dir);
429 git.setProcessEnvironment(gitEnvironment(dir, s));
430 auto run = [&](const QStringList &args) -> bool {
431 return runGit(git, s, dir, args, note);
432 };
433 QStringList files{QFileInfo(gpgIdFile).fileName()};
434 if (!sigFile.isEmpty()) {
435 files << QFileInfo(sigFile).fileName();
436 }
437 // initialise() checked before writing these; again here, as signing can
438 // take a while and the folder is shared.
439 if (refusedForGit(dir, files, note)) {
440 return false;
441 }
442 const QStringList add =
443 QStringList{QStringLiteral("add"), QStringLiteral("--")} + files;
444 if (!run({QStringLiteral("init")}) || !run(add)) {
445 return false;
446 }
447 return run(commitArgs(git, s) +
448 QStringList{QStringLiteral("-m"),
449 QStringLiteral("Added .gpg-id using QtPass."),
450 QStringLiteral("--")} +
451 files);
452}

454auto ProfileInit::initGit(const QString &dir, const AppSettings &s,
455 QString *note) -> bool {
456 QProcess git;
457 git.setWorkingDirectory(dir);
458 git.setProcessEnvironment(gitEnvironment(dir, s));
459 auto run = [&](const QStringList &args) -> bool {
460 return runGit(git, s, dir, args, note);
461 };
462 // Stage only store files (as the re-encryption backup commit does): an
463 // existing folder may hold plaintext exports or swap files. Regular files
464 // only: a link or junction must not have its target staged.
465 QStringList files;
466 const QDir base(dir);
467 // Hidden directories (.git among them) are not walked, hidden files are:
468 // .gpg-id is one.
469 const QStringList found = Util::regularFilesUnder(
470 dir,
471 {QStringLiteral("*.gpg"), QStringLiteral(".gpg-id"),
472 QStringLiteral(".gpg-id.sig")},
473 nullptr, true);
474 for (const QString &path : found) {
475 files << base.relativeFilePath(path);
476 }
477 if (refusedForGit(dir, files, note)) {
478 return false;
479 }
480 if (!run({QStringLiteral("init")})) {
481 return false;
482 }
483 if (!files.isEmpty() &&
484 !run(QStringList{QStringLiteral("add"), QStringLiteral("--")} + files)) {
485 return false;
486 }
487 return run(commitArgs(git, s) +
488 QStringList{QStringLiteral("--allow-empty"), QStringLiteral("-m"),
489 QStringLiteral("Added password store using QtPass.")});
490}
