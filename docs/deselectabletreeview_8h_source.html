78 void mousePressEvent(QMouseEvent *event) override {
79 // A new press belongs to a new click: whatever the previous release left
80 // pending must not clear the selection this press is about to make.

83 selectionModel()->isSelected(indexAt(event->position().toPoint()));
84 QTreeView::mousePressEvent(event);
85 }

93 void mouseReleaseEvent(QMouseEvent *event) override {
94 // Persistent: QTreeView::mouseReleaseEvent() emits clicked(), and what
95 // that runs can change the model (a message box for a link, the file
96 // system model reloading meanwhile); a plain index then names another
97 // row, or memory that is gone.
98 const QPersistentModelIndex item(indexAt(event->position().toPoint()));
99 const bool pressWasOnSelection = clickSelected;
100 clickSelected = false;
101
102 // Handle the release immediately: waiting here would delay selection —
103 // and therefore every decrypt — by the double-click interval.
104 QTreeView::mouseReleaseEvent(event);
105
106 const bool emptyOrSelected =
107 !item.isValid() || selectionModel()->isSelected(item);
108 if (!pressWasOnSelection || !emptyOrSelected)
109 return;
110
111 // Deselecting is the one decision that must wait: a double-click starts
112 // with a release, and opening the editor must win over clearing the
113 // selection. Ask the platform how long a double-click may take instead
114 // of assuming 200 ms.
115 doubleClickHappened = false;
116 deselectTimer.start(
117 QGuiApplication::styleHints()->mouseDoubleClickInterval());
118 }
119
124 void deselect() {
125 if (doubleClickHappened)
126 return;
127 clearSelection();
128 selectionModel()->setCurrentIndex(QModelIndex(),
129 QItemSelectionModel::Select);
130 emit emptyClicked();
131 }
132
138 void mouseDoubleClickEvent(QMouseEvent *event) override {
139 doubleClickHappened = true;
140 deselectTimer.stop();
141 QTreeView::mouseDoubleClickEvent(event);
142 }
