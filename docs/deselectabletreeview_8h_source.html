58 explicit DeselectableTreeView(QWidget *parent) : QTreeView(parent) {
59 deselectTimer.setSingleShot(true);

74private:
75 bool doubleClickHappened = false;
76 bool clickSelected = false;
77 QTimer deselectTimer;
78
83 void mousePressEvent(QMouseEvent *event) override {
84 // A new press belongs to a new click: whatever the previous release left
85 // pending must not clear the selection this press is about to make.
86 deselectTimer.stop();
87 clickSelected =
88 selectionModel()->isSelected(indexAt(event->position().toPoint()));
89 QTreeView::mousePressEvent(event);
90 }
91
96 void mouseReleaseEvent(QMouseEvent *event) override {
97 const QModelIndex item = indexAt(event->position().toPoint());
98 const bool pressWasOnSelection = clickSelected;
99 clickSelected = false;
100
101 // Handle the release immediately: waiting here would delay selection —
102 // and therefore every decrypt — by the double-click interval.
103 QTreeView::mouseReleaseEvent(event);
104
105 const bool emptyOrSelected =
106 !item.isValid() || selectionModel()->isSelected(item);
107 if (!pressWasOnSelection || !emptyOrSelected)
108 return;
109
110 // Deselecting is the one decision that must wait: a double-click starts
111 // with a release, and opening the editor must win over clearing the
112 // selection. Ask the platform how long a double-click may take instead
113 // of assuming 200 ms.
114 doubleClickHappened = false;
115 deselectTimer.start(
116 QGuiApplication::styleHints()->mouseDoubleClickInterval());
117 }
118
119 void deselect() {
120 if (doubleClickHappened)
121 return;
122 clearSelection();
123 selectionModel()->setCurrentIndex(QModelIndex(),
124 QItemSelectionModel::Select);
125 emit emptyClicked();
126 }
127
132 void mouseDoubleClickEvent(QMouseEvent *event) override {
133 doubleClickHappened = true;
134 deselectTimer.stop();
135 QTreeView::mouseDoubleClickEvent(event);
136 }
137};
