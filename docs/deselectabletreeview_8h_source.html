77 void mousePressEvent(QMouseEvent *event) override {
78 // A new press belongs to a new click: whatever the previous release left
79 // pending must not clear the selection this press is about to make.
80 deselectTimer.stop();
81 clickSelected =
82 selectionModel()->isSelected(indexAt(event->position().toPoint()));
83 QTreeView::mousePressEvent(event);
84 }

92 void mouseReleaseEvent(QMouseEvent *event) override {
93 const QModelIndex item = indexAt(event->position().toPoint());
94 const bool pressWasOnSelection = clickSelected;
95 clickSelected = false;
96
97 // Handle the release immediately: waiting here would delay selection —
98 // and therefore every decrypt — by the double-click interval.
99 QTreeView::mouseReleaseEvent(event);
100
101 const bool emptyOrSelected =
102 !item.isValid() || selectionModel()->isSelected(item);
103 if (!pressWasOnSelection || !emptyOrSelected)
104 return;
105
106 // Deselecting is the one decision that must wait: a double-click starts
107 // with a release, and opening the editor must win over clearing the
108 // selection. Ask the platform how long a double-click may take instead
109 // of assuming 200 ms.
110 doubleClickHappened = false;
111 deselectTimer.start(
112 QGuiApplication::styleHints()->mouseDoubleClickInterval());
113 }
114
119 void deselect() {
120 if (doubleClickHappened)
121 return;
122 clearSelection();
123 selectionModel()->setCurrentIndex(QModelIndex(),
124 QItemSelectionModel::Select);
125 emit emptyClicked();
126 }
127
133 void mouseDoubleClickEvent(QMouseEvent *event) override {
134 doubleClickHappened = true;
135 deselectTimer.stop();
136 QTreeView::mouseDoubleClickEvent(event);
137 }
