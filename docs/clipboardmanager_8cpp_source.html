40auto markedSecret(const QMimeData *data) -> bool {
41 if (data == nullptr) {
42 return false;
43 }
44 const QStringList formats = data->formats();
45 for (const QString &format : formats) {
46 for (const char *mark :
47 {"passwordManagerHint", "nspasteboard-concealed-type",
48 "nspasteboard.ConcealedType",
49 "ExcludeClipboardContentFromMonitorProcessing"}) {
50 if (format.contains(QLatin1String(mark), Qt::CaseInsensitive)) {
51 return true;
52 }
53 }
54 }
55 return false;
56}

59void ClipboardManager::copyIfAlways(const QString &text,
60 const QString &p_output) {
61 const AppSettings s = QtPassSettings::load();
62 // Only "always copy" mode actually places text on the clipboard here; in
63 // on-demand mode the copy happens later from the field's copy button.
64 // Crucially, do not touch m_clippedText otherwise — it tracks what is
65 // currently on the clipboard so the autoclear timer knows what to clear.
66 // Overwriting it when merely showing another entry left a previously copied
67 // password on the clipboard past its autoclear window (#1607).
68 if (s.clipBoardType == Enums::CLIPBOARD_ALWAYS && !p_output.isEmpty()) {
69 copyText(text);
70 }
71}

73void ClipboardManager::copyText(const QString &text) {
74 const AppSettings s = QtPassSettings::load();
75 QClipboard *clip = QApplication::clipboard();
76
77 QClipboard::Mode mode = QClipboard::Clipboard;
78 if (s.useSelection && clip->supportsSelection()) {
79 mode = QClipboard::Selection;
80 }
81
82 // What was there before the first secret, to put back when autoclear
83 // runs (as pass -c does). Text only, as pass keeps it: rich content comes
84 // back as its plain text. Where Qt cannot read the clipboard without focus
85 // (some Wayland compositors, a copy from the tray), "before" reads as
86 // nothing and autoclear clears instead: the safe way round.
87 if (s.useAutoclear && s.restoreClipboard) {
88 std::optional<QString> &previous = mode == QClipboard::Selection
89 ? m_previousSelection
90 : m_previousClipboard;
91 const QString now = clip->text(mode);
92 // Taken once per window, at the first copy, whatever is there: a text
93 // of ours, or the very text being copied, is saved as nothing (restoring
94 // it would leave a secret on the clipboard for good), and a later copy
95 // in the window cannot take a snapshot instead.
96 if (!previous) {
97 // Another password manager's secret is marked as one; put back as
98 // plain text it would lose that mark (and go into clipboard history)
99 // and nobody would clear it again.
100 const bool safe = now != text && now != m_clippedText &&
101 !m_otherCopies.contains(now) &&
102 !markedSecret(clip->mimeData(mode));
103 previous = safe ? now : QString();
104 }
105 }
106 forgetIfSaved(text);
107 clip->setMimeData(buildClipboardMimeData(text), mode);
108
109 // The text copied before may still be in the other buffer (a PIN copied to
110 // the clipboard, then selected, so also in the X11 selection): replacing
111 // it here would leave it there past autoclear.
112 if (!m_clippedText.isEmpty() && m_clippedText != text &&
113 !m_otherCopies.contains(m_clippedText) &&
114 (clip->text(QClipboard::Clipboard) == m_clippedText ||
115 clip->text(QClipboard::Selection) == m_clippedText)) {
116 m_otherCopies.append(m_clippedText);
117 }
118 m_clippedText = text;
119 emit statusMessage(tr("Copied to clipboard"));
120 if (s.useAutoclear) {
121 m_clearTimer.start();
122 }
123}

125void ClipboardManager::forgetIfSaved(const QString &text) {
126 // The saved text is copied as a secret now: putting it back would leave
127 // that secret on the clipboard. Emptied, not dropped, so no later copy
128 // takes a new snapshot in this window either; clear() then clears.
129 for (std::optional<QString> *previous :
130 {&m_previousClipboard, &m_previousSelection}) {
131 if (*previous && **previous == text) {
132 *previous = QString();
133 }
134 }

137void ClipboardManager::trackCopied(const QString &text) {
138 if (text.isEmpty()) {
139 return;
140 }
141 forgetIfSaved(text);
142 // Kept beside m_clippedText, not instead of it: replacing it would leave a
143 // password copied earlier on the clipboard past its autoclear (#1607).
144 constexpr qsizetype kMaxOtherCopies = 16;
145 if (text != m_clippedText && !m_otherCopies.contains(text)) {
146 // A mouse selection reports itself on every change while it grows or
147 // shrinks (the X11 primary selection); only where it ends counts, or a
148 // passing "a" would stay tracked and autoclear could take someone
149 // else's "a" later.
150 // An earlier copy still on the clipboard proper (Ctrl+C) stays tracked.
151 const QString &last =
152 m_otherCopies.isEmpty() ? QString() : m_otherCopies.last();
153 if (!last.isEmpty() && (text.contains(last) || last.contains(text)) &&
154 QApplication::clipboard()->text(QClipboard::Clipboard) != last &&
155 QApplication::clipboard()->text(QClipboard::Selection) != last) {
156 m_otherCopies.last() = text;
157 } else {
158 if (m_otherCopies.size() >= kMaxOtherCopies) {
159 // Make room, but never by forgetting a text that is still on the
160 // clipboard or in the selection: it would then outlive autoclear.
161 // At most two are, so the oldest of the rest can always go.
162 const QString onClipboard =
163 QApplication::clipboard()->text(QClipboard::Clipboard);
164 const QString inSelection =
165 QApplication::clipboard()->text(QClipboard::Selection);
166 for (qsizetype i = 0; i < m_otherCopies.size(); ++i) {
167 if (m_otherCopies.at(i) != onClipboard &&
168 m_otherCopies.at(i) != inSelection) {
169 m_otherCopies.removeAt(i);
170 break;
171 }
172 }
173 }
174 m_otherCopies.append(text);
175 }
176 }
177 // Not pushed back: a selection reports itself on every change, and a drag
178 // too, so restarting here gave a password copied earlier a new full
179 // period each time. The first deadline clears everything tracked.
180 if (QtPassSettings::load().useAutoclear && !m_clearTimer.isActive()) {
181 m_clearTimer.start();
182 }
183}

185void ClipboardManager::clear() {
186 // A clear before the deadline (a deselect) ends this window: a copy
187 // tracked after it starts a full one, not the rest of the old one.
188 m_clearTimer.stop();
189 // Whatever happens below, the saved text is forgotten: it is the user's,
190 // and no later autoclear may put it back.
191 const std::optional<QString> previousClipboard =
192 std::exchange(m_previousClipboard, std::nullopt);
193 const std::optional<QString> previousSelection =
194 std::exchange(m_previousSelection, std::nullopt);
195 if (m_clippedText.isEmpty() && m_otherCopies.isEmpty()) {
196 // Nothing of ours is out there. The old code compared the empty tracker
197 // with an empty selection, "cleared" that and said so on every deselect.
198 return;
199 }
200 const auto ours = [this](const QString &text) {
201 return !text.isEmpty() &&
202 (text == m_clippedText || m_otherCopies.contains(text));
203 };
204 QClipboard *clipboard = QApplication::clipboard();
205 const bool restore = QtPassSettings::load().restoreClipboard;
206 bool cleared = false;
207 bool restored = false;
208 // Ours still there: the text from before comes back if there was one and
209 // what is there is copyText()'s own copy, otherwise it is cleared. A copy
210 // from a shown field (trackCopied()) is cleared on purpose, not restored
211 // over: it can come long after copyText(), when "before" is stale, and the
212 // user chose that copy themselves. Something else copied meanwhile is left
213 // alone. The setting is read now: switched off during the window, the
214 // saved text was still held until here, and is dropped unused.
215 const auto putBack = [&](QClipboard::Mode mode,
216 const std::optional<QString> &previous) {
217 const QString now = clipboard->text(mode);
218 if (!ours(now)) {
219 return;
220 }
221 if (restore && previous && !previous->isEmpty() && now == m_clippedText) {
222 clipboard->setText(*previous, mode);
223 restored = true;
224 return;
225 }
226 clipboard->clear(mode);
227 if (mode == QClipboard::Selection) {
228 clipboard->setText(QString(""), QClipboard::Selection);
229 }
230 cleared = true;
231 };
232 putBack(QClipboard::Selection, previousSelection);
233 putBack(QClipboard::Clipboard, previousClipboard);
234 emit statusMessage(restored ? tr("Clipboard restored")
235 : cleared ? tr("Clipboard cleared")
236 : tr("Clipboard not cleared"));
237 m_clippedText.clear();
238 m_otherCopies.clear();
239}

241auto buildClipboardMimeData(const QString &text) -> QMimeData * {
242 auto *mimeData = new QMimeData();
243 mimeData->setText(text);
244// The KDE hint (what wl-copy --sensitive sets) keeps the secret out of
245// Klipper, CopyQ and cliphist history, on the BSDs as well as on Linux.
246#if defined(Q_OS_UNIX) && !defined(Q_OS_DARWIN)
247 mimeData->setData("x-kde-passwordManagerHint", QByteArray("secret"));
248#endif
249#ifdef Q_OS_MAC
250 mimeData->setData("application/x-nspasteboard-concealed-type", QByteArray());
251#endif
252#ifdef Q_OS_WIN
253 mimeData->setData("ExcludeClipboardContentFromMonitorProcessing",
254 dwordBytes(1));
255 mimeData->setData("CanIncludeInClipboardHistory", dwordBytes(0));
256 mimeData->setData("CanUploadToCloudClipboard", dwordBytes(0));
257#endif
258 return mimeData;
259}
