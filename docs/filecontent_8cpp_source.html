37static auto bareOtpUriOf(const QString &line) -> QString {
38 const QString trimmed = line.trimmed();
39 if (isOtpUri(trimmed)) {
40 return trimmed;
41 }
42 static const QString scheme = QStringLiteral("otpauth:");
43 if (!trimmed.startsWith(scheme, Qt::CaseInsensitive)) {
44 return {};
45 }
46 qsizetype i = scheme.size();
47 while (i < trimmed.size() && trimmed.at(i).isSpace()) {
48 ++i;
49 }
50 if (trimmed.mid(i, 2) != QLatin1String("//")) {
51 return {};
52 }
53 return trimmed.left(scheme.size()) + trimmed.mid(i);
54}

79static auto isBase32Seed(const QString &value) -> bool {
80 qsizetype letters = 0;
81 for (const QChar c : value) {
82 const char16_t u = c.unicode();
83 // Grouping and padding, as seeds are written and pasted: spaces, tabs, a
84 // no-break space, dashes, dots, underscores, "=". Base32::sanitizeInput()
85 // drops them too, and Totp::parse() accepts them.
86 if (c.isSpace() || u == u'-' || u == u'.' || u == u'_' || u == u'=') {
87 continue;
88 }
89 if (!((u >= u'A' && u <= u'Z') || (u >= u'a' && u <= u'z') ||
90 (u >= u'2' && u <= u'7'))) {
91 return false;
92 }
93 ++letters;
94 }
95 constexpr qsizetype kMinSeedLetters = 16;
96 return letters >= kMinSeedLetters && Totp::isValid(value);
97}

117static auto holdsOtpConfig(const QString &line) -> bool {
118 if (!bareOtpUriOf(line).isEmpty()) {
119 return true;
120 }
121 const qsizetype colon = line.indexOf(u':');
122 if (colon < 0) {
123 return false;
124 }
125 const QString value = line.mid(colon + 1);
126 // Any field holding an otpauth URI is a shared secret; an OTP-named field
127 // only when what it holds would give a code: the line is the password
128 // otherwise, as the pass format has it.
129 return isOtpUri(value) ||
130 (FileContent::isOtpFieldName(line.left(colon)) && isBase32Seed(value));
131}

152struct OtpCandidate {
153 QString first;
154 QString usable;
155
160 void offer(const QString &config) {
161 if (!usable.isEmpty()) {
162 return;
163 }
164 if (first.isEmpty()) {
165 first = config;
166 }
167 if (Totp::isValid(config)) {
168 usable = config;
169 }
170 }
171};

178struct OtpCandidates {
179 OtpCandidate field;
180 OtpCandidate bareLine;
181 OtpCandidate remainingField;
182
190 auto pick() const -> QString {
191 for (const OtpCandidate *kind : {&field, &bareLine, &remainingField}) {
192 if (!kind->usable.isEmpty()) {
193 return kind->usable;
194 }
195 }
196 for (const OtpCandidate *kind : {&field, &bareLine, &remainingField}) {
197 if (!kind->first.isEmpty()) {
198 return kind->first;
199 }
200 }
201 return {};
202 }
203};

217auto takeNamedValue(const QString &line, const QStringList &templateFields,
218 bool allFields,
219 const std::function<bool(const QString &)> &alsoField,
220 NamedValues &namedValues, OtpCandidates &otp) -> bool {
221 const qsizetype colon = line.indexOf(u':');
222 if (colon < 0) {
223 return false;
224 }
225 const QString name = line.left(colon);
226 const QString value = line.right(line.length() - colon - 1);
227 // A value starting with // means the colon is probably from a URL.
228 if (!templateFields.contains(name) &&
229 !(allFields && !value.startsWith(QStringLiteral("//"))) &&
230 !(alsoField && alsoField(name.trimmed()))) {
231 // An `OTP:` line the template did not promote to a field.
232 if (FileContent::isOtpFieldName(name)) {
233 otp.remainingField.offer(value.trimmed());
234 }
235 return false;
236 }
237 namedValues.append({name.trimmed(), value.trimmed(), line});
238 // Keyed on the value as well as the name: a field called anything at all
239 // whose value is an otpauth URI is still a shared secret.
240 if (FileContent::isOtpFieldName(name) || FileContent::isOtpUriValue(value)) {
241 otp.field.offer(value.trimmed());
242 }
243 return true;
244}

263auto FileContent::parse(const QString &fileContent,
264 const QStringList &templateFields, bool allFields,
265 const std::function<bool(const QString &)> &alsoField)
266 -> FileContent {
267 // Line by line over the text, never a list of every line: a compressed
268 // entry can expand into tens of millions of lines within the plaintext
269 // limit, and a QString each (empty ones too) would take gigabytes.
270 QStringTokenizer tokens = QStringView(fileContent).tokenize(u'\n');
271 auto it = tokens.begin();
272 QString password;
273 if (it != tokens.end()) {
274 password = passwordLine(*it);
275 ++it;
276 }
277 QString remainingData;
278 QString remainingDataDisplay;
279 qsizetype remainingLines = 0;
280 qsizetype displayedLines = 0;
281 qsizetype notShown = 0;
282 // Past kMaxNamedFields a line is still offered to takeNamedValue(), into
283 // this throwaway list, for what it notes about OTP: the line stays text.
284 NamedValues pastTheCap;
285 NamedValues namedValues;
286 OtpCandidates otp;
287 // An entry from `pass otp insert` has the URI as its only (password) line,
288 // which the loop below never sees; an `OTP:`/`TOTP:` field written there by
289 // hand holding a usable secret is the same, and yields a code as it does
290 // further down. Read only from what is written as a configuration
291 // (holdsOtpConfig()); a first line with a URI elsewhere in it is kept off
292 // the screen all the same (isOtpPasswordLine()).
293 if (holdsOtpConfig(password)) {
294 otp.bareLine.offer(otpConfigOf(password));
295 }
296 for (; it != tokens.end(); ++it) {
297 const QString line = it->toString();
298 bool secretPastTheCap = false;
299 // A line with an otpauth URI in it is never a field unless the URI is
300 // its value ("2fa: otpauth://..."), also not when the template lists
301 // "otpauth": split at a colon, the value would lose the scheme that
302 // marks it as a shared secret and be shown seed and all ("otpauth://...",
303 // "AWS: backup phone: otpauth://..."). Past kMaxNamedFields, lines stay
304 // text.
305 if (!carriesOtpUri(line) || isOtpUri(afterFirstColon(line))) {
306 if (namedValues.size() < kMaxNamedFields) {
307 if (takeNamedValue(line, templateFields, allFields, alsoField,
308 namedValues, otp)) {
309 continue;
310 }
311 } else {
312 takeNamedValue(line, templateFields, allFields, alsoField, pastTheCap,
313 otp);
314 // A secret past the cap stays text, but not on screen: the panel
315 // would have hidden it.
316 secretPastTheCap = !pastTheCap.isEmpty() && alsoField &&
317 alsoField(pastTheCap.first().name);
318 pastTheCap.clear();
319 }
320 }
321 // Joined with "\n" as they go, as split() and join() did.
322 if (remainingLines > 0) {
323 remainingData += QLatin1Char('\n');
324 }
325 remainingData += line;
326 ++remainingLines;
327 // A bare otpauth:// line, the convention the pass-otp extension uses,
328 // or the split form older versions saved.
329 if (const QString uri = bareOtpUriOf(line); !uri.isEmpty()) {
330 otp.bareLine.offer(uri);
331 }
332 if (!isLineHidden(line) && !secretPastTheCap) {
333 if (displayedLines == kMaxDisplayLines) {
334 ++notShown;
335 continue;
336 }
337 if (displayedLines > 0) {
338 remainingDataDisplay += QLatin1Char('\n');
339 }
340 remainingDataDisplay += line;
341 ++displayedLines;
342 }
343 }
344 if (notShown > 0) {
345 remainingDataDisplay += QLatin1Char('\n') + moreLinesNotShown(notShown);
346 }
347 // The OTP value stays in namedValues/remainingData: the edit dialog reads it
348 // from there and PasswordDialog::getPassword() drops empty fields, so
349 // stripping it here would delete the secret on the next save. Suppression
350 // happens in getRemainingDataForDisplay() and PasswordDisplayPanel.
351 return {password, namedValues, remainingData, remainingDataDisplay,
352 otp.pick(), remainingLines};
353}

361auto FileContent::isSecretFieldName(const QString &name,
362 const QStringList &secrets) -> bool {
363 // One pattern for the whole list, compiled once per list: the panel asks
364 // for every field of every entry shown, and the list rarely changes.
365 thread_local QStringList cachedSecrets;
366 thread_local QStringList cachedWords;
367 thread_local QRegularExpression cachedPattern;
368 thread_local bool cachedYet = false;
369 constexpr auto options = QRegularExpression::CaseInsensitiveOption |
370 QRegularExpression::UseUnicodePropertiesOption;
371 if (!cachedYet || secrets != cachedSecrets) {
372 cachedWords.clear();
373 for (const QString &secret : secrets) {
374 const QString word = secret.trimmed();
375 if (word.isEmpty()) {
376 continue;
377 }
378 // A word counts when no letter touches it: digits and `_` end a word
379 // as a blank does ("PIN2", "api_key"), and a plural "s" may follow
380 // ("tokens", "Security answers"); "key" is still not in "keyboard".
381 // A word that starts or ends with a digit is not part of a longer
382 // number either ("tan1" not in "tan12").
383 const QString letters = QStringLiteral("\\p{L}\\p{M}");
384 const QString before =
385 word.front().isDigit() ? letters + QStringLiteral("\\d") : letters;
386 const QString after =
387 word.back().isDigit() ? letters + QStringLiteral("\\d") : letters;
388 cachedWords << QStringLiteral("(?<![%1])%2s?(?![%3])")
389 .arg(before, QRegularExpression::escape(word), after);
390 }
391 cachedSecrets = secrets;
392 cachedYet = true;
393 cachedPattern = QRegularExpression(
394 QStringLiteral("(?:%1)").arg(cachedWords.join(QLatin1Char('|'))),
395 options);
396 }
397 if (cachedWords.isEmpty()) {
398 return false;
399 }
400 // camelCase names its words with capitals, which a case-insensitive match
401 // does not see, so the name is also tried with a blank at each word start:
402 // "apiKey" as "api Key", "APIKey" as "API Key". The acronym rule cuts a
403 // capitals word that a lower-case letter follows ("cardPINs" as "card PI
404 // Ns"), so a split at lower-to-upper starts alone is tried too ("card
405 // PINs", "sim PINcode"), and the name as written ("PINcode").
406 static const QRegularExpression lowerThenUpper(
407 QStringLiteral("(?<=[\\p{Ll}\\d])(?=\\p{Lu})"),
408 QRegularExpression::UseUnicodePropertiesOption);
409 static const QRegularExpression acronymThenWord(
410 QStringLiteral("(?<=\\p{Lu})(?=\\p{Lu}\\p{Ll})"),
411 QRegularExpression::UseUnicodePropertiesOption);
412 QString camel = name;
413 camel.replace(lowerThenUpper, QStringLiteral(" "));
414 QString words = camel;
415 words.replace(acronymThenWord, QStringLiteral(" "));
416 const auto matches = [&](const QRegularExpression &pattern) {
417 return pattern.match(name).hasMatch() ||
418 (camel != name && pattern.match(camel).hasMatch()) ||
419 (words != camel && pattern.match(words).hasMatch());
420 };
421 if (cachedPattern.isValid()) {
422 return matches(cachedPattern);
423 }
424 // Too many words for one pattern (PCRE2's size limit): word by word,
425 // rather than every secret field shown in plain view.
426 return std::any_of(cachedWords.cbegin(), cachedWords.cend(),
427 [&matches, options](const QString &word) {
428 return matches(QRegularExpression(word, options));
429 });
430}

437auto FileContent::carriesOtpUri(const QString &text) -> bool {
438 // Not only at the start: "- otpauth://", "\"otpauth://...\"", "old token
439 // otpauth://", "AWS: backup phone: otpauth://" and the split "otpauth:
440 // //totp/" all showed the seed, being neither a bare line nor the value
441 // after the first colon. By hand, not a regular expression: parse() asks
442 // this for every line of an entry, up to 64 MiB of them.
443 static const QString scheme = QStringLiteral("otpauth:");
444 for (qsizetype at = text.indexOf(scheme, 0, Qt::CaseInsensitive); at >= 0;
445 at = text.indexOf(scheme, at + 1, Qt::CaseInsensitive)) {
446 qsizetype i = at + scheme.size();
447 while (i < text.size() && text.at(i).isSpace()) {
448 ++i;
449 }
450 if (text.mid(i, 2) == QLatin1String("//")) {
451 return true;
452 }
453 }
454 return false;
455}

506auto FileContent::boundedForDisplay(const QString &text) -> QString {
507 // Where the line after the last one shown starts, if there is one.
508 qsizetype end = -1;
509 for (int line = 0; line < kMaxDisplayLines; ++line) {
510 end = text.indexOf(QLatin1Char('\n'), end + 1);
511 if (end < 0) {
512 return text;
513 }
514 }
515 const qsizetype more =
516 text.count(QLatin1Char('\n'), Qt::CaseSensitive) - (kMaxDisplayLines - 1);
517 return text.left(end) + QLatin1Char('\n') + moreLinesNotShown(more);
518}
