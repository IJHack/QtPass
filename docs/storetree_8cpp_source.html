26void StoreTree::setStore(const QString &path) {
27 m_store = path;
28 const QModelIndex rootDir = m_fs.setRootPath(path);
29 m_fs.fetchMore(rootDir);
30 if (m_proxy.sourceModel() == nullptr) {
31 m_proxy.setModelAndStore(&m_fs, path);
32 // Columns exist only once a source model is attached: hiding the size,
33 // type and date ones is a no-op before that point, and addressing header
34 // section 0 without any sections crashes on Qt 6.8.
35 m_view->setColumnHidden(1, true);
36 m_view->setColumnHidden(2, true);
37 m_view->setColumnHidden(3, true);
38 m_view->header()->setSectionResizeMode(0, QHeaderView::Stretch);
39 m_view->sortByColumn(0, Qt::AscendingOrder);
40 } else {
41 m_proxy.setStore(path);
42 }
43 m_proxy.setFilterRegularExpression(QRegularExpression());
44 m_view->setRootIndex(m_proxy.rootIndexFor(path));
45}

55auto StoreTree::dirFor(const QModelIndex &index, bool forPass) const
56 -> QString {
57 const QString abspath = QDir(m_store).absolutePath() + QDir::separator();
58 if (!index.isValid()) {
59 return forPass ? QString() : abspath;
60 }
61 const QFileInfo info = m_fs.fileInfo(m_proxy.mapToSource(index));
62 QString filePath =
63 info.isFile() ? info.absolutePath() : info.absoluteFilePath();
64 if (forPass) {
65 filePath = QDir(abspath).relativeFilePath(filePath);
66 }
67 filePath += QDir::separator();
68 return filePath;
69}

75auto StoreTree::fileFor(const QModelIndex &index, bool forPass) const
76 -> QString {
77 if (!index.isValid() || !m_fs.fileInfo(m_proxy.mapToSource(index)).isFile()) {
78 return {};
79 }
80 QString filePath = m_fs.filePath(m_proxy.mapToSource(index));
81 if (forPass) {
82 filePath = QDir(m_store).relativeFilePath(filePath);
83 filePath.replace(Util::endsWithGpg(), "");
84 }
85 return filePath;
86}
