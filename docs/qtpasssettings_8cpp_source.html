29// Portable mode: a qtpass.ini next to the executable wins over the platform
30// settings store.
31auto QtPassSettings::getInstance() -> QtPassSettings * {
32 if (!QtPassSettings::initialized) {
33 QString portable_ini = QCoreApplication::applicationDirPath() +
34 QDir::separator() + "qtpass.ini";
35 if (QFile(portable_ini).exists()) {
36 m_instance = new QtPassSettings(portable_ini, QSettings::IniFormat);
37 } else {
38 m_instance = new QtPassSettings("IJHack", "QtPass");
39 }
40
41 initialized = true;
42 }
43
44 return m_instance;
45}

47auto QtPassSettings::load() -> AppSettings {
48 AppSettings s = SettingsSerializer::load(*getInstance());
49 if (!s.passStore.isEmpty()) {
50 s.passStore = QDir::cleanPath(QDir(s.passStore).absolutePath());
51 if (!s.passStore.endsWith('/'))
52 s.passStore += '/';
53 }
54 return s;
55}

67auto QtPassSettings::getProfiles() -> Profiles {
68 getInstance()->beginGroup(SettingsConstants::profile);
69 Profiles profiles;
70
71 // migration from version <= v1.3.2: profiles datastructure
72 QStringList childKeys = getInstance()->childKeys();
73 for (const auto &key : std::as_const(childKeys)) {
74 Profile profile;
75 profile.path = getInstance()->value(key).toString();
76 profiles.insert(key, profile);
77 }
78 // /migration from version <= v1.3.2
79
80 QStringList childGroups = getInstance()->childGroups();
81 for (const auto &group : std::as_const(childGroups)) {
82 Profile profile;
83 profile.path = getInstance()->value(group + "/path").toString();
84 profile.signingKey = getInstance()->value(group + "/signingKey").toString();
85 profile.useGit = Profile::flagFromString(
86 getInstance()->value(group + "/useGit").toString());
87 profile.autoPush = Profile::flagFromString(
88 getInstance()->value(group + "/autoPush").toString());
89 profile.autoPull = Profile::flagFromString(
90 getInstance()->value(group + "/autoPull").toString());
91 profiles.insert(group, profile);
92 }
93
94 getInstance()->endGroup();
95
96 return profiles;
97}

99void QtPassSettings::setProfiles(const Profiles &profiles) {
100 // "profile" is both the active-profile scalar (getProfile()) and the
101 // profile/<name>/ group; remove() wipes both, which would switch the active
102 // store on every save, so preserve the active name across the rewrite.
103 const QString activeProfile =
104 getInstance()->value(SettingsConstants::profile).toString();
105 getInstance()->remove(SettingsConstants::profile);
106 if (!activeProfile.isEmpty()) {
107 getInstance()->setValue(SettingsConstants::profile, activeProfile);
108 }
109 getInstance()->beginGroup(SettingsConstants::profile);
110
111 for (auto i = profiles.cbegin(); i != profiles.cend(); ++i) {
112 getInstance()->setValue(i.key() + "/path", i.value().path);
113 getInstance()->setValue(i.key() + "/signingKey", i.value().signingKey);
114 getInstance()->setValue(i.key() + "/useGit",
115 Profile::flagToString(i.value().useGit));
116 getInstance()->setValue(i.key() + "/autoPush",
117 Profile::flagToString(i.value().autoPush));
118 getInstance()->setValue(i.key() + "/autoPull",
119 Profile::flagToString(i.value().autoPull));
120 }
121
122 getInstance()->endGroup();
123}

181// Creates the directory if missing; the result always ends in a separator.
182auto QtPassSettings::getPassStore(const QString &defaultValue) -> QString {
183 QString returnValue = getInstance()
184 ->value(SettingsConstants::passStore, defaultValue)
185 .toString();
186
187 returnValue = QDir(returnValue).absolutePath();
188
189 // ensure directory exists if never used pass or misconfigured.
190 // otherwise process->setWorkingDirectory(passStore); will fail on execution.
191 if (!QDir(returnValue).exists()) {
192 if (!Util::mkpathForStore(returnValue)) {
193 qWarning() << "Failed to create password store directory:" << returnValue;
194 }
195 }
196
197 if (!returnValue.endsWith("/") && !returnValue.endsWith(QDir::separator())) {
198 returnValue += QDir::separator();
199 }
200
201 return returnValue;
202}

206void QtPassSettings::initExecutables() {
207 AppSettings s = QtPassSettings::load();
208 if (s.passExecutable.isEmpty())
209 s.passExecutable = Util::findBinaryInPath("pass");
210 if (s.gitExecutable.isEmpty())
211 s.gitExecutable = Util::findBinaryInPath("git");
212 if (s.gpgExecutable.isEmpty()) {
213 s.gpgExecutable = Util::findBinaryInPath("gpg2");
214 if (s.gpgExecutable.isEmpty())
215 s.gpgExecutable = Util::findBinaryInPath("gpg");
216 }
217 if (s.pwgenExecutable.isEmpty())
218 s.pwgenExecutable = Util::findBinaryInPath("pwgen");
219 QtPassSettings::save(s);
220}

232// With a true default and nothing stored, a store with a .git dir means Git.
233auto QtPassSettings::isUseGit(const bool &defaultValue) -> bool {
234 bool storedValue =
235 getInstance()->value(SettingsConstants::useGit, defaultValue).toBool();
236 if (storedValue == defaultValue && defaultValue) {
237 QString passStore = getPassStore();
238 if (QFileInfo(passStore).isDir() &&
239 QFileInfo(passStore + QDir::separator() + ".git").isDir()) {
240 return true;
241 }
242 }
243 return storedValue;
244}
