35auto readTemplateFile(const QString &path) -> std::optional<QString> {
36 QFile file;
37 if (!Util::openRegularFile(path, file)) {
38 return std::nullopt;
39 }
40 const QByteArray bytes = file.read(kMaxTemplateFileBytes + 1);
41 if (bytes.size() > kMaxTemplateFileBytes) {
42 qWarning() << "Ignoring" << path << "; a template file is at most"
43 << kMaxTemplateFileBytes << "bytes";
44 return std::nullopt;
45 }
46 // QTextStream skipped a UTF-8 byte order mark (older Notepad writes one);
47 // fromUtf8() keeps it as U+FEFF, which trimmed() leaves in place.
48 const QByteArray bom("\xEF\xBB\xBF");
49 return QString::fromUtf8(bytes.startsWith(bom) ? bytes.mid(bom.size())
50 : bytes);
51}

55auto sectionHeader(const QString &line) -> std::optional<QString> {
56 if (!line.startsWith('[') || !line.endsWith(']')) {
57 return std::nullopt;
58 }
59 return line.mid(1, line.length() - 2).trimmed();
60}

75auto TemplateIO::readTemplates(const QString &storePath)
76 -> QHash<QString, QStringList> {
77 QHash<QString, QStringList> result;
78 std::optional<QString> contents =
79 readTemplateFile(QDir(storePath).filePath(".templates"));
80 if (!contents) {
81 return result;
82 }
83 QTextStream in(&*contents, QIODevice::ReadOnly);
84 // No section before the first header, nor for an empty `[]`: fields there
85 // belong to nothing and are dropped.
86 std::optional<QString> section;
87 QStringList fields;
88 const auto flush = [&] {
89 if (section) {
90 result.insert(*section, fields);
91 }
92 fields.clear();
93 };
94 while (!in.atEnd()) {
95 const QString line = in.readLine().trimmed();
96 if (const std::optional<QString> header = sectionHeader(line)) {
97 flush();
98 section = header->isEmpty() ? std::nullopt : header;
99 if (!section) {
100 qWarning()
101 << "Empty template section in .templates file, ignoring fields";
102 }
103 } else if (section && !line.isEmpty() && !line.startsWith('#')) {
104 fields.append(line);
105 }
106 }
107 flush();
108 return result;
109}

111auto TemplateIO::secretFieldsIn(const QString &templateText) -> QStringList {
112 static const QRegularExpression marker(
113 QStringLiteral("^#\\s*secret\\s*:(.*)$"),
114 QRegularExpression::CaseInsensitiveOption);
115 QStringList names;
116 const QStringList lines = templateText.split(QLatin1Char('\n'));
117 for (const QString &line : lines) {
118 const QRegularExpressionMatch match = marker.match(line.trimmed());
119 if (!match.hasMatch()) {
120 continue;
121 }
122 const QStringList listed = match.captured(1).split(QLatin1Char(','));
123 for (const QString &name : listed) {
124 const QString trimmed = name.trimmed();
125 if (!trimmed.isEmpty()) {
126 names << trimmed;
127 }
128 }
129 }
130 return names;
131}

147auto TemplateIO::writeTemplates(const QString &storePath,
148 const QHash<QString, QStringList> &templates)
149 -> bool {
150 QSaveFile saveFile(QDir(storePath).filePath(".templates"));
151 if (!saveFile.open(QIODevice::WriteOnly | QIODevice::Text)) {
152 return false;
153 }
154 QTextStream out(&saveFile);
155 out.setEncoding(QStringConverter::Utf8);
156 out << "# QtPass templates configuration\n";
157 out << "# Format: INI-style with [template_name] sections,\n";
158 out << "# followed by field names (one per line)\n\n";
159
160 QStringList sortedKeys = templates.keys();
161 std::sort(sortedKeys.begin(), sortedKeys.end());
162 for (const QString &key : sortedKeys) {
163 out << "[" << key << "]\n";
164 for (const QString &field : templates.value(key)) {
165 out << field << "\n";
166 }
167 out << "\n";
168 }
169 out.flush();
170 if (out.status() != QTextStream::Ok) {
171 return false;
172 }
173 return saveFile.commit();
174}

176auto TemplateIO::getFolderTemplate(const QString &folderPath,
177 const QString &storePath) -> QString {
178 const QString store = QDir::cleanPath(QDir(storePath).absolutePath());
179 // Only the store root and its descendants are probed for
180 // .default_template; a path outside the store opens nothing.
181 const auto inStore = [&store](const QString &path) {
182 return path == store ||
183 (path.startsWith(store) && path.length() > store.length() &&
184 path.at(store.length()) == QChar('/'));
185 };
186 QDir dir(folderPath);
187 while (inStore(QDir::cleanPath(dir.absolutePath()))) {
188 const QString name = defaultTemplateIn(dir);
189 if (!name.isEmpty()) {
190 return name;
191 }
192 if (QDir::cleanPath(dir.absolutePath()) == store || !dir.cdUp()) {
193 break;
194 }
195 }
196 return {};
197}
