158void PasswordDialog::save() {
159 // Show is asynchronous: refuse to save an existing entry before its
160 // content loads. getPassword() always returns at least a newline, so an
161 // isEmpty() check cannot stand in for this.
162 if (!m_isNew && !m_contentLoaded) {
163 return;
164 }
165
166 // Canonicalise before serialising: getPassword() stays a dumb serializer.
167 normalizeOtpField();
168
169 QString newValue = getPassword();
170 if (newValue.right(1) != "\n") {
171 newValue += "\n";
172 }
173
174 setSaving(true);
175 // A refusal (the name is taken, the .gpg-id is unusable, a linked path)
176 // comes through critical() before Insert() returns, with nothing started
177 // that could still report.
178 QString refusal;
179 const QMetaObject::Connection refused =
180 connect(m_pass, &Pass::critical, this,
181 [&refusal](const QString &, const QString &message) {
182 refusal = message;
183 });
184 // Reports from inserts already queued (an earlier dialog that a hanging
185 // queue let close while its save was still pending) must pass before one is
186 // taken as this dialog's own. The count is read before our own Insert()
187 // queues, so it does not include it.
188 m_insertsAhead = m_pass->insertsPending();
189 m_pass->Insert(m_file, newValue, !m_isNew);
190 disconnect(refused);
191 if (m_saving && !refusal.isEmpty()) {
192 setSaving(false);
193 showNotSaved(refusal);
194 }
195}

210auto PasswordDialog::isOwnInsertReport() -> bool {
211 if (!m_saving) {
212 return false;
213 }
214 // Since a hanging queue can let a dialog close while its insert is still
215 // queued (#1965), more than one insert can be in flight, and neither
216 // finishedInsert nor insertFailed says which it is. The Executor runs the
217 // queue in order, so the first m_insertsAhead reports belong to inserts
218 // queued before this dialog's; let them pass before taking one as our own.
219 if (m_insertsAhead > 0) {
220 --m_insertsAhead;
221 return false;
222 }
223 return true;
224}

234void PasswordDialog::onInsertFailed(const QString &err, bool written) {
235 if (!isOwnInsertReport()) {
236 return;
237 }
238 setSaving(false);
239 // The entry is in the store and a later step (git add or commit) failed,
240 // which the backend reports ("Saved, not committed"). Saved is saved, so
241 // the dialog closes, also for an existing entry: "Not saved" would be
242 // wrong, and a new entry could only be tried again under a name it now
243 // has.
244 if (written) {
245 QDialog::accept();
246 return;
247 }
248 showNotSaved(err);
249}

251void PasswordDialog::reject() {
252 if (m_saving) {
253 const QMessageBox::StandardButton answer = QMessageBox::question(
254 this, tr("Still saving"),
255 tr("The entry is still being saved. Close the dialog anyway? If "
256 "saving then fails, what you typed is lost."),
257 QMessageBox::Yes | QMessageBox::No, QMessageBox::No);
258 // The insert may have reported while the question was up.
259 if (answer != QMessageBox::Yes || !m_saving) {

263 m_saving = false;
264 }
265 QDialog::reject();

306auto PasswordDialog::resolveNewEntry(QString *problem) const -> QString {
307 QString name = QDir::fromNativeSeparators(ui->nameEdit->text().trimmed());
308 // The suffix belongs to the file, not the entry: Insert() appends it, so a
309 // typed "github.gpg" would otherwise become github.gpg.gpg.
310 if (name.endsWith(QStringLiteral(".gpg"), Qt::CaseInsensitive)) {
311 name.chop(4);
312 }
313 if (name.isEmpty()) {
314 *problem = tr("Give the entry a name.");
315 return {};
316 }
317 if (name.endsWith(u'/')) {
318 *problem = tr("A name cannot end in /.");
319 return {};
320 }
321 const QString folder = ui->folderBox->currentData().toString();
322 const QString rel =
323 QDir::cleanPath(folder.isEmpty() ? name : folder + u'/' + name);
324 const QString absolute = QDir(m_storeRoot).filePath(rel);
325 if (rel.isEmpty() || rel == QStringLiteral(".") ||
326 !PathValidator::isPathInStore(m_storeRoot, absolute)) {
327 *problem = tr("That name would resolve outside the password store.");
328 return {};
329 }
330 // The tree, like `pass ls`, hides names that start with a dot: such an
331 // entry would be saved but never listed.
332 const QStringList parts = rel.split(u'/');
333 if (std::any_of(parts.cbegin(), parts.cend(),
334 [](const QString &part) { return part.startsWith(u'.'); })) {
335 *problem = tr("A name or folder that starts with a dot would be hidden "
336 "from the list.");
337 return {};
338 }
339 // Checked here, not only by Insert(): accept() creates the entry's folder
340 // first, and behind a link that folder is outside the store.
341 if (Util::isUnderLink(absolute, m_storeRoot)) {
342 *problem = tr("%1 lies behind a symbolic link or junction, outside the "
343 "password store.")
344 .arg(rel);
345 return {};
346 }
347 if (QFileInfo::exists(absolute + QStringLiteral(".gpg"))) {
348 *problem = tr("An entry called %1 already exists.").arg(rel);
349 return {};
350 }
351 if (QFileInfo(absolute).isDir()) {
352 *problem = tr("%1 is a folder.").arg(rel);
353 return {};
354 }
355 problem->clear();
356 return rel;
357}

367void PasswordDialog::accept() {
368 if (m_saving) {
369 return;
370 }
371 if (!m_storeRoot.isEmpty()) {
372 QString problem;
373 const QString rel = resolveNewEntry(&problem);
374 if (rel.isEmpty()) {
375 ui->statusLabel->setText(problem);
376 return;
377 }
378 // A name like work/vpn may need its folder; gpg does not create it.
379 const QString parent = QFileInfo(QDir(m_storeRoot).filePath(rel)).path();
380 if (!QDir(parent).exists() && !Util::mkpathForStore(parent)) {
381 ui->statusLabel->setText(
382 tr("Could not create the folder %1.").arg(parent));
383 return;
384 }
385 m_file = rel;
386 setWindowTitle(tr("Password") + QStringLiteral(" ") + m_file);
387 }
388 save();
389}

391void PasswordDialog::setPassword(const QString &password) {
392 // Which lines become fields is the user's choice: forcing allFields
393 // (#1138, for #132) made every `key: value` line a label-locked widget,
394 // even with templates off (#1766). MainWindow renders by the same rule.
395 FileContent fileContent =
396 FileContent::parse(password, m_templating ? m_fields : QStringList(),
397 m_templating && m_allFields);
398 ui->lineEditPassword->setText(fileContent.getPassword());
399
400 QWidget *previous = ui->checkBoxShow;
401 NamedValues namedValues = fileContent.getNamedValues();
402 for (QLineEdit *line : std::as_const(m_templateLines)) {
403 const NamedValue nv = namedValues.takeNamedValue(line->objectName());
404 line->setText(nv.value);
405 line->setProperty(kOriginalLineProperty, nv.originalLine);
406 // The value is the entry's again, whatever the user typed before this
407 // reload: not a value normalizeOtpField() may rewrite.
408 line->setProperty(kOtpEditedProperty, QVariant());
409 previous = line;
410 }
411 // Remove previously created dynamic widgets to prevent duplicates and leaks
412 for (QLineEdit *line : std::as_const(m_otherLines)) {
413 ui->formLayout->removeRow(line);
414 }
415 m_otherLines.clear();
416 for (const NamedValue &nv : std::as_const(namedValues)) {
417 auto *line = new SecretLineEdit();
418 reportCopies(line);
419 line->setObjectName(nv.name);
420 line->setText(nv.value);
421 line->setProperty(kOriginalLineProperty, nv.originalLine);
422 auto *label = new FieldLabel(nv.name);
423 label->setBuddy(line);
424 ui->formLayout->addRow(label, line);
425 setTabOrder(previous, line);
426 m_otherLines.append(line);
427 previous = line;
428 // Both widgets are rows of the form; removeRow() deletes them together,
429 // so a rename or removal that arrives afterwards finds no receiver.
430 connect(label, &FieldLabel::renamed, this,
431 [this, line, label](const QString &, const QString &to) {
432 renameField(line, label, to);
433 });
434 connect(label, &FieldLabel::removeRequested, this,
435 [this, line] { removeField(line); });
436 // The visible way to remove the field; the context menu is the other.
437 auto *remove = line->addAction(
438 QIcon::fromTheme(QStringLiteral("edit-delete"),
439 QIcon(QStringLiteral(":/icons/edit-delete.svg"))),
440 QLineEdit::TrailingPosition);
441 remove->setObjectName(QStringLiteral("removeFieldAction"));
442 remove->setToolTip(tr("Remove field"));
443 connect(remove, &QAction::triggered, this,
444 [this, line] { removeField(line); });
445 }
446
447 // setPlainText, not insertPlainText: re-populating must replace the body,
448 // not save back a duplicated copy.
449 ui->plainTextEdit->setPlainText(fileContent.getRemainingData());
450
451 // m_otherLines was just rebuilt, so the OTP field may be a new widget.
452 hookOtpField();
453}

455void PasswordDialog::renameField(QLineEdit *line, FieldLabel *label,
456 const QString &to) {
457 QList<QLineEdit *> allLines(m_templateLines);
458 allLines.append(m_otherLines);
459 for (QLineEdit *other : std::as_const(allLines)) {
460 if (other != line && other->objectName() == to) {
461 ui->statusLabel->setText(tr("A field called %1 already exists.").arg(to));
462 return;
463 }
464 }
465 ui->statusLabel->clear();
466 line->setObjectName(to);
467 label->setText(to);
468 // The field may have become, or stopped being, the OTP one.
469 hookOtpField();
470}

472void PasswordDialog::removeField(QLineEdit *line) {
473 m_otherLines.removeAll(line);
474 // takeRow(), not removeRow(): removeRow() deletes the widgets while the
475 // label's contextMenuEvent() or the action's triggered() is still on the
476 // stack. They are deleted once the stack has unwound.
477 const QFormLayout::TakeRowResult row = ui->formLayout->takeRow(line);
478 // hide() takes the focus from a focused line, and that emits
479 // editingFinished(): not into normalizeOtpField() for a field that is
480 // being removed.
481 disconnect(line, nullptr, this, nullptr);
482 if (m_otpLine == line) {
483 m_otpLine = nullptr;
484 }
485 for (QLayoutItem *item : {row.labelItem, row.fieldItem}) {
486 if (item == nullptr) {
487 continue;
488 }
489 if (QWidget *widget = item->widget()) {
490 widget->hide();
491 widget->deleteLater();
492 }
493 delete item;
494 }
495 hookOtpField();
496}

498auto PasswordDialog::otpLineEdit() const -> QLineEdit * {
499 QList<QLineEdit *> allLines(m_templateLines);
500 allLines.append(m_otherLines);
501 QLineEdit *firstMatch = nullptr;
502 for (QLineEdit *line : std::as_const(allLines)) {
503 if (!FileContent::isOtpFieldName(line->objectName())) {
504 continue;
505 }
506 // Prefer a populated field: the default template's empty `OTP` widget
507 // would otherwise shadow a `TOTP:` value in m_otherLines, leaving the
508 // real secret unvalidated and unnormalised.
509 if (!line->text().trimmed().isEmpty()) {
510 return line;
511 }
512 if (firstMatch == nullptr) {
513 firstMatch = line;
514 }
515 }
516 return firstMatch;
517}

519void PasswordDialog::hookOtpField() {
520 // Only m_otherLines widgets are deleted by removeRow(); a template widget
521 // keeps its action, and the next validate would add a second icon.
522 // m_otpWarning is a QPointer: null when setPassword() deleted its owner.
523 if (!m_otpWarning.isNull()) {
524 if (auto *owner = qobject_cast<QWidget *>(m_otpWarning->parent())) {
525 owner->removeAction(m_otpWarning);
526 }
527 delete m_otpWarning;
528 }
529 m_otpWarning = nullptr;
530
531 QLineEdit *otp = otpLineEdit();
532 // The field that was the OTP one may not be any more (renamed away, or
533 // another one now comes first): it stops being validated and normalised
534 // here, so its OTP hooks go with the role. Its copies are still reported.
535 if (!m_otpLine.isNull() && m_otpLine != otp) {
536 disconnect(m_otpLine, &QLineEdit::textChanged, this,
537 &PasswordDialog::validateOtpField);
538 disconnect(m_otpLine, &QLineEdit::textEdited, this,
539 &PasswordDialog::markOtpFieldEdited);
540 disconnect(m_otpLine, &QLineEdit::editingFinished, this,
541 &PasswordDialog::normalizeOtpField);
542 }
543 m_otpLine = otp;
544 if (otp == nullptr) {
545 return;
546 }
547 otp->setPlaceholderText(tr("otpauth:// URI or base32 secret"));
548 connect(otp, &QLineEdit::textChanged, this, &PasswordDialog::validateOtpField,
549 Qt::UniqueConnection);
550 // textEdited, not textChanged: it fires only for user input, so programmatic
551 // population does not count as the user having touched the field.
552 connect(otp, &QLineEdit::textEdited, this,
553 &PasswordDialog::markOtpFieldEdited, Qt::UniqueConnection);
554 // Canonicalise as soon as the user leaves the field, so the value they end
555 // up saving is the value they were shown.
556 connect(otp, &QLineEdit::editingFinished, this,
557 &PasswordDialog::normalizeOtpField, Qt::UniqueConnection);
558 validateOtpField();
559}

567void PasswordDialog::validateOtpField() {
568 // m_otpLine, not otpLineEdit(): the signals that bring us here and the
569 // edit mark are that line's. The OTP role is settled when fields change.
570 QLineEdit *otp = m_otpLine;
571 if (otp == nullptr) {
572 return;
573 }
574
575 const QString text = otp->text().trimmed();
576 const bool bad = !text.isEmpty() && !Totp::isValid(text);
577 if (bad) {
578 if (m_otpWarning.isNull()) {
579 // Theme-aware indicator: no hardcoded colours and no extra layout row.
580 m_otpWarning =
581 otp->addAction(QIcon::fromTheme(QStringLiteral("dialog-warning")),
582 QLineEdit::TrailingPosition);
583 }
584 otp->setToolTip(tr("Invalid OTP secret"));
585 } else {
586 if (!m_otpWarning.isNull()) {
587 otp->removeAction(m_otpWarning);
588 delete m_otpWarning;
589 m_otpWarning = nullptr;
590 }
591 otp->setToolTip(QString());
592 }
593}

595// Only an otpauth:// URI or a value typed into this field is rewritten.
596// Base32::sanitizeInput() maps 1 to L and 8 to B, so a backup code like
597// `12345678` passes as base32 and was silently rewritten on any OK. A bare
598// secret left alone still works: Totp::parse() accepts one.
599void PasswordDialog::normalizeOtpField() {
600 QLineEdit *otp = m_otpLine;
601 if (otp == nullptr) {
602 return;
603 }
604 const QString text = otp->text().trimmed();
605 if (text.isEmpty()) {
606 return;
607 }
608 if (!otp->property(kOtpEditedProperty).toBool() &&
609 !FileContent::isOtpUriValue(text)) {
610 return;
611 }
612 // fileName(), not completeBaseName(): the latter truncates at the last dot,
613 // turning an entry called github.com into a label of "github". m_file already
614 // has its .gpg suffix stripped by MainWindow::getFile().
615 const QString canonical = Totp::normalize(text, QFileInfo(m_file).fileName());
616 if (!canonical.isEmpty()) {
617 otp->setText(canonical);
618 }
619}

621auto PasswordDialog::getPassword() -> QString {
622 QString passFile = ui->lineEditPassword->text() + "\n";
623 QList<QLineEdit *> allLines(m_templateLines);
624 allLines.append(m_otherLines);
625 for (QLineEdit *line : std::as_const(allLines)) {
626 const QString text = line->text();
627 const QString name = line->objectName();
628 const QString original = line->property(kOriginalLineProperty).toString();
629 // A field whose name and value are still exactly what was parsed keeps its
630 // source line verbatim: reserialising would turn `key:value` into
631 // `key: value`, trim whitespace the value carried, and drop a field with an
632 // empty value (`recovery:`) entirely. The widget holds trimmed name/value,
633 // so compare against the original line split and trimmed the same way.
634 if (!original.isEmpty()) {
635 const qsizetype colon = original.indexOf(QLatin1Char(':'));
636 if (colon >= 0 && original.left(colon).trimmed() == name &&
637 original.right(original.length() - colon - 1).trimmed() == text) {
638 passFile += original + "\n";
639 continue;
640 }
641 }
642 if (text.isEmpty()) {
643 continue;
644 }
645 passFile += name + ": " + text + "\n";
646 }
647 // Not toPlainText(): it saves a blank for every no-break space (U+00A0) in
648 // the notes, so any OK rewrites even lines the user never touched and a
649 // passphrase or recovery code with no-break spaces is corrupted. The raw
650 // document text keeps the no-break spaces; only the block and line
651 // separators it uses are normalised back to '\n'.
652 QString notes = ui->plainTextEdit->document()->toRawText();
653 notes.replace(QChar(QChar::ParagraphSeparator), QLatin1Char('\n'));
654 notes.replace(QChar(QChar::LineSeparator), QLatin1Char('\n'));
655 passFile += notes;
656 return passFile;
657}

777void PasswordDialog::cycleTemplate() {
778 if (m_availableTemplates.isEmpty()) {
779 return;
780 }
781 QStringList names = m_availableTemplates.keys();
782 std::sort(names.begin(), names.end());
783
784 qsizetype currentIdx = names.indexOf(m_currentTemplateName);
785 qsizetype nextIdx;
786 if (currentIdx < 0) {
787 nextIdx = 0;
788 } else {
789 nextIdx = (currentIdx + 1) % names.size();
790 }
791 applyTemplate(names.at(nextIdx));
792}

794void PasswordDialog::setPass(const QString &output, const QString &file) {
795 // finishedShow names its file: a decrypt of any other entry (a tree click
796 // that was still queued when the dialog opened) is not ours.
797 if (!file.isEmpty() && file != m_file) {
798 return;
799 }
800 // An editor block per line: an entry of millions of lines (a small,
801 // compressed one in a shared store can be one) would take QtPass down.
802 // Not loaded at all, so nothing can be saved over it from here either.
803 // Counted as an editor of the whole entry would hold it: the final
804 // newline pass and QtPass write ends the last line, it starts no new one.
805 // Password and fields count too, so this is stricter than the display's
806 // limit on the remaining text alone.
807 if (const qsizetype lines =
808 output.count(QLatin1Char('\n')) +
809 (output.isEmpty() || output.endsWith(QLatin1Char('\n')) ? 0 : 1);
810 lines > FileContent::kMaxDisplayLines) {
811 // The count in a variable first, as in FileContent: older lupdate
812 // versions lose a tr() with template arguments among its arguments.
813 const int n = static_cast<int>(
814 std::min<qsizetype>(lines, std::numeric_limits<int>::max()));
815 ui->statusLabel->setText(
816 tr("This entry has %n line(s), too many to edit here.", nullptr, n));
817 return;
818 }
819 setPassword(output);
820 m_contentLoaded = true;
821 // Drop the "Decrypting…" note so it cannot be mistaken for a warning.
822 ui->statusLabel->clear();
823 setEditorEnabled(true);
824 ui->buttonBox->button(QDialogButtonBox::Ok)->setEnabled(true);
825}
