125auto normalisedKeyId(const QString &id) -> QString {
126 const QString trimmed = id.trimmed();
127 return (trimmed.startsWith(QLatin1String("0x"), Qt::CaseInsensitive)
128 ? trimmed.mid(2)
129 : trimmed)
130 .toUpper();
131}
132
139auto answersTo(const UserInfo &key, const QString &recipient) -> bool {
140 static const QRegularExpression hexId(
141 QStringLiteral("^(0x)?[0-9A-Fa-f]{8,}$"));
142 if (hexId.match(recipient.trimmed()).hasMatch()) {
143 const QString id = normalisedKeyId(recipient);
144 const QString keyId = normalisedKeyId(key.key_id);
145 return keyId.endsWith(id) || (keyId.size() >= 8 && id.endsWith(keyId));
146 }
147 QString term = recipient.trimmed();
148 if (term.startsWith(u'=') || term.startsWith(u'@') || term.startsWith(u'*')) {
149 term.remove(0, 1);
150 } else if (term.startsWith(u'<') && term.endsWith(u'>')) {
151 term = term.mid(1, term.size() - 2);
152 }
153 return !term.isEmpty() && key.name.contains(term, Qt::CaseInsensitive);
154}

158void UsersDialog::selectRecipients(const QStringList &recipients) {
159 // Each per-line gpg lookup is a blocking call on the dialog's way up, so a
160 // store with many keys missing from the keyring (a fresh clone, a team
161 // store) is capped at this many across both the fallback below and the
162 // missing-key check; the rest resolve by the batched call or stay unchecked.
163 constexpr int kMaxLookups = 10;
164 int lookups = 0;
165 // Remembers the per-line lookups the fallback already ran, so the
166 // missing-key loop below reuses them instead of asking gpg a second time.
167 QHash<QString, bool> perLineHasKey;
168
169 // One gpg call resolves every recipient, by key ID or by email/UID.
170 QList<UserInfo> resolved = m_pass->listKeys(recipients);
171 if (resolved.isEmpty() && recipients.size() > 1) {
172 // One line gpg cannot read as a key search ("0x", a fingerprint one
173 // digit short) aborts that whole call with exit code 2, so nothing is
174 // listed for the other lines, whose keys are in the keyring all the same.
175 // Asked for one by one they still resolve; only the unreadable line is
176 // reported missing by the loop below, as an unknown key already is.
177 for (const QString &recipient : recipients) {
178 if (lookups++ >= kMaxLookups) {
179 break;
180 }
181 const QList<UserInfo> keys = m_pass->listKeys(recipient);
182 perLineHasKey.insert(recipient, !keys.isEmpty());
183 resolved += keys;
184 }
185 }
186 for (UserInfo &user : m_userList) {
187 if (std::any_of(resolved.cbegin(), resolved.cend(),
188 [&user](const UserInfo &key) {
189 return key.key_id == user.key_id;
190 })) {
191 user.enabled = true;
192 }
193 }
194 // Only what gpg itself cannot find is reported missing. The fallback's
195 // per-line answer, when it has one, is gpg's own verdict for that line and
196 // decides it outright. Otherwise the heuristic match is a fast path (a
197 // secondary UID or a subkey ID resolves in gpg without showing in what it
198 // lists), and a per-line gpg call settles anything it does not catch. Taking
199 // the cached answer first also keeps a genuinely unreadable line ("0x")
200 // reported missing even when a resolved key's UID happens to contain it, so
201 // the heuristic answersTo() would otherwise match.
202 for (const QString &recipient : recipients) {
203 bool found = false;
204 const auto cached = perLineHasKey.constFind(recipient);
205 if (cached != perLineHasKey.constEnd()) {
206 found = cached.value();
207 } else {
208 found = std::any_of(resolved.cbegin(), resolved.cend(),
209 [&recipient](const UserInfo &key) {
210 return answersTo(key, recipient);
211 });
212 if (!found && lookups++ < kMaxLookups) {
213 found = !m_pass->listKeys(recipient).isEmpty();
214 }
215 }
216 if (!found) {
217 UserInfo missing;
218 missing.enabled = true;
219 missing.key_id = recipient;
220 missing.name = " ?? " + tr("Key not found in keyring");
221 m_userList.append(missing);
222 }
223 }
224}

226void UsersDialog::loadRecipients() {
227 // A store that had no list when it was chosen (a new profile): a list
228 // that appeared since is nobody's choice, and was read under the active
229 // backend's signing policy, not the new profile's.
230 if (m_preselect == Preselect::Nothing) {
231 return;
232 }
233 // Through the backend: with a signing key only a verified list may be
234 // preselected, since OK signs whatever is selected.
235 const Pass::RecipientsForEditing loaded =
236 m_pass->recipientsForEditing(m_dir, m_passStore);
237 if (!loaded.warning.isEmpty()) {
238 showRecipientWarning(loaded.warning);
239 }
240 // A folder without .gpg-id (new store, new profile) has no recipients yet;
241 // listKeys() with no filter would return the whole keyring, preselected.
242 if (loaded.state != Pass::RecipientsForEditing::State::Rejected &&
243 !loaded.recipients.isEmpty()) {
244 selectRecipients(loaded.recipients);
245 }
246}

305void UsersDialog::populateList(const QString &filter) {
306 // Invalidate cached datetime so expiry checks use fresh current time
307 m_cachedDateTimeValid = false;
308
309 QString patternString = "*" + filter + "*";
310 if (m_cachedPatternString != patternString) {
311 QRegularExpression re(
312 QRegularExpression::wildcardToRegularExpression(patternString),
313 QRegularExpression::CaseInsensitiveOption);
314 if (re.isValid()) {
315 m_cachedNameFilter = re;
316 m_cachedPatternString = patternString;
317 }
318 }
319 const QRegularExpression &nameFilter = m_cachedNameFilter;
320 ui->listWidget->clear();
321
322 for (int i = 0; i < m_userList.size(); ++i) {
323 const auto &user = m_userList.at(i);
324 if (!passesFilter(user, filter, nameFilter)) {
325 continue;
326 }
327
328 auto *item = new QListWidgetItem(buildUserText(user), ui->listWidget);
329 applyUserStyling(item, user);
330 item->setCheckState(user.enabled ? Qt::Checked : Qt::Unchecked);
331 item->setData(Qt::UserRole, QVariant::fromValue(i));
332 ui->listWidget->addItem(item);
333 }
334 updateOkButton();
335}

337bool UsersDialog::passesFilter(const UserInfo &user, const QString &filter,
338 const QRegularExpression &nameFilter) const {
339 if (!filter.isEmpty() && !nameFilter.match(user.name).hasMatch()) {
340 return false;
341 }
342 // A ticked key is one the folder is encrypted to: it is shown whatever
343 // its state, or an expired or missing recipient stays in .gpg-id unseen and
344 // cannot be unticked. "Show unusable keys" is about keys to add.
345 if (user.enabled || ui->checkBox->isChecked()) {
346 return true;
347 }
348 return user.isValid() && !isUserExpired(user);
349}

351auto UsersDialog::isUserExpired(const UserInfo &user) const -> bool {
352 if (!m_cachedDateTimeValid) {
353 m_cachedCurrentDateTime = QDateTime::currentDateTime();
354 m_cachedDateTimeValid = true;
355 }
356 // gpg marks a key it already judged expired with validity 'e'.
357 return user.validity == 'e' ||
358 (user.expiry.isValid() && m_cachedCurrentDateTime > user.expiry);
359}

374void UsersDialog::applyUserStyling(QListWidgetItem *item,
375 const UserInfo &user) const {
376 const QString originalText = item->text();
377 // Status badge first: an own key can be expired too, and then the marker
378 // matters more than the "you can decrypt with this" colour.
379 bool badged = true;
380 // Expired before invalid: gpg gives an expired key validity 'e', which is
381 // not a valid one, and "expired" says what to do about it.
382 if (isUserExpired(user)) {
383 // A lone dark-red foreground is unreadable on dark themes, and gpg
384 // refuses to encrypt to expired keys anyway.
385 item->setBackground(Qt::darkRed);
386 item->setForeground(Qt::white);
387 item->setText(tr("[EXPIRED] ") + originalText);
388 } else if (!user.isValid()) {
389 item->setBackground(Qt::darkRed);
390 item->setForeground(Qt::white);
391 item->setText(tr("[INVALID] ") + originalText);
392 } else if (!user.fullyValid()) {
393 item->setBackground(Qt::darkYellow);
394 item->setForeground(Qt::white);
395 item->setText(tr("[PARTIAL] ") + originalText);
396 } else {
397 item->setText(originalText);
398 badged = false;
399 }
400 if (user.have_secret) {
401 if (!badged) {
402 item->setForeground(QApplication::palette().color(QPalette::Link));
403 }
404 QFont font = item->font();
405 font.setBold(true);
406 item->setFont(font);
407 }
408}

416void UsersDialog::selectImportedKey(const QString &importedKey) {
417 // Suffixes shorter than 16 characters would give false positives.
418 const auto sameKey = [&importedKey](const QString &keyId) {
419 return (keyId.length() >= 16 &&
420 importedKey.endsWith(keyId, Qt::CaseInsensitive)) ||
421 (importedKey.length() >= 16 &&
422 keyId.endsWith(importedKey, Qt::CaseInsensitive));
423 };
424 // The stored key_id, not the item text.
425 for (int i = 0; i < ui->listWidget->count(); ++i) {
426 QListWidgetItem *item = ui->listWidget->item(i);
427 bool ok = false;
428 const int idx = item != nullptr ? item->data(Qt::UserRole).toInt(&ok) : -1;
429 if (ok && idx >= 0 && idx < m_userList.size() &&
430 !m_userList[idx].key_id.isEmpty() && sameKey(m_userList[idx].key_id)) {
431 ui->listWidget->setCurrentItem(item);
432 return;
433 }
434 }
435}

437void UsersDialog::on_importKeyButton_clicked() {
438 ImportKeyDialog dialog(m_gpgExe, this);
439 if (dialog.exec() != QDialog::Accepted) {
440 return;
441 }
442
443 // Non-empty after Accepted, but an empty id would make the endsWith()
444 // below match the first listed key.
445 const QString importedKey = dialog.importedKeyId();
446 if (importedKey.isEmpty()) {
447 return;
448 }
449
450 // loadGpgKeys() rebuilds the key list from the keyring and drops every tick.
451 // Remember which recipients were selected (the .gpg-id list and the user's
452 // own ticks, including any "key not found" rows) and re-apply them after, so
453 // importing a key does not silently remove the other recipients.
454 QStringList selected;
455 for (const UserInfo &user : std::as_const(m_userList)) {
456 if (user.enabled) {
457 selected.append(user.key_id);
458 }
459 }
460
461 if (!loadGpgKeys()) {
462 return;
463 }
464
465 // Re-apply the remembered selection. Most saved identifiers are a key ID that
466 // is in the rebuilt list, so match those exactly first (no gpg lookup). An
467 // identifier that does not match a row is one that was listed as "key not
468 // found" - a short ID or email that gpg could not resolve, or whose key has
469 // just been imported and now shows under its fingerprint. Hand those to
470 // selectRecipients(), which resolves each against the refreshed keyring,
471 // ticks the matching row, and re-adds a "key not found" row for any that
472 // still does not resolve.
473 QStringList unresolved;
474 for (const QString &keyId : std::as_const(selected)) {
475 bool matched = false;
476 for (UserInfo &user : m_userList) {
477 if (user.key_id == keyId) {
478 user.enabled = true;
479 matched = true;
480 }
481 }
482 if (!matched) {
483 unresolved.append(keyId);
484 }
485 }
486 if (!unresolved.isEmpty()) {
487 selectRecipients(unresolved);
488 }
489
490 // Clear the filter so the just-imported key is visible. setText("") still
491 // emits textChanged once, so don't double-populate.
492 {
493 const QSignalBlocker blocker(ui->lineEdit);
494 ui->lineEdit->clear();
495 }
496 populateList(QString());
497
498 selectImportedKey(importedKey);
499}
