108void UsersDialog::showRecipientWarning(const QString &text,
109 const QString &name) {
110 // Plain text before the text is set: it can quote the store's recipient
111 // list, which as rich text could make Qt load a file (<img src=...>).
112 auto *banner = new QLabel(this);
113 banner->setTextFormat(Qt::PlainText);
114 // And with nothing in it that changes how the rest reads: a folder name
115 // with a bidi override mirrored the warning after it.
116 banner->setText(Util::displayable(text));
117 banner->setObjectName(name);
118 banner->setWordWrap(true);
119 // Palette, not a stylesheet colour, so it reads on dark themes too; red as

122 bold.setBold(true);
123 banner->setFont(bold);
124 QPalette palette = banner->palette();
125 palette.setColor(QPalette::WindowText, QColor(Qt::red));
126 banner->setPalette(palette);

128 // In a scroll area of a few lines: the text quotes folder paths a
129 // co-writer chose, and a word-wrapped label is as wide as its longest

142void UsersDialog::showNestedLists() {
143 // The walk re-encryption takes: hidden folders and links are not entered.
144 const QString dir = QDir::cleanPath(m_dir);
145 if (!QFileInfo(dir).isDir()) {
146 return;
147 }
148 QStringList unlisted;
149 const QStringList lists = Util::regularFilesUnder(
150 dir, {QStringLiteral(".gpg-id")}, nullptr, true, &unlisted);
151 const auto relative = [&dir](const QString &folder) {
152 return QDir::toNativeSeparators(QDir(dir).relativeFilePath(folder));
153 };
154 QStringList nested;
155 for (const QString &list : lists) {
156 const QString folder = QFileInfo(list).path();
157 if (QDir::cleanPath(folder) != dir) {
158 nested << relative(folder);
159 }
160 }
161 QStringList unread;
162 for (const QString &folder : std::as_const(unlisted)) {
163 unread << relative(folder);
164 }
165 constexpr qsizetype kMaxNamed = 5;
166 const auto named = [](const QStringList &folders) {
167 QString names = folders.mid(0, kMaxNamed).join(QStringLiteral(", "));
168 if (folders.size() > kMaxNamed) {
169 names += QStringLiteral(", …");
170 }
171 return names;
172 };
173 QStringList text;
174 if (!nested.isEmpty()) {
175 //: %1 lists folders, such as "work, work/bank". Shown when changing the
176 //: recipients of a folder that has folders with their own list below it.
177 text << tr("%n folder(s) below this one have their own recipients, which "
178 "saving here does not change: %1",
179 nullptr, static_cast<int>(nested.size()))
180 .arg(named(nested));
181 }
182 if (!unread.isEmpty()) {
183 //: %1 lists folders, such as "work, work/bank", that QtPass could not
184 //: read: no permission, or a name in them that is not valid UTF-8.
185 text << tr("%n folder(s) below this one could not be read, so whether "
186 "they have their own recipients is unknown: %1",
187 nullptr, static_cast<int>(unread.size()))
188 .arg(named(unread));
189 }
190 if (!text.isEmpty()) {
191 showRecipientWarning(text.join(QLatin1Char('\n')),
192 QStringLiteral("nestedListsWarning"));
193 }
194}

199auto normalisedKeyId(const QString &id) -> QString {
200 const QString trimmed = id.trimmed();
201 return (trimmed.startsWith(QLatin1String("0x"), Qt::CaseInsensitive)
202 ? trimmed.mid(2)
203 : trimmed)
204 .toUpper();
205}
206
210auto keyIdsMatch(const QString &first, const QString &second) -> bool {
211 static const QRegularExpression hexId(QStringLiteral("^[0-9A-F]{8,}$"));
212 const QString a = normalisedKeyId(first);
213 const QString b = normalisedKeyId(second);
214 if (!hexId.match(a).hasMatch() || !hexId.match(b).hasMatch()) {
215 return false;
216 }
217 if (a.size() == 64) {
218 return b == a || (b.size() == 16 && a.startsWith(b)) ||
219 (b.size() == 8 && a.mid(8, 8) == b);
220 }
221 if (b.size() == 64) {
222 return a == b || (a.size() == 16 && b.startsWith(a)) ||
223 (a.size() == 8 && b.mid(8, 8) == a);
224 }
225 return a.endsWith(b) || b.endsWith(a);
226}
227
231auto keyIdOfLine(const QString &recipient) -> QString {
232 static const QRegularExpression hexLine(
233 QStringLiteral("^(?:0[xX])?([0-9A-Fa-f]+)!?$"));
234 const QRegularExpressionMatch match = hexLine.match(recipient.trimmed());
235 if (!match.hasMatch()) {
236 return {};
237 }
238 const QString digits = match.captured(1);
239 const qsizetype size = digits.size();
240 return size == 8 || size == 16 || size == 40 || size == 64 ? digits
241 : QString();
242}
243
247auto nameTermOfLine(const QString &recipient) -> QString {
248 const QString line = recipient.trimmed();
249 if (line.isEmpty()) {
250 return {};
251 }
252 // A leading 0x, digits only, or groups of four digits split by spaces (a
253 // fingerprint as gpg prints it) is a key ID gpg reads in its own way
254 // (keyIdOfLine() takes the lengths it accepts); ".", "+", "#", "&", "^",
255 // "/", "[" and ":" start search modes this dialog does not model.
256 static const QRegularExpression hexish(QStringLiteral(
257 "^(?:0[xX]|[0-9A-Fa-f]+!?$|[0-9A-Fa-f]{4}(?:\\s+[0-9A-Fa-f]{4})+!?$)"));
258 if (hexish.match(line).hasMatch() ||
259 QStringLiteral(".+#&^/[:!\\").contains(line.front())) {
260 return {};
261 }
262 QString term = line;
263 if (term.startsWith(u'<')) {
264 if (!term.endsWith(u'>')) {
265 return {};
266 }
267 term = term.mid(1, term.size() - 2);
268 } else if (term.startsWith(u'=') || term.startsWith(u'@') ||
269 term.startsWith(u'*')) {
270 term.remove(0, 1);
271 }
272 return term;
273}
274
281auto judgedLocally(const QString &recipient) -> bool {
282 return !keyIdOfLine(recipient).isEmpty() ||
283 !nameTermOfLine(recipient).isEmpty();
284}
285
293auto answersTo(const UserInfo &key, const QString &recipient) -> bool {
294 const QString keyId = keyIdOfLine(recipient);
295 if (!keyId.isEmpty()) {
296 return keyIdsMatch(key.key_id, keyId) ||
297 std::any_of(key.subkeys.cbegin(), key.subkeys.cend(),
298 [&keyId](const QString &subkey) {
299 return keyIdsMatch(subkey, keyId);
300 });
301 }
302 const QString term = nameTermOfLine(recipient);
303 if (term.isEmpty()) {
304 return false;
305 }
306 const auto holds = [&term](const QString &uid) {
307 return uid.contains(term, Qt::CaseInsensitive);
308 };
309 return holds(key.name) ||
310 std::any_of(key.uids.cbegin(), key.uids.cend(), holds);
311}

315void UsersDialog::selectRecipients(const QStringList &listed) {
316 // A line listed twice is one recipient, and one row.
317 QStringList recipients;
318 QSet<QString> seen;
319 for (const QString &line : listed) {
320 if (!seen.contains(line)) {
321 seen.insert(line);
322 recipients.append(line);
323 }
324 }
325
326 // One gpg call resolves every recipient, by key ID or by email/UID.
327 QList<UserInfo> resolved = m_pass->listKeys(recipients);
328 if (resolved.isEmpty() && recipients.size() > 1) {
329 // One line gpg cannot read as a key search ("0x", a fingerprint one
330 // digit short) aborts that whole call with exit code 2, so nothing is
331 // listed for the other lines, whose keys are in the keyring all the same.
332 // Every form judgedLocally() accepts gpg reads, so ask again for those;
333 // the rest get a call of their own below.
334 QStringList readable;
335 std::copy_if(recipients.cbegin(), recipients.cend(),
336 std::back_inserter(readable), judgedLocally);
337 if (!readable.isEmpty() && readable.size() < recipients.size()) {
338 resolved = m_pass->listKeys(readable);
339 }
340 }
341
342 // A line answersTo() cannot judge gets gpg's own verdict, one blocking
343 // call each on the dialog's way up. Only unusual forms need it, so a few
344 // are asked; past that a line is listed unticked as not looked up, so it
345 // does not stay in the list unseen (a member could otherwise hide behind
346 // it a key the user unticks).
347 constexpr int kMaxLookups = 10;
348 int lookups = 0;
349 QHash<QString, bool> perLineHasKey;
350 QSet<QString> notLookedUp;
351 for (const QString &recipient : std::as_const(recipients)) {
352 if (judgedLocally(recipient)) {
353 continue;
354 }
355 if (lookups++ >= kMaxLookups) {
356 notLookedUp.insert(recipient);
357 continue;
358 }
359 const QList<UserInfo> keys = m_pass->listKeys(recipient);
360 perLineHasKey.insert(recipient, !keys.isEmpty());
361 resolved += keys;
362 }
363
364 for (UserInfo &user : m_userList) {
365 if (std::any_of(resolved.cbegin(), resolved.cend(),
366 [&user](const UserInfo &key) {
367 return key.key_id == user.key_id;
368 })) {
369 user.enabled = true;
370 }
371 }
372 // Only what gpg itself cannot find is reported missing. A per-line answer
373 // is gpg's own verdict for that line and decides it outright; that also
374 // keeps a genuinely unreadable line ("0x") reported missing. Any other
375 // line is missing when no key gpg resolved answers to it: the full listing
376 // carries every user ID and subkey, so nothing is left for gpg to add.
377 for (const QString &recipient : std::as_const(recipients)) {
378 UserInfo missing;
379 missing.key_id = recipient;
380 if (notLookedUp.contains(recipient)) {
381 missing.name = " ?? " + tr("Not looked up in keyring");
382 m_notLookedUp.insert(recipient);
383 m_userList.append(missing);
384 continue;
385 }
386 const auto cached = perLineHasKey.constFind(recipient);
387 const bool found = cached != perLineHasKey.constEnd()
388 ? cached.value()
389 : std::any_of(resolved.cbegin(), resolved.cend(),
390 [&recipient](const UserInfo &key) {
391 return answersTo(key, recipient);
392 });
393 if (!found) {
394 missing.enabled = true;
395 missing.name = " ?? " + tr("Key not found in keyring");
396 m_userList.append(missing);
397 }
398 }
399}

401void UsersDialog::loadRecipients() {
402 // A store that had no list when it was chosen (a new profile): a list
403 // that appeared since is nobody's choice, and was read under the active
404 // backend's signing policy, not the new profile's.
405 if (m_preselect == Preselect::Nothing) {
406 return;
407 }
408 // Through the backend: with a signing key only a verified list may be
409 // preselected, since OK signs whatever is selected.
410 const Pass::RecipientsForEditing loaded =
411 m_pass->recipientsForEditing(m_dir, m_passStore);
412 m_listRead = true;
413 m_readState = static_cast<int>(loaded.state);
414 m_readRecipients = loaded.recipients;
415 if (!loaded.warning.isEmpty()) {
416 showRecipientWarning(loaded.warning);
417 }
418 // A folder without .gpg-id (new store, new profile) has no recipients yet;
419 // listKeys() with no filter would return the whole keyring, preselected.
420 if (loaded.state != Pass::RecipientsForEditing::State::Rejected &&
421 !loaded.recipients.isEmpty()) {
422 selectRecipients(loaded.recipients);
423 }
424}

432 now.recipients != m_readRecipients;

435void UsersDialog::accept() {
436 if (!hasSelection()) {
437 // Nothing to encrypt to: an empty .gpg-id would make every insert fail
438 // and the wizard never offers this dialog again once the file exists.
439 return;
440 }
441 // OK writes the whole selection, ticked from the list as it was read. A
442 // newer list that arrived meanwhile (a pull the watchdog let run on under
443 // the dialog, a sync) would be overwritten by the old one plus the user's
444 // change: someone just removed came back, signed as the next generation.
445 if (m_listRead && listChangedSinceRead()) {
446 QMessageBox::warning(
447 this, tr("Recipients not changed"),
448 tr("The recipient list of this folder changed while this dialog was "
449 "open, for example through an update from Git. OK would have "
450 "written the old list back. Close this dialog and open it again "
451 "to see the new list."));
452 return;
453 }
454 if (m_initOnAccept) {
455 if (m_listRead) {
456 // Compared again by the backend at the last point before it writes:
457 // a pull in a terminal can still land between here and there.
458 Pass::RecipientsForEditing read;
459 read.state = static_cast<Pass::RecipientsForEditing::State>(m_readState);
460 read.recipients = m_readRecipients;
461 // Refused there (changed after all, or another operation runs): the
462 // dialog stays open with the selection, as the message asks to save
463 // again.
464 if (!m_pass->initIfUnchanged(m_dir, m_userList, m_dir, m_passStore,
465 read)) {
466 return;
467 }
468 } else {
469 m_pass->Init(m_dir, m_userList);
470 }
471 }
472
473 QDialog::accept();
474}

518void UsersDialog::populateList(const QString &filter) {
519 // Invalidate cached datetime so expiry checks use fresh current time
520 m_cachedDateTimeValid = false;
521
522 QString patternString = "*" + filter + "*";
523 if (m_cachedPatternString != patternString) {
524 // A user ID is not a path: "*" and "?" must match a "/" as well, or a
525 // key named "ACME A/S" or "(dev/ops)" drops out of the search and stays
526 // ticked unseen while the listed rows are unticked.
527 QRegularExpression re(
528 QRegularExpression::wildcardToRegularExpression(
529 patternString, QRegularExpression::NonPathWildcardConversion),
530 QRegularExpression::CaseInsensitiveOption);
531 if (re.isValid()) {
532 m_cachedNameFilter = re;
533 m_cachedPatternString = patternString;
534 }
535 }
536 const QRegularExpression &nameFilter = m_cachedNameFilter;
537 // Sorted once at the end: with sorting on, each badge and tick below
538 // re-sorted every row added so far, and a .gpg-id of thousands of
539 // unknown recipients kept the dialog from opening for minutes.
540 ui->listWidget->setSortingEnabled(false);
541 ui->listWidget->clear();
542
543 for (int i = 0; i < m_userList.size(); ++i) {
544 const auto &user = m_userList.at(i);
545 if (!passesFilter(user, filter, nameFilter)) {
546 continue;
547 }
548
549 auto *item = new QListWidgetItem(buildUserText(user), ui->listWidget);
550 applyUserStyling(item, user);
551 item->setCheckState(user.enabled ? Qt::Checked : Qt::Unchecked);
552 item->setData(Qt::UserRole, QVariant::fromValue(i));
553 ui->listWidget->addItem(item);
554 }
555 // setSortingEnabled() only sets the flag; sortItems() sorts.
556 ui->listWidget->sortItems();
557 ui->listWidget->setSortingEnabled(true);
558 updateOkButton();
559}

561bool UsersDialog::passesFilter(const UserInfo &user, const QString &filter,
562 const QRegularExpression &nameFilter) const {
563 if (!filter.isEmpty() && !nameFilter.match(user.name).hasMatch()) {
564 return false;
565 }
566 // A ticked key is one the folder is encrypted to: it is shown whatever
567 // its state, or an expired or missing recipient stays in .gpg-id unseen and
568 // cannot be unticked. "Show unusable keys" is about keys to add. A line
569 // listed unticked without a lookup is shown too: OK would drop it unseen.
570 if (user.enabled || ui->checkBox->isChecked() ||
571 m_notLookedUp.contains(user.key_id)) {
572 return true;
573 }
574 return user.isValid() && !isUserExpired(user);
575}

577auto UsersDialog::isUserExpired(const UserInfo &user) const -> bool {
578 if (!m_cachedDateTimeValid) {
579 m_cachedCurrentDateTime = QDateTime::currentDateTime();
580 m_cachedDateTimeValid = true;
581 }
582 // gpg marks a key it already judged expired with validity 'e'.
583 return user.validity == 'e' ||
584 (user.expiry.isValid() && m_cachedCurrentDateTime > user.expiry);
585}

599auto shortened(const QString &text) -> QString {
600 constexpr qsizetype kMaxShown = 256;
601 if (text.size() <= kMaxShown) {
602 return text;
603 }
604 qsizetype cut = kMaxShown;
605 if (text.at(cut - 1).isHighSurrogate()) {
606 --cut;
607 }
608 return text.left(cut) + QChar(0x2026);
609}

612QString UsersDialog::buildUserText(const UserInfo &user) const {
613 // One line for the user ID: a line separator in it (gpg escapes only
614 // ASCII) drew a line that looked like the key's fingerprint.
615 // The key ID too: a recipient no key answers to is the .gpg-id line as
616 // written, which may hold the same separators.
617 QString text = Util::displayable(shortened(user.name), false) + "\n" +
618 Util::displayable(shortened(user.key_id), false);
619 if (user.created.isValid()) {
620 text += " " + tr("created") + " " +
621 QLocale::system().toString(user.created, QLocale::ShortFormat);
622 }
623 if (user.expiry.isValid()) {
624 text += " " + tr("expires") + " " +
625 QLocale::system().toString(user.expiry, QLocale::ShortFormat);
626 }
627 return text;
628}

630void UsersDialog::applyUserStyling(QListWidgetItem *item,
631 const UserInfo &user) const {
632 const QString originalText = item->text();
633 // Status badge first: an own key can be expired too, and then the marker
634 // matters more than the "you can decrypt with this" colour.
635 bool badged = true;
636 // Expired before invalid: gpg gives an expired key validity 'e', which is
637 // not a valid one, and "expired" says what to do about it.
638 if (isUserExpired(user)) {
639 // A lone dark-red foreground is unreadable on dark themes, and gpg
640 // refuses to encrypt to expired keys anyway.
641 item->setBackground(Qt::darkRed);
642 item->setForeground(Qt::white);
643 item->setText(tr("[EXPIRED] ") + originalText);
644 } else if (!user.isValid()) {
645 item->setBackground(Qt::darkRed);
646 item->setForeground(Qt::white);
647 item->setText(tr("[INVALID] ") + originalText);
648 } else if (!user.fullyValid()) {
649 item->setBackground(Qt::darkYellow);
650 item->setForeground(Qt::white);
651 item->setText(tr("[PARTIAL] ") + originalText);
652 } else {
653 item->setText(originalText);
654 badged = false;
655 }
656 if (user.have_secret) {
657 if (!badged) {
658 item->setForeground(QApplication::palette().color(QPalette::Link));
659 }
660 QFont font = item->font();
661 font.setBold(true);
662 item->setFont(font);
663 }

679 !m_userList[idx].key_id.isEmpty() &&

687void UsersDialog::on_importKeyButton_clicked() {
688 ImportKeyDialog dialog(m_gpgExe, this);
689 if (dialog.exec() != QDialog::Accepted) {
690 return;
691 }
692
693 // Non-empty after Accepted, but an empty id would make the endsWith()
694 // below match the first listed key.
695 const QString importedKey = dialog.importedKeyId();
696 if (importedKey.isEmpty()) {
697 return;
698 }
699
700 // loadGpgKeys() rebuilds the key list from the keyring and drops every tick.
701 // Remember which recipients were selected (the .gpg-id list and the user's
702 // own ticks, including any "key not found" rows) and re-apply them after, so
703 // importing a key does not silently remove the other recipients.
704 QStringList selected;
705 for (const UserInfo &user : std::as_const(m_userList)) {
706 if (user.enabled) {
707 selected.append(user.key_id);
708 }
709 }
710
711 if (!loadGpgKeys()) {
712 return;
713 }
714
715 // Re-apply the remembered selection. Most saved identifiers are a key ID that
716 // is in the rebuilt list, so match those exactly first (no gpg lookup). An
717 // identifier that does not match a row is one that was listed as "key not
718 // found" - a short ID or email that gpg could not resolve, or whose key has
719 // just been imported and now shows under its fingerprint. Hand those to
720 // selectRecipients(), which resolves each against the refreshed keyring,
721 // ticks the matching row, and re-adds a "key not found" row for any that
722 // still does not resolve.
723 QStringList unresolved;
724 for (const QString &keyId : std::as_const(selected)) {
725 bool matched = false;
726 for (UserInfo &user : m_userList) {
727 if (user.key_id == keyId) {
728 user.enabled = true;
729 matched = true;
730 }
731 }
732 if (!matched) {
733 unresolved.append(keyId);
734 }
735 }
736 if (!unresolved.isEmpty()) {
737 selectRecipients(unresolved);
738 }
739
740 // Clear the filter so the just-imported key is visible. setText("") still
741 // emits textChanged once, so don't double-populate.
742 {
743 const QSignalBlocker blocker(ui->lineEdit);
744 ui->lineEdit->clear();
745 }
746 populateList(QString());
747
748 selectImportedKey(importedKey);
749}
